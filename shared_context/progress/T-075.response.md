import SwiftUI
import UIKit
import MetalKit
import simd

@MainActor
public final class OrderedQuadCanvasView: MTKView, MTKViewDelegate {
    private let metalContext: MetalContext
    private var orderedQuadPass: OrderedQuadPass?
    private var lastReportedError: String?

    public var quads: [OrderedTextureQuad] = []
    public var onError: (String) -> Void

    public init(metalContext: MetalContext, onError: @escaping (String) -> Void) {
        self.metalContext = metalContext
        self.onError = onError

        super.init(frame: .zero, device: metalContext.device)

        colorPixelFormat = .bgra8Unorm
        depthStencilPixelFormat = .invalid
        clearColor = MTLClearColor(red: 0, green: 0, blue: 0, alpha: 0)
        isOpaque = false
        backgroundColor = .clear
        layer.isOpaque = false
        isPaused = false
        enableSetNeedsDisplay = false
        preferredFramesPerSecond = 120
        delegate = self

        do {
            orderedQuadPass = try OrderedQuadPass(
                device: metalContext.device,
                shaderLibrary: metalContext.shaderLibrary,
                colorPixelFormat: colorPixelFormat
            )
        } catch {
            orderedQuadPass = nil
            report(error: error)
        }
    }

    @available(*, unavailable)
    required init(coder: NSCoder) {
        fatalError("init(coder:) has not been implemented")
    }

    public func mtkView(_ view: MTKView, drawableSizeWillChange size: CGSize) {
    }

    public func draw(in view: MTKView) {
        guard let orderedQuadPass = orderedQuadPass else { return }

        let width = bounds.width
        let height = bounds.height
        guard width.isFinite, height.isFinite, width > 0, height > 0 else {
            return
        }

        guard let renderPassDescriptor = currentRenderPassDescriptor,
              let drawable = currentDrawable,
              let commandBuffer = metalContext.commandQueue.makeCommandBuffer(),
              let encoder = commandBuffer.makeRenderCommandEncoder(
                descriptor: renderPassDescriptor
              ) else {
            report(message: "OrderedQuadCanvasView: unable to create render pass resources.")
            return
        }

        do {
            try orderedQuadPass.encode(
                encoder: encoder,
                quads: quads,
                viewportSize: SIMD2<Float>(Float(width), Float(height))
            )
            encoder.endEncoding()
            commandBuffer.present(drawable)
            commandBuffer.commit()
            lastReportedError = nil
        } catch {
            encoder.endEncoding()
            commandBuffer.commit()
            report(error: error)
        }
    }

    private func report(error: Error) {
        report(message: error.localizedDescription)
    }

    private func report(message: String) {
        guard lastReportedError != message else { return }
        lastReportedError = message
        onError(message)
    }
}

@MainActor
public struct OrderedQuadMetalView: UIViewRepresentable {
    public let metalContext: MetalContext
    public let quads: [OrderedTextureQuad]
    public let onError: (String) -> Void

    public init(

    ) {
        self.metalContext = metalContext
        self.quads = quads
        self.onError = onError
    }

    public func makeUIView(context: Context) -> OrderedQuadCanvasView {
        let view = OrderedQuadCanvasView(metalContext: metalContext, onError: onError)
        view.quads = quads
        return view
    }

    public func updateUIView(_ uiView: OrderedQuadCanvasView, context: Context) {
        uiView.quads = quads
        uiView.onError = onError
    }
}

// App/ChallengeMapQuadAdapter.swift

import Foundation
import CoreGraphics
import Metal
import simd
import FNPBRender
import FNPBUI

@MainActor
final class ChallengeMapQuadAdapter {
    private let textureLoader: TextureLoader
    private var textureCache: [ChallengeMapTextureKey: TextureHandle] = [:]

    init(device: MTLDevice) {
        self.textureLoader = TextureLoader(device: device)
    }

    func makeQuads(
        components: [ChallengeMapLayoutComponent],
        viewportSize: CGSize
    ) throws -> [OrderedTextureQuad] {
        guard viewportSize.width.isFinite,
              viewportSize.height.isFinite,
              viewportSize.width > 0,
              viewportSize.height > 0 else {
            throw ChallengeMapQuadAdapterError.invalidViewport(viewportSize)
        }

        var orderedQuads: [OrderedTextureQuad] = []
        orderedQuads.reserveCapacity(components.count)

        for (componentIndex, component) in components.enumerated() {
            guard let texturePath = component.texturePath else { continue }
            guard component.screenVisibility else { continue }
            guard component.absoluteRGBA.w != 0 else { continue }

            let resolved: ChallengeMapResolvedTexture
            switch ChallengeMapTextureResolver.resolve(
                texturePath: texturePath,
                iconIndex: component.iconIndex
            ) {
            case .success(let value):
                resolved = value
            case .failure(let error):
                throw ChallengeMapQuadAdapterError.textureResolutionFailed(
                    componentIndex: componentIndex,
                    error: error
                )
            }

            let sourcePositions = ChallengeMapLayoutGeometry.positions(
                for: component,
                deviceWidth: viewportSize.width,
                deviceHeight: viewportSize.height
            )
            let sourceTexCoords = ChallengeMapLayoutGeometry.texCoords(
                for: component,
                resolvedUV: resolved.resolvedUV
            )

            guard sourcePositions.count == 4,
                  sourceTexCoords.count == 4,
                  sourcePositions.allSatisfy({ $0.x.isFinite && $0.y.isFinite }),
                  sourceTexCoords.allSatisfy({ $0.x.isFinite && $0.y.isFinite }) else {
                throw ChallengeMapQuadAdapterError.invalidGeometry(
                    componentIndex: componentIndex
                )
            }

            let positions = sourcePositions.map {
                SIMD2<Float>(Float($0.x), Float($0.y))
            }
            let uvs = sourceTexCoords.map {
                SIMD2<Float>(Float($0.x), Float($0.y))
            }

            guard positions.allSatisfy({ $0.x.isFinite && $0.y.isFinite }),
                  uvs.allSatisfy({ $0.x.isFinite && $0.y.isFinite }) else {
                throw ChallengeMapQuadAdapterError.invalidGeometry(
                    componentIndex: componentIndex
                )
            }

            guard let vertices = OrderedTextureQuad.makeVertices(
                positions: positions,
                uvs: uvs,
                absoluteRGBA: component.absoluteRGBA,
                screenVisibility: true
            ) else {
                throw ChallengeMapQuadAdapterError.invalidGeometry(
                    componentIndex: componentIndex
                )
            }

            let texture = try textureHandle(for: resolved.textureKey)

            guard let quad = OrderedTextureQuad(
                texture: texture,
                componentIndex: componentIndex,
                vertices: vertices
            ) else {
                throw ChallengeMapQuadAdapterError.invalidQuad(
                    componentIndex: componentIndex
                )
            }

            orderedQuads.append(quad)
        }

        return orderedQuads
    }

    private func textureHandle(for key: ChallengeMapTextureKey) throws -> TextureHandle {
        if let cached = textureCache[key] {
            return cached
        }

        guard let url = UIArtwork.url(for: key.rawValue) else {
            throw ChallengeMapQuadAdapterError.missingTextureURL(key.rawValue)
        }

        let asset = TextureAsset(
            id: "challengeMap/" + key.rawValue,
            source: .url(url),
            srgb: false,
            generateMipmaps: false,
            premultipliedAlpha: false
        )

        let handle = try textureLoader.load(asset)
        textureCache[key] = handle
        return handle
    }
}

enum ChallengeMapQuadAdapterError: LocalizedError {
    case invalidViewport(CGSize)
    case invalidGeometry(componentIndex: Int)
    case textureResolutionFailed(componentIndex: Int, error: ChallengeMapTextureResolutionError)
    case missingTextureURL(String)
    case invalidQuad(componentIndex: Int)

    var errorDescription: String? {
        switch self {
        case .invalidViewport(let size):
            return "ChallengeMapQuadAdapter: invalid viewport \(size.width)x\(size.height)."
        case .invalidGeometry(let componentIndex):
            return "ChallengeMapQuadAdapter: invalid geometry for component \(componentIndex)."
        case .textureResolutionFailed(let componentIndex, let error):
            return "ChallengeMapQuadAdapter: texture resolution failed for component \(componentIndex): \(error)."
        case .missingTextureURL(let name):
            return "ChallengeMapQuadAdapter: missing texture URL for \(name)."
        case .invalidQuad(let componentIndex):
            return "ChallengeMapQuadAdapter: failed to construct quad for component \(componentIndex)."
        }
    }
}
