//
//  OriginalStringHash.swift
//  FNPBCore
//
//  T-049: 原版 FUN_002f69c4 字符串哈希。
//  纯核心实现，不依赖 Foundation，不接 UI / 存档。
//

public enum OriginalStringHash {

    /// 原版 FUN_002f69c4：返回最后混合后的 c。
    ///
    /// - 空串或首字节为 NUL：返回 0。
    /// - ASCII `A...Z` 先转小写；其他字节原样。
    /// - 12 字节一组走主循环，尾部按剩余 1...11 字节混合。
    /// - 全部为 UInt32 模运算，使用 `&+` / `&-` 避免 Swift 溢出 trap。
    public static func hash(_ value: String) -> UInt32 {
        hash(utf8: Array(value.utf8))
    }

    @inline(__always)
    private static func hash(utf8 bytes: [UInt8]) -> UInt32 {
        // 汇编起始处：r0 == 0 或 *r0 == 0 直接返回 0。
        guard let first = bytes.first, first != 0 else { return 0 }

        // 汇编对每个字节用 ldrsb 读、仅对 ASCII A-Z 加 0x20，再 strb 写回。
        var lower = bytes
        for i in lower.indices {
            let byte = lower[i]
            if byte >= 0x41 && byte <= 0x5A {
                lower[i] = byte &+ 0x20
            }
        }

        var a: UInt32 = 0x9e37_79b9
        var b: UInt32 = 0x9e37_79b9
        var c: UInt32 = 0x0000_0805

        let totalLength = UInt32(lower.count)
        var offset = 0

        // 主循环：剩余 >= 12 时处理一个小端 12 字节块。
        while lower.count - offset >= 12 {
            a = a &+ word(lower, at: offset)
            b = b &+ word(lower, at: offset + 4)
            c = c &+ word(lower, at: offset + 8)
            mix(&a, &b, &c)
            offset += 12
        }

        // 002f6b2a / 002f6b30：尾部处理前 c += 原始总长度。
        c = c &+ totalLength

        let remaining = lower.count - offset
        if remaining > 0 {
            // 002f6b5c...002f6bec：按剩余长度跳转。
            // byte0/1/2、byte4/5/6、byte8/9/10 为有符号扩展；
            // byte3/7/11 为无符号高位。
            switch remaining {
            case 11:
                c = c &+ (UInt32(lower[offset + 10]) << 24)
                fallthrough
            case 10:
                c = c &+ (signed(lower[offset + 9]) << 16)
                fallthrough
            case 9:
                c = c &+ (signed(lower[offset + 8]) << 8)
                fallthrough
            case 8:
                b = b &+ (UInt32(lower[offset + 7]) << 24)
                fallthrough
            case 7:
                b = b &+ (signed(lower[offset + 6]) << 16)
                fallthrough
            case 6:
                b = b &+ (signed(lower[offset + 5]) << 8)
                fallthrough
            case 5:
                b = b &+ signed(lower[offset + 4])
                fallthrough
            case 4:
                a = a &+ (UInt32(lower[offset + 3]) << 24)
                fallthrough
            case 3:
                a = a &+ (signed(lower[offset + 2]) << 16)
                fallthrough
            case 2:
                a = a &+ (signed(lower[offset + 1]) << 8)
                fallthrough
            case 1:
                a = a &+ signed(lower[offset + 0])
            default:
                break
            }
        }

        // 002f6bee...002f6c38：最终 mix。
        mix(&a, &b, &c)
        return c
    }

    /// 主循环中每 4 字节的拼装：
    /// b0 有符号低字节 + b1 有符号<<8 + b2 有符号<<16 + b3 无符号<<24。
    @inline(__always)
    private static func word(_ bytes: [UInt8], at offset: Int) -> UInt32 {
        signed(bytes[offset])
        &+ (signed(bytes[offset + 1]) << 8)
        &+ (signed(bytes[offset + 2]) << 16)
        &+ (UInt32(bytes[offset + 3]) << 24)
    }

    /// ARM ldrsb 的 32 位符号扩展语义。
    @inline(__always)
    private static func signed(_ byte: UInt8) -> UInt32 {
        UInt32(bitPattern: Int32(Int8(bitPattern: byte)))
    }

    /// Jenkins lookup2 的 9 步混合。全部使用 UInt32 模运算。
    @inline(__always)
    private static func mix(_ a: inout UInt32, _ b: inout UInt32, _ c: inout UInt32) {
        a = a &- b
        a = a &- c
        a ^= c >> 13

        b = b &- c
        b = b &- a
        b ^= a << 8

        c = c &- a
        c = c &- b
        c ^= b >> 13

        a = a &- b
        a = a &- c
        a ^= c >> 12

        b = b &- c
        b = b &- a
        b ^= a << 16

        c = c &- a
        c = c &- b
        c ^= b >> 5

        a = a &- b
        a = a &- c
        a ^= c >> 3

        b = b &- c
        b = b &- a
        b ^= a << 10

        c = c &- a
        c = c &- b
        c ^= b >> 15
    }
}

final class OriginalStringHashTests: XCTestCase {

    // MARK: 空串

    func testEmptyStringReturnsZero() {
        XCTAssertEqual(OriginalStringHash.hash(""), 0)
    }

    // MARK: ASCII 大小写等价

    func testASCIICaseEquivalence() {
        // 手算常量；"APPLE"/"Apple" 经 A-Z 转小写后与 "apple" 完全一致。
        XCTAssertEqual(OriginalStringHash.hash("apple"), 0xAA39_7683)
        XCTAssertEqual(OriginalStringHash.hash("APPLE"), 0xAA39_7683)
        XCTAssertEqual(OriginalStringHash.hash("Apple"), 0xAA39_7683)
    }

    // MARK: 长度 1...13 边界

    func testLengthBoundaries1Through13() {
        for n in 1...13 {
            let s = String(repeating: "a", count: n)
            XCTAssertEqual(OriginalStringHash.hash(s),
                           ReferenceHash.hash(s),
                           "length \(n)")
        }

        // 长度 1 的手算常量。
        XCTAssertEqual(OriginalStringHash.hash("a"), 0x25AB_5C50)
    }

    // MARK: 12 字节主循环 / 13 字节主循环 + 1 字节尾部

    func testTwelveByteMainLoop() {
        let s12 = "abcdefghijkl"
        let s13 = "abcdefghijklm"

        XCTAssertEqual(OriginalStringHash.hash(s12), ReferenceHash.hash(s12))
        XCTAssertEqual(OriginalStringHash.hash(s13), ReferenceHash.hash(s13))
    }

    // MARK: 真实名称

    func testRealNames() {
        let names = [
            "apple",
            "FastFruit1",
            "bombPairs1",
            "LEFT_RIGHT"
