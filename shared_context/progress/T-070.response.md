```swift
//
//  ChallengeMapTextureResolver.swift
//  FNPBUI
//
//  挑战地图（UIScreen: challengeMap）原始 texturePath 到资源语义 key 的精确解析。
//
//  约束：
//  - 只接受资源生成脚本已经验证过的精确路径字符串（区分大小写），不做前缀匹配、
//    大小写归一化、basename 猜测或任何模糊回退。
//  - challengeAtlas.tex 只有在携带 iconIndex 时才使用运行时 UV（8×8 图集）；
//    非 challengeAtlas 路径携带 iconIndex 视为明确错误，绝不把静态 UV 当成运行时图标。
//  - 图片装载复用 `UIArtwork.image`；重复解码可留待后续缓存，但不改变资源语义。
//

import UIKit

// MARK: - 资源语义 key

/// 挑战地图所需、且已由资源生成脚本验证过的资源语义 key。
public enum ChallengeMapTextureKey: String, Sendable, Equatable, CaseIterable {
    case challengeFlipper = "challengeFlipper"
    case challengeBasicPoster = "challengeBasicPoster"
    case challengePosterDeco01 = "challengePosterDeco01"
    case challengeIconSlot = "challengeIconSlot"
    case challengeAtlas = "challengeAtlas"
    case challengeTick = "challengeTick"
    case challengeCross = "challengeCross"
    case challengePosterDeco02 = "challengePosterDeco02"
    case challengeLocations = "challengeLocations"
}

// MARK: - 解析结果值类型

/// 运行时解析出的 UV 起点与尺寸（归一化到 0...1）。
/// 负的尺寸分量在后续几何计算中原样保留，本类型不对其做任何修正。
public struct ChallengeMapResolvedUV: Sendable, Equatable {
    public var u: Float
    public var v: Float
    public var width: Float
    public var height: Float

    public init(u: Float, v: Float, width: Float, height: Float) {
        self.u = u
        self.v = v
        self.width = width
        self.height = height
    }
}

/// 一次成功解析的结果。
public struct ChallengeMapResolvedTexture {
    /// 明确的资源语义 key（用于加载 PNG / asset catalog）。
    public let textureKey: ChallengeMapTextureKey

    /// 已装载的 UIImage。
    public let image: UIImage

    /// - `nil`：调用方应使用组件自身的 `texCoordPos` / `texCoordSize` 静态 UV。
    /// - 非 `nil`：使用图集运行时图标 UV，调用方不得再用静态 UV 覆盖。
    public let resolvedUV: ChallengeMapResolvedUV?

    public init(textureKey: ChallengeMapTextureKey,
                image: UIImage,
                resolvedUV: ChallengeMapResolvedUV?) {
        self.textureKey = textureKey
        self.image = image
        self.resolvedUV = resolvedUV
    }
}

// MARK: - 明确错误

public enum ChallengeMapTextureResolutionError: Error, Equatable, Sendable {
    /// 组件没有 texturePath（例如纯色块或布局占位组件）。
    case missingTexturePath

    /// 原始 texturePath 不在已验证映射表中。
    case unsupportedTexturePath(String)

    /// 资源 key 已知，但 bundle / asset catalog 中没有对应图片。
    case imageUnavailable(textureKey: ChallengeMapTextureKey)

    /// 非 challengeAtlas 路径携带了 iconIndex：
    /// 无法计算图集 UV，也不能把静态 UV 当成运行时图标。
    case iconIndexRequiresChallengeAtlas(texturePath: String)

    /// challengeAtlas 的 iconIndex 超出 0..<64。
    case iconIndexOutOfRange(Int32)
}

// MARK: - 解析器

public enum ChallengeMapTextureResolver {

    /// 已验证的精确映射表。键为原始 texturePath，必须逐字符（含大小写）匹配。
    ///
    /// 注意 `textures/adventureprogress/iconSlot.tex` 与
    /// `textures/adventureProgress/iconSlot.tex` 是两条独立条目，二者都映射到
    /// `challengeIconSlot`；不得用大小写不敏感查找合并。
    public static let texturePathToKey: [String: ChallengeMapTextureKey] = [
        "textures/adventureprogress/flipper.tex": .challengeFlipper,
        "textures/menus/basicPoster.tex": .challengeBasicPoster,
        "textures/menus/posterDeco01.tex": .challengePosterDeco01,
        "textures/adventureprogress/iconSlot.tex": .challengeIconSlot,
        "textures/adventureProgress/iconSlot.tex": .challengeIconSlot,
        "textures/adventureprogress/challengeAtlas.tex": .challengeAtlas,
        "textures/adventureprogress/tick.tex": .challengeTick,
        "textures/adventureprogress/cross.tex": .challengeCross,
        "textures/menus/posterDeco02.tex": .challengePosterDeco02,
        "textures/adventureProgress/locations.tex": .challengeLocations
    ]

    /// challengeAtlas 的图集网格配置。
    /// 原控制器以 `(index & 7) / 8`、`(index >> 3) / 8` 设置 UV，每格 1/8 × 1/8。
    public static let atlasGridSize: Int32 = 8
    public static let atlasCellCount: Int32 = 64

    /// 解析原始 texturePath（及可选 iconIndex），返回资源 key、UIImage 与运行时 UV。
    public static func resolve(
        texturePath: String?,
        iconIndex: Int32?
    ) -> Result<ChallengeMapResolvedTexture, ChallengeMapTextureResolutionError> {

        guard let texturePath else {
            return .failure(.missingTexturePath)
        }

        guard let textureKey = texturePathToKey[texturePath] else {
            return .failure(.unsupportedTexturePath(texturePath))
        }

        let resolvedUV: ChallengeMapResolvedUV?

        if let iconIndex = iconIndex {
            guard textureKey == .challengeAtlas else {
                return .failure(.iconIndexRequiresChallengeAtlas(texturePath: texturePath))
            }
            guard iconIndex >= 0, iconIndex < atlasCellCount else {
                return .failure(.iconIndexOutOfRange(iconIndex))
            }

            let index = Int(iconIndex)
            resolvedUV = ChallengeMapResolvedUV(
                u: Float(index & 7) / 8,
                v: Float(index >> 3) / 8,
                width: 1.0 / 8.0,
                height: 1.0 / 8.0
            )
        } else {
            // 非图集运行时图标：显式表示“使用组件自带的静态 UV”。
            resolvedUV = nil
        }

        guard let image = UIArtwork.image(textureKey.rawValue) else {
            return .failure(.imageUnavailable(textureKey: textureKey))
        }

        return .success(ChallengeMapResolvedTexture(
            textureKey: textureKey,
            image: image,
            resolvedUV: resolvedUV
        ))
    }

    /// 可选便捷入口：失败时返回 nil。
    /// 需要区分“路径不支持 / 图片缺失 / iconIndex 非法”时请直接使用 `resolve`。
    public static func resolveIfAvailable(
        texturePath: String?,
        iconIndex: Int32?
    ) -> ChallengeMapResolvedTexture? {
        try? resolve(texturePath: texturePath, iconIndex: iconIndex).get()
    }
}
```

```swift
//
//  ChallengeMapLayoutGeometry.swift
//  FNPBUI
//
//  挑战地图组件的二维屏幕四边形几何。
//
//  计算顺序固定为：
//  1. 组件局部顶点 (0,0), (width,0), (width,height), (0,height) 先减去 effectiveOrigin；
//  2. 应用世界矩阵的二维仿射部分：
//         CGAffineTransform(a: m0, b: m1, c: m4, d: m5, tx: m12, ty: m13)
//  3. 再应用各轴独立的设备缩放：x = deviceWidth / 480，y = deviceHeight / 320。
//
//  组合顺序通过 `local.concatenating(deviceScale)` 表达（先局部，后设备缩放）。
//  这里不做等比缩放，也不做居中缩放，不交换矩阵顺序。
//
//  UV 四角固定为：
//      pos, pos + [size.x, 0], pos + size, pos + [0, size.y]
//  负的 size 分量原样保留，不取绝对值、不交换角点。
//

import CoreGraphics

// MARK: - 输出值类型

/// 四边形上的一个顶点：屏幕位置与对应 UV。
public struct ChallengeMapLayoutVertex: Sendable, Equatable {
    public var position: CGPoint
    public var texCoord: CGPoint

    public init(position: CGPoint, texCoord: CGPoint) {
        self.position = position
        self.texCoord = texCoord
    }
}

/// 挑战地图单个组件的四边形。
/// `vertices` 恒为 4 个，顺序固定（与几何/UV 构造顺序一一对应）：
///   0: 局部 (0, 0)
///   1: 局部 (width, 0)
///   2: 局部 (width, height)
///   3: 局部 (0, height)
public struct ChallengeMapLayoutQuad: Sendable, Equatable {
    public var vertices: [ChallengeMapLayoutVertex]

    public init(vertices: [ChallengeMapLayoutVertex]) {
        self.vertices = vertices
    }
}

// MARK: - 几何计算

public enum ChallengeMapLayoutGeometry {

    /// 原版挑战地图 UIScreen 的基准尺寸。
    public static let referenceWidth: CGFloat = 480
    public static let referenceHeight: CGFloat = 320

    /// 组件的二维局部仿射矩阵。
    ///
    /// 列主序 4×4 中：
    ///   m0  = column0.x
    ///   m1  = column0.y
    ///   m4  = column1.x
    ///   m5  = column1.y
    ///   m12 = column3.x
    ///   m13 = column3.y
    ///
    /// CGAffineTransform 的点变换为：
    ///   x' = a * x + c * y + tx
    ///   y' = b * x + d * y + ty
    public static func localAffineTransform(
        for component: ChallengeMapLayoutComponent
    ) -> CGAffineTransform {
        let matrix = component.worldMatrix
        return CGAffineTransform(
            a: CGFloat(matrix.column0.x),
            b: CGFloat(matrix.column0.y),
            c: CGFloat(matrix.column1.x),
            d: CGFloat(matrix.column1.y),
            tx: CGFloat(matrix.column3.x),
            ty: CGFloat(matrix.column3.y)
        )
    }

    /// 各轴独立的设备缩放。
    /// 不做等比缩放，也不做居中缩放。
    public static func deviceScaleTransform(
        deviceWidth: CGFloat,
        deviceHeight: CGFloat
    ) -> CGAffineTransform {
        CGAffineTransform(
            scaleX: deviceWidth / referenceWidth,
            y: deviceHeight / referenceHeight
        )
    }

    /// 先局部矩阵、后设备缩放的组合矩阵。
    ///
    /// `CGAffineTransform.concatenating(_:)` 的语义是：点先经过 self，再经过参数。
    /// 因此这里使用 `local.concatenating(deviceScale)`，不能反过来。
    public static func deviceSpaceAffineTransform(
        for component: ChallengeMapLayoutComponent,
        deviceWidth: CGFloat,
        deviceHeight: CGFloat
    ) -> CGAffineTransform {
        localAffineTransform(for: component)
            .concatenating(deviceScaleTransform(
                deviceWidth: deviceWidth,
                deviceHeight: deviceHeight
            ))
    }

    /// 生成四个屏幕位置。
    /// 顺序固定为局部 (0,0), (width,0), (width,height), (0,height)；
    /// 局部顶点先减去 `effectiveOrigin`，再应用组合矩阵。
    public static func positions(
        for component: ChallengeMapLayoutComponent,
        deviceWidth: CGFloat,
        deviceHeight: CGFloat
    ) -> [CGPoint] {
        let originX = CGFloat(component.effectiveOrigin.x)
        let originY = CGFloat(component.effectiveOrigin.y)
        let width = CGFloat(component.width)
        let height = CGFloat(component.height)

        let localPositions: [CGPoint] = [
            CGPoint(x: -originX, y: -originY),
            CGPoint(x: width - originX, y: -originY),
            CGPoint(x: width - originX, y: height - originY),
            CGPoint(x: -originX, y: height - originY)
        ]

        let transform = deviceSpaceAffineTransform(
            for: component,
            deviceWidth: deviceWidth,
            deviceHeight: deviceHeight
        )

        return localPositions.map { $0.applying(transform) }
    }

    /// 生成四个 UV。
    /// 若 `resolvedUV` 非 nil（challengeAtlas 运行时图标），使用解析出的 UV；
    /// 否则使用组件自身的静态 `texCoordPos` / `texCoordSize`。
    ///
    /// 顺序：
    ///   pos,
    ///   pos + [size.x, 0],
    ///   pos + size,
    ///   pos + [0, size.y]
    ///
    /// 负的 size 分量原样保留。
    public static func texCoords(
        for component: ChallengeMapLayoutComponent,
        resolvedUV: ChallengeMapResolvedUV?
    ) -> [CGPoint] {
        let pos: SIMD2<Float>
        let size: SIMD2<Float>

        if let resolvedUV {
            pos = SIMD2(resolvedUV.u, resolvedUV.v)
            size = SIMD2(resolvedUV.width, resolvedUV.height)
        } else {
            pos = component.texCoordPos
            size = component.texCoordSize
        }

        let u = CGFloat(pos.x)
        let v = CGFloat(pos.y)
        let sizeX = CGFloat(size.x)
        let sizeY = CGFloat(size.y)

        return [
            CGPoint(x: u, y: v),
            CGPoint(x: u + sizeX, y: v),
            CGPoint(x: u + sizeX, y: v + sizeY),
            CGPoint(x: u, y: v + sizeY)
        ]
    }

    /// 组合位置与 UV，生成可直接供绘制层使用的四边形。
    /// 本函数不决定显示树遍历顺序或绘制顺序。
    public static func quad(
        for component: ChallengeMapLayoutComponent,
        resolvedUV: ChallengeMapResolvedUV?,
        deviceWidth: CGFloat,
        deviceHeight: CGFloat
    ) -> ChallengeMapLayoutQuad {
        let positions = positions(
            for: component,
            deviceWidth: deviceWidth,
            deviceHeight: deviceHeight
        )
        let texCoords = texCoords(
            for: component,
            resolvedUV: resolvedUV
        )

        let vertices = zip(positions, texCoords).map { pair in
            ChallengeMapLayoutVertex(position: pair.0, texCoord: pair.1)
        }
        return ChallengeMapLayoutQuad(vertices: vertices)
    }
}
```

补充说明：

- 解析表严格按任务给定字符串逐字匹配，`adventureprogress/iconSlot.tex` 与 `adventureProgress/iconSlot.tex` 分别保留，不做大小写模糊。
- `challengeAtlas` 带 `iconIndex` 时 UV 固定为 `Float(index & 7)/8`、`Float(index >> 3)/8`、`1/8 × 1/8`，并校验 `0..<64`；非 atlas 带 `iconIndex` 返回明确错误，避免把静态 UV 误当运行时图标。
- 几何组合使用 `local.concatenating(deviceScale)`，先局部矩阵后设备非等比缩放；局部点先减 `effectiveOrigin`；UV 顺序与负 size 均原样保留。
