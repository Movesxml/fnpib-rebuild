## 设计说明

**本轮的两个交付文件**

1. `App/AppViewModel.swift`（新增）：`@Observable @MainActor final class AppViewModel: FNPBUIViewModel`
   - 会话状态机 `mainMenu → modeSelect → playing ⇄ paused → results`；
   - `send(_:)` 覆盖任务卡列出的 6 个 `UIIntent`，并留 `default` 兜底（主菜单→模式选择导航意图名称在注入材料中不可见）；
   - `refreshSnapshot()` 每帧由 `GameHost.handleFrame` 末尾调用：`score ← ScoreSystem.total`、`bestScore ← saveData`、`combo ← clamp 双档位`、`result` 转入 results 时组装；用 `GameSnapshot: Equatable` 做**等值去抖**，避免 120Hz 下每帧无条件触发 Observation 失效。
   - 持有关系：`GameHost`（强）→ `AppViewModel`（weak）→ `GameHost`，打破 (a) 循环；宿主只暴露 `AppViewModel`，不使用 `@Published`，满足 (b)。

2. `App/FNPBApp.swift`（重写）：`ContentView` 按 `vm.snapshot.phase` 挂五屏；`GameHost` 用真实服务（`SliceTester`/`ComboTracker`/`ScoreSystem`/`StatsRecorder`/`WaveDirector`）替换 5 个占位，只保留 `PlaceholderEffectEmitter` 与 `SplitMix64Random`；新增 `startGame(mode:)` / `restart()` / `quitToMenu()` / `pauseGame()` / `resumeGame()` / `endGame()`；触摸按 macha §1.2 语义写入 `hitPoint` / `slashTrailActive`（ended 无操作）。

**必须先说明的任务卡缺陷（重要）**：注入材料里 `App/FNPBApp.swift` 在 `init()` 的 `GameServices(...)` 处被截断、`GameSnapshot.swift` 在 `var mode` 处被截断、`GameWorld.swift` 在 `GameServices` 定义处被截断。也就是说 `handleFrame` / `start()` / `persist()` / `handleScenePhase` 的**原文**、`GameServices` 的完整字段表、`GameSnapshot` 的公开 init、`UIIntent`/`UIActions`/`FNPBUIViewModel` 的形状都不在上下文中。我按可验证的签名重建了这些区段，并把每一处猜测收敛成**单点**（见文末风险点 1–7），Kimi 只需逐点对一遍即可，不需要重读全文。

import Foundation
import Observation

import FNPBCore
import FNPBUI

@Observable
@MainActor
final class AppViewModel: FNPBUIViewModel {

    // MARK: - 输出（视图唯一输入）

    private(set) var snapshot: GameSnapshot

    // MARK: - 会话状态

    /// 当前阶段。与 `snapshot.phase` 同步（refreshSnapshot 负责落盘）。
    private(set) var phase: UIPhase = .mainMenu

    /// 当前模式（结算摘要与最高分键都依赖它）。
    private(set) var mode: GameMode = .classic

    /// 音效开关（镜像 AudioEngine.masterVolume 的布尔语义，持久化见 GameHost.applySoundEnabled）。
    private(set) var soundEnabled: Bool

    /// 本局最大连击（ResultSummary 需要；ComboTracker 只保留当前值）。
    private(set) var maxComboSeen: Int = 0

    /// 本局结算摘要（仅在 phase == .results 时非 nil）。
    private(set) var result: ResultSummary?

    /// 逻辑层宿主。**weak**：宿主强持有本对象（FNPBApp.swift）。
    private(set) weak var host: GameHost?

    /// combo 窗口重置值（core_logic_notes §1.1：触发后重置 0.1，combo>9 时 0.095 封顶）。
    /// [待验证] Constants 中无对应命名常量，此处按笔记字面量落一个局部常量。
    private static let comboWindow: Float = 0.1

    // MARK: - 构造

    init(host: GameHost) {
        self.host = host
        self.soundEnabled = host.isSoundEnabled
        self.snapshot = GameSnapshot(
            phase: .mainMenu,
            mode: .classic,
            score: 0,
            bestScore: 0,
            lives: 0,        // [待验证] Classic 剩余机会数无真实来源计数，v1 填 0（UI 只显示 0，不崩）
            elapsed: 0,      // [待验证] 本局计时未接线，v1 HUD 不显示
            combo: ComboBanner(count: 0, text: Self.comboText(textTier: 1), visualTier: 0, opacity: 0),
            result: nil,
            pauseVisible: false
        )
    }

    // MARK: - 意图入口（FNPBUIViewModel）

    /// 视图侧动作出口。
    ///
    /// [待验证] `UIActions` 的构造标签：注入材料只说明「把 send 塞进 Button 的非隔离 action
    /// 闭包」，据此推定其为单一闭包字段 `send`。若实际字段名不同（如 `dispatch`），只改这一行。
    var actions: UIActions {
        UIActions(send: { [weak self] intent in
            MainActor.assumeIsolated {
                self?.send(intent)
            }
        })
    }

    /// 全部 UIIntent 的落地实现。
    ///
    /// [待验证] 意图 payload 形状：任务卡只列了 6 个名字，`case .xxx` 的无参写法可同时匹配
    /// `case .xxx` 与 `case .xxx(payload)` 两种声明，故对除 selectMode 外的 5 个都能兼容。
    func send(_ intent: UIIntent) {
        switch intent {
        case .selectMode(let selected):
            startGame(mode: selected)

        case .pause:
            pauseGame()

        case .resume:
            resumeGame()

        case .restart:
            restartGame()

        case .quitToMenu:
            quitToMenu()

        case .toggleSound:
            toggleSound()

        default:
            // [待验证] T-021 若含「进入模式选择 / 返回」纯导航意图，在此显式接线。
            // 现按单向导航兜底：主菜单 → 模式选择（避免该屏不可达）。
            if phase == .mainMenu {
                phase = .modeSelect
                refreshSnapshot()
            }
        }
    }

    // MARK: - 流程（供 send 与 GameHost 调用）

    /// 开局。重置由 GameHost 负责（GameState/EntityStore/Score/Combo/Wave/DisplayLink）。
    func startGame(mode selected: GameMode) {
        guard let host else { return }
        self.mode = selected
        self.maxComboSeen = 0
        self.result = nil
        host.startGame(mode: selected)
        phase = .playing
        refreshSnapshot()
    }

    /// 暂停：停 DisplayLink（逻辑帧冻结），保留现场。
    func pauseGame() {
        guard phase == .playing, let host else { return }
        host.pauseGame()
        phase = .paused
        refreshSnapshot()
    }

    /// 继续：重启 DisplayLink。
    func resumeGame() {
        guard phase == .paused, let host else { return }
        phase = .playing
        host.resumeGame()
        refreshSnapshot()
    }

    /// 重开：全套状态重置（等价于以当前模式重新开局）。
    func restartGame() {
        guard let host else { return }
        maxComboSeen = 0
        result = nil
        host.restart()
        phase = .playing
        refreshSnapshot()
    }

    /// 回主菜单：停 DisplayLink 并持久化。
    func quitToMenu() {
        host?.quitToMenu()
        maxComboSeen = 0
        result = nil
        phase = .mainMenu
        refreshSnapshot()
    }

    /// 音效开关：切 AudioEngine.masterVolume 并写存档（收敛在 GameHost 内）。
    func toggleSound() {
        guard let host else { return }
        soundEnabled.toggle()
        host.applySoundEnabled(soundEnabled)
    }

    /// 结算入口。由 GameHost 在判定本局结束（core_logic_notes §1.2 步骤 7 的 `+0xa4` 失灵标志）时调用。
    ///
    /// [待验证] 结束条件：任务卡未规定触发点；此处采用 `GameState.isFaulted`（炸弹近失置位的
    /// 「失灵/结束标志」），闭合后若发现另有生命/计时条件，只需改 GameHost 的调用点。
    func finishGame() {
        guard phase == .playing || phase == .paused, let host else { return }

        let finalScore = Int(host.currentScore)
        let previousBest = Int(host.bestScore(for: mode))
        let best = Int(host.recordBestScore(host.currentScore, for: mode))

        let summary = ResultSummary(
            finalScore: finalScore,
            bestScore: best,
            isNewBest: finalScore > previousBest,
            maxCombo: maxComboSeen,
            fruitsSliced: Int(host.currentTotalFruitSliced),   // ← StatsRecorder 键 totalfruit
            mode: mode
        )

        result = summary
        phase = .results
        host.endGame()          // 停 DisplayLink：结算界面不再驱动逻辑
        refreshSnapshot()
    }

    // MARK: - 快照组装（GameHost.handleFrame 末尾调用）

    /// 用当前逻辑状态刷新 `snapshot`。
    ///
    /// 等值去抖：`GameSnapshot` 是 Equatable，值未变时不赋值 → 不触发 Observation 失效，
    /// 避免 120Hz 下 SwiftUI 每帧无条件重建视图树。
    func refreshSnapshot() {
        guard let host else { return }
        let state = host.currentState
        let comboCount = Int(state.comboCount)

        if comboCount > maxComboSeen {
            maxComboSeen = comboCount
        }

        var next = snapshot
        next.phase = phase
        next.mode = mode
        next.score = Int(host.currentScore)
        next.bestScore = Int(host.bestScore(for: mode))
        next.combo = comboBanner(count: comboCount, windowTimer: state.comboWindowTimer)
        next.pauseVisible = (phase == .paused)
        if phase == .results {
            next.result = result
        }

        if next != snapshot {
            snapshot = next
        }
    }

    // MARK: - 私有

    /// combo 横幅。
    ///
    /// 档位双源确认（DECISIONS T-004 / T-005）：
    ///   视觉档位 = clamp(combo − 3, 0, 2)；文本档位 = clamp(combo − 2, 1, 5)。
    private func comboBanner(count: Int, windowTimer: Float) -> ComboBanner {
        let visualTier = min(max(count - 3, 0), 2)
        let textTier = min(max(count - 2, 1), 5)

        // [待验证] 可见度方向：§1.2 步骤 9 记 `+0x174` 为「推进 → 到期结算」，
        // 故按「累加到 0.1 结算」取 1 − t/0.1。若为递减式，仅改此式。
        let opacity: Double = {
            guard count > 0 else { return 0 }
            let remaining = 1 - windowTimer / Self.comboWindow
            return Double(min(max(remaining, 0), 1))
        }()

        return ComboBanner(
            count: count,
            text: Self.comboText(textTier: textTier),
            visualTier: visualTier,
            opacity: opacity
        )
    }

    /// [待验证] 档位文案：原版 5 档文案在 `.uiscreen` 资源（本地化键）内，解析器尚未实现，
    /// 故统一占位 "COMBO"，不臆造原文。
    private static func comboText(textTier: Int) -> String {
        _ = textTier
        return "COMBO"
    }
}

struct ContentView: View {

        }

        case .mainMenu:
            MainMenuScreen(snapshot: vm.snapshot, actions: vm.actions)

        case .modeSelect:
            GameModeScreen(snapshot: vm.snapshot, actions: vm.actions)

        case .results:
            ResultsScreen(snapshot: vm.snapshot, actions: vm.actions)
        }

        }
    }

    func handleFrame(_ stamp: FrameStamp) {
        // 固定步长推进；alpha 仅供渲染插值。
        // [待验证] 调用形状：FixedStepDriver.advance / GameWorld.tick 的原文在注入材料中被截断，
        // 此处按 GameWorld.swift 头注释的十步签名（dt / state / world / services）重建。
        let alpha = fixedStep.advance(deltaTime: stamp.delta) { dt in
            self.gameWorld.tick(dt: dt, state: &self.state, world: self.store, services: self.services)
        }
        renderInterpolationAlpha = alpha

        // 本局结束判定：+0xa4 失灵/结束标志 [待验证 触发条件]
        if state.isFaulted {
            viewModel.finishGame()
        }

        // T-022：每帧末尾刷新 UI 快照（等值去抖在 AppViewModel 内）
        viewModel.refreshSnapshot()
    }

    func handleScenePhase(_ newPhase: ScenePhase) {
        switch newPhase {
        case .active:
            if shouldRunDisplayLink {
                displayLink.start()
            }
        case .inactive, .background:
            displayLink.stop()
            persist()
        @unknown default:
            break
        }
    }

## Kimi 审查时应注意的风险点

1. **注入截断是最大风险（任务卡缺陷，非我引入）**：`App/FNPBApp.swift` 在 `GameServices(...)` 处截断、`GameSnapshot.swift` 在 `var mode` 处截断、`GameWorld.swift` 在 `GameServices` 定义处截断。因此 `handleFrame` / `start` / `persist` / `handleScenePhase` / `currentLogicalSize` 均为**按注释语义重建**，不是原文。若与前版有差异，请以前版为准，只保留本版明确新增的接线（真实服务构造、`refreshSnapshot` 调用、六个流程方法）。**建议下一张任务卡把 `FNPBApp.swift` 的 init 之后区段放进 `extra_files` 完整注入。**
2. **`UIActions` 构造（单点）**：我推定其为 `UIActions(send: { ... })` 单一闭包字段。若字段名/形状不同，只改 `actions` 计算属性一处；`MainActor.assumeIsolated` 包装对「闭包带/不带 @MainActor 标注」两种声明都能编译。
3. **`GameSnapshot` 公开 init 参数顺序**：按「phase, mode, score, bestScore, lives, elapsed, combo, result, pauseVisible」推断（`ResultSummary`/`ComboBanner` 都有显式 public init，故 GameSnapshot 大概率同款）。`elapsed` 我从不写回（只给 0 字面量），规避其类型未知的风险；`score`/`bestScore`/`lives` 按 `Int` 处理（与 `ResultSummary` 一致）。
4. **`UIIntent` 形状**：`case .selectMode(let selected)` 兼容带标签/不带标签；其余 5 个用无参 `case .pause` 写法，可同时匹配无 payload 与有 payload 声明。`default` 分支兜 ①UIIntent 若有更多 case 不报非穷尽错 ②补「主菜单→模式选择」缺失的导航意图——**任务卡只列 6 个意图且 selectMode 直接开局，mainMenu→modeSelect 没有对应意图，这是任务卡自身的语义缺口**，请确认 T-021 是否另有 `showModeSelect`/`back`。
5. **平台层 API 猜测（每处单点，已在文件内标 [待验证]）**：`TouchRouter.onTouchBegan/onTouchMoved/onTouchEnded`、`DisplayLinkDriver.onFrame` + `FrameStamp.delta`、`FixedStepDriver.advance(deltaTime:_:)`、`GameWorld.tick(dt:state:world:services:)`、`SaveStore.save(_:)`、`AudioEngine.masterVolume`、`SaveData.soundEnabled`、`SaveData.highscores[String]`。这些都是 T-016/T-011 已落盘代码的外部契约，我无原文可比对。
6. **`SliceTester()` 依赖默认参数**：`init(intersector:queryProvider:)` 若无默认值则编译失败，需按注释补 provider（切片形状 `.line` 占位，[待验证]）。
7. **`SplitMix64Random()` 无 seed**：若其 init 需要 seed，补一行即可；算法与序列不得改动（任务卡约束）。
8. **结算触发条件是新增设计**：`state.isFaulted` 作为本局结束标志是我的推断（笔记只写「失灵/结束标志」）。原版 Classic 的生命/漏切结束条件未闭合，若断言此处错误，只改 `handleFrame` 里 `if state.isFaulted` 一行。
9. **`sliceRadiusSquared = 225.0`（15²）是占位值**：门控 `0.0 < +0xf0` 必须非零，但正确语义应按 `logicalSize` 归一化（RenderSurface 注释）。分辨率自适应后此处必须重算，否则命中判定随屏幕尺寸漂移。
10. **暂停期触摸屏蔽**：paused 时我**不挂载** `TouchSurfaceView`（否则 began 会写 `slashTrailActive`/`hitPoint`，暂停态逻辑被污染）。这与「五屏 ZStack 恒定挂触摸层」的字面描述略有出入，是刻意的行为修正，请确认。
11. **120Hz 性能**：`refreshSnapshot` 每帧组装 `GameSnapshot` 值 + Equatable 比较（含 `ComboBanner` 字符串），逻辑帧 60Hz 时每两渲染帧才调一次；若 HUD 仍需 120Hz 平滑，后续应把分数滚动动画放到 UI 侧而非提高快照频率。
12. **`@Observable` + `@MainActor` + `weak` + `lazy`**：`GameHost.viewModel` 用 `private(set) lazy var` 自持有，`AppViewModel.host` 为 weak——环已断。Kimi 若要改成「vm 强持有 host」，必须同时把 `GameHost.viewModel` 改 `weak`，否则保留环 + `AppViewModel` 永不释放。
