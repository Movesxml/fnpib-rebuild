import Foundation

// 发售版 Challenge 控制器的状态子集。
// 来源：FUN_0005fd7c / FUN_00060108 / FUN_00060514 / FUN_0006059c /
// FUN_000606a4 / FUN_00060e54 / FUN_0006104c / FUN_00061128 /
// FUN_000613e4 / FUN_0005edc4 / FUN_0005ecf4。
// tier/challenge 必须显式注入；真实持久化选择器闭合前不伪造随机选取。

public enum ChallengeModePhase: Int32, Sendable, Equatable {
    case inactive = 0
    case bonusIntro = 1
    case running = 2
    case settling = 3
    case suspended = 5
}

public enum ChallengeSettlementSubstate: Int32, Sendable, Equatable {
    case bonus = 0
    case nextRound = 2
    case finished = 5
    case other = -1
}

public enum ChallengeModeEvent: Sendable, Equatable {
    case playAnimation(target: String, name: String)
    case setVisible(target: String, Bool)
    case playMusic(String)
    case playSound(String)
    case roundFinished(ChallengeOutcome)
    case nextRound
    case cleanedUp
}

public struct ChallengeFrame: Sendable, Equatable {
    public var evaluation: ChallengeEvaluationContext
    public var waveCount: Int32
    public var fruitEntityCount: Int32

    public init(evaluation: ChallengeEvaluationContext = ChallengeEvaluationContext(),
                waveCount: Int32 = 0,

public struct ChallengeModeRuntime: Sendable, Equatable {
    public private(set) var tier: ChallengeTierDefinition
    public private(set) var challenge: ChallengeDefinition
    public private(set) var conditions: ChallengeConditionsRuntime
    /// controller+0x88 的本回合选择；先保存它，再激活 challengeFruit 条件。
    public private(set) var selectedFruitName: String?
    public private(set) var phase: ChallengeModePhase = .running
    public private(set) var countdown: Float
    public private(set) var wavesEnded = false
    public private(set) var roundsPlayed: Int32 = 0
    public private(set) var roundsPassed: Int32 = 0
    public private(set) var roundResults: [ChallengeOutcome] = []
    private var events: [ChallengeModeEvent] = []

    public var isBonusRound: Bool { tier.isBonusRound }

    /// FUN_0006104c：TIME 条件优先，否则才使用 challenge.timeLimit。
    public var endsOnTimer: Bool {
        phase == .running && (conditions.timerDuration != 0 || challenge.timeLimit != 0)
    }

    public init(tier: ChallengeTierDefinition,
                challenge: ChallengeDefinition,
                conditionDefinitions: [ChallengeConditionDefinition],
                selectedFruitName: String? = nil) {
        self.tier = tier
        self.challenge = challenge
        self.selectedFruitName = selectedFruitName
        conditions = ChallengeConditionsRuntime(definitions: conditionDefinitions)
        conditions.select(conditionNames: challenge.conditionNames,
                          selectedFruitName: selectedFruitName)
        countdown = conditions.timerDuration != 0
            ? conditions.timerDuration : challenge.timeLimit
    }

    public mutating func drainEvents() -> [ChallengeModeEvent] {
        defer { events.removeAll(keepingCapacity: false) }
        return events
    }

    public mutating func evaluateImmediate(context: ChallengeEvaluationContext) {
        guard phase == .running else { return }
        conditions.evaluateImmediate(context: context)
    }

    public mutating func recordConsecutive(count: Int32, specificType: String?) {
        guard phase == .running else { return }
        conditions.recordConsecutive(count: count, specificType: specificType)
    }

    public mutating func checkComparisons(context: ChallengeEvaluationContext) {
        guard phase == .running else { return }
        conditions.checkComparisons(context: context)
    }

    /// FUN_00060e54：只锁存一次；FUN_0005edc4 在发售版为空函数。
    public mutating func markWavesEnded() {
        guard phase == .running, !wavesEnded else { return }
        wavesEnded = true
    }

    /// FUN_0005fd7c 状态 2：倒计时直接减 dt，不 clamp。
    /// 波次耗尽的直接结束只发生于非 bonus、实际波次数已达 numWaves、且无水果实体时；
    /// 其余路径仍按“计时到期或条件结果非 pending”的优先级判定。
    public mutating func update(dt: Float, frame: ChallengeFrame) {
        guard phase == .running else { return }
        countdown -= dt
        conditions.update(dt: dt, context: frame.evaluation)
        // FUN_0005fd7c 在 TIME 更新后同帧调用 FUN_0005ecf4。
        conditions.checkComparisons(context: frame.evaluation)

        let timedOrConditionEnd = (endsOnTimer && countdown <= 0)
            || conditions.result != .pending
        let shouldFinish: Bool
        if !wavesEnded {
            shouldFinish = timedOrConditionEnd
        } else if isBonusRound
                    || frame.waveCount < challenge.numWaves
                    || frame.fruitEntityCount != 0 {
            shouldFinish = timedOrConditionEnd
        } else {
            shouldFinish = true
        }
        if shouldFinish {
            finishRound(context: frame.evaluation)
            wavesEnded = false
        }
    }

    /// FUN_0005fd7c 状态 3：只响应结算控制器 state==7 的真实子状态。
    /// 新回合配置与水果由外部选择器显式提供；缺失时不推进、不虚构选择。
    /// bonusIntro 之前同样先完成外部回合 setup，避免沿用上一轮已冻结的聚合结果。
    ///
    /// 注意：子状态 3（bonus 完成后的 3）属于“其余值”，只写运行时状态 5
    /// （本类型中即 .suspended），不得伪造 cleanup、下一回合或 Results。
    public mutating func updateSettlement(controllerState: Int32,
                                          substate: Int32,
                                          nextChallenge: ChallengeDefinition? = nil,
                                          nextSelectedFruitName: String? = nil) {
        guard phase == .settling, controllerState == 7 else { return }
        switch ChallengeSettlementSubstate(rawValue: substate) ?? .other {
        case .bonus:
            guard let nextChallenge else { return }
            prepareRound(challenge: nextChallenge,
                         selectedFruitName: nextSelectedFruitName)
            startBonusIntro()
        case .nextRound:
            guard let nextChallenge else { return }
            beginRound(challenge: nextChallenge,
                       selectedFruitName: nextSelectedFruitName)
            events.append(.nextRound)
        case .finished:
            cleanup()
        case .other:
            // 含子状态 3：精确落到运行时状态 5，不产生任何事件。
            phase = .suspended
        }
    }

    /// The normal branch of FUN_00061128 makes Timer and challengeMods visible
    /// and starts their real animate_in timelines after the first-round setup.
    /// 首个 bonus 回合不得走这里；它必须由 startInitialBonusIntro() 显式进入 intro。
    public mutating func startInitialRound() {
        guard phase == .running, roundsPlayed == 0, !isBonusRound else { return }
        emitRunningHUDEvents()
    }

    /// 首个回合即 bonus 的显式入口：只启动 FUN_00060514 的 bonusIntro/play
    /// 与 Music-bonus，不结算 bonus，也不重置累计回合结果。
    public mutating func startInitialBonusIntro() {
        guard phase == .running, roundsPlayed == 0, isBonusRound else { return }
        startBonusIntro()
    }

    /// FUN_0005ffa8：状态 1 只由 bonusIntro 的真实播放完成回调推进；
    /// 条件与倒计时已在此前的外部回合 setup 中复位，这里不再次结算或选择。
    public mutating func completeBonusIntro() {
        guard phase == .bonusIntro else { return }
        phase = .running
        emitRunningHUDEvents()
    }

    private mutating func prepareRound(challenge nextChallenge: ChallengeDefinition,
                                       selectedFruitName: String?) {
        challenge = nextChallenge
        self.selectedFruitName = selectedFruitName
        countdown = 0
        wavesEnded = false
        conditions.select(conditionNames: challenge.conditionNames,
                          selectedFruitName: selectedFruitName)
        countdown = conditions.timerDuration != 0
            ? conditions.timerDuration : challenge.timeLimit
    }

    private mutating func beginRound(challenge nextChallenge: ChallengeDefinition,
                                     selectedFruitName: String?) {
        prepareRound(challenge: nextChallenge, selectedFruitName: selectedFruitName)
        phase = .running
        emitRunningHUDEvents()
    }

    /// FUN_0005ffa8 restores both controls hidden by FUN_000606a4. Playing
    /// animate_in alone does not undo the original explicit visibility write.
    private mutating func emitRunningHUDEvents() {
        events.append(.setVisible(target: "Timer", true))
        events.append(.playAnimation(target: "Timer", name: "animate_in"))
        events.append(.setVisible(target: "challengeMods", true))
        events.append(.playAnimation(target: "challengeMods", name: "animate_in"))
    }

    private mutating func startBonusIntro() {
        phase = .bonusIntro
        events.append(.playAnimation(target: "bonusIntro", name: "play"))
        events.append(.playMusic("Music-bonus"))
    }

    private mutating func cleanup() {
        phase = .inactive
        events.append(.setVisible(target: "inGameChallengeHud", false))
        events.append(.cleanedUp)
    }

    private mutating func finishRound(context: ChallengeEvaluationContext) {
        events.append(.playAnimation(target: "inGameChallengeHud", name: "animate_out"))
        let outcome = conditions.finish(context: context)
        guard outcome != .pending else { return }

        roundsPlayed += 1
        if outcome == .pass { roundsPassed += 1 }
        roundResults.append(outcome)
        events.append(.roundFinished(outcome))
        if outcome == .fail {
            events.append(.playSound("gank"))
        }
        if isBonusRound {
            events.append(.playMusic("Music-bonus-end"))
        }
        phase = .settling
        events.append(.playAnimation(target: "instructionsText", name: "instructions_out"))
        events.append(.setVisible(target: "Timer", false))
        events.append(.setVisible(target: "challengeMods", false))
    }
}

    public func tick(dt: Float, state: inout GameState,
                     world: inout any EntityStore, services: GameServices) {
        let step = step1_resolveDelta(dt)
        guard step > 0 else { return }              // 首帧零步进（未 prime 时）

        step2_normalizeCollisionRadius(state: &state)
        step3_advanceSlashTrail(dt: step, state: &state)

        // 步骤 4 / 5 只门控「实体遍历」，不终止整帧（见文件头 T-023 控制流修正）。
        let levelGateOK = step4_applyLevelEffects(state: &state)
        let traverseOK = levelGateOK ? step5_shouldTraverseEntities(world: world) : false
        if traverseOK {
            step6_processFruits(dt: step, state: &state, world: &world, services: services)
            step7_processBombs(dt: step, state: &state, world: &world, services: services)
            step8_processSpecials(dt: step, state: &state, world: &world, services: services)
        }

        step9_settleCombo(dt: step, state: &state, services: services)
        step10_updateSubsystems(dt: step, state: &state, world: &world, services: services)
        classic.update(state: &state)

        // FUN_0001f018 先由 FUN_0001b384 调每玩家 vtable+0x14，再推进 +0x24。
        // 放在整个实体主循环之后，保证命中帧也消耗一次 dt，下一帧开始关闭遍历。
        if explosionTimeRemaining > 0 {
            let old = explosionTimeRemaining
            let new = old - step
            if old > Constants.Bomb.classicGameOverThreshold,
               new <= Constants.Bomb.classicGameOverThreshold {
                classic.onExplosionThreshold(state: &state)
            }
            explosionTimeRemaining = max(0, new)
        }
    }

    // MARK: 1 — dt 兜底 [both]
    /// §1.2 步骤 1：`dt==0` → 用上一帧缓存值（exe `param_1!=0` 分支 / macho fVar29）。
    /// 缓存初值 0.0（exe 自证），故首帧未 prime 时返回 0 → tick 直接返回（R7）。
    private func step1_resolveDelta(_ dt: Float) -> Float {
        let resolved = (dt == 0) ? lastDelta : dt
        lastDelta = resolved
        return resolved
    }

    // MARK: 2 — 碰撞半径归一化 [both]
    /// §1.2 步骤 2：d=|(x,y,z)|；d/15.0<=1.0 → w=0.5*(d/15.0)+0.5，否则 1.0。
    ///
    /// 原式输出到**全局最大值** `DAT_005573b8 = max(全局, w)`（↔ exe `DAT_019514c0`）；
    /// 重建无全局单例，直接写入 `state.sliceRadiusSquared`（= w²，供
    /// `SliceTester` 的二次确认半径² 使用，见 GameState.swift `sliceRadiusSquared` 注释）。
    /// **差异：原版是跨帧单调不减的最大值，重建是每帧覆盖**——[待验证] 若原版确实单调
    /// 递增，切刀半径会随时间膨胀；从 `0.0 < game+0xf0` 的门控看，更可能是每帧重算。
    ///
    /// ⚠️ 原版本步受 `game +0x50`（`collisionRadiusActive`）门控，且该字段默认 false；
    ///    照抄门控会让 `sliceRadiusSquared` 恒为 0 → 切割门控 `0.0 < radius²` 永远失败
    ///    → 游戏锁死。本卡按任务卡字面**无条件计算**，门控留待字段默认值闭合后再引入。
    private func step2_normalizeCollisionRadius(state: inout GameState) {
        // 完整反编译步骤 2 读取 game+0xc0/+0xc4/+0xc8（当前触摸点），
        // 不是 +0x104 的最后命中实体位置。
        let distance = simd_length(state.slashPosition)
        let ratio = distance / Constants.Simulation.collisionNormalizeDivisor

        let weight: Float
        if ratio <= Constants.Simulation.collisionNormalizeClamp {
            // 原式两个 0.5 同值：0.5 * (d/15) + 0.5，复用 collisionNormalizeWeight
            weight = Constants.Simulation.collisionNormalizeWeight * ratio
                   + Constants.Simulation.collisionNormalizeWeight
        } else {
            weight = Constants.Simulation.collisionNormalizeClamp
        }

        state.sliceRadiusSquared = weight * weight
    }

    // MARK: 3 — slash 轨迹 [both]
    /// §1.2 步骤 3：+0xa5!=0 → +0x17c += dt；超轨迹时长 → 槽位 (slot+1)%8 写入采样点、
    /// 清 +0xa5、+0x17c=0。容量 8 / 采样间隔 0.05 [macho H]。
    ///
    /// 实现：轨迹环形缓冲已收进 `GameState.slashTrail`（T-023），本步只做推进 +
    /// 镜像字段同步（`slashTrailActive` ↔ +0xa5，`slashTrailTimer` ↔ +0x17c），
    /// 保持与原偏移语义的可对账性。
    private func step3_advanceSlashTrail(dt: Float, state: inout GameState) {
        // 原 FUN_00019f10 的输入是 +0xb8/+0xb4 关联的触摸采样，不是命中实体位置。
        let samplePoint = state.slashPosition
        state.slashTrail.advance(dt: dt, point: samplePoint)

        // 镜像字段同步（原 +0xa5 / +0x17c）
        state.slashTrailActive = state.slashTrail.isActive
        state.slashTrailTimer = state.slashTrail.elapsed
    }

    // MARK: 4 — 等级特效 [both]
    /// §1.2 步骤 4：`+0xb4 < 4` → `+0xf4 = 0` 并**跳过实体遍历**；返回 false 表示本帧不遍历。
    ///
    /// ⚠️ [待验证] `levelTier`(+0xb4) 默认 0 < 4 → 默认永远跳过遍历，会锁死游戏。
    ///    原版开局应把 +0xb4 写成 ≥4（或由关卡加载写入），但该写入点未从反编译定位。
    ///    本卡只实现判定本身，**不改 GameState 默认值**（避免掩盖未闭合事实）；
    ///    默认值问题列入 T-023 风险点交 Kimi 裁决。
    private func step4_applyLevelEffects(state: inout GameState) -> Bool {
        if state.levelTier < Constants.Simulation.lowLevelEffectCutoff {
            state.levelEffectParam = 0
            return false
        }
        return true
    }

    // MARK: 5 — 遍历门控 [macho]
    /// §1.2 步骤 5：爆炸/冻结计时 <=0，且全局位掩码 & 0x40 == 0，
    /// 且（单例 vtable+0x28 返回 1 或单例为空）。
    ///
    /// 已闭合爆炸计时门控：`PTR+0x24 <= 0.0` 才允许实体遍历。
    /// `DAT_005d4d74 & 0x40` 是发售版正常路径关闭的 Debug 位，不迁移。
    /// 另一单例 `vtable+0x28` 身份尚未闭合，不能用猜测条件阻断正常游玩。
    private func step5_shouldTraverseEntities(world: any EntityStore) -> Bool {
        _ = world
        return explosionTimeRemaining <= 0
    }

    // MARK: 6 — 循环A（type=1 水果）[both]
    /// §1.2 步骤 6，每实体：
    ///  6.1 全局锁 / +0x2b>0 → 跳过
    ///  6.2 门控：+0x6f==0 && +0x41<=-1.0 && +0x34<=0 && (flags&1)==0 && +0x30!=0
    ///  6.3 isSliced==true →（a）+0xa5=1 开轨迹（b）(+0x33==0 || +0x32>3) && +0xc1==0 → +0x3c+=2
    ///      （c）清 +0x40/+0x44，写 +0xf8~+0x100（旧 slash 位）、+0x110=type、+0x104~+0x10c=位置
    ///  6.4 非炸弹：entity[0x21]==2 或同实体 → 跳过窗口检查；否则 +0x180=entity、+0x174=0、
    ///      comboArr[+0x8c]=type、+0x90=(entity[0x21]!=0)、按首发分支上报统计、+0x51=1、combo++；
    ///      combo>9 → +0x174=0.095；+0x54 -= combo*(0.5*rand/8+0.75)；vtable+0x44 回调
    ///  6.5 +0x1ba!=0 → +0x110 += specialBonus；+0x54 += -3.0*(0.5*rand/8+0.75)
    ///
    /// ── 实现边界（逐条与任务卡对齐）────────────────────────────────────────
    /// • 6.1 全局锁 `DAT_00557390`：无协议入口 → 不实现 [待验证]。
    /// • 6.2 已实现 `+0x6f`、`+0x30`、`+0x2b` 与 `(flags&1)==0`；`+0x41` 和 `+0x34`
    ///   的生成初值仍未闭合，暂不引入会阻止默认实体参与判定的条件。
    /// • 6.4 的 `entity[0x21]==2` 与同实体门控由 ComboTracker 执行；生产调用必须传
    ///   真实 `Entity.stateSlot0`，不能用水果 typeID 代替。
    /// • 首发 `slashes` / `slashesScoring` 统计：**已由 ComboTracker 内部上报**
    ///   （其 `stats` 注入 = `services.stats`），本步不重复上报，避免双倍计数。
    /// • 单果得分由 `FruitOnSliced` 每次命中独立执行，不绑定 combo 事件；
    ///   杨桃、noScore 与暴击加分按发售版 OnSliced 的已闭合分支处理。
    /// • `state.lastSliceScore`(+0x110) 先写水果 typeID；实体 +0x1ba 后处理再加
    ///   `PTR_DAT_004d90b4 -> DAT_005d4cd0` 指向的水果类型总数，不是额外得分。
    private func step6_processFruits(dt: Float, state: inout GameState,
                                     world: inout any EntityStore, services: GameServices) {
        let tuning = motionTuning

        // 闭包内不访问 `world`（它是 inout，闭包内再访问会触发独占访问冲突），
        // 收集后统一 despawn。
        var pendingDespawn: [EntityID] = []
        var hitPositions: [SIMD3<Float>] = []

        world.forEach(kind: .fruit) { (entity: inout Entity) -> Bool in
            // 先运动、后判定；正常出界走 FUN_0000f2f8(param_2 == 1) 的完整回收筛选。
            guard FruitMotion.integrate(&entity, dt: dt, tuning: tuning) else {
                classic.retireFruit(entity,
                                    reason: .normalDrop,
                                    state: &state,
                                    services: services)
                pendingDespawn.append(entity.id)
                return true
            }

            // 6.2 参与判定门控。dropState 与 +0x34 的生成初值仍未闭合，保持既有子集；
            // flags 默认 0，可以无锁死风险地恢复发售版 (flags&1)==0 条件。
            guard !entity.isMarked else { return true }       // +0x6f==0
            guard entity.isVisible else { return true }       // +0x30!=0
            guard entity.fallDelay <= 0 else { return true }  // +0x2b>0 → 跳过
            guard (entity.flags & 1) == 0 else { return true }

            // 6.3 两段式切割判定（SliceTesting 内部已做门控 + 形状相交 + 球体二次确认）
            guard services.slice.isSliced(game: state, entity: entity) else { return true }

            // 6.3.1 开轨迹（原 DAT_005d4d80 != 0 → +0xa5 = 1）
            state.slashTrail.begin()
            state.slashTrailActive = true

            let fruitType = entity.fruit?.typeID ?? Constants.EntityPool.emptySlot
            let entityPosition = entity.transform.position

            // 旧 slash 位置来自当前触摸输入；hitPoint 将在本调用中改写为实体位置。
            state.recordSliceHit(entityPosition: entityPosition,
                                 slashPosition: state.slashPosition,
                                 fruitType: fruitType)

            // (+0x33==0 || +0x32>3) && entity+0xc1==0 → +0x3c+=2。
            if (entity.stateSlot2 == 0 || entity.stateSlot0x32 > 3)
                && !entity.specialOrBombMarked {
                state.missPenaltyCount &+= 2
            }

            // +0x6e==0 才进入普通 combo / 主循环命中特效路径。
            if !entity.skipsNormalFruitPath {
                let previousVisualTier = ComboTier.visual(for: state.comboCount)
                let event = services.combo.registerSlice(fruitType: fruitType,
                                                         entity: entity.id,
                                                         stateSlot0: entity.stateSlot0,
                                                         state: &state,
                                                         rng: services.rng)

                if case let .incremented(_, visualTier, _)? = event {
                    if visualTier > previousVisualTier {
                        services.audio.play("combo_\(visualTier)")  // 音效 key [待验证]
                        services.effects.emit(visualTier, at: entityPosition)
                    }
                }
                hitPositions.append(entityPosition)
            }

            // 原顺序是先实体 vtable+0x24，再消费实体 +0x1ba。单果得分不依赖
            // combo 是否递增，也不依赖 stateSlot0 或 +0x6e 的普通路径门控。
            onSliced.apply(entity: &entity, state: &state, services: services,
                           explosionTimeRemaining: explosionTimeRemaining,
                           consecutive: &challengeConsecutive) { count, specificType in
                challenge?.recordConsecutive(count: count, specificType: specificType)
            }
            classic.onFruitSliced(entity, state: &state, services: services)
            evaluateChallengeImmediate(services: services)

            pendingDespawn.append(entity.id)
            return true
        }

        // 6.3.4 / 6.3.5 回收 + 命中特效 + 音效（原 FUN_00033ca0/FUN_00035a08）
        for id in pendingDespawn {
            world.despawn(id)
        }
        for position in hitPositions {
            services.effects.emit(Self.hitEffectKind, at: position)
            services.audio.play("splash")                            // 音效 key [待验证]
        }
    }

    // MARK: 7 — 循环B（type=2 炸弹）[macho]
    /// 完整 FUN_00016214 的顺序是：先 FUN_00017d74 命中测试；命中时 OnSliced、
    /// 按 +0x7e && entity[0x26]==0 写 game+0xa4；只有未命中才做 FUN_00017fdc
    /// 近失并上报 near_miss。近失本身绝不写结束标志，也没有每帧去重门。
    private func step7_processBombs(dt: Float, state: inout GameState,
                                    world: inout any EntityStore, services: GameServices) {
        let tuning = motionTuning
        var pendingDespawn: [EntityID] = []
        world.forEach(kind: .bomb) { (entity: inout Entity) -> Bool in
            // 炸弹出界只回收，不计水果漏切。
            guard FruitMotion.integrate(&entity, dt: dt, tuning: tuning) else {
                pendingDespawn.append(entity.id)
                return true
            }
            guard entity.isBombSliceEnabled,
                  entity.bombState <= 0,
                  (entity.flags & 1) == 0 else { return true }

            if services.slice.isSliced(game: state, entity: entity) {
                pendingDespawn.append(entity.id)

                // FUN_00020c38：统计 bomb、写 +0x24=3.2、创建爆炸并播放音效。
                // 爆炸实体/阈值回调另卡闭合；这里只接入已经逐项核实的公共链。
                services.stats.add(.bomb, 1)
                explosionTimeRemaining = Constants.Bomb.explosionTimerSeconds
                services.audio.play(Constants.Bomb.explodeAudioKey)
                if entity.isFaultingBomb && entity.stateSlot1 == 0 {
                    state.isFaulted = true
                }
                // DAT_00557390 只终止本帧后续炸弹遍历；跨帧禁止切割由 +0x24 负责。
                return false
            }

            if services.slice.isNearMiss(game: state, entity: entity) {
                services.stats.add(.nearMiss, 1)
            }
            return true
        }
        for id in pendingDespawn {
            world.despawn(id)
        }
    }

    // MARK: 8 — 循环C（type=0 其他）[macho]
    /// 完整反编译已闭合筛选：entity[0x17] 低字节非零、(flags&0x11)==0、
    /// FUN_00017d74==1 → OnSliced；若 +0x51==0 则首发 slashesScoring 并置位。
    /// 具体 coin/特殊果的 OnSliced 奖励仍属各自实体实现，不在此臆造。
    private func step8_processSpecials(dt: Float, state: inout GameState,
                                       world: inout any EntityStore, services: GameServices) {
        _ = dt // 运动由实体自身更新函数负责；主循环此处只做切割分发。
        var pendingDespawn: [EntityID] = []
        world.forEach(kind: .other) { (entity: inout Entity) -> Bool in
            guard entity.isSpecialSliceEnabled,
                  (entity.flags & 0x11) == 0,
                  services.slice.isSliced(game: state, entity: entity) else { return true }

            pendingDespawn.append(entity.id)
            if !state.slashesScoringEmitted {
                services.stats.add(.slashesScoring, 1)
            }
            state.slashesScoringEmitted = true
            return true
        }
        for id in pendingDespawn {
            world.despawn(id)
        }
    }

    // MARK: 9 — combo 结算 [both]
    /// §1.2 步骤 9：+0x174 推进（触发重置 0.1；>9 封顶 0.095）→ 到期结算 →
    /// clearComboTracking（comboArr memset 0xff）→ 分发 ComboEvent 到计分/统计/特效。
    ///
    /// 分工：ComboTracker 只保存清零前快照；本步消费完整快照，避免计分、统计
    /// 分别读取已清零的 GameState。2 连起更新最高连击，3 连起计分和上报 combos。
    private func step9_settleCombo(dt: Float, state: inout GameState, services: GameServices) {
        let events = services.combo.tickWindow(dt: dt, state: &state)

        for event in events {
            switch event {
            case let .settled(count, fruitTypes, comboType):
                if count >= 3 {
                    // 发售版 FUN_0001cd6c 完整函数体为 return 0，当前路径无模式阻断。
                    services.score.dispatch(score: count,
                                            comboType: comboType,
                                            isScoring: true,
                                            mode: 0)
                }
                if count >= 2 {
                    services.stats.recordCombo(count: count,
                                               fruitTypes: fruitTypes,
                                               modePrefix: state.comboStatsPrefix)
                }

                let tier = ComboTier.visual(for: count)
                services.audio.play("combo_settle_\(tier)")         // 音效 key [待验证]
                services.effects.emit(tier, at: state.hitPoint)
            case .started, .incremented:
                break
            }
        }
    }

    // MARK: 10 — 子系统更新 [macho]
    /// §1.2 步骤 10：波次（services.waves.tick）→ 计分分发（services.score）→
    /// 统计（services.stats）→ 音效（services.audio）→ 特效（services.effects）。
    ///
    /// ⚠️ [待验证] 后四个子系统**没有逐帧 API**：
    ///  - `ScoreSystem.dispatch` 是事件驱动（由 step6/step9 调用），无帧级入口
    ///  - `StatsRecorder` 只有计数写入，原逐帧上报点（如 `all` / `score_without_beans`）
    ///    未从反编译定位
    ///  - `AudioPlaying` / `EffectEmitting` 由命中与结算事件驱动
    /// 故本步仅调用 `waves.tick`；其余调用点待各自任务卡闭合后回填。
    private func step10_updateSubsystems(dt: Float, state: inout GameState,
                                         world: inout any EntityStore, services: GameServices) {
        services.waves.tick(dt: dt, state: &state, world: world)
        updateChallenge(dt: dt, world: world, services: services)
    }

    /// 原 ChallengeManager 在水果 OnSliced 后读取同一玩家分数与统计对象；
    /// 不在挑战控制器中另建计数副本。单刀多果的 CONSECUTIVE 分组仍待闭合，
    /// 因此这里只走即时条件，不虚构 comboCount。
    private func evaluateChallengeImmediate(services: GameServices) {
        guard challenge != nil else { return }
        challenge!.evaluateImmediate(context: ChallengeEvaluationContext(
            score: services.score.total, stats: services.stats))
    }

    /// 波次结束只由真实 WaveManager 状态触发；水果数量也直接查询 ECS，
    /// 避免控制器持有与世界脱节的影子计数。
    private func updateChallenge(dt: Float, world: any EntityStore,
                                 services: GameServices) {
        guard challenge != nil else { return }
        let waveProvider = services.waves as? any ChallengeWaveStateProviding
        if waveProvider?.hasEndedWaveForChallenge == true {
            challenge!.markWavesEnded()
        }
        var fruitEntityCount: Int32 = 0
        world.forEach(kind: .fruit) { _ in
            fruitEntityCount += 1

    func start() {
        // [待验证] DisplayLinkDriver 的帧回调注入面（属性 or start(handler:)）。
        // [待验证] FrameStamp 的字段名（delta / deltaTime）。

    func handleFrame(_ stamp: FrameStamp) {
        // R7：首帧 dt == 0，显式 prime 让 GameWorld 用兜底步长推进
        if stamp.isFirstFrame {
            gameWorld.prime()
        }

        // 固定步长推进；alpha 仅供渲染插值。
        // [Kimi 修正 T-022 审查] 签名以 TimeDriver.swift 为准：advance(delta:) /
        // GameWorld.tick(dt:state:world:services:) 且 world 为 inout。
        let alpha = fixedStep.advance(delta: stamp.clampedDelta()) { step in
            // FUN_00025164 的宽高比必须来自当前真实渲染视口；未布局或 Metal
            // 不可用时才沿用宿主尺寸兜底，无效值由 FruitMotionTuning 防护。
            if let size = renderDriver?.logicalSize,
               size.x > 0, size.y > 0, size.x.isFinite, size.y.isFinite {
                gameWorld.motionTuning.actualSize = size
            } else {
                gameWorld.motionTuning.actualSize = GameHost.currentLogicalSize()
            }
            waveSpawnBridge?.tuning.actualSize = gameWorld.motionTuning.actualSize
            // 先推进已有 UISF 时间线，再跑控制器并启动本帧新动画；新动画从 0 ms 开始，
            // 完成 game_over 时产生的 game_over_real 会在同一逻辑帧被消费。
            advanceClassicAnimations(deltaSeconds: step)
            advanceChallengeAnimations(deltaSeconds: step)
            gameWorld.tick(dt: step, state: &state, world: &store, services: services)
            // 此时本逻辑帧的 HUD 时间线已推进一次；仅在整个递归动画 query
            // 确认 bonusIntro/play 已完成时，才让运行时进入 running。
            // 完成事件在这里入队，随后由 consumeChallengeEvents() 于同帧消费；
            // Timer/challengeMods 的 animate_in 从 0 ms 开始，下一逻辑帧再推进。
            completeInitialChallengeBonusIntroIfNeeded()
            consumeChallengeEvents()
            consumeClassicEvents()
        }
        renderInterpolationAlpha = alpha

        // 三振或炸弹阈值只进入 Classic 结束动画；两段动画完成后才进入结算。
        // isFaulted(+0xa4) 是命中时写入的原始状态，不再是独立的即时结算条件。
        if state.classicPhase == .results {
            viewModel.finishGame()
        }

        // T-024：渲染层不走「帧末通知」通路——`MTKView` 以 `isPaused = false` 自驱
        // （preferredFramesPerSecond = 120），每次 draw 时直接读
        // `currentState` / `currentEntities`。因此这里只刷新 UI 快照。
        // 选该通路的理由：注入文件里 `MetalCanvasView.onDraw` 是唯一帧入口，
        // 不存在「由逻辑层主动触发一次渲染」的 API。
        viewModel.refreshSnapshot()
    }

    func handleScenePhase(_ newPhase: ScenePhase) {
        switch newPhase {
        case .active:
            if shouldRunDisplayLink {
                displayLink.start()
            }
        case .inactive, .background:
            displayLink.stop()
            persist()
        @unknown default:
            break
        }
    }

    /// 只有真实内容可用才重置并开局；失败保留当前状态并返回 false。
    @discardableResult
    func startGame(mode: GameMode) -> Bool {
        guard let content = ensureContent() else {
            Self.contentLogger.error(
                "开局被拒绝：原始内容不可用 — \(self.contentLoadError ?? "unknown", privacy: .public)")
            return false
        }

        if mode == .classic, !ensureClassicAnimations() {
            Self.contentLogger.error("经典开局被拒绝：动画资源不可用")
            return false
        }
        if mode == .challenge, !ensureChallengeAnimations() {
            Self.contentLogger.error("挑战开局被拒绝：动画资源不可用")
            return false
        }

        currentMode = mode
        state = GameState()
        switch mode {
        case .classic:
            state.comboStatsPrefix = "CLASSIC"
        case .adventure:
            state.comboStatsPrefix = "NEW_MODE"
        case .challenge:
            state.comboStatsPrefix = nil // 原模式表索引未闭合，不猜成 ALL。
        }

        // [待验证] 切割半径²：初值 0 会让 HitTest 门控（`0.0 < +0xf0`）永不通过。
        // 原式按 15.0 归一化碰撞半径（core_logic_notes §1.1 / RenderSurface 注释），
        // 暂取 15² = 225；归一化语义闭合后应改为按 logicalSize 换算。
        state.sliceRadiusSquared = 225.0

        // [待验证] 原版 levelTier 开局写入点尚未闭合。
        state.levelTier = Constants.Simulation.lowLevelEffectCutoff

        // store.clear 对应 param_2==0 的强制清理，不经过正常漏切计数。
        store.clear()
        // 必须先读档再重置：FUN_0002504c 每局从存档 +0x100 读取整数初值。
        saveData = saveStore.load()
        gameWorld.resetForNewGame(isClassicMode: mode == .classic,
                                  initialCriticalChance: saveData.criticalChance)
        classicAnimationPlayer?.reset()
        // inGameClassic 开局三颗 active heart 直接可见，不播放入场动画。
        visibleClassicCrosses = mode == .classic ? [1, 2, 3] : []
        scoreSystem.reset()
        comboTracker.reset(state: &state)
        statsRecorder.reset()
        guard configureMode(content: content, mode: mode) else { return false }
        shouldRunDisplayLink = true
        displayLink.start()
        return true
    }

    /// 挑战接线顺序与 FUN_00061128 / WaveManager setup 对齐：
    /// 回合装配 → 注入挑战水果 → beginGame → customWave → 激活条件运行时。
    /// 任一步失败都拒绝开局，不退回普通波、不用默认 tier/difficulty 冒充存档配置。
    private func configureMode(content: OriginalContent, mode: GameMode) -> Bool {
        challengeConfigurationError = nil
        challengeSettlement = nil
        challengeHUDAnimationPlayer?.reset()
        challengeResultAnimationPlayer?.reset()
        challengeBackgroundAnimationPlayer?.reset()
        gameBackgroundAnimationPlayer?.reset()
        challengeVisibleTargets.removeAll(keepingCapacity: false)
        challengeBackgroundVisible = false
        switch mode {
        case .classic, .adventure:
            challengeAssembler = nil
            currentChallengeRound = nil
            waveDirector.setChallengeFruit(nil)
            gameWorld.configureChallenge(nil)
            waveDirector.beginGame(waves: content.waves.waves, fruits: content.fruits.fruits)
            return true

        case .challenge:
            do {
                let tierIndex = saveData.challengeSession.initialTierIndex(
                    validTierCount: content.challengeCatalog.tiers.count)
                guard tierIndex >= 0, Int(tierIndex) < content.challengeCatalog.tiers.count else {
                    challengeConfigurationError = "挑战 tier 越界：\(tierIndex)"
                    return false
                }
                let tier = content.challengeCatalog.tiers[Int(tierIndex)]
                let roundCount = ChallengeTierSelector.rollRoundCount(tier: tier, rng: rng)
                let difficulty = saveData.challengeSession.clampedDifficulty(roundCount: roundCount)
                var assembler = try ChallengeRoundAssembler(
                    catalog: content.challengeCatalog,
                    fruits: content.fruits.fruits,
                    conditionDefinitions: content.challengeConditions,
                    persistence: saveStore,
                    random: rng,
                    tierIndex: tierIndex,
                    difficulty: difficulty,
                    roundCount: roundCount)
                let round = try assembler.startFirstRound()

                waveDirector.setChallengeFruit(round.fruitID)
                waveDirector.beginGame(waves: content.challengeWaves.waves,
                                       fruits: content.fruits.fruits)
                guard waveDirector.configureChallengeWave(
                    customWave: round.challenge.customWave,
                    forceWave: round.challenge.forceWave) else {
                    challengeConfigurationError =
                        "挑战 \(round.challenge.name) 的 customWave 无法配置"
                    return false
                }

                gameWorld.configureChallenge(round.runtime)
                if round.runtime.isBonusRound {
                    // 首个回合即 bonus：显式进入真实 bonusIntro，
                    // 不调用普通初始回合 HUD 入口。
                    gameWorld.startInitialChallengeBonusIntro()
                } else {
                    gameWorld.startInitialChallengeRound()
                }
                challengeAssembler = assembler
                currentChallengeRound = round
                return true
            } catch {
                challengeConfigurationError = String(describing: error)
                Self.contentLogger.error(
                    "挑战开局被拒绝：\(error.localizedDescription, privacy: .public)")
                return false
            }
        }
    }

    private func ensureChallengeAnimations() -> Bool {
        if challengeHUDAnimationPlayer != nil,
           challengeResultAnimationPlayer != nil,
           challengeBackgroundAnimationPlayer != nil,
           gameBackgroundAnimationPlayer != nil {
            return true
        }
        do {

            challengeBackgroundAnimationPlayer = try UIScreenAnimationPlayer(

                data: OriginalContentBundle.readResource(named: "backgroundAnimations"))

        } catch {
            challengeHUDAnimationPlayer = nil
            challengeResultAnimationPlayer = nil
            challengeBackgroundAnimationPlayer = nil
            gameBackgroundAnimationPlayer = nil
            challengeConfigurationError = "挑战动画加载失败：\(error.localizedDescription)"
            Self.contentLogger.error("挑战动画加载失败：\(error.localizedDescription, privacy: .public)")
            return false
        }
    }

    /// 仅当真实 bonusIntro/play 递归动画 query 返回完成且运行时仍为 bonusIntro 时，
    /// 才调用运行时完成回调。这里不使用 `.completed` 字符串事件猜测多个子动画组合。
    private func completeInitialChallengeBonusIntroIfNeeded() {
        guard currentMode == .challenge,
              let runtime = gameWorld.challenge,
              runtime.phase == .bonusIntro,
              let player = challengeHUDAnimationPlayer else { return }
        let completed = (try? player.isCompleted(
            component: "bonusIntro", animation: "play")) ?? false
        guard completed else { return }
        gameWorld.completeChallengeBonusIntro()
    }

    /// 与 Classic 相同，先推进已有时间线，再让 GameWorld 产生本帧事件。
    /// state 7 的回合分发只发生一次；成功后旧结果控制器立即退出装配。
    private func advanceChallengeAnimations(deltaSeconds: Float) {
        guard currentMode == .challenge else { return }
        do {
            if var player = gameBackgroundAnimationPlayer {
                let output = try player.advance(deltaSeconds: deltaSeconds)
                gameBackgroundAnimationPlayer = player
                processChallengeAnimationOutput(output)
            }
            if var player = challengeHUDAnimationPlayer {
                let output = try player.advance(deltaSeconds: deltaSeconds)
                challengeHUDAnimationPlayer = player
                processChallengeAnimationOutput(output)
            }
            guard var controller = challengeSettlement,
                  var resultPlayer = challengeResultAnimationPlayer,
                  var backgroundPlayer = challengeBackgroundAnimationPlayer else { return }
            let output = try controller.update(
                deltaSeconds: deltaSeconds,
                resultPlayer: &resultPlayer,
                backgroundPlayer: &backgroundPlayer,
                canAnimateResultExit: canAnimateChallengeResultExit())
            challengeSettlement = controller
            challengeResultAnimationPlayer = resultPlayer
            challengeBackgroundAnimationPlayer = backgroundPlayer
            processChallengeAnimationOutput(output)
            if controller.state == 7 {
                try dispatchChallengeSettlement(controller)
            }
        } catch {
            rejectChallenge("挑战动画／结算推进失败：\(error.localizedDescription)")
        }
    }

    private func canAnimateChallengeResultExit() -> Bool {
        guard let runtime = gameWorld.challenge else { return false }
        // FUN_00062a28(parent)：普通 tier 且失败回合数==3 时跳过结果退场。
        let suppressOutro = !runtime.isBonusRound
            && (runtime.roundsPlayed - runtime.roundsPassed) == 3
        return !suppressOutro
    }

    private func dispatchChallengeSettlement(_ controller: ChallengeSettlementController) throws {
        challengeBackgroundVisible = false
        switch controller.substate {
        case 2:
            guard var assembler = challengeAssembler, let content = originalContent else {
                throw ChallengeHostError("下一回合缺少真实装配器或原始内容")
            }
            let round = try assembler.advanceToNextRound()
            waveDirector.setChallengeFruit(round.fruitID)
            waveDirector.beginGame(waves: content.challengeWaves.waves,
                                   fruits: content.fruits.fruits)
            guard waveDirector.configureChallengeWave(
                customWave: round.challenge.customWave,
                forceWave: round.challenge.forceWave) else {
                throw ChallengeHostError("挑战 \(round.challenge.name) 的下一回合 customWave 无法配置")
            }
            // 保留 GameWorld 内同一 runtime 的累计回合结果，只替换当前回合定义与条件。
            gameWorld.updateChallengeSettlement(
                controllerState: controller.state,
                substate: controller.substate,
                nextChallenge: round.challenge,
                nextSelectedFruitName: round.fruitName)
            challengeAssembler = assembler
            currentChallengeRound = round
            challengeSettlement = nil
        case 5:
            gameWorld.updateChallengeSettlement(
                controllerState: controller.state,
                substate: controller.substate)
            challengeSettlement = nil
        default:
            throw ChallengeHostError("未知挑战结算分发子状态：\(controller.substate)")
        }
    }

    private func consumeChallengeEvents() {
        for event in gameWorld.drainChallengeEvents() {
            switch event {
            case let .roundFinished(outcome):
                finishChallengeRound(outcome: outcome)
            case let .playSound(key), let .playMusic(key):
                audio.play(key)
            case let .playAnimation(target, name):
                playChallengeAnimation(target: target, name: name)
            case let .setVisible(target, visible):
                challengeVisibleTargets[target] = visible
            case .nextRound:
                // state 7 已完成真实波次、水果和条件配置；此事件只是运行时通知。
                break
            case .cleanedUp:
                viewModel.finishGame()
            }
        }
    }

    private func finishChallengeRound(outcome: ChallengeOutcome) {
        guard var assembler = challengeAssembler, let round = currentChallengeRound else {
            rejectChallenge("回合结束时缺少真实挑战装配状态")
            return
        }
        assembler.recordCurrentRoundOutcome(outcome)
        challengeAssembler = assembler
        saveData.challengeSession.recordRound(
            challengeName: round.challenge.name,
            tierIndex: assembler.tierIndex,
            roundIndex: round.roundIndex,
            outcome: outcome)
        guard saveStore.saveMergingChallengeData(saveData) else {
            rejectChallenge("挑战回合存档失败：\(saveStore.lastError?.localizedDescription ?? "unknown")")
            return
        }
        // FUN_0006d80c 是独立 bonus 状态机，禁止误入普通结果控制器。
        if round.runtime.isBonusRound {
            rejectChallenge("bonus 回合尚未接入其独立结算状态机")
            return
        }
        do {
            guard var resultPlayer = challengeResultAnimationPlayer,
                  var backgroundPlayer = challengeBackgroundAnimationPlayer else {
                throw ChallengeHostError("普通回合结算缺少真实动画资源")
            }
            var controller = ChallengeSettlementController()
            let output = try controller.beginNormalResults(
                outcome: outcome,
                resultPlayer: &resultPlayer,
                backgroundPlayer: &backgroundPlayer)
            challengeSettlement = controller
            challengeResultAnimationPlayer = resultPlayer
            challengeBackgroundAnimationPlayer = backgroundPlayer
            challengeBackgroundVisible = true
            processChallengeAnimationOutput(output)
        } catch {
            rejectChallenge("无法开始普通回合结算：\(error.localizedDescription)")
        }
    }

    private func playChallengeAnimation(target: String, name: String) {
        do {
            if target == "Background" {
                guard var player = gameBackgroundAnimationPlayer else {
                    throw ChallengeHostError("独立 Background 屏幕动画资源不可用")
                }
                // PlayAnimation 的第一个参数是屏幕名；根组件递归定位 inGame。
                let output = try player.playRecursively(component: "", animation: name)
                gameBackgroundAnimationPlayer = player
                processChallengeAnimationOutput(output)
                return
            }
            guard var player = challengeHUDAnimationPlayer else {
                throw ChallengeHostError("挑战 HUD 动画资源不可用")
            }
            // inGameChallengeHud 是屏幕名，生成资源的根组件名为空字符串。
            let component = target == "inGameChallengeHud" ? "" : target
            let output = try player.playRecursively(component: component, animation: name)
            challengeHUDAnimationPlayer = player
            processChallengeAnimationOutput(output)
        } catch {
            rejectChallenge("挑战动画 \(target)/\(name) 启动失败：\(error.localizedDescription)")
        }
    }

    private func processChallengeAnimationOutput(_ output: [UIScreenAnimationOutput]) {
        for item in output {
            switch item {
            case let .event(_, _, command, arguments):
                switch command {
                case "PlaySound":
                    if let key = arguments.first { audio.play(key) }
                case "PlayAnimation":
                    if arguments.count == 2 {
                        playChallengeAnimation(target: arguments[0], name: arguments[1])
                    }
                default:
                    rejectChallenge("未支持的原始挑战动画事件：\(command)")
                }
            case .completed:
                // bonusIntro/play 的完成必须走递归 query；“completed”事件
                // 只用于已有资源事件，不拿来猜测多子动画是否全部完成。
                break
            }
        }
    }

    func requestNextChallengeRound() {
        guard var controller = challengeSettlement,
              var background = challengeBackgroundAnimationPlayer,
              controller.isNormalResultFinished,
              let assembler = challengeAssembler,
              assembler.currentRoundIndex + 1 < assembler.plan.count else { return }
        do {
            let output = try controller.requestNextRound(backgroundPlayer: &background)
            challengeSettlement = controller
            challengeBackgroundAnimationPlayer = background
            processChallengeAnimationOutput(output)
        } catch {
            rejectChallenge("无法请求下一挑战回合：\(error.localizedDescription)")
        }
    }

    func requestChallengeCleanup() {
        guard var controller = challengeSettlement,
              var background = challengeBackgroundAnimationPlayer,
              controller.isNormalResultFinished else { return }
        do {
            let output = try controller.requestCleanup(backgroundPlayer: &background)
            challengeSettlement = controller
            challengeBackgroundAnimationPlayer = background
            processChallengeAnimationOutput(output)
        } catch {
            rejectChallenge("无法退出挑战回合：\(error.localizedDescription)")
        }
    }

        challengeConfigurationError = message
        Self.contentLogger.error("挑战流程已终止：\(message, privacy: .public)")
        viewModel.finishGame()
    }

    private func consumeClassicEvents() {
        for event in gameWorld.drainClassicEvents() {
            switch event {
            case let .missCross(index):
                guard (1...3).contains(index) else { continue }
                visibleClassicCrosses.insert(Int(index))
                startClassicAnimation(component: "misscounters.cross\(index)", animation: "out")
            case let .restoreCross(index):
                guard (1...3).contains(index) else { continue }
                visibleClassicCrosses.insert(Int(index))
                startClassicAnimation(component: "misscounters.cross\(index)", animation: "in")
            case let .magicBeanPopup(kind, _):
                // FUN_00074254 的位置参数已保留在事件中；原控件坐标投影待 HUD 整组接入。
                startClassicAnimation(component: kind.rawValue, animation: "trigger")
            case .gameOver:
                startClassicAnimation(component: "gameOverContainer", animation: "game_over")
            case .gameOverReal:
                startClassicAnimation(component: "gameOverContainer", animation: "game_over_real")
            case .results:
                break
            }
        }
    }

    private func startClassicAnimation(component: String, animation: String) {

        } catch {

            Self.contentLogger.error("Classic 动画启动失败：\(error.localizedDescription, privacy: .public)")
        }
    }

    private func processClassicAnimationOutput(_ output: [UIScreenAnimationOutput]) {
        for item in output {
            switch item {
            case let .event(_, _, command, arguments):
                if command == "PlaySound", let key = arguments.first {
                    audio.play(key)
                }
            case let .completed(component, animation):
                if component == "gameOverContainer", animation == "game_over",
                   state.classicPhase == .gameOver {
                    gameWorld.completeClassicAnimation(state: &state)
                    consumeClassicEvents()
                } else if component == "gameOverContainer", animation == "game_over_real",
                          state.classicPhase == .gameOverReal {
                    gameWorld.completeClassicAnimation(state: &state)
                    consumeClassicEvents()
                } else if component.hasPrefix("misscounters.cross"), animation == "out",
                          let index = Int(component.dropFirst("misscounters.cross".count)) {
                    // FUN_00071838_1：out 完成回调在这里隐藏控件，而不是漏切瞬间隐藏。
                    // 同回调的 heartLostSml 粒子创建仍属于尚未重建的特效系统。
                    visibleClassicCrosses.remove(index)
                }
            }
        }
    }

    /// 逻辑像素尺寸（兜底用）。TouchCoordinateMapper 要求 SIMD2<Float>。
    ///
    /// T-024：正常路径已改接 `RenderSurface.logicalSize`（逻辑点）；

    static func currentLogicalSize() -> SIMD2<Float> {
        let scenes = UIApplication.shared.connectedScenes.compactMap { $0 as? UIWindowScene }
        let scene = scenes.first { $0.activationState == .foregroundActive } ?? scenes.first
        guard let screen = scene?.screen else {
            return SIMD2<Float>(1024, 768)          // 兜底（不应命中）
        }
        let bounds = screen.bounds
        return SIMD2<Float>(Float(bounds.width * screen.scale),
                            Float(bounds.height * screen.scale))
    }
