PROP_GROUP_HEADER = b"\x00\x07\xaa\xa1"
PROP_GROUP_COUNT = b"\x00\x07\xbb\xb1"
PROP_GROUP_END = b"\x00\x07\xff\xf1"
PROP_RECORD_HEADER = b"\x00\x07\xde\xe1"
PROP_RECORD_HEADER_2 = b"\xff\xff\xff\xff"

PROPERTY_CATEGORIES: dict[str, str] = {}


def _register_properties(category: str, names: tuple[str, ...]) -> None:
    for name in names:
        if name in PROPERTY_CATEGORIES:
            raise RuntimeError(f"duplicate static property mapping: {name!r}")
        PROPERTY_CATEGORIES[name] = category

def property_category(name: str) -> str:
    category = PROPERTY_CATEGORIES.get(name)
    if category is None:
        raise FormatError(f"unknown static property name {name!r}")
    return category


def _require_property_range(reader: Reader, offset: int, size: int) -> None:
    if offset < 0 or size < 0 or offset + size > len(reader.data):
        raise FormatError(f"truncated property record at 0x{offset:x}")


def _read_property_u32(reader: Reader, offset: int) -> int:
    _require_property_range(reader, offset, 4)
    return reader.u32(offset)


def _read_property_f32(reader: Reader, offset: int) -> float:
    _require_property_range(reader, offset, 4)
    return reader.f32(offset)


def _parse_vec_metadata(
    reader: Reader,
    cursor: int,
    name: str,
    record_offset: int,
) -> tuple[list[int], int]:
    metadata: list[int] = []
    pending: int | None = None

    while True:
        if pending is None:
            token = _read_property_u32(reader, cursor)
            cursor += 4
        else:
            token = pending
            pending = None

        if token == 0:
            raise FormatError(
                f"{name!r}: zero in vec metadata before 1,0 at 0x{record_offset:x}"
            )

        if token == 1:
            following = _read_property_u32(reader, cursor)
            cursor += 4

            if following == 0:
                if not metadata:
                    raise FormatError(
                        f"{name!r}: vec metadata is empty at 0x{record_offset:x}"
                    )
                if len(metadata) > 8:
                    raise FormatError(
                        f"{name!r}: more than 8 vec metadata words at 0x{record_offset:x}"
                    )
                return metadata, cursor

            metadata.append(token)
            if len(metadata) > 8:
                raise FormatError(
                    f"{name!r}: more than 8 vec metadata words at 0x{record_offset:x}"
                )
            pending = following
        else:
            metadata.append(token)
            if len(metadata) > 8:
                raise FormatError(
                    f"{name!r}: more than 8 vec metadata words at 0x{record_offset:x}"
                )


def parse_property_record(reader: Reader, offset: int) -> tuple[dict[str, Any], int]:
    record_offset = offset
    name, cursor = reader.string(offset)
    category = property_category(name)

    reader.require(cursor, PROP_RECORD_HEADER)
    cursor += 4
    reader.require(cursor, PROP_RECORD_HEADER_2)
    cursor += 4

    metadata: list[int] = []
    raw_value: int | None = None
    trailer: int | None = None

    if category in ("vec2", "vec3", "vec4"):
        dimension = {"vec2": 2, "vec3": 3, "vec4": 4}[category]

        metadata, cursor = _parse_vec_metadata(reader, cursor, name, record_offset)

        value: list[float] = []
        for _ in range(dimension):
            value.append(_read_property_f32(reader, cursor))
            cursor += 4

        trailer = _read_property_u32(reader, cursor)
        cursor += 4
        if trailer != 0:
            raise FormatError(
                f"{name!r}: invalid vec trailer at 0x{record_offset:x}"
            )

    elif category in ("f32", "u32", "bool", "string"):
        header: list[int] = []
        for _ in range(3):
            header.append(_read_property_u32(reader, cursor))
            cursor += 4

        if header != [0, 1, 0]:
            raise FormatError(
                f"{name!r}: invalid scalar/string/bool header {header!r} "
                f"at 0x{record_offset:x}"
            )

        if category == "string":
            value, cursor = reader.string(cursor)

            trailer = _read_property_u32(reader, cursor)
            cursor += 4
            if trailer != 0:
                raise FormatError(
                    f"{name!r}: invalid string trailer at 0x{record_offset:x}"
                )

        elif category == "bool":
            raw_value = _read_property_u32(reader, cursor)
            cursor += 4
            if raw_value not in (0, 1):
                raise FormatError(
                    f"{name!r}: invalid bool payload {raw_value} "
                    f"at 0x{record_offset:x}"
                )
            value = bool(raw_value)

            trailer = _read_property_u32(reader, cursor)
            cursor += 4
            if trailer != 0:
                raise FormatError(
                    f"{name!r}: invalid bool trailer at 0x{record_offset:x}"
                )

        elif category == "f32":
            value_offset = cursor
            raw_value = _read_property_u32(reader, cursor)
            cursor += 4
            value = _read_property_f32(reader, value_offset)

            trailer = _read_property_u32(reader, cursor)
            cursor += 4
            if trailer != 0:
                raise FormatError(
                    f"{name!r}: invalid f32 trailer at 0x{record_offset:x}"
                )

        elif category == "u32":
            raw_value = _read_property_u32(reader, cursor)
            cursor += 4
            value = raw_value

            trailer = _read_property_u32(reader, cursor)
            cursor += 4
            if trailer != 0:
                raise FormatError(
                    f"{name!r}: invalid u32 trailer at 0x{record_offset:x}"
                )

        else:
            raise RuntimeError(f"unhandled scalar category {category!r}")

    else:
        raise RuntimeError(f"unhandled property category {category!r}")

    return {
        "name": name,
        "category": category,
        "offset": record_offset,
        "end_offset": cursor,
        "metadata": metadata,
        "raw_value": raw_value,
        "trailer": trailer,
        "value": value,
    }, cursor
