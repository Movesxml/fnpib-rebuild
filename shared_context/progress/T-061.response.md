下面是一份可直接替换原脚本的完整版本：按数据驱动规格表处理 4 张 PVRTC4 纹理解码，以及 2 张 RGBA32 PNG 逐字节复制；所有断言均在写入前/后执行。

```python
"""重新生成挑战地图所需的纹理 PNG。

输入（只读）：
  assets_raw/textures/adventureProgress/*.tex   12 字节 legacy 头 + PVRTC4 负载
  assets_png/textures/adventureProgress/*.png   已由资产管线验证的 RGBA32 PNG

输出（全部进入 Content）：
  challengeAtlas.png
  challengeCross.png
  challengeTick.png
  challengeLocations.png
  challengeIconSlot.png
  challengeIconShadow.png
"""

from pathlib import Path
import sys

from PIL import Image

ROOT = Path(__file__).resolve().parents[1]
sys.path.insert(0, str(ROOT / "tools/AssetPipeline/vendor"))
import texture2ddecoder  # noqa: E402

RAW_DIR = ROOT / "assets_raw/textures/adventureProgress"
PNG_DIR = ROOT / "assets_png/textures/adventureProgress"
OUT_DIR = ROOT / "Content"

PVRTC4_HEADER_LEN = 12
PVRTC4_FORMAT_ID = 12

# 数据驱动规格表：12 字节 legacy 头 + PVRTC4 负载。
# header[2:4] = format_id(12)，header[4:6] = width，header[6:8] = height，均为小端。
PVRTC_SPECS = (
    {
        "name": "challengeAtlas",
        "source": "challengeAtlas.tex",
        "output": "challengeAtlas.png",
        "size": 1024,
        "header": bytes.fromhex("0a 0a 0c 00 00 04 00 04 ff ff ff ff"),
    },
    {
        "name": "challengeCross",
        "source": "challengeCross.tex",
        "output": "challengeCross.png",
        "size": 128,
        "header": bytes.fromhex("07 07 0c 00 80 00 80 00 ff ff ff ff"),
    },
    {
        "name": "challengeTick",
        "source": "challengeTick.tex",
        "output": "challengeTick.png",
        "size": 128,
        "header": bytes.fromhex("07 07 0c 00 80 00 80 00 ff ff ff ff"),
    },
    {
        "name": "challengeLocations",
        "source": "challengeLocations.tex",
        "output": "challengeLocations.png",
        "size": 512,
        "header": bytes.fromhex("09 09 0c 00 00 02 00 02 ff ff ff ff"),
    },
)

# 这两张已经是 RGBA32 PNG，只做逐字节复制，禁止重解码或重新压缩。
COPY_SPECS = (
    {
        "name": "challengeIconSlot",
        "source": "challengeIconSlot.png",
        "output": "challengeIconSlot.png",
    },
    {
        "name": "challengeIconShadow",
        "source": "challengeIconShadow.png",
        "output": "challengeIconShadow.png",
    },
)


def decode_pvrtc_spec(spec):
    source = RAW_DIR / spec["source"]
    output = OUT_DIR / spec["output"]
    size = spec["size"]

    data = source.read_bytes()

    # 精确头/长度断言。
    expected_len = PVRTC4_HEADER_LEN + size * size // 2
    assert len(data) == expected_len, (source, len(data), expected_len)
    header = data[:PVRTC4_HEADER_LEN]
    assert header == spec["header"], (source, header.hex(), spec["header"].hex())
    assert int.from_bytes(header[2:4], "little") == PVRTC4_FORMAT_ID
    assert int.from_bytes(header[4:6], "little") == size
    assert int.from_bytes(header[6:8], "little") == size

    payload = data[PVRTC4_HEADER_LEN:]
    assert len(payload) == size * size // 2

    rgba = texture2ddecoder.decode_pvrtc(payload, size, size, True)
    assert len(rgba) == size * size * 4, (source, len(rgba), size * size * 4)

    output.parent.mkdir(parents=True, exist_ok=True)
    Image.frombytes("RGBA", (size, size), rgba).save(output)
    assert output.is_file() and output.stat().st_size > 0
    return output


def copy_png_spec(spec):
    source = PNG_DIR / spec["source"]
    output = OUT_DIR / spec["output"]

    source_data = source.read_bytes()
    assert source_data[:8] == b"\x89PNG\r\n\x1a\n", source

    output.parent.mkdir(parents=True, exist_ok=True)
    output.write_bytes(source_data)

    # 复制前后字节相等断言。
    output_data = output.read_bytes()
    assert output_data == source_data, output
    assert output.stat().st_size == len(source_data)
    return output


def main():
    outputs = []

    for spec in PVRTC_SPECS:
        outputs.append(decode_pvrtc_spec(spec))

    for spec in COPY_SPECS:
        outputs.append(copy_png_spec(spec))

    for output in outputs:
        print(output)


if __name__ == "__main__":
    main()
```
