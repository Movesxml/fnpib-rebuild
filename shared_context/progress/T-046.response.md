// MARK: - SpawnKind

/// Spawn.type 的显式类型；水果 ID 使用 FruitInfo 文档零基序号。
/// bomb/random 不使用水果 ID 或 EntityKindValue 冒充。
///
/// T-046 新增：
///  - `.empty`：XML `Spawn.type="fruit"` 的显式空槽。
///    原版 FUN_00055670 的 else 分支调用 FUN_0000aae0("fruit", 0)；
///    水果表没有该名称时返回 -1，不生成实体。
///  - `.challengeFruit`：XML `Spawn.type="challengeFruit"` 的显式绑定。
///    原版直接取 FUN_000617a8() 的 controller+0x88，不做名称查找。
public enum SpawnKind: Equatable, Sendable {
    case fruit(Int32)
    case bomb
    case random
    case empty
    case challengeFruit
}

extension SpawnKind: Codable {
    public init(from decoder: Decoder) throws {
        let container = try decoder.singleValueContainer()
        let raw = try container.decode(String.self)
        switch raw {
        case "bomb": self = .bomb
        case "random": self = .random
        case "empty": self = .empty
        case "challengeFruit": self = .challengeFruit
        default:
            let prefix = "fruit:"
            guard raw.hasPrefix(prefix), let id = Int32(raw.dropFirst(prefix.count)) else {
                throw DecodingError.dataCorruptedError(
                    in: container, debugDescription: "未知 SpawnKind 字面量: \(raw)")
            }
            self = .fruit(id)
        }
    }

    public func encode(to encoder: Encoder) throws {
        var container = encoder.singleValueContainer()
        switch self {
        case .fruit(let id): try container.encode("fruit:\(id)")
        case .bomb: try container.encode("bomb")
        case .random: try container.encode("random")
        case .empty: try container.encode("empty")
        case .challengeFruit: try container.encode("challengeFruit")
        }
    }
}

    private static func parseSpawn(_ node: ContentNode,
                                   path: String,
                                   fruitIndex: [String: Int32]) throws -> Spawner {
        var spawner = Spawner()
        if let value = try floatValue(node, "min", path) { spawner.min = value }
        if let value = try floatValue(node, "max", path) { spawner.max = value }
        let minInc = try floatValue(node, "mininc", path)
        let maxInc = try floatValue(node, "maxinc", path)
        spawner.minInc = maxInc ?? minInc ?? 0
        spawner.maxInc = maxInc ?? 0
        if let value = try floatValue(node, "delay", path) { spawner.delay = value }
        if let value = try floatValue(node, "delayinc", path) { spawner.delayInc = value }

        // FUN_000537ec：velscale 先同时写两轴，显式单轴缩放随后覆盖。
        if let value = try floatValue(node, "velX", path) { spawner.velocityX = value }
        if let value = try floatValue(node, "velY", path) { spawner.velocityY = value }
        if let value = try floatValue(node, "velscale", path) {
            spawner.velocityScaleX = value
            spawner.velocityScaleY = value
        } else {
            spawner.velocityScaleY = spawner.velocityScaleX
        }
        if let value = try floatValue(node, "velXscale", path) { spawner.velocityScaleX = value }
        if let value = try floatValue(node, "velYscale", path) { spawner.velocityScaleY = value }

        // 原版只采纳非空 gravity；这里要求与 FUN_000394a0 对应的三分量逗号格式。
        if let raw = node.attrs["gravity"], !raw.isEmpty {
            let components = raw.split(separator: ",", omittingEmptySubsequences: false)
            guard components.count == 3 else {
                throw OriginalContentError.invalidNumber(node: node.tag, attribute: "gravity",
                                                         raw: raw, path: path)
            }
            var values: [Float] = []
            values.reserveCapacity(3)
            for component in components {
                let trimmed = component.trimmingCharacters(in: .whitespacesAndNewlines)
                guard let value = Float(trimmed) else {
                    throw OriginalContentError.invalidNumber(node: node.tag, attribute: "gravity",
                                                             raw: raw, path: path)
                }
                values.append(value)
            }
            spawner.gravity = SIMD3<Float>(values[0], values[1], values[2])
        }

        guard let rawType = node.attrs["type"] else {
            throw OriginalContentError.missingAttribute(node: node.tag, attribute: "type", path: path)
        }
        let tokens = rawType.split(separator: ",", omittingEmptySubsequences: false)
            .map { $0.trimmingCharacters(in: .whitespacesAndNewlines) }
        for (index, token) in tokens.enumerated() {
            guard !token.isEmpty else {
                throw OriginalContentError.emptySpawnTypeToken(index: index, path: path)
            }

            // FUN_00055670：先比较 bomb/random/challengeFruit 的 ASCII 大小写不敏感
            // 名称哈希；"fruit" 不是特殊名称，走水果表查找，未命中才返回 -1。
            if asciiCaseInsensitiveEquals(token, "bomb") {
                spawner.types.append(.bomb)
            } else if asciiCaseInsensitiveEquals(token, "random") {
                spawner.types.append(.random)
            } else if asciiCaseInsensitiveEquals(token, "challengeFruit") {
                // FUN_00055670 的显式分支：直接取 controller+0x88。
                spawner.types.append(.challengeFruit)
            } else if let typeID = fruitIndex[token] {
                spawner.types.append(.fruit(typeID))
            } else if asciiCaseInsensitiveEquals(token, "fruit") {
                // FUN_0000aae0("fruit", 0)：水果表无此名时返回 -1 空槽。
                spawner.types.append(.empty)
            } else {
                // 普通未知名字维持现有抛错，不扩大行为。
                throw OriginalContentError.unknownSpawnType(raw: token, path: path)
            }
        }
        return spawner
    }

    private func resolveTypeSequence(spawner: Spawner) -> [Int32] {

        kinds.reserveCapacity(spawner.types.count)
        for kind in spawner.types {
            switch kind {
            case .fruit(let id):
                kinds.append(id)
            case .bomb:
                kinds.append(Constants.EntityPool.bombSlot)
            case .empty:
                // XML `Spawn.type="fruit"`：FUN_0000aae0 未命中水果表 → -1。
                kinds.append(Constants.EntityPool.emptySlot)

                kinds.append(challengeFruitID ?? Constants.EntityPool.emptySlot)

            }

// 4b) 替换 drawRandomFruit：

    /// 普通模式权重抽取：按 `FruitInfo.chance` 累计和、保持文档顺序。
    /// 挑战模式（challengeFruitID 非 nil）按 FUN_00055670 反复权重抽取，
    /// 直到结果不等于 controller+0x88；普通模式没有此排除。
    /// 无正权重候选 → 空槽（-1），不以水果 0 兜底。
    private func drawRandomFruit() -> Int32 {
        var cumulative: [Int64] = []
        var total: Int64 = 0
        for fruit in fruits {
            total += max(0, Int64(fruit.chance ?? 0))
            cumulative.append(total)
        }
        guard total > 0 else { return Constants.EntityPool.emptySlot }

        let bound = total >= Int64(Int32.max) ? Int32.max : Int32(total)
        let excluded = challengeFruitID

        while true {
            let roll = Int64(rng.nextInt(bound))
            var picked: Int32 = Constants.EntityPool.emptySlot
            for (index, boundary) in cumulative.enumerated() where roll < boundary {
                picked = fruits[index].typeID
                break
            }

            if let excluded, picked == excluded {
                // FUN_00055670 挑战分支：反复权重抽取直到不等于 controller+0x88。
                // 原机无次数上限；挑战水果池必须至少包含一个非排除 ID。
                continue
            }
            return picked
        }
    }

/// 测试专用 RandomSource：按顺序返回预设 nextInt 值。
/// 方法签名按现有 RandomSource 接口；若仓库协议签名不同，仅调整
/// `nextInt(_:)` / `nextFloat()` 的声明，不改变生产逻辑。
private final class QueueRandomSource: RandomSource {
    private var ints: [Int32]

    init(ints: [Int32] = []) {
        self.ints = ints
    }

    func nextInt(_ upperBound: Int32) -> Int32 {
        guard upperBound > 0, !ints.isEmpty else { return 0 }
        let value = ints.removeFirst()
        return min(max(0, value), upperBound - 1)
    }

    func nextFloat() -> Float { 0 }
}

## 建议审查重点

1. `SpawnKind.empty` 对应 XML 字面量 `"fruit"`，`SpawnKind.challengeFruit` 对应 `"challengeFruit"`；普通未知名称仍抛 `unknownSpawnType`。
2. `WaveInfo.name` 保持 Optional 以兼容旧 Codable；真实加载时由 loader 填 `debugName` 或 `"waveNo. <documentIndex>"`。
3. `setChallengeFruit(_:)` 必须由调用方在 `beginGame` 前调用；草案未从 wave/challenge 名称推断该 ID。
4. `drawRandomFruit` 的挑战排除分支严格保留原机“反复重抽直到不等于”的语义；真实挑战水果池需至少包含一个非排除 ID，否则原机同样不会终止。
5. 测试辅助 `QueueRandomSource` 仅为草案测试可控 RNG；生产代码未新增 RandomSource 方法。
