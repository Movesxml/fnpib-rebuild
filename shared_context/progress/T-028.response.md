简短设计说明：

- **ContentNode**：手写 `init(from:)` 容错匹配 `tag / attrs / text / tail / children`（缺失字段给默认），`encode` 由编译器合成；root 校验、数字/布尔/必填属性解析全部集中在 `OriginalContentLoader`，错误枚举带节点路径。
- **加载结果**：`OriginalFruitList`（fruits/bomb/critical）与 `OriginalWaveList`（waves/defaults）分离，避免 critical 字段被灌进每个 FruitInfo。
- **Spawn.type**：拆分为 `[SpawnKind]`（`.fruit(Int32)/.bomb/.random`），挂在 `Spawner.types` 新字段上，旧 `SpawnerFruit.kind` 哨兵不改；未知水果名 throw。
- **maxinc 覆盖**：`minInc` 取 maxinc 运行时值，`maxInc` 保留原始值（maxinc 缺失时按「第二次写 +0x68」的语义，minInc 保持 mininc）。
- **WaveInfo**：新增 `nextWait / nextWaitSpInc`（不把 wait 冒充 delay/inc）；`until` 原始 tokens 交 `postProcessUntil`；defaults 按原二进制键名 `waveChanceGrowth` 读取。
- **兼容性**：纯 Swift 逻辑，无 UIKit/GLES 依赖；Swift 6 严格并发下全部 Sendable，无共享可变状态。

public extension WaveInfo {

    /// `until` 语义后处理（core_logic_notes §4.4 + T-027 精修；
    /// **[macho] 独有**，exe 未见；精确符合 `FUN_00053cb8` 末尾循环）。
    ///
    /// 规则（按序执行，顺序不可交换）：
    ///  1. 非空且精确等于小写 `"forever"` → `-2`；原版使用 `strcmp`，不忽略大小写。
    ///  2. 其他非空串按 C `atoi` 读取数字前缀；结果 `>= 0 && >= waveNo[i]` 才采纳。
    ///  3. 仅对**仍为 -1**（空串 / 未采纳）的条目：
    ///     a. 先设为 `-2`（forever 兜底）
    ///     b. 从**所有** `waveNo[j] > waveNo[i]` 中取最小 `waveNo - 1`
    ///     c. 若无严格更大的 waveNo（当前是最大 / 全部重复），保持 `-2`
    ///  4. **显式有效的 `until` 绝不做旧 step3 改写**。
    ///
    /// 注意：`atoi("garbage") == 0`。因此 waveNo=0 的非空非法串会采纳 0，
    /// 这与空串缺省不同；不能用 Swift 的严格 `Int32(token)` 替代。
    ///
    /// - Parameters:
    ///   - rawTokens: XML `until` 原始串数组（空串表示属性缺失）
    ///   - waveNos:   每波原始 `waveNo`（严格与 rawTokens 等长；可重复）
    /// - Returns: 与 `rawTokens` 等长的后处理结果
    static func postProcessUntil(rawTokens: [String], waveNos: [Int32]) -> [Int32] {
        let n = rawTokens.count
        guard n > 0 else { return [] }
        precondition(waveNos.count == n,
                     "postProcessUntil: waveNos 必须与 rawTokens 等长（禁止下标兜底）")

        var result = [Int32](repeating: Constants.WaveDefaults.untilUnparsed, count: n)
        for i in 0..<n {
            let token = rawTokens[i]
            guard !token.isEmpty else { continue }
            if token == "forever" {
                result[i] = Constants.WaveDefaults.untilForever
                continue
            }
            let parsed = cAtoi(token)
            if parsed >= 0 && parsed >= waveNos[i] {
                result[i] = parsed
            }
        }

        for i in 0..<n where result[i] == Constants.WaveDefaults.untilUnparsed {
            result[i] = Constants.WaveDefaults.untilForever
            let currentNo = waveNos[i]
            var minGreater: Int32?
            for otherNo in waveNos where otherNo > currentNo {
                if minGreater == nil || otherNo < minGreater! {
                    minGreater = otherNo
                }
            }
            if let g = minGreater {
                result[i] = g - 1
            }
        }
        return result
    }

    /// 本任务所需的 C `atoi` 语义：ASCII 空白、可选符号、数字前缀；无数字返回 0。
    /// 超出 Int32 时做饱和保护；原版 C 溢出属未定义行为，不把它宣称为官方规则。
    private static func cAtoi(_ token: String) -> Int32 {
        let bytes = Array(token.utf8)
        var index = 0
        while index < bytes.count {
            switch bytes[index] {
            case 9, 10, 11, 12, 13, 32: index += 1
            default: break
            }
            if index >= bytes.count || !(9...13).contains(bytes[index]) && bytes[index] != 32 {
                break
            }
        }

        var negative = false
        if index < bytes.count, bytes[index] == 45 || bytes[index] == 43 {
            negative = bytes[index] == 45
            index += 1
        }

        let limit = negative ? Int64(Int32.max) + 1 : Int64(Int32.max)
        var value: Int64 = 0
        while index < bytes.count, bytes[index] >= 48 && bytes[index] <= 57 {
            let digit = Int64(bytes[index] - 48)
            value = value > (limit - digit) / 10 ? limit : value * 10 + digit
            index += 1
        }
        if negative {
            return value >= Int64(Int32.max) + 1 ? .min : Int32(-value)
        }
        return Int32(value)
    }

    /// 就地应用后处理结果（辅助入口）。
    ///
    /// - Parameters:
    ///   - waves:     待后处理的 WaveInfo 数组（要求每项 `waveNo >= 0` 已由解析层填好）
    ///   - rawTokens: XML `until` 原始串数组（与 `waves` 顺序一致）
    ///
    /// ⚠️ 本 API 不再用 `String($0.until)` 反推 token —— Int32 → String 会丢失原始字面量
    ///    （如 `forever` / 前导空串），必须由调用方持有原始串。
    static func applyUntilPostProcess(to waves: inout [WaveInfo], rawTokens: [String]) {
        let waveNos = waves.map { $0.waveNo }
        let processed = postProcessUntil(rawTokens: rawTokens, waveNos: waveNos)
        for i in 0..<min(waves.count, processed.count) {
            waves[i].until = processed[i]
        }
    }
}

/// `Spawn.type` 逗号列表的显式表示（T-028）。
///
/// 原版 XML：`type="apple, watermelon, bomb, random"`。
/// - `.fruit(Int32)`：水果零基 typeID（与 `FruitInfo.typeID` 同一空间）
/// - `.bomb`：炸弹（**禁止**用 `Constants.EntityPool.bombSlot` 等哨兵值冒充）

public enum SpawnKind: Equatable, Sendable {
    case fruit(Int32)
    case bomb
    case random
}

extension SpawnKind: Codable {

    public init(from decoder: Decoder) throws {
        let container = try decoder.singleValueContainer()
        let raw = try container.decode(String.self)
        switch raw {
        case "bomb":
            self = .bomb
        case "random":
            self = .random
        default:
            let prefix = "fruit:"
            guard raw.hasPrefix(prefix), let id = Int32(raw.dropFirst(prefix.count)) else {
                throw DecodingError.dataCorruptedError(
                    in: container,
                    debugDescription: "未知 SpawnKind 字面量: \(raw)")
            }
            self = .fruit(id)
        }
    }

    public func encode(to encoder: Encoder) throws {
        var container = encoder.singleValueContainer()
        switch self {
        case .fruit(let id):
            try container.encode("fruit:\(id)")
        case .bomb:
            try container.encode("bomb")
        case .random:
            try container.encode("random")
        }
    }
}

//
// 输入：T-026 转换器产出的 JSON（原 XML 元素树，节点字段 tag/attrs/text/tail/children）。

// 设计要点（对应任务卡 T-028）：
//  1. ContentNode 手写解码：tag 必需；attrs/text/tail/children 缺失给安全默认。
//  2. 严格解析：root tag 校验；数字/布尔非法 throw；Fruit 必需 name；Spawn 必需 type。
//     所有 throw 均带节点路径（如 "FruitList/Fruit[1]"）。
//  3. 水果 typeID = Fruit 节点在 XML 文档中的零基顺序（apple 0 / watermelon 3 / pear_gold 23）；
//     noScore 由 score==0 派生。

//  5. Spawn.type 按逗号拆分并 trim，保序映射为 [SpawnKind]；未知水果名 throw。
//  6. Spawn min/max/mininc/maxinc/delay/delayinc → T-027 字段；

//  7. waveNo 缺失时按文档序号；until 原始 tokens 交给 WaveInfo.postProcessUntil（T-027 修正版）；
//     Wave_dt → dt/inc/spinc；NextWaveDelay → delay/inc/wait/waitSpinc/speedLoss/

//     不做 waveChanceRegrowth 之类的擅自别名。
//
// 未知子节点处理：忽略（不参与解析）。这是为了让真实原版 XML 中的非玩法节点
// （若有）不阻断加载；但**不**用默认值兜底任何已识别字段的非法值。

// MARK: - 通用 XML 节点

/// 原 XML 元素树的 JSON 表示（T-026 产物）。
public struct ContentNode: Codable, Equatable, Sendable {

    public var tag: String
    public var attrs: [String: String]
    public var text: String?
    public var tail: String?
    public var children: [ContentNode]

    public init(tag: String,
                attrs: [String: String] = [:],
                text: String? = nil,
                tail: String? = nil,
                children: [ContentNode] = []) {
        self.tag = tag
        self.attrs = attrs
        self.text = text
        self.tail = tail
        self.children = children
    }

    private enum CodingKeys: String, CodingKey {
        case tag, attrs, text, tail, children
    }

    public init(from decoder: Decoder) throws {
        let container = try decoder.container(keyedBy: CodingKeys.self)
        self.tag = try container.decode(String.self, forKey: .tag)
        self.attrs = try container.decodeIfPresent([String: String].self, forKey: .attrs) ?? [:]
        self.text = try container.decodeIfPresent(String.self, forKey: .text)
        self.tail = try container.decodeIfPresent(String.self, forKey: .tail)
        self.children = try container.decodeIfPresent([ContentNode].self, forKey: .children) ?? []
    }
}

    /// 同时加载 fruitlist / wavelist 两份 JSON。
    public static func load(fruitsData: Data, wavesData: Data) throws -> OriginalContent {
        let fruitList = try loadFruits(from: fruitsData)
        let waves = try loadWaves(from: wavesData, fruits: fruitList.fruits)
        return OriginalContent(fruits: fruitList, waves: waves)

    /// 加载 fruitlist.xml 对应的 JSON。
    public static func loadFruits(from data: Data) throws -> OriginalFruitList {
        let root = try decodeNode(from: data, source: "FruitList")
        guard root.tag.lowercased() == "fruitlist" else {
            throw OriginalContentError.unexpectedRoot(expected: "FruitList", actual: root.tag)
        }

        var fruits: [FruitInfo] = []
        var bomb: BombInfo?
        var critical: CriticalConfig?
        var seenNames = Set<String>()

        for (index, child) in root.children.enumerated() {
            let path = "FruitList/\(child.tag)[\(index)]"
            switch child.tag.lowercased() {
            case "fruit":
                let info = try parseFruit(child, typeID: Int32(fruits.count), path: path)
                guard !seenNames.contains(info.name) else {
                    throw OriginalContentError.duplicateFruit(name: info.name, path: path)
                }
                seenNames.insert(info.name)
                fruits.append(info)
            case "critical":
                guard critical == nil else {
                    throw OriginalContentError.duplicateNode(tag: child.tag, path: path)
                }
                critical = try parseCritical(child, path: path)
            case "bomb":
                guard bomb == nil else {
                    throw OriginalContentError.duplicateNode(tag: child.tag, path: path)
                }
                bomb = try parseBomb(child, path: path)
            default:
                // 未知节点：不参与解析（见文件头说明），不静默兜底任何已识别字段。
                continue
            }
        }

        return OriginalFruitList(fruits: fruits,
                                 bomb: bomb ?? BombInfo(),
                                 critical: critical)
    }

    /// 加载 wavelist.xml 对应的 JSON（水果表由水果条目提供）。
    public static func loadWaves(from data: Data, fruits: [FruitInfo]) throws -> OriginalWaveList {
        let index = try makeFruitIndex(fruits)
        return try loadWaves(from: data, fruitIndex: index)
    }

    /// 加载 wavelist.xml 对应的 JSON（水果名 → 零基 typeID）。
    public static func loadWaves(from data: Data, fruitIndex: [String: Int32]) throws -> OriginalWaveList {
        let root = try decodeNode(from: data, source: "WaveList")
        guard root.tag.lowercased() == "wavelist" else {
            throw OriginalContentError.unexpectedRoot(expected: "WaveList", actual: root.tag)
        }

        // 1) Defaults（最多一个）
        var defaults = OriginalWaveDefaults()
        var sawDefaults = false
        for (index, child) in root.children.enumerated() where child.tag.lowercased() == "defaults" {
            let path = "WaveList/\(child.tag)[\(index)]"
            guard !sawDefaults else {
                throw OriginalContentError.duplicateNode(tag: child.tag, path: path)
            }
            sawDefaults = true
            defaults = try parseDefaults(child, path: path)
        }

        // 2) Wave（文档顺序）
        var waves: [WaveInfo] = []
        var untilTokens: [String] = []
        var documentIndex: Int32 = 0
        for (index, child) in root.children.enumerated() {
            guard child.tag.lowercased() == "wave" else { continue }
            let path = "WaveList/\(child.tag)[\(index)]"
            let parsed = try parseWave(child,
                                       documentIndex: documentIndex,
                                       path: path,
                                       fruitIndex: fruitIndex,
                                       defaults: defaults)
            waves.append(parsed.wave)
            untilTokens.append(parsed.untilToken)
            documentIndex += 1
        }

        // 3) until 后处理（T-027 修正版语义：只用原始 token + 原始 waveNo）
        let processedUntil = WaveInfo.postProcessUntil(rawTokens: untilTokens,
                                                       waveNos: waves.map { $0.waveNo })
        for i in waves.indices {
            waves[i].until = processedUntil[i]
        }

        return OriginalWaveList(waves: waves, defaults: defaults)
    }

        }

    private static func parseFruit(_ node: ContentNode,

    private static func parseWave(_ node: ContentNode,
                                  documentIndex: Int32,
                                  path: String,
                                  fruitIndex: [String: Int32],
                                  defaults: OriginalWaveDefaults) throws -> (wave: WaveInfo, untilToken: String) {
        var wave = WaveInfo()

        wave.waveNo = try int32Value(node, "waveNo", path) ?? documentIndex
        wave.chance = try int32Value(node, "chance", path)

        if let growth = defaults.waveChanceGrowth {
            wave.chanceRegrowth = growth
        }
        if let critical = defaults.criticalChance {
            wave.criticalChance = critical
        }

        let untilToken = node.attrs["until"] ?? ""

        for (index, child) in node.children.enumerated() {
            let childPath = "\(path)/\(child.tag)[\(index)]"
            switch child.tag.lowercased() {
            case "wave_dt":
                if let value = try floatValue(child, "dt", childPath) {
                    wave.dt = value
                }
                if let value = try floatValue(child, "inc", childPath) {
                    wave.dtInc = value
                }
                if let value = try floatValue(child, "spinc", childPath) {
                    wave.dtSpInc = value
                }
            case "nextwavedelay":
                if let value = try floatValue(child, "delay", childPath) {
                    wave.nextDelay = value
                }
                if let value = try floatValue(child, "inc", childPath) {
                    wave.nextDelayInc = value
                }
                if let value = try floatValue(child, "wait", childPath) {
                    wave.nextWait = value
                }
                if let value = try floatValue(child, "waitSpinc", childPath) {
                    wave.nextWaitSpInc = value
                }
                if let value = try floatValue(child, "speedLoss", childPath) {
                    wave.speedLoss = value
                }
                if let value = try boolValue(child, keys: ["waitForEntities"], path: childPath) {
                    wave.waitForEntities = value
                }
                if let value = try boolValue(child, keys: ["waitForProcessing"], path: childPath) {
                    wave.waitForProcessing = value
                }
            case "spawn":
                let spawner = try parseSpawn(child, path: childPath, fruitIndex: fruitIndex)
                wave.spawners.append(spawner)
            default:
                // 未知节点：不参与解析。
                continue
            }
        }

        return (wave, untilToken)
    }

    private static func parseSpawn(_ node: ContentNode,
                                   path: String,
                                   fruitIndex: [String: Int32]) throws -> Spawner {
        var spawner = Spawner()

        if let name = node.attrs["name"] {
            spawner.name = name
        }
        if let mode = try int32Value(node, "mode", path) {
            spawner.mode = mode
        }
        if let value = try floatValue(node, "min", path) {
            spawner.min = value
        }
        if let value = try floatValue(node, "max", path) {
            spawner.max = value
        }

        let minIncRaw = try floatValue(node, "mininc", path)
        let maxIncRaw = try floatValue(node, "maxinc", path)
        if let maxInc = maxIncRaw {
            // 原版 FUN_000537ec：maxinc 最后也写 +0x68，运行时值覆盖 mininc；
            // maxInc 保留原始 maxinc 值，便于后续调度器复现写槽顺序。
            spawner.minInc = maxInc
            spawner.maxInc = maxInc
        } else {
            // maxinc 缺失时 +0x68 保持 mininc 的值（不写第二次）。
            spawner.minInc = minIncRaw ?? 0
            spawner.maxInc = 0
        }

        if let value = try floatValue(node, "delay", path) {
            spawner.delay = value
        }
        if let value = try floatValue(node, "delayinc", path) {
            spawner.delayInc = value
        }
        if let value = try floatValue(node, "minWait", path) {
            spawner.minWait = value
        }
        if let value = try floatValue(node, "maxWait", path) {
            spawner.maxWait = value
        }
        if let value = try boolValue(node, keys: ["canSpawnWithPowers"], path: path) {
            spawner.canSpawnWithPowers = value
        }
        if let value = try floatValue(node, "dontSpawnBeforeWave", path) {
            spawner.dontSpawnBeforeWave = value
        }

        guard let rawType = node.attrs["type"] else {
            throw OriginalContentError.missingAttribute(node: node.tag, attribute: "type", path: path)
        }
        let tokens = rawType.split(separator: ",", omittingEmptySubsequences: false)
            .map { $0.trimmingCharacters(in: .whitespacesAndNewlines) }

        var kinds: [SpawnKind] = []
        kinds.reserveCapacity(tokens.count)
        for (index, token) in tokens.enumerated() {
            guard !token.isEmpty else {
                throw OriginalContentError.emptySpawnTypeToken(index: index, path: path)
            }
            switch token.lowercased() {
            case "bomb":
                kinds.append(.bomb)
            case "random":
                kinds.append(.random)
            default:
                guard let typeID = fruitIndex[token] else {
                    throw OriginalContentError.unknownSpawnType(raw: token, path: path)
                }
                kinds.append(.fruit(typeID))
            }
        }
        spawner.types = kinds

        return spawner
    }

    private static func int32Value(_ node: ContentNode,
                                   _ key: String,
                                   _ path: String) throws -> Int32? {
        guard let raw = node.attrs[key] else { return nil }
        let trimmed = raw.trimmingCharacters(in: .whitespacesAndNewlines)
        guard let value = Int32(trimmed) else {
            throw OriginalContentError.invalidNumber(node: node.tag,
                                                     attribute: key,
                                                     raw: raw,
                                                     path: path)
        }
        return value
    }

    private static func floatValue(_ node: ContentNode,
                                   _ key: String,
                                   _ path: String) throws -> Float? {
        guard let raw = node.attrs[key] else { return nil }
        let trimmed = raw.trimmingCharacters(in: .whitespacesAndNewlines)
        guard let value = Float(trimmed) else {
            throw OriginalContentError.invalidNumber(node: node.tag,
                                                     attribute: key,
                                                     raw: raw,
                                                     path: path)
        }
        return value
    }

    private static func boolValue(_ node: ContentNode,
                                  keys: [String],
                                  path: String) throws -> Bool? {
        for key in keys {
            guard let raw = node.attrs[key] else { continue }
            switch raw.trimmingCharacters(in: .whitespacesAndNewlines).lowercased() {
            case "true", "1":
                return true
            case "false", "0":
                return false
            default:
                throw OriginalContentError.invalidBoolean(node: node.tag,
                                                          attribute: key,
                                                          raw: raw,
                                                          path: path)
            }
        }
        return nil
    }

    /// 在仓库 Content/ 目录递归查找 root tag 匹配的 JSON（不依赖具体文件名）。
    private func findContentJSON(rootTag: String) -> Data? {
        let contentDirectory = repositoryRoot.appendingPathComponent("Content", isDirectory: true)
        let fileManager = FileManager.default
        var isDirectory: ObjCBool = false
        guard fileManager.fileExists(atPath: contentDirectory.path, isDirectory: &isDirectory),
              isDirectory.boolValue,
              let enumerator = fileManager.enumerator(at: contentDirectory,
                                                      includingPropertiesForKeys: [.isRegularFileKey]) else {
            return nil
        }

        let target = rootTag.lowercased()
        for case let url as URL in enumerator where url.pathExtension.lowercased() == "json" {
            guard let data = try? Data(contentsOf: url) else { continue }
            guard let node = try? JSONDecoder().decode(ContentNode.self, from: data) else { continue }
            if node.tag.lowercased() == target {
                return data
            }
        }
        return nil

### 认为 Kimi 审查时应注意的风险点

1. **T-026 JSON 实际结构未在本轮上下文中提供**：`ContentNode` 字段名（tag/attrs/text/tail/children）与 `attrs` 值必须全是字符串，这是我按任务卡文字与常见 XML→JSON 转换约定实现的。若 T-026 实际输出 `attrib`、数字型 attrs、或额外字段（如 `line`），`ContentNode` 解码会失败（`invalidJSON`）。**必须先跑真实测试确认。**
2. **真实文件名未知**：测试不依赖文件名，递归扫描 `Content/` 并靠 root tag 区分；但如果 T-026 把两个 XML 合并到一个 JSON 或放在子目录以外位置，测试会 `XCTUnwrap` 失败。
3. **「已知节点白名单 vs 未知节点忽略」取舍**：我选择「未知子节点忽略、不 throw」，理由是真实原版 XML 可能含非玩法节点（如 `<Misc>`），忽略能保证完整加载；但若 Kimi 认为未知节点也算「错误数据明确 throw」，需要改成 `unexpectedNode`。
4. **「必需 name/type」的语义解释**：我实现为「Fruit 必需 name，Spawn 必需 type」，未要求 Fruit 自身有 `type` 属性。若任务卡原意是 Fruit 必须有 `type`（如 "fruit"/"bomb"），需要补校验；目前 Fruit 的 `type` 属性会被忽略。
5. **defaults 键名**：只读取 `waveChanceGrowth` / `criticalChance`（按任务卡要求不写成 `waveChanceRegrowth`）。若真实 Defaults 节点使用其它键名（或没有 Defaults 节点），则 `chanceRegrowth` 保持 `Constants.WaveDefaults` 默认值，不会报错；这是「不擅自别名」的代价。
6. **maxinc 缺失分支**：当 XML 没有 `maxinc` 时，我按「第二次写 +0x68 不发生」处理（`minInc` 保持 `mininc`，`maxInc = 0`）。若原版实际是无条件写 0，需要改成始终覆盖。
7. **真实数值断言是任务卡给定值的直译**：`wave0.dt == 0.7`、`watermelon` 的 `min==max==1`、38 波次、重复 waveNo 存在——这些我无法在本机验证（Windows 无 Xcode，且 Content JSON 未在本轮上下文中出现），若真实转换结果与任务卡描述有出入，测试会失败而非实现错误，需按真实数据校准断言。
8. **`FruitInfo.chance` 的 XML 字面量**：我用严格 `Int32` 解析，若原版 chance 写成浮点字面量（如 `"100.0"`）会 throw；这是「严格解析」的预期行为，但可能与真实数据冲突，需确认。
