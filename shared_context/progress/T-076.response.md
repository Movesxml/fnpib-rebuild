import SwiftUI
import CoreGraphics

public enum ChallengeMapHitGeometry {
    public static func contains(point: CGPoint, quad: [CGPoint]) -> Bool {
        guard quad.count == 4 else { return false }
        guard point.x.isFinite, point.y.isFinite else { return false }
        guard quad.allSatisfy({ $0.x.isFinite && $0.y.isFinite }) else { return false }

        var twiceArea: CGFloat = 0
        for index in 0..<4 {
            let current = quad[index]
            let next = quad[(index + 1) % 4]
            twiceArea += current.x * next.y - next.x * current.y
        }
        guard twiceArea.isFinite, abs(twiceArea) > 1e-12 else { return false }

        var hasPositiveCross = false
        var hasNegativeCross = false

        for index in 0..<4 {
            let start = quad[index]
            let end = quad[(index + 1) % 4]
            let edgeX = end.x - start.x
            let edgeY = end.y - start.y
            let pointX = point.x - start.x
            let pointY = point.y - start.y
            let cross = edgeX * pointY - edgeY * pointX

            if cross > 0 {
                hasPositiveCross = true
            } else if cross < 0 {
                hasNegativeCross = true
            }

            if hasPositiveCross && hasNegativeCross {
                return false
            }
        }

        return true
    }
}

public struct ChallengeMapInteraction: View {
    public let map: ChallengeMapHUDState
    public let actions: UIActions

    public init(map: ChallengeMapHUDState, actions: UIActions) {
        self.map = map
        self.actions = actions
    }

    public var body: some View {
        GeometryReader { proxy in
            Color.clear
                .contentShape(Rectangle())
                .gesture(
                    SpatialTapGesture()
                        .onEnded { value in
                            handleTap(at: value.location, viewportSize: proxy.size)
                        }
                )
        }
    }

    private func handleTap(at point: CGPoint, viewportSize: CGSize) {
        for component in map.components.reversed() {
            guard let round = matchingRound(for: component) else { continue }

            let quad = ChallengeMapLayoutGeometry.positions(
                for: component,
                deviceWidth: viewportSize.width,
                deviceHeight: viewportSize.height
            )

            guard ChallengeMapHitGeometry.contains(point: point, quad: quad) else {
                continue
            }

            if map.canSelect && round.outcomeRawValue == nil {
                actions.send(.selectChallengeMapRound(round.id))
            }
            return
        }
    }

    private func matchingRound(for component: ChallengeMapLayoutComponent) -> ChallengeMapRoundHUDState? {
        map.rounds.first { round in
            component.name.hasSuffix(".spinner.adventureNode_\(round.id + 1)")
        }
    }
}

// App/ChallengeMapTextureOverlay.swift

import SwiftUI
import OSLog
import FNPBRender
import FNPBUI

@MainActor
struct ChallengeMapTextureOverlay: View {
    let metalContext: MetalContext
    let adapter: ChallengeMapQuadAdapter
    let components: [ChallengeMapLayoutComponent]

    @State private var quads: [OrderedTextureQuad] = []
    @State private var errorMessage: String?

    private let logger = Logger(
        subsystem: "com.fnpb.app",
        category: "ChallengeMapTextureOverlay"
    )

    var body: some View {
        GeometryReader { proxy in
            Group {
                if let errorMessage {
                    Text(errorMessage)
                        .font(.footnote)
                        .foregroundStyle(.red)
                        .multilineTextAlignment(.center)
                        .frame(width: proxy.size.width, height: proxy.size.height)
                } else {
                    OrderedQuadMetalView(
                        metalContext: metalContext,
                        quads: quads,
                        onError: { message in
                            logger.error(
                                "ChallengeMapTextureOverlay Metal error: \(message, privacy: .public)"
                            )
                            errorMessage = message
                        }
                    )
                    .frame(width: proxy.size.width, height: proxy.size.height)
                }
            }
            .onAppear {
                updateQuads(viewportSize: proxy.size)
            }
            .onChange(of: proxy.size) { _, newSize in
                updateQuads(viewportSize: newSize)
            }
        }
        .allowsHitTesting(false)
    }

    private func updateQuads(viewportSize: CGSize) {
        do {
            quads = try adapter.makeQuads(
                components: components,
                viewportSize: viewportSize
            )
            errorMessage = nil
        } catch {
            quads = []
            errorMessage = error.localizedDescription
            logger.error(
                "ChallengeMapTextureOverlay quad creation failed: \(error.localizedDescription, privacy: .public)"
            )
        }
    }
}

// Tests/FNPBUITests/ChallengeMapInteractionTests.swift

import XCTest
import CoreGraphics
@testable import FNPBUI

final class ChallengeMapInteractionTests: XCTestCase {
    func testRectangleInteriorBoundaryAndExterior() {
        let quad = [
            CGPoint(x: 0, y: 0),
            CGPoint(x: 10, y: 0),
            CGPoint(x: 10, y: 10),
            CGPoint(x: 0, y: 10),
        ]

        XCTAssertTrue(ChallengeMapHitGeometry.contains(point: CGPoint(x: 5, y: 5), quad: quad))
        XCTAssertTrue(ChallengeMapHitGeometry.contains(point: CGPoint(x: 0, y: 0), quad: quad))
        XCTAssertTrue(ChallengeMapHitGeometry.contains(point: CGPoint(x: 10, y: 5), quad: quad))
        XCTAssertTrue(ChallengeMapHitGeometry.contains(point: CGPoint(x: 5, y: 10), quad: quad))
        XCTAssertTrue(ChallengeMapHitGeometry.contains(point: CGPoint(x: 0, y: 10), quad: quad))

        XCTAssertFalse(ChallengeMapHitGeometry.contains(point: CGPoint(x: 11, y: 5), quad: quad))
        XCTAssertFalse(ChallengeMapHitGeometry.contains(point: CGPoint(x: 5, y: -1), quad: quad))
        XCTAssertFalse(ChallengeMapHitGeometry.contains(point: CGPoint(x: -1, y: 5), quad: quad))
        XCTAssertFalse(ChallengeMapHitGeometry.contains(point: CGPoint(x: 5, y: 11), quad: quad))
    }

    func testRotatedQuadInteriorAndExterior() {
        let quad = [
            CGPoint(x: 0, y: -10),
            CGPoint(x: 10, y: 0),
            CGPoint(x: 0, y: 10),
            CGPoint(x: -10, y: 0),
        ]

        XCTAssertTrue(ChallengeMapHitGeometry.contains(point: CGPoint(x: 0, y: 0), quad: quad))
        XCTAssertTrue(ChallengeMapHitGeometry.contains(point: CGPoint(x: 2, y: -2), quad: quad))
        XCTAssertTrue(ChallengeMapHitGeometry.contains(point: CGPoint(x: 0, y: 10), quad: quad))
        XCTAssertTrue(ChallengeMapHitGeometry.contains(point: CGPoint(x: 10, y: 0), quad: quad))

        XCTAssertFalse(ChallengeMapHitGeometry.contains(point: CGPoint(x: 0, y: 11), quad: quad))
        XCTAssertFalse(ChallengeMapHitGeometry.contains(point: CGPoint(x: 8, y: 8), quad: quad))
        XCTAssertFalse(ChallengeMapHitGeometry.contains(point: CGPoint(x: -8, y: -8), quad: quad))
    }

    func testNegativeScaleQuadInteriorAndExterior() {
        let quad = [
            CGPoint(x: 10, y: 10),
            CGPoint(x: 10, y: 0),
            CGPoint(x: 0, y: 0),
            CGPoint(x: 0, y: 10),
        ]

        XCTAssertTrue(ChallengeMapHitGeometry.contains(point: CGPoint(x: 5, y: 5), quad: quad))
        XCTAssertTrue(ChallengeMapHitGeometry.contains(point: CGPoint(x: 0, y: 0), quad: quad))
        XCTAssertTrue(ChallengeMapHitGeometry.contains(point: CGPoint(x: 10, y: 10), quad: quad))

        XCTAssertFalse(ChallengeMapHitGeometry.contains(point: CGPoint(x: -1, y: 5), quad: quad))
        XCTAssertFalse(ChallengeMapHitGeometry.contains(point: CGPoint(x: 5, y: 11), quad: quad))
    }

    func testNonFinitePointReturnsFalse() {
        let quad = [
            CGPoint(x: 0, y: 0),
            CGPoint(x: 10, y: 0),
            CGPoint(x: 10, y: 10),
            CGPoint(x: 0, y: 10),
        ]

        XCTAssertFalse(
            ChallengeMapHitGeometry.contains(
                point: CGPoint(x: CGFloat.nan, y: 5),
                quad: quad
            )
        )
        XCTAssertFalse(
            ChallengeMapHitGeometry.contains(
                point: CGPoint(x: 5, y: CGFloat.infinity),
                quad: quad
            )
        )
        XCTAssertFalse(
            ChallengeMapHitGeometry.contains(
                point: CGPoint(x: -CGFloat.infinity, y: 5),
                quad: quad
            )
        )
    }

    func testNonFiniteVerticesReturnFalse() {
        let nanQuad = [
            CGPoint(x: 0, y: 0),
            CGPoint(x: 10, y: 0),
            CGPoint(x: 10, y: CGFloat.nan),
            CGPoint(x: 0, y: 10),
        ]
        let infiniteQuad = [
            CGPoint(x: 0, y: 0),
            CGPoint(x: CGFloat.infinity, y: 0),
            CGPoint(x: 10, y: 10),
            CGPoint(x: 0, y: 10),
        ]

        XCTAssertFalse(
            ChallengeMapHitGeometry.contains(point: CGPoint(x: 5, y: 5), quad: nanQuad)
        )
        XCTAssertFalse(
            ChallengeMapHitGeometry.contains(point: CGPoint(x: 5, y: 5), quad: infiniteQuad)
        )
    }

    func testInvalidVertexCountReturnsFalse() {
        let threePoints = [
            CGPoint(x: 0, y: 0),
            CGPoint(x: 10, y: 0),
            CGPoint(x: 0, y: 10),
        ]
        let fivePoints = [
            CGPoint(x: 0, y: 0),
            CGPoint(x: 10, y: 0),
            CGPoint(x: 10, y: 10),
            CGPoint(x: 0, y: 10),
            CGPoint(x: 5, y: 5),
        ]

        XCTAssertFalse(
            ChallengeMapHitGeometry.contains(point: CGPoint(x: 1, y: 1), quad: threePoints)
        )
        XCTAssertFalse(
            ChallengeMapHitGeometry.contains(point: CGPoint(x: 5, y: 5), quad: fivePoints)
        )
    }

    func testDegenerateQuadReturnsFalse() {
        let collinearQuad = [
            CGPoint(x: 0, y: 0),
            CGPoint(x: 1, y: 0),
            CGPoint(x: 2, y: 0),
            CGPoint(x: 3, y: 0),
        ]
        let duplicatedQuad = [
            CGPoint(x: 0, y: 0),
            CGPoint(x: 0, y: 0),
            CGPoint(x: 0, y: 0),
            CGPoint(x: 0, y: 0),
        ]

        XCTAssertFalse(
            ChallengeMapHitGeometry.contains(point: CGPoint(x: 1, y: 0), quad: collinearQuad)
        )
        XCTAssertFalse(
            ChallengeMapHitGeometry.contains(point: CGPoint(x: 0, y: 0), quad: duplicatedQuad)
        )
    }
}
