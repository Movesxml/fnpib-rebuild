#!/usr/bin/env python3
# -*- coding: utf-8 -*-
"""T-026 Halfbrick bxml 二进制 XML 解码器。

格式实证：Kimi 对 assets_raw/xml/*.xml 的字节级分析（见 DECISIONS / BXML_FORMAT.md）。
本模块只实现已实证的两种节点 marker：

  * 0x01 = 元素节点
  * 0x04 = 文本叶节点

未知 marker 一律报错，不做猜测性扩展。

对外 API::

    is_bxml(data) -> bool
    decode_bxml(data, *, max_depth=64, max_nodes=200_000,
                max_attrs=65_535) -> xml.etree.ElementTree.Element
    BxmlError

严格性
------
* 头、字符串表、字节码全部做越界检查；
* 字符串 offset 必须 < 字符串表大小，且必须能在表内找到 NUL；
  offset 允许落在某个字符串内部（后缀复用），读取到下一个 NUL 为止；
* 只接受单根文档（rootCount == 1），顶层节点必须是元素；
* 解析结束后必须恰好消费到文件末尾，否则报“尾部有未消费字节”；
* 递归深度、节点总数、单元素属性数均有上限，超限抛 BxmlError。

依赖：仅标准库。
"""

from __future__ import annotations

import struct
import xml.etree.ElementTree as ET

MAGIC = b"bxml"
HEADER_SIZE = 12

MARKER_ELEMENT = 0x01
MARKER_TEXT = 0x04

DEFAULT_MAX_DEPTH = 64
DEFAULT_MAX_NODES = 200_000
DEFAULT_MAX_ATTRS = 65_535


class BxmlError(Exception):
    """bxml 结构/边界错误。offset 为出错位置的字节偏移（有则带上）。"""

    def __init__(self, message: str, offset: int | None = None):
        self.offset = offset
        if offset is not None:
            message = f"{message}（字节偏移 {offset}）"
        super().__init__(message)


def is_bxml(data: bytes) -> bool:
    """仅按 magic 判断，不验证长度/内容。"""
    return len(data) >= 4 and data[:4] == MAGIC


class _Decoder:
    def __init__(
        self,
        data: bytes,
        *,
        max_depth: int,
        max_nodes: int,
        max_attrs: int,
    ):
        if not isinstance(data, (bytes, bytearray)):
            raise TypeError(f"bxml 数据必须是 bytes/bytearray，实得 {type(data)!r}")

        self.data = bytes(data)
        self.max_depth = max_depth
        self.max_nodes = max_nodes
        self.max_attrs = max_attrs
        self._node_count = 0

        n = len(self.data)
        if n < HEADER_SIZE:
            raise BxmlError(
                f"文件长度 {n} 小于 bxml 头长度 {HEADER_SIZE}"
            )
        if self.data[:4] != MAGIC:
            raise BxmlError(
                f"magic 不是 bxml（实得 {self.data[:4]!r}）"
            )

        self.header_attr_count = struct.unpack_from("<H", self.data, 4)[0]
        self.header_unknown = struct.unpack_from("<H", self.data, 6)[0]
        self.string_table_size = struct.unpack_from("<I", self.data, 8)[0]

        self.string_table_start = HEADER_SIZE
        self.string_table_end = self.string_table_start + self.string_table_size
        if self.string_table_end > n:
            raise BxmlError(
                f"字符串表越界：12 + {self.string_table_size} = "
                f"{self.string_table_end} > 文件长度 {n}",
                HEADER_SIZE,
            )

        self.cursor = self.string_table_end
        if self.cursor + 4 > n:
            raise BxmlError(
                "字符串表之后缺少 rootCount（u32）",
                self.cursor,
            )
        self.root_count = self._read_u32()

    # ------------------------------------------------------------------ #
    # 基础读取
    # ------------------------------------------------------------------ #

    def _require(self, size: int) -> None:
        if self.cursor + size > len(self.data):
            raise BxmlError(
                f"字节码越界：需要 {size} 字节，剩余 "
                f"{len(self.data) - self.cursor} 字节",
                self.cursor,
            )

    def _read_u8(self) -> int:
        self._require(1)
        value = self.data[self.cursor]
        self.cursor += 1
        return value

    def _read_u32(self) -> int:
        self._require(4)
        value = struct.unpack_from("<I", self.data, self.cursor)[0]
        self.cursor += 4
        return value

    def _read_string(self, offset: int, what: str) -> str:
        if offset >= self.string_table_size:
            raise BxmlError(
                f"{what} 的字符串 offset={offset} 越界"
                f"（字符串表大小 {self.string_table_size}）"
            )

        start = self.string_table_start + offset
        end = self.data.find(b"\x00", start, self.string_table_end)
        if end < 0:
            raise BxmlError(
                f"{what} 的字符串 offset={offset} 在字符串表内没有 NUL 终止符"
            )

        raw = self.data[start:end]
        try:
            return raw.decode("utf-8")
        except UnicodeDecodeError as exc:
            raise BxmlError(
                f"{what} 的字符串 offset={offset} 不是合法 UTF-8：{exc}"
            ) from exc

    # ------------------------------------------------------------------ #
    # 节点解析
    # ------------------------------------------------------------------ #

    def _parse_node(self, depth: int) -> tuple[int, ET.Element | str]:
        if depth > self.max_depth:
            raise BxmlError(
                f"递归深度 {depth} 超过上限 {self.max_depth}（疑似环/恶意数据）",
                self.cursor,
            )

        self._node_count += 1
        if self._node_count > self.max_nodes:
            raise BxmlError(
                f"节点总数超过上限 {self.max_nodes}",
                self.cursor,
            )

        marker_offset = self.cursor
        marker = self._read_u8()

        if marker == MARKER_ELEMENT:
            return MARKER_ELEMENT, self._parse_element_body(depth, marker_offset)

        if marker == MARKER_TEXT:
            text_offset = self._read_u32()
            return MARKER_TEXT, self._read_string(text_offset, "文本叶")

        raise BxmlError(
            f"未知节点 marker=0x{marker:02x}"
            "（仅实证 0x01=元素、0x04=文本叶；不猜测其他 marker）",
            marker_offset,
        )

    def _parse_element_body(self, depth: int, marker_offset: int) -> ET.Element:
        name_offset = self._read_u32()
        tag = self._read_string(name_offset, "元素名")
        if not tag:
            raise BxmlError("元素名为空字符串", marker_offset)

        attr_count = self._read_u32()
        if attr_count > self.max_attrs:
            raise BxmlError(
                f"元素属性数 {attr_count} 超过上限 {self.max_attrs}",
                marker_offset,
            )

        attrib: dict[str, str] = {}
        for index in range(attr_count):
            attr_name_offset = self._read_u32()
            attr_value_offset = self._read_u32()
            attr_name = self._read_string(attr_name_offset, f"属性名[{index}]")
            attr_value = self._read_string(attr_value_offset, f"属性值[{index}]")
            if not attr_name:
                raise BxmlError(f"属性名[{index}] 为空字符串", marker_offset)
            attrib[attr_name] = attr_value

        child_count = self._read_u32()
        element = ET.Element(tag, attrib)

        # 按 ElementTree 语义合并文本叶：
        #   所有元素子节点之前 → element.text
        #   某个元素子节点之后 → 该子节点.tail
        # 连续文本叶直接拼接。
        last_child: ET.Element | None = None
        for _ in range(child_count):
            kind, value = self._parse_node(depth + 1)
            if kind == MARKER_TEXT:
                text = value  # type: ignore[assignment]
                if last_child is None:
                    element.text = (element.text or "") + text
                else:
                    last_child.tail = (last_child.tail or "") + text
            else:
                child = value  # type: ignore[assignment]
                element.append(child)
                last_child = child

        return element

    # ------------------------------------------------------------------ #
    # 入口
    # ------------------------------------------------------------------ #

    def decode(self) -> ET.Element:
        n = len(self.data)

        if self.root_count == 0:
            raise BxmlError(
                "rootCount=0：文档没有根节点",
                self.string_table_end,
            )
        if self.root_count != 1:
            raise BxmlError(
                f"rootCount={self.root_count}：bxml_decode 只支持单根文档"
                "（多根语义未实证，明确拒绝而不是猜测）",
                self.string_table_end,
            )

        kind, value = self._parse_node(0)
        if kind != MARKER_ELEMENT:
            raise BxmlError(
                "顶层节点必须是元素（marker=1），实际是文本叶（marker=4）",
                self.string_table_end,
            )

        if self.cursor != n:
            raise BxmlError(
                f"尾部有 {n - self.cursor} 字节未消费",
                self.cursor,
            )

        return value  # type: ignore[return-value]


def decode_bxml(
    data: bytes,
    *,
    max_depth: int = DEFAULT_MAX_DEPTH,
    max_nodes: int = DEFAULT_MAX_NODES,
    max_attrs: int = DEFAULT_MAX_ATTRS,
) -> ET.Element:
    """把 bxml 字节解码为 xml.etree.ElementTree 元素。

    失败抛 BxmlError（带出错字节偏移）；成功时保证完整消费输入。
    """
    return _Decoder(
        data,
        max_depth=max_depth,
        max_nodes=max_nodes,
        max_attrs=max_attrs,
    ).decode()


__all__ = [
    "BxmlError",
    "DEFAULT_MAX_ATTRS",
    "DEFAULT_MAX_DEPTH",
    "DEFAULT_MAX_NODES",
    "MARKER_ELEMENT",
    "MARKER_TEXT",
    "decode_bxml",
    "is_bxml",
]

def parse_xml(data: bytes) -> tuple[ET.Element | None, str | None, str | None]:
    """返回 (root, error, encoding_tag)。

    encoding_tag:
      * "as-declared"      —— 文本 XML，按声明编码解析
      * "gb18030-fallback" —— 文本 XML，gb18030 兜底
      * "bxml"             —— Halfbrick bxml 二进制 XML（T-026）
    """
    if is_bxml(data):
        try:
            return decode_bxml(data), None, "bxml"
        except BxmlError as exc:
            return None, f"BxmlError: {exc}", None

    try:
        return ET.fromstring(data), None, "as-declared"
    except ET.ParseError as first_err:
        try:
            text = data.decode("gb18030")
        except UnicodeDecodeError as dec_err:
            return None, (f"ParseError(as-declared): {first_err} | "
                          f"gb18030 decode failed: {dec_err}"), None
        try:
            root = ET.fromstring(text)
            return root, None, "gb18030-fallback"
        except ET.ParseError as second_err:
            return None, (f"ParseError(as-declared): {first_err} | "
                          f"ParseError(gb18030): {second_err}"), None


def elem_to_obj(el: ET.Element, depth_limit: int | None, depth: int = 0) -> dict:
    obj: dict = {"tag": el.tag}
    if el.attrib:
        obj["attrs"] = dict(el.attrib)

    text = (el.text or "").strip()
    if text:
        obj["text"] = text

    tail = (el.tail or "").strip()
    if tail:
        obj["tail"] = tail

    kids = list(el)
    if kids:
        if depth_limit is not None and depth >= depth_limit:
            obj["children_omitted"] = len(kids)
        else:
            obj["children"] = [elem_to_obj(k, depth_limit, depth + 1) for k in kids]
    return obj


def walk_stats(root: ET.Element) -> tuple[int, int, set[str]]:
    nodes = 0
    max_depth = 0
    attrs: set[str] = set()
    stack: list[tuple[ET.Element, int]] = [(root, 0)]
    while stack:
        node, depth = stack.pop()
        nodes += 1
        max_depth = max(max_depth, depth)
        attrs.update(node.attrib.keys())
        stack.extend((child, depth + 1) for child in node)
    return nodes, max_depth, attrs

    try:
        return path.relative_to(in_path)

def collect_xml(in_path: Path) -> list[Path]:
    if in_path.is_file():
        return [in_path]
    files = [p for p in in_path.rglob("*") if p.is_file() and p.suffix.lower() == ".xml"]
    files.sort()
    return files


def convert_one(src: Path, in_path: Path, out_root: Path, *,
                depth_limit: int | None, indent: int, force: bool) -> dict:
    rel = _rel_of(src, in_path)
    out_rel = rel.with_suffix(".json")
    entry: dict = {
        "source": rel.as_posix(),
        "output": out_rel.as_posix(),
    }

    try:
        data = src.read_bytes()
    except OSError as exc:
        entry.update({"status": "error", "error": f"read: {exc}"})
        return entry
    entry["bytes"] = len(data)

    root, err, encoding = parse_xml(data)
    if root is None:
        entry.update({"status": "error", "error": err})
        return entry
    entry["encoding"] = encoding
    if encoding == "gb18030-fallback":
        entry["warning"] = "encoding_fallback"

    nodes, max_depth, attrs = walk_stats(root)
    entry.update({
        "root_tag": root.tag,
        "nodes": nodes,
        "max_depth": max_depth,
        "attrs_sample": sorted(attrs)[:MAX_PREVIEW_ATTRS],
    })
    if root.tag.startswith("{"):
        entry["warning"] = (entry.get("warning", "") + " namespaced_tag").strip()

    doc = {
        "source": rel.as_posix(),
        "bytes": len(data),
        "encoding": encoding,
        "root": elem_to_obj(root, depth_limit),
    }

    dst = out_root / out_rel
    if dst.exists() and not force:
        entry.update({"status": "skipped", "reason": "exists"})
        return entry

    try:
        dst.parent.mkdir(parents=True, exist_ok=True)
        dst.write_text(json.dumps(doc, ensure_ascii=False, indent=indent),
                       encoding="utf-8")
    except OSError as exc:
        entry.update({"status": "error", "error": f"write: {exc}"})
        return entry

    entry["status"] = "ok"
    entry["json_bytes"] = dst.stat().st_size
    return entry


def convert_tree(in_path: Path, out_root: Path, *,

def _bxml_elem(name_offset: int, attrs: list[tuple[int, int]],
               children: list[bytes]) -> bytes:
    out = bytearray()
    out.append(0x01)
    out += struct.pack("<I", name_offset)
    out += struct.pack("<I", len(attrs))
    for attr_name_offset, attr_value_offset in attrs:
        out += struct.pack("<II", attr_name_offset, attr_value_offset)
    out += struct.pack("<I", len(children))
    for child in children:
        out += child
    return bytes(out)


def _bxml_text(text_offset: int) -> bytes:
    return b"\x04" + struct.pack("<I", text_offset)


def _bxml_sample() -> tuple[bytes, dict[str, int], int]:
    """合成一份覆盖 属性/文本叶/后缀引用 的 bxml。

    结构：
      <fruitInfoFile version="1.0.4">
        <fruitInfo name="watermelon">12</fruitInfo>
        <InfoFile>hello<note/>tail</InfoFile>   # InfoFile 是 "fruitInfoFile" 的后缀引用
      </fruitInfoFile>
    """
    strings = [
        "fruitInfoFile",
        "version",
        "1.0.4",
        "fruitInfo",
        "name",
        "watermelon",
        "note",
        "hello",
        "tail",
        "12",
    ]
    strtab = bytearray()
    offsets: dict[str, int] = {}
    for text in strings:
        offsets[text] = len(strtab)
        strtab += text.encode("utf-8") + b"\x00"

    # 后缀复用：offset 指向 "fruitInfoFile" 内部的 "InfoFile"
    suffix_offset = offsets["fruitInfoFile"] + len("fruit")

    fruit_info = _bxml_elem(
        offsets["fruitInfo"],
        [(offsets["name"], offsets["watermelon"])],
        [_bxml_text(offsets["12"])],
    )
    info_file = _bxml_elem(
        suffix_offset,
        [],
        [
            _bxml_text(offsets["hello"]),
            _bxml_elem(offsets["note"], [], []),
            _bxml_text(offsets["tail"]),
        ],
    )
    root = _bxml_elem(
        offsets["fruitInfoFile"],
        [(offsets["version"], offsets["1.0.4"])],
        [fruit_info, info_file],
    )

    data = (
        b"bxml"
        + struct.pack("<HHI", 1, 0, len(strtab))
        + bytes(strtab)
        + struct.pack("<I", 1)  # rootCount
        + root
    )
    return data, offsets, len(strtab)


def _selftest_bxml(check, tmp: Path) -> None:

    check(root[0].text == "12",
          f"bxml 首个文本叶 → 父 text（got {root[0].text!r}）")
    check(root[1].text == "hello",
          f"bxml 文本叶 → 父 text（got {root[1].text!r}）")
    check(root[1][0].tag == "note", "bxml 元素子节点")
    check(root[1][0].tail == "tail",
          f"bxml 尾随文本叶 → 前兄弟 tail（got {root[1][0].tail!r}）")
    check(root[1][0].text is None, "bxml note 无前导文本")

    check(root2 is not None and err2 is None and encoding2 == "bxml",
          f"parse_xml 识别 bxml（encoding={encoding2!r}, err={err2!r}）")

    def expect_error(label: str, bad: bytes, **kwargs) -> None:

            decode_bxml(bad, **kwargs)

        except Exception as exc:  # noqa: BLE001 - selftest 需暴露非预期异常

        else:
            check(False, f"{label} 未被拒绝")

    bad = bytearray(data)
    struct.pack_into("<I", bad, body_start + 4 + 1, 0xFFFFFFFF)
    expect_error("非法元素名 offset", bytes(bad))

    bad[body_start + 4] = 0x02
    expect_error("未知 marker=0x02", bytes(bad))

    bad = bytearray(data)
    struct.pack_into("<I", bad, body_start, 0)
    expect_error("rootCount=0", bytes(bad))

    bad = bytearray(data)
    struct.pack_into("<I", bad, body_start, 2)
    expect_error("rootCount=2（多根）", bytes(bad))

    expect_error("尾部多余字节", data + b"\x00")

    bad = bytearray(data)
    struct.pack_into("<I", bad, 8, len(data))
    expect_error("字符串表越界", bytes(bad))

    bad = bytearray(data)
    bad[12 + offsets["hello"]] = 0xFF
    expect_error("非法 UTF-8", bytes(bad))

    expect_error("递归深度上限", data, max_depth=1)
    expect_error("节点总数上限", data, max_nodes=2)

    # ---- 3. 文件级：bxml 内容 + .xml 后缀，走完整转换管线 ----
    raw_bx = tmp / "RawBxml"

    check(len(entries) == 1 and entries[0]["status"] == "ok",
          f"bxml 文件转换成功（entries={entries!r}）")

    if entries and entries[0]["status"] == "ok":
        check(entries[0]["encoding"] == "bxml",
              f"index 中 encoding=bxml（got {entries[0].get('encoding')!r}）")

        json_path = out_bx / "xml" / "fruitList.json"
        doc = json.loads(json_path.read_text(encoding="utf-8"))
        root_obj = doc["root"]
        check(root_obj["tag"] == "fruitInfoFile", "bxml JSON 根 tag")
        check(root_obj["children"][1]["tag"] == "InfoFile",
              "bxml JSON 后缀引用 tag")
        check(root_obj["children"][1]["text"] == "hello",
              "bxml JSON text 字段")
        check(root_obj["children"][1]["children"][0]["tail"] == "tail",
              "bxml JSON tail 字段")

    check(summary["encodings"].get("bxml") == 1,
          f"summary 编码统计（got {summary['encodings']!r}）")

```markdown
# Halfbrick bxml 格式说明（T-026 实证）

状态：对 `settings.xml`、`originalModeInfo.xml`、`fruitList.xml`、
`originalWaveList.xml` 做过字节级消费验证（Kimi 执行）；本文件描述
`tools/AssetPipeline/bxml_decode.py` 实际实现并验证过的子集。

> 真实资产 `assets_raw/xml/*.xml` 是 bxml 二进制容器，不是文本 XML。
> 判断方式：文件前 4 字节为 ASCII `bxml`。

## 1. 总体布局

| 偏移 | 类型 | 字段 | 说明 |
|---:|---|---|---|
| 0 | char[4] | magic | 固定 `bxml` |
| 4 | u16 LE | `headerAttrCount` | 语义未实证；当前解码器不读取、不依赖 |
| 6 | u16 LE | `unknown` | 语义未实证；不参与消费 |
| 8 | u32 LE | `stringTableByteSize` | 字符串表字节数 |
| 12 | bytes[size] | 字符串表 | NUL 结尾的 UTF-8 字符串拼接 |
| 12+size | u32 LE | `rootCount` | 根节点数；已实证样本为 1 |
| 12+size+4 | nodes | 字节码 | 根节点开始，递归结构见下 |

## 2. 字符串表

- 每条字符串以 `0x00` 结尾，内容为 UTF-8。
- 引用使用 **u32 LE 偏移**，相对字符串表起点（偏移 0 指向表内第一个字节）。
- **后缀复用**：偏移允许落在另一条字符串的内部。读取规则是「从 offset 开始，
  取到下一个 `0x00` 为止」，不要求 offset 恰好是某条字符串的起点。
  例：表内 `fruitInfoFile\0`，offset=5 会解析为 `InfoFile`。
- 解码器要求 `offset < stringTableByteSize`，且表内必须存在终止 NUL；
  不满足即 `BxmlError`。
- UTF-8 按 strict 解码，非法字节立即报错（不静默替换）。

## 3. 节点文法

```
node       := element | text_leaf

element    := 0x01
              u32 nameOffset
              u32 attrCount
              attrCount * (u32 attrNameOffset, u32 attrValueOffset)
              u32 childCount
              childCount * node

text_leaf  := 0x04
              u32 textOffset
```

- `0x01` 元素：名字、属性表、按顺序排列的子节点。
- `0x04` 文本叶：**只有 5 字节**（marker + u32），后面不再跟元素体。
- childCount 包含文本叶；文本叶和元素可以在同一层任意交错。

## 4. rootCount 与根节点

- `rootCount` 是顶层节点数量。
- 已实证样本 `rootCount == 1`，且顶层节点是元素（marker=1）。
- 当前解码器只接受单根文档：
  - `rootCount == 0` → `BxmlError`
  - `rootCount > 1` → `BxmlError`（多根语义未实证，拒绝猜测）
  - 顶层是文本叶 → `BxmlError`
- 单根文档解析结束后，游标必须恰好等于文件长度；否则报「尾部有未消费字节」。
  这样可以捕获结构理解偏差，而不是静默忽略尾部数据。

## 5. 文本叶与 ElementTree 语义

解码器在构建元素时按 ET 规则合并文本叶：

1. 文本叶出现在所有元素子节点之前 → 设为父元素的 `.text`；
2. 文本叶出现在某个元素子节点之后 → 设为该子元素的 `.tail`；
3. 连续多个文本叶直接拼接（ET 无法区分相邻文本分段）。

`xml_convert.py` 的 `elem_to_obj` 会把非空 `.text` / `.tail` 分别输出为
JSON 的 `"text"` / `"tail"` 字段（纯空白丢弃）。

## 6. 严格性与限制

实现中明确拒绝而不猜测的情况：

- magic 不是 `bxml`；
- 文件短于 12 字节头；
- 字符串表越界；
- 字符串 offset 越界或表内无 NUL；
- 字符串不是合法 UTF-8；
- 未知节点 marker（除 `0x01` / `0x04` 外全部报错）；
- 元素名/属性名为空；
- `rootCount != 1`；
- 顶层节点是文本叶；
- 解析结束后仍有未消费字节；
- 递归深度、节点总数、单元素属性数超过上限。

| 参数 | 默认值 |
|---|---:|
| `max_depth` | 64 |
| `max_nodes` | 200,000 |
| `max_attrs` | 65,535 |

## Kimi 审查风险点

1. **`headerAttrCount` / `unknown` 未参与消费**：任务卡给的头字段语义未实证，我按「不依赖」处理。若 60 个文件中有文件根属性数与头字段强相关，可能出现「尾部未消费」或过早读子节点。实跑第一优先看这条。

4. **严格完整消费**：字符串表若在真实文件中带尾部填充字节，会触发「尾部有未消费字节」。任务卡声称 4 个样本已精确消费，但 60 个文件可能有个别差异。

6. **`ET.Element` 不校验 XML 名称合法性**：bxml 标签/属性名直接来自字符串表，非法命名不会在转换阶段报错，会进入 JSON。若下游有 XML 语法依赖需注意。
7. **导入方式**：`xml_convert.py` 先相对导入、失败再绝对导入 `bxml_decode`。脚本直跑和包导入两种方式都覆盖，但若部署时只拷 `xml_convert.py` 会直接 ImportError（本任务要求新增文件，属预期）。
