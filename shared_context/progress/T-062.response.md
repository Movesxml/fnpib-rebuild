def property_category(name: str) -> str:
    category = PROPERTY_CATEGORIES.get(name)
    if category is not None:
        return category
    if name.endswith("Colour") or name.endswith("Color"):
        return "vec4"
    raise FormatError(f"unknown static property name {name!r}")


def parse_property_record(reader: Reader, offset: int) -> tuple[dict[str, Any], int]:
    record_offset = offset
    name, cursor = reader.string(offset)
    category = property_category(name)

    reader.require(cursor, PROP_RECORD_HEADER)
    cursor += 4
    reader.require(cursor, PROP_RECORD_HEADER_2)
    cursor += 4

    raw_type: int | None = None
    type_info: int | None = None
    string_flag: int | None = None
    trailer: int | None = None
    raw_value: int | None = None

    if category in ("vec2", "vec3", "vec4"):
        expected_type = {"vec2": 2, "vec3": 1, "vec4": VEC4_TYPE}[category]
        dimension = {"vec2": 2, "vec3": 3, "vec4": 4}[category]

        raw_type = reader.u32(cursor)
        cursor += 4
        if raw_type != expected_type:
            raise FormatError(
                f"{name!r}: expected {category} type {expected_type}, "
                f"got {raw_type} at 0x{record_offset:x}"
            )

        type_info = reader.u32(cursor)
        cursor += 4

        one = reader.u32(cursor)
        cursor += 4
        zero = reader.u32(cursor)
        cursor += 4
        if one != 1 or zero != 0:
            raise FormatError(f"{name!r}: invalid vec header at 0x{record_offset:x}")

        value = [reader.f32(cursor + 4 * index) for index in range(dimension)]
        cursor += 4 * dimension

        trailer = reader.u32(cursor)
        cursor += 4
        if trailer != 0:
            raise FormatError(f"{name!r}: invalid vec trailer at 0x{record_offset:x}")

    elif category in ("f32", "u32"):
        raw_type = reader.u32(cursor)
        cursor += 4
        string_flag = reader.u32(cursor)
        cursor += 4
        one = reader.u32(cursor)
        cursor += 4
        zero = reader.u32(cursor)
        cursor += 4
        if raw_type != 0 or string_flag != 0 or one != 1 or zero != 0:
            raise FormatError(f"{name!r}: invalid scalar header at 0x{record_offset:x}")

        raw_value = reader.u32(cursor)
        value = reader.f32(cursor) if category == "f32" else raw_value
        cursor += 4

        trailer = reader.u32(cursor)
        cursor += 4
        if trailer != 0:
            raise FormatError(f"{name!r}: invalid scalar trailer at 0x{record_offset:x}")

    elif category == "bool":
        raw_type = reader.u32(cursor)
        cursor += 4
        string_flag = reader.u32(cursor)
        cursor += 4
        zero = reader.u32(cursor)
        cursor += 4
        raw_value = reader.u32(cursor)
        cursor += 4
        trailer = reader.u32(cursor)
        cursor += 4
        if raw_type != 0 or string_flag != 1 or zero != 0 or raw_value not in (0, 1) or trailer != 0:
            raise FormatError(f"{name!r}: invalid bool record at 0x{record_offset:x}")
        value = bool(raw_value)

    elif category == "string":
        raw_type = reader.u32(cursor)
        cursor += 4
        string_flag = reader.u32(cursor)
        cursor += 4
        zero = reader.u32(cursor)
        cursor += 4
        if raw_type != 0 or string_flag != 1 or zero != 0:
            raise FormatError(f"{name!r}: invalid string header at 0x{record_offset:x}")

        value, cursor = reader.string(cursor)

        trailer = reader.u32(cursor)
        cursor += 4
        if trailer != 0:
            raise FormatError(f"{name!r}: invalid string trailer at 0x{record_offset:x}")

    else:
        raise RuntimeError(f"unhandled property category {category!r}")

    return {
        "name": name,
        "category": category,
        "offset": record_offset,
        "end_offset": cursor,
        "type_info": type_info,
        "raw_type": raw_type,
        "string_flag": string_flag,
        "trailer": trailer,
        "raw_value": raw_value,
        "value": value,
    }, cursor


def parse_static_properties(
    reader: Reader,
    start: int,
    limit: int,
) -> tuple[dict[str, Any], list[dict[str, Any]], list[dict[str, Any]], int]:
    cursor = start
    properties: dict[str, Any] = {}
    records: list[dict[str, Any]] = []
    groups: list[dict[str, Any]] = []
    seen: dict[str, int] = {}

    while True:
        if cursor + 4 > limit:
            raise FormatError(f"static property area has no UIAM before 0x{limit:x}")

        if reader.data[cursor:cursor + 4] == b"UIAM":
            return properties, records, groups, cursor

        if reader.data[cursor:cursor + 4] != PROP_GROUP_HEADER:
            raise FormatError(f"expected property group header or UIAM at 0x{cursor:x}")

        if cursor + 20 > limit:
            raise FormatError(f"truncated property group header at 0x{cursor:x}")

        group_offset = cursor
        group_flags = reader.u32(cursor + 4)
        group_id = reader.u32(cursor + 8)

        reader.require(cursor + 12, PROP_GROUP_COUNT)
        property_count = reader.u32(cursor + 16)
        cursor += 20

        group_property_names: list[str] = []

        for _ in range(property_count):
            record, cursor = parse_property_record(reader, cursor)
            if cursor > limit:
                raise FormatError(
                    f"property {record['name']!r} at 0x{record['offset']:x} "
                    f"extends past static area limit 0x{limit:x}"
                )

            name = record["name"]
            if name in seen:
                raise FormatError(
                    f"duplicate static property {name!r} at 0x{record['offset']:x}; "
                    f"first at 0x{seen[name]:x}"
                )
            seen[name] = record["offset"]

            record["group_id"] = group_id
            record["group_flags"] = group_flags

            properties[name] = record["value"]
            records.append(record)
            group_property_names.append(name)

        reader.require(cursor, PROP_GROUP_END)
        cursor += 4

        if cursor > limit:
            raise FormatError(f"property group end extends past 0x{limit:x}")

        groups.append({
            "offset": group_offset,
            "flags": group_flags,
            "group_id": group_id,
            "property_count": property_count,
            "end_offset": cursor,
            "properties": group_property_names,
        })

def parse_file(path: Path) -> dict[str, Any]:
    data = path.read_bytes()
    reader = Reader(data)
    reader.require(0, b"UISF")
    if len(reader.positions(b"UISF")) != 1 or len(reader.positions(b"UISB")) != 1:
        raise FormatError("expected exactly one UISF and UISB")

    component_positions = reader.positions(b"UICH")
    if not component_positions:
        raise FormatError("no UICH components")

    components = []
    for index, component_offset in enumerate(component_positions):
        component_end = component_positions[index + 1] if index + 1 < len(component_positions) else len(data)

        reader.require(component_offset, b"UICH")
        name, after_name = reader.string(component_offset + 20)
        parent = reader.i32(after_name)

        post_parent_start = after_name + 4
        post_parent_end = post_parent_start + 16
        if post_parent_end > component_end:
            raise FormatError(f"UICH {name!r}: post-parent fields outside component")

        post_parent_16 = data[post_parent_start:post_parent_end].hex()
        static_start = post_parent_end

        properties, property_records, property_groups, map_offset = parse_static_properties(
            reader, static_start, component_end
        )
        reader.require(map_offset, b"UIAM")

        component_ends = reader.positions(b"UIME", map_offset, component_end)
        if len(component_ends) != 1:
            raise FormatError(f"UICH {name!r}: UIME count mismatch")
        if map_offset >= component_ends[0]:
            raise FormatError(f"UICH {name!r}: UIAM is not before UIME")

        reader.require(map_offset + 4, b"UIAB")
        animation_count = reader.u32(map_offset + 8)
        animation_positions = reader.positions(b"USAM", map_offset + 12, component_ends[0])
        if len(animation_positions) != animation_count:
            raise FormatError(
                f"UICH {name!r}: UIAB={animation_count}, USAM={len(animation_positions)}"
            )

        animations = []
        for animation_index, animation_offset in enumerate(animation_positions):
            animation_end = (
                animation_positions[animation_index + 1]
                if animation_index + 1 < len(animation_positions)
                else component_ends[0]
            )
            animations.append(parse_animation(reader, animation_offset, animation_end))

        components.append({
            "name": name,
            "parent_index": parent,
            "component_offset": component_offset,
            "post_parent_16": post_parent_16,
            "static_start": static_start,
            "uiam_offset": map_offset,
            "properties": properties,
            "property_groups": property_groups,
            "property_records": property_records,
            "animations": [
                {**asdict(animation), "end_time_ms": animation.end_time_ms}
                for animation in animations
            ],
        })

    for index, component in enumerate(components):
        parent = component["parent_index"]
        if parent < -1 or parent >= len(components) or parent == index:
            raise FormatError(f"invalid parent index {parent} for {component['name']!r}")

    return {
        "file": path.name,
        "bytes": len(data),
        "component_count": len(components),
        "animation_count": sum(len(component["animations"]) for component in components),
        "components": components,
    }

# ===== verify_layout 与 main 参数接入 =====

def _require_property(component: dict[str, Any], name: str) -> Any:
    properties = component.get("properties")
    if not isinstance(properties, dict) or name not in properties:
        raise FormatError(f"component {component.get('name')!r} missing property {name!r}")
    return properties[name]


def verify_layout(path: Path) -> None:
    report = parse_file(path)

    for component in report["components"]:
        uiam_offset = component["uiam_offset"]
        static_start = component["static_start"]
        for record in component["property_records"]:
            if not (
                static_start <= record["offset"] < record["end_offset"] <= uiam_offset
            ):
                raise FormatError(
                    f"property {record['name']!r} at 0x{record['offset']:x}..0x{record['end_offset']:x} "
                    f"is not fully before UIAM 0x{uiam_offset:x} in {component['name']!r}"
                )

    matches = [component for component in report["components"] if component["name"] == "iconSlot1"]
    if len(matches) != 1:
        raise FormatError(f"expected exactly one iconSlot1 component, got {len(matches)}")

    component = matches[0]
    expected_values = {
        "texture": "textures/adventureProgress/iconSlot.tex",
        "scale": [1.0, 1.0, 1.0],
        "origin": [83.5, 0.0, 0.0],
        "position": [126.0, 42.5, 0.0],
        "width": 83.0,
        "height": 83.0,
    }

    for name, expected in expected_values.items():
        actual = _require_property(component, name)
        if actual != expected:
            raise FormatError(
                f"iconSlot1 property {name!r}: expected {expected!r}, got {actual!r}"
            )


def main() -> None:
    parser = argparse.ArgumentParser(description="Extract and validate release UISF animation timelines.")
    parser.add_argument("paths", nargs="*", type=Path)
    parser.add_argument("--all", action="store_true")
    parser.add_argument("--output", type=Path)
    parser.add_argument("--verify-layout", type=Path, metavar="PATH")
    args = parser.parse_args()

    if args.verify_layout is not None:
        verify_layout(args.verify_layout)
        print(f"OK layout {args.verify_layout}")
        return

    root = Path(__file__).resolve().parents[1]
    paths = sorted((root / "assets_raw/xml/screens").glob("*.uiscreen")) if args.all else args.paths
    if not paths:
        raise SystemExit("no .uiscreen inputs")

    reports = [parse_file(path) for path in paths]
    output = args.output or root / "reverse/uiscreen_reports"
    output.mkdir(parents=True, exist_ok=True)
    for report in reports:
        (output / f"{Path(report['file']).stem}.json").write_text(
            json.dumps(report, ensure_ascii=False, indent=2), encoding="utf-8")
        print(f"OK {report['file']}: {report['component_count']} components, "
              f"{report['animation_count']} animations, {report['bytes']} bytes")
    print(f"Validated {len(reports)} files; reports: {output}")


if __name__ == "__main__":
    main()
