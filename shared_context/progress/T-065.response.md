LAYOUT_BOOL_KEYS = {

def is_finite_number(value: object) -> bool:
    return (
        isinstance(value, (int, float))
        and not isinstance(value, bool)
        and math.isfinite(value)
    )


def validate_layout(component_name: str, stem: str, component: dict) -> dict:
    properties = component.get("properties")
    if not isinstance(properties, dict):
        raise FormatError(f"{stem}/{component_name}: missing UIAM properties object")

    layout = {}
    for key in LAYOUT_KEYS:
        if key not in properties:
            continue
        value = properties[key]
        layout[key] = value

        if key in LAYOUT_VECTOR_SIZES:
            expected = LAYOUT_VECTOR_SIZES[key]
            if not isinstance(value, list) or len(value) != expected:
                raise FormatError(
                    f"{stem}/{component_name}: layout {key!r} must have {expected} values"
                )
            if any(not is_finite_number(item) for item in value):
                raise FormatError(
                    f"{stem}/{component_name}: layout {key!r} contains a non-finite value"
                )
        elif key in LAYOUT_NONNEGATIVE_NUMBERS:
            if not is_finite_number(value) or value < 0:
                raise FormatError(
                    f"{stem}/{component_name}: layout {key!r} must be finite and non-negative"
                )
        elif key in LAYOUT_STRING_KEYS:
            if not isinstance(value, str):
                raise FormatError(
                    f"{stem}/{component_name}: layout {key!r} must be a string"
                )
        elif key in LAYOUT_BOOL_KEYS:
            if not isinstance(value, bool):
                raise FormatError(
                    f"{stem}/{component_name}: layout {key!r} must be a boolean"
                )
        elif key in LAYOUT_INT_KEYS:
            if not isinstance(value, int) or isinstance(value, bool):
                raise FormatError(
                    f"{stem}/{component_name}: layout {key!r} must be an integer"
                )
        else:
            raise AssertionError(f"unhandled layout key {key!r}")

    return layout


def validate_parent_indices(components: list[dict], stem: str) -> None:
    count = len(components)
    for index, component in enumerate(components):
        parent_index = component["parentIndex"]
        if not isinstance(parent_index, int) or isinstance(parent_index, bool):
            raise FormatError(f"{stem}: parentIndex at {index} is not an integer")
        if parent_index < -1 or parent_index >= count:
            raise FormatError(
                f"{stem}: parentIndex {parent_index} at {index} is outside -1..<{count}"
            )
        if parent_index == index:
            raise FormatError(f"{stem}: component {index} is its own parent")
        if index == 0 and parent_index != -1:
            raise FormatError(f"{stem}: root parentIndex must be -1")


def validate_map_icon_slot1(components: list[dict], stem: str) -> None:
    if stem != "inGameChallengeMapScreen":
        return

    matches = [
        component
        for component in components
        if component["name"] == "iconSlot1"
        or component["name"].endswith(".iconSlot1")
    ]
    if len(matches) != 1:
        raise FormatError(
            f"{stem}: expected exactly one iconSlot1 component, got {len(matches)}"
        )

    layout = matches[0]["layout"]
    expected = {
        "texture": "textures/adventureprogress/iconSlot.tex",
        "position": [126, 42.5, 0],
        "origin": [83.5, 0, 0],
        "scale": [1, 1, 1],
        "width": 83,
        "height": 83,
    }
    for key, value in expected.items():
        if key not in layout or layout[key] != value:
            raise FormatError(
                f"{stem}/iconSlot1: layout {key!r} mismatch: "
                f"{layout.get(key)!r} != {value!r}"
            )

def build(stem: str) -> tuple[dict, dict]:
    output_name, expected_components, expected_animations, expected_interpolations = SPECS[stem]
    source = SCREEN_ROOT / f"{stem}.uiscreen"
    report = parse_file(source)
    names = qualified_names(report["components"])
    if len(names) != len(set(names)):
        raise FormatError("duplicate qualified component name")
    components = []
    interpolation_counts: collections.Counter[int] = collections.Counter()
    comment_count = 0
    for component, name in zip(report["components"], names):
        animations = []
        for animation in component["animations"]:
            tracks = []
            for track in animation["tracks"]:
                if track["name"] not in VALUE_COUNTS:
                    raise FormatError(
                        f"unhandled track {track['name']!r} in {name}/{animation['name']}"
                    )
                keyframes = []
                for key in sorted(track["keyframes"], key=lambda item: item["time_ms"]):
                    interpolation_counts[key["interpolation"]] += 1
                    keyframe = {
                        "timeMS": key["time_ms"],
                        "interpolation": key["interpolation"],
                        "values": crop_values(track["name"], key["values"], key["offset"]),
                        "events": key["events"],
                    }
                    if track["name"] == "comment":
                        keyframe["comment"] = key["values"][0]
                        comment_count += 1
                    keyframes.append(keyframe)
                tracks.append({"name": track["name"], "keyframes": keyframes})
            animations.append({
                "name": animation["name"],
                "endTimeMS": animation["end_time_ms"],
                "tracks": tracks,
            })

        layout = validate_layout(name, stem, component)
        parent_index = component.get("parent_index")
        if not isinstance(parent_index, int) or isinstance(parent_index, bool):
            raise FormatError(f"{stem}/{name}: parent_index must be an integer")

        # Keep every source component and animation, including the root and empty ones.
        # New fields are appended after the existing name/animations payload so the
        # pre-existing animation JSON remains semantically unchanged.
        components.append({
            "name": name,
            "animations": animations,
            "parentIndex": parent_index,
            "layout": layout,
        })

    validate_parent_indices(components, stem)
    validate_map_icon_slot1(components, stem)

    animation_count = sum(len(component["animations"]) for component in components)
    if len(components) != expected_components:
        raise FormatError(f"{stem}: expected {expected_components} components, got {len(components)}")
    if animation_count != expected_animations:
        raise FormatError(f"{stem}: expected {expected_animations} animations, got {animation_count}")
    if dict(interpolation_counts) != expected_interpolations:
        raise FormatError(f"{stem}: interpolation census mismatch: {dict(interpolation_counts)}")
    if components[0]["name"] != "":
        raise FormatError(f"{stem}: the UISF screen root must have an empty qualified name")
    if any(name.startswith(f"{stem}.") for name in names):
        raise FormatError(f"{stem}: screen root was not removed from a qualified name")
    content = {
        "formatVersion": 1,
        "source": {
            "file": source.name,
            "sha256": hashlib.sha256(source.read_bytes()).hexdigest(),
        },
        "components": components,
    }
    summary = {
        "output": output_name,
        "components": len(components),
        "animations": animation_count,
        "comments": comment_count,
        # JSON object keys are strings; keep the in-memory manifest identical to its round-trip.
        "interpolations": {str(key): value for key, value in sorted(interpolation_counts.items())},
    }
    return content, summary

    private static func validateParentIndices(
        _ components: [UIScreenAnimationContent.Component]
    ) throws {
        for (index, component) in components.enumerated() {
            guard let parentIndex = component.parentIndex else { continue }

            let inRange = parentIndex >= -1 && parentIndex < components.count
            guard inRange, parentIndex != index else {
                throw UIScreenAnimationError.invalidParentIndex(
                    component: component.name,
                    parentIndex: parentIndex)
            }
            if index == 0, parentIndex != -1 {
                throw UIScreenAnimationError.invalidParentIndex(
                    component: component.name,
                    parentIndex: parentIndex)
            }
        }
    }

    private static func validateLayout(
        _ layout: UIScreenAnimationContent.Component.Layout,
        componentName: String
    ) throws {
        func requireVector(
            _ values: [Float]?,
            count: Int,
            name: String
        ) throws {
            guard let values else { return }
            guard values.count == count,
                  values.allSatisfy({ $0.isFinite }) else {
                throw UIScreenAnimationError.invalidLayout(
                    component: componentName,
                    reason: "\(name) must have \(count) finite values")
            }
        }

        try requireVector(layout.position, count: 3, name: "position")
        try requireVector(layout.scale, count: 3, name: "scale")
        try requireVector(layout.origin, count: 3, name: "origin")
        try requireVector(layout.rotation, count: 3, name: "rotation")
        try requireVector(layout.texCoordPos, count: 2, name: "texCoordPos")
        try requireVector(layout.texCoordSize, count: 2, name: "texCoordSize")
        try requireVector(layout.colour, count: 4, name: "colour")

        let nonnegativeValues: [(String, Float?)] = [
            ("width", layout.width),
            ("height", layout.height),
            ("fontSize", layout.fontSize),
        ]
        for (name, value) in nonnegativeValues {
            guard let value else { continue }
            guard value.isFinite, value >= 0 else {
                throw UIScreenAnimationError.invalidLayout(
                    component: componentName,
                    reason: "\(name) must be finite and non-negative")
            }
        }
    }

    private static func validate(_ content: UIScreenAnimationContent) throws {
        guard content.formatVersion == 1 else {
            throw UIScreenAnimationError.unsupportedFormat(content.formatVersion)
        }

        try validateParentIndices(content.components)

        var componentNames = Set<String>()
        for component in content.components {
            if let layout = component.layout {
                try validateLayout(layout, componentName: component.name)
            }

            guard componentNames.insert(component.name).inserted else {
                throw UIScreenAnimationError.duplicateComponent(component.name)
            }
            var animationNames = Set<String>()
            for animation in component.animations {
                guard animationNames.insert(animation.name).inserted else {
                    throw UIScreenAnimationError.duplicateAnimation(component: component.name,
                                                                     animation: animation.name)
                }
                guard animation.endTimeMS >= 0 else {
                    throw UIScreenAnimationError.invalidTimeline(component: component.name,
                                                                  animation: animation.name,
                                                                  reason: "negative endTimeMS")
                }
                for track in animation.tracks {
                    guard let expectedValues = expectedValueCount(for: track.name) else {
                        throw UIScreenAnimationError.invalidTimeline(
                            component: component.name,
                            animation: animation.name,
                            reason: "unknown track \(track.name)")
                    }
                    var previousTime = -1
                    for keyframe in track.keyframes {
                        guard keyframe.timeMS >= 0,
                              // Release UISF resources contain equal-time keys.
                              // Their order is significant: the last key at the
                              // current timestamp wins, so only regression is invalid.
                              keyframe.timeMS >= previousTime,
                              keyframe.timeMS <= animation.endTimeMS,
                              keyframe.values.allSatisfy({ $0.isFinite }) else {
                            throw UIScreenAnimationError.invalidTimeline(
                                component: component.name,
                                animation: animation.name,
                                reason: "unordered/out-of-range/non-finite keyframe in \(track.name)")
                        }
                        previousTime = keyframe.timeMS
                        guard keyframe.values.count == expectedValues else {
                            throw UIScreenAnimationError.invalidTimeline(
                                component: component.name,
                                animation: animation.name,
                                reason: "invalid value count in \(track.name)")
                        }
                        guard (track.name == "comment") == (keyframe.comment != nil) else {
                            throw UIScreenAnimationError.invalidTimeline(
                                component: component.name,
                                animation: animation.name,
                                reason: "invalid comment payload in \(track.name)")
                        }
                        for event in keyframe.events {
                            let validEvent = (event.command == "PlaySound" && event.arguments.count == 1)
                                || (event.command == "PlayAnimation" && event.arguments.count == 2)
                            guard validEvent else {
                                throw UIScreenAnimationError.invalidTimeline(
                                    component: component.name,
                                    animation: animation.name,
                                    reason: "unsupported resource event")
                            }
                        }
                        // FUN_003e5554 uses each keyframe's outgoing interpolation.
                        // Validate the first key too: it controls the first segment.
                        if track.name != "enabled", track.name != "_Events_", track.name != "comment",
                           !UIScreenInterpolation.isSupported(mode: keyframe.interpolation) {
                            throw UIScreenAnimationError.unsupportedInterpolation(
                                component: component.name,
                                animation: animation.name,
                                track: track.name,
                                timeMS: keyframe.timeMS)
                        }
                    }
                }
            }
        }
    }
