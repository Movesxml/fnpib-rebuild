```swift
import Foundation

/// Verified UIScreen root-viewport scale modes from FUN_003ec940 / FUN_003ec8bc.
/// Raw-value construction is intentionally open so unknown values can follow the
/// original switch default branch instead of being coerced to KeepAspect.
public struct UIScreenViewportScaleMode: RawRepresentable, Equatable, Sendable {
    public let rawValue: Int

    public init(rawValue: Int) {
        self.rawValue = rawValue
    }

    public static let none = Self(rawValue: 0)
    public static let keepAspect = Self(rawValue: 1)
    public static let scaleX = Self(rawValue: 2)
    public static let scaleY = Self(rawValue: 3)
    public static let scale = Self(rawValue: 4)

    public static let verifiedModes: [Self] = [
        .none, .keepAspect, .scaleX, .scaleY, .scale,
    ]

    public var isVerified: Bool {
        (0...4).contains(rawValue)
    }
}

public enum UIScreenRootViewportError: Error, Equatable, Sendable {
    case invalidDimension(name: String, value: Float)
}

/// Root viewport for the release UIScreen projection.
/// Device and design dimensions are Float; all four values must be finite and > 0.
public struct UIScreenRootViewport: Equatable, Sendable {
    public static let iPhoneDesignWidth: Float = 480
    public static let iPhoneDesignHeight: Float = 320

    public let deviceWidth: Float
    public let deviceHeight: Float
    public let designWidth: Float
    public let designHeight: Float
    public let scaleToViewportMode: UIScreenViewportScaleMode

    public init(
        deviceWidth: Float,
        deviceHeight: Float,
        designWidth: Float = UIScreenRootViewport.iPhoneDesignWidth,
        designHeight: Float = UIScreenRootViewport.iPhoneDesignHeight,
        scaleToViewportMode: UIScreenViewportScaleMode = .scale
    ) throws {
        try Self.validateDimension(deviceWidth, name: "deviceWidth")
        try Self.validateDimension(deviceHeight, name: "deviceHeight")
        try Self.validateDimension(designWidth, name: "designWidth")
        try Self.validateDimension(designHeight, name: "designHeight")

        self.deviceWidth = deviceWidth
        self.deviceHeight = deviceHeight
        self.designWidth = designWidth
        self.designHeight = designHeight
        self.scaleToViewportMode = scaleToViewportMode
    }

    public var xRatio: Float {
        deviceWidth / designWidth
    }

    public var yRatio: Float {
        deviceHeight / designHeight
    }

    /// FUN_003cd778:
    /// 0 None       -> [1, 1]
    /// 1 KeepAspect -> [min(xRatio, yRatio), min(xRatio, yRatio)]
    /// 2 ScaleX     -> [xRatio, xRatio]
    /// 3 ScaleY     -> [yRatio, yRatio]
    /// 4 Scale      -> [xRatio, yRatio]
    /// default      -> [xRatio, yRatio]
    /// z is always 1. No cap, centering or letterboxing is applied here.
    public var rootScale: UIScreenPoint3 {
        let xRatio = self.xRatio
        let yRatio = self.yRatio

        switch scaleToViewportMode.rawValue {
        case 0:
            return UIScreenPoint3(x: 1, y: 1, z: 1)
        case 1:
            let aspect = min(xRatio, yRatio)
            return UIScreenPoint3(x: aspect, y: aspect, z: 1)
        case 2:
            return UIScreenPoint3(x: xRatio, y: xRatio, z: 1)
        case 3:
            return UIScreenPoint3(x: yRatio, y: yRatio, z: 1)
        case 4:
            return UIScreenPoint3(x: xRatio, y: yRatio, z: 1)
        default:
            // Original switch default: independent two-axis ratios, not KeepAspect.
            return UIScreenPoint3(x: xRatio, y: yRatio, z: 1)
        }
    }

    public var rootScaleMatrix: UIScreenMatrix4 {
        UIScreenMatrix4.scale(rootScale)
    }

    private static func validateDimension(_ value: Float, name: String) throws {
        guard value.isFinite, value > 0 else {
            throw UIScreenRootViewportError.invalidDimension(name: name, value: value)
        }
    }
}
```

## 2. 新增文件：`Sources/FNPBCore/Content/UIScreenLayoutProjection+DevicePixels.swift`

```swift
import Foundation

extension UIScreenLayoutProjection {
    /// Device-pixel projection using an already validated root viewport.
    /// The existing logical project(player) is called unchanged; only world-space
    /// values are wrapped by the common root scale. Local values stay logical.
    public static func projectDevicePixels(
        _ player: UIScreenAnimationPlayer,
        viewport: UIScreenRootViewport
    ) throws -> UIScreenLayoutProjection {
        let logical = try project(player)
        let rootMatrix = viewport.rootScaleMatrix

        let components = logical.components.map { component in
            UIScreenProjectedComponent(
                name: component.name,
                index: component.index,
                parentIndex: component.parentIndex,
                position: component.position,
                scale: component.scale,
                rotation: component.rotation,
                width: component.width,
                height: component.height,
                origin: component.origin,
                originFromCenter: component.originFromCenter,
                effectiveOrigin: component.effectiveOrigin,
                localMatrix: component.localMatrix,
                worldMatrix: rootMatrix * component.worldMatrix,
                worldOrigin: rootMatrix.transformed(component.worldOrigin),
                worldCorners: UIScreenProjectedCorners(
                    minimumXMinimumY: rootMatrix.transformed(
                        component.worldCorners.minimumXMinimumY),
                    maximumXMinimumY: rootMatrix.transformed(
                        component.worldCorners.maximumXMinimumY),
                    maximumXMaximumY: rootMatrix.transformed(
                        component.worldCorners.maximumXMaximumY),
                    minimumXMaximumY: rootMatrix.transformed(
                        component.worldCorners.minimumXMaximumY)))
        }

        return UIScreenLayoutProjection(components: components)
    }

    /// Convenience entry that validates device/design dimensions through
    /// UIScreenRootViewport and then applies the common root scale once.
    public static func projectDevicePixels(
        _ player: UIScreenAnimationPlayer,
        deviceWidth: Float,
        deviceHeight: Float,
        designWidth: Float = UIScreenRootViewport.iPhoneDesignWidth,
        designHeight: Float = UIScreenRootViewport.iPhoneDesignHeight,
        scaleToViewportMode: UIScreenViewportScaleMode = .scale
    ) throws -> UIScreenLayoutProjection {
        let viewport = try UIScreenRootViewport(
            deviceWidth: deviceWidth,
            deviceHeight: deviceHeight,
            designWidth: designWidth,
            designHeight: designHeight,
            scaleToViewportMode: scaleToViewportMode)
        return try projectDevicePixels(player, viewport: viewport)
    }

    public static func project(
        _ player: UIScreenAnimationPlayer,
        viewport: UIScreenRootViewport
    ) throws -> UIScreenLayoutProjection {
        try projectDevicePixels(player, viewport: viewport)
    }

    public static func project(
        _ player: UIScreenAnimationPlayer,
        deviceWidth: Float,
        deviceHeight: Float,
        designWidth: Float = UIScreenRootViewport.iPhoneDesignWidth,
        designHeight: Float = UIScreenRootViewport.iPhoneDesignHeight,
        scaleToViewportMode: UIScreenViewportScaleMode = .scale
    ) throws -> UIScreenLayoutProjection {
        try projectDevicePixels(
            player,
            deviceWidth: deviceWidth,
            deviceHeight: deviceHeight,
            designWidth: designWidth,
            designHeight: designHeight,
            scaleToViewportMode: scaleToViewportMode)
    }
}
```

```swift
    func testRootViewportScaleModesUseVerifiedRatiosAndUnknownFallsBackToScale() throws {
        let deviceWidth: Float = 960
        let deviceHeight: Float = 480
        let cases: [(UIScreenViewportScaleMode, UIScreenPoint3)] = [
            (UIScreenViewportScaleMode.none, UIScreenPoint3(x: 1, y: 1, z: 1)),
            (UIScreenViewportScaleMode.keepAspect, UIScreenPoint3(x: 1.5, y: 1.5, z: 1)),
            (UIScreenViewportScaleMode.scaleX, UIScreenPoint3(x: 2, y: 2, z: 1)),
            (UIScreenViewportScaleMode.scaleY, UIScreenPoint3(x: 1.5, y: 1.5, z: 1)),
            (UIScreenViewportScaleMode.scale, UIScreenPoint3(x: 2, y: 1.5, z: 1)),
        ]

        for (mode, expected) in cases {
            let viewport = try UIScreenRootViewport(
                deviceWidth: deviceWidth,
                deviceHeight: deviceHeight,
                scaleToViewportMode: mode)
            assertPoint(viewport.rootScale, expected)
            XCTAssertEqual(viewport.rootScale.z, 1)
            XCTAssertEqual(viewport.xRatio, 2)
            XCTAssertEqual(viewport.yRatio, 1.5)
        }

        let unknown = UIScreenViewportScaleMode(rawValue: 99)
        let unknownViewport = try UIScreenRootViewport(
            deviceWidth: deviceWidth,
            deviceHeight: deviceHeight,
            scaleToViewportMode: unknown)
        assertPoint(unknownViewport.rootScale, UIScreenPoint3(x: 2, y: 1.5, z: 1))
        XCTAssertNotEqual(unknownViewport.rootScale,
                          UIScreenPoint3(x: 1.5, y: 1.5, z: 1))
        XCTAssertFalse(unknown.isVerified)
        XCTAssertEqual(UIScreenViewportScaleMode.verifiedModes.map(\.rawValue),
                       [0, 1, 2, 3, 4])
    }

    func testRootViewportRejectsNonFiniteAndNonPositiveDimensions() throws {
        let invalidDeviceCases: [(String, Float, Float)] = [
            ("deviceWidth", .nan, 320),
            ("deviceWidth", .infinity, 320),
            ("deviceWidth", 0, 320),
            ("deviceWidth", -1, 320),
            ("deviceHeight", 480, .nan),
            ("deviceHeight", 480, .infinity),
            ("deviceHeight", 480, 0),
            ("deviceHeight", 480, -1),

    func testDeviceProjectionAppliesRootScaleOnceAndKeepsLocalMatricesLogical() throws {
        let viewport = try UIScreenRootViewport(deviceWidth: 960, deviceHeight: 480)
        XCTAssertEqual(viewport.scaleToViewportMode, UIScreenViewportScaleMode.scale)
        XCTAssertEqual(viewport.scaleToViewportMode.rawValue, 4)
        assertPoint(viewport.rootScale, UIScreenPoint3(x: 2, y: 1.5, z: 1))

        let staticLayout = UIScreenAnimationContent.Component.Layout(
            position: [10, 20, 0],
            scale: [2, 3, 1],
            origin: [1, 2, 0],
            rotation: [0, 0, 0],
            width: 4,
            height: 6,
            originFromCenter: false)

        let player = try UIScreenAnimationPlayer(content: makeContent([
            UIScreenAnimationContent.Component(
                name: "static",
                animations: [],
                parentIndex: -1,
                layout: staticLayout),
            UIScreenAnimationContent.Component(
                name: "child",
                animations: [],
                parentIndex: 0,
                layout: UIScreenAnimationContent.Component.Layout(position: [5, 10, 0])),
        ]))

        let logical = try UIScreenLayoutProjection.project(player)
        let device = try UIScreenLayoutProjection.projectDevicePixels(
            player, viewport: viewport)
        XCTAssertEqual(device.components.count, logical.components.count)

        let staticLogical = logical.components[0]
        let staticDevice = device.components[0]
        XCTAssertEqual(staticDevice.position, staticLogical.position)
        XCTAssertEqual(staticDevice.localMatrix, staticLogical.localMatrix)
        XCTAssertEqual(staticDevice.worldMatrix,
                       viewport.rootScaleMatrix * staticLogical.worldMatrix)
        assertPoint(staticDevice.worldOrigin, UIScreenPoint3(x: 16, y: 21, z: 0))
        assertPoint(staticDevice.worldCorners.minimumXMinimumY,
                    UIScreenPoint3(x: 16, y: 21, z: 0))
        assertPoint(staticDevice.worldCorners.maximumXMinimumY,
                    UIScreenPoint3(x: 32, y: 21, z: 0))
        assertPoint(staticDevice.worldCorners.maximumXMaximumY,
                    UIScreenPoint3(x: 32, y: 48, z: 0))
        assertPoint(staticDevice.worldCorners.minimumXMaximumY,
                    UIScreenPoint3(x: 16, y: 48, z: 0))

        let childLogical = logical.components[1]
        let childDevice = device.components[1]
        XCTAssertEqual(childDevice.position, childLogical.position)
        XCTAssertEqual(childDevice.localMatrix, childLogical.localMatrix)
        XCTAssertEqual(childDevice.worldMatrix,
                       viewport.rootScaleMatrix * childLogical.worldMatrix)
        XCTAssertNotEqual(
            childDevice.worldMatrix,
            viewport.rootScaleMatrix
                * viewport.rootScaleMatrix
                * childLogical.worldMatrix)
    }
```

```swift
    private func scaled(_ point: UIScreenPoint3,
                        by scale: UIScreenPoint3) -> UIScreenPoint3 {
        UIScreenPoint3(
            x: point.x * scale.x,
            y: point.y * scale.y,
            z: point.z * scale.z)
    }
```
