## 设计说明

**证据到实现的映射**

| Mach-O 事实 | 草案实现 |
|---|---|
| 计分水果路径更新 `DAT_005d4ce0`（单槽） | `ChallengeConsecutiveTracker`（`lastPlayerIndex` + `count`，单槽，**不**用 `comboCount`） |
| 同玩家且 `noScore==false` → `count+=1`；否则 `count=1` 并记录玩家 | `recordSlice(playerIndex:)`：同玩家 +1，换玩家/首切置 1 |
| `noScore` 在计分早退路径返回，不进入该事件 | `ChallengeConsecutiveDispatch.emit(isNoScore:)` 早退：不写计数、不上报 |
| `FUN_00003fb4(manager, count, fruitDef+0x250)` → `FUN_0005ee50` | `emit` 内**先**更新计数、**再**调 `send(count, fruitName)` |
| `r2` = 水果类型哈希 | 值语义传 `FruitInfo.name`，大小写无关匹配留给现有 `ChallengeModeRuntime.recordConsecutive` |
| 激活条件虚表 `+0x24` | 由 runtime 内部处理，`GameWorld.step6` 的 `evaluateChallengeImmediate` 保持不动 |

**顺序约束**：插入点固定在 `FruitOnSliced.apply` 内「`resolveCritical` 成功、计分已写入」之后，即原版「先计分 → 再更新紧接计数 → 再分发条件」的次序；不触碰 combo、不触碰触摸事件、不改 `step6`。

//
//  ChallengeConsecutiveTracker.swift
//  FNPBCore
//
//  T-038：原版连续切果挑战（CONSECUTIVE）的计数与事件分发。
//
//  证据链 [macho]：
//    FUN_00010bec（切片计分路径，先按玩家与 fruitDefinition+0x330(noScore) 更新 DAT_005d4ce0）
//      → DAT_005d4ce0（单槽：{ 玩家, 连续计数 }）
//      → FUN_00003fb4(manager, count, fruitDefinition + 0x250)
//      → FUN_0005ee50 → 激活条件虚表 + 0x24（0x5f21c）
//
//  语义 [both]：同一玩家且 noScore == false → count += 1；否则（首切 / 换玩家）→ count = 1
//  并记录玩家。noScore 水果在原版更早的计分早退路径返回，不写 DAT_005d4ce0、不触发条件。
//
//  ⚠️ 本计数器与 combo 无关：原版 combo 走 GameState 的 combo 计数与结算窗口，
//     不得用 ComboTracker / comboCount / 触摸事件替代本模块。
//

import Foundation

/// 原版 `DAT_005d4ce0` 的单槽状态：{ 最近切果的玩家, 连续切果计数 }。
///
/// 原版为**单槽**：同一时刻只记录一位玩家。多人同屏时后一位玩家会覆盖前一位，
/// 先前玩家再切果时计数从 1 重新开始——此处逐字保留该行为，不做多玩家聚合。
public struct ChallengeConsecutiveTracker: Equatable, Sendable {

    /// 最近一次触发连续计数的玩家索引；`nil` 表示本局尚未记录过切片。
    public private(set) var lastPlayerIndex: Int?

    /// 当前连续计数（原版为 32 位有符号整数）。
    public private(set) var count: Int32

    public init() {
        self.lastPlayerIndex = nil
        self.count = 0
    }

    public var hasRecordedSlice: Bool { lastPlayerIndex != nil }

    /// 记录一次「计分水果」切片，返回更新后的连续计数。
    ///
    /// 对应原版：
    /// - 同一玩家 → `count += 1`
    /// - 首次 / 换玩家 → `count = 1` 并记录玩家
    ///
    /// `noScore` 的判定在调用方（`ChallengeConsecutiveDispatch`）先做，
    /// 因为原版 noScore 水果在计分早退路径就已返回，不会走到本函数。
    @discardableResult
    public mutating func recordSlice(playerIndex: Int) -> Int32 {
        if lastPlayerIndex == playerIndex {
            // 原版为裸 int 加法；正常对局不可达溢出，饱和仅作防御。
            if count < Int32.max { count += 1 }
        } else {
            lastPlayerIndex = playerIndex
            count = 1
        }
        return count
    }

    /// 清空。新局入口必须调用（对应原版新对局重建 `DAT_005d4ce0`）。
    public mutating func reset() {
        lastPlayerIndex = nil
        count = 0
    }

    /// 仅当记录中的玩家匹配时清空。原版单槽，效果等同 `reset()`；
    /// 保留参数以便将来多槽扩展时调用点语义不破。
    public mutating func reset(playerIndex: Int) {
        if lastPlayerIndex == playerIndex { reset() }
    }
}

/// 连续切果事件分发（原版 `FUN_00003fb4` + `FUN_0005ee50` 那一步）。
///
/// 抽成独立函数是为了让两条顺序契约可被 XCTest 覆盖：
/// 1) noScore 不上报也不改计数；2) 先更新计数、再分发事件。
/// 生产调用点只有 `FruitOnSliced.apply` 一处。
public enum ChallengeConsecutiveDispatch {

    /// - Parameters:
    ///   - tracker: 对局内单槽计数器（存放在 `GameState`）。
    ///   - playerIndex: 原版记录玩家；本作为单本地玩家，生产固定传 `0`。
    ///   - fruitName: 水果类型名（`FruitInfo.name`），替代原版 `fruitDefinition + 0x250` 的类型哈希；
    ///     大小写无关匹配由 `ChallengeModeRuntime.recordConsecutive` 负责，这里原样透传。
    ///   - isNoScore: `FruitInfo.noScore`（原版 `fruitDefinition + 0x330`）。
    ///     `true` 时走原版早退语义：**不更新计数、不上报**。
    ///   - send: 上报闭包；生产实现为 `services.challengeMode?.recordConsecutive(count:specificType:)`。
    /// - Returns: 更新后的连续计数；`isNoScore == true` 时返回 `nil`。
    @discardableResult
    public static func emit(
        tracker: inout ChallengeConsecutiveTracker,
        playerIndex: Int,
        fruitName: String,
        isNoScore: Bool,
        send: (Int32, String?) -> Void
    ) -> Int32? {
        // 原版 noScore 早退：不写 DAT_005d4ce0，也不触发条件虚表 +0x24。
        guard !isNoScore else { return nil }

        // 先更新单槽计数（DAT_005d4ce0），再分发（FUN_00003fb4）。
        let count = tracker.recordSlice(playerIndex: playerIndex)

        // r2 = 水果类型哈希；本项目值语义传 name。
        send(count, fruitName)
        return count
    }
}

import XCTest
@testable import FNPBCore

/// T-038 连续切果（CONSECUTIVE）计数与事件分发测试。
///
/// 覆盖任务卡要求的场景：
/// 1. 普通水果逐次 1 / 2 / 3
/// 2. noScore 水果不触发（且不重置既有计数）
/// 3. 换玩家重置为 1（含原版单槽语义）
/// 4. 新局复位
/// 5. 特定水果类型名原样透传（供 runtime 做大小写无关过滤）
/// 6. 非 consecutive 条件不影响既有计分/连击，且事件顺序为 计分 → 连续上报 → 条件求值
final class ChallengeConsecutiveTrackerTests: XCTestCase {

    // MARK: - 1. 普通水果 1,2,3 逐次

    func testPlainFruitCountsOneTwoThree() {
        var tracker = ChallengeConsecutiveTracker()

        XCTAssertEqual(tracker.recordSlice(playerIndex: 0), 1)
        XCTAssertEqual(tracker.recordSlice(playerIndex: 0), 2)
        XCTAssertEqual(tracker.recordSlice(playerIndex: 0), 3)

        XCTAssertEqual(tracker.count, 3)
        XCTAssertEqual(tracker.lastPlayerIndex, 0)
        XCTAssertTrue(tracker.hasRecordedSlice)
    }

    // MARK: - 2. noScore 不触发

    func testNoScoreFruitDoesNotEmitOrCount() {
        var tracker = ChallengeConsecutiveTracker()
        var events: [(count: Int32, type: String?)] = []

        XCTAssertEqual(emit(&tracker, name: "banana", noScore: false) { events.append(($0, $1)) }, 1)
        XCTAssertEqual(events.count, 1)

        XCTAssertNil(emit(&tracker, name: "coin", noScore: true) { events.append(($0, $1)) })
        XCTAssertEqual(events.count, 1, "noScore 水果不得上报 CONSECUTIVE 事件")
        XCTAssertEqual(tracker.count, 1, "noScore 水果不得更新计数")

        // 关键边界：noScore 不重置既有计数，紧接的普通水果继续 +1
        XCTAssertEqual(emit(&tracker, name: "apple", noScore: false) { events.append(($0, $1)) }, 2)
        XCTAssertEqual(events.map { $0.count }, [1, 2])
    }

    func testNoScoreFruitAsFirstSliceLeavesTrackerEmpty() {
        var tracker = ChallengeConsecutiveTracker()

        XCTAssertNil(emit(&tracker, name: "coin", noScore: true) { _, _ in })
        XCTAssertFalse(tracker.hasRecordedSlice)
        XCTAssertEqual(tracker.count, 0)
    }

    // MARK: - 3. 换玩家重置为 1

    func testPlayerSwitchResetsToOne() {
        var tracker = ChallengeConsecutiveTracker()

        XCTAssertEqual(tracker.recordSlice(playerIndex: 0), 1)
        XCTAssertEqual(tracker.recordSlice(playerIndex: 0), 2)
        XCTAssertEqual(tracker.recordSlice(playerIndex: 1), 1)
        XCTAssertEqual(tracker.lastPlayerIndex, 1)
        XCTAssertEqual(tracker.recordSlice(playerIndex: 1), 2)
        // 原版单槽：切回先前玩家同样从 1 开始（不是 3）
        XCTAssertEqual(tracker.recordSlice(playerIndex: 0), 1)
    }

    // MARK: - 4. 新局复位

    func testNewGameReset() {
        var tracker = ChallengeConsecutiveTracker()
        tracker.recordSlice(playerIndex: 0)
        tracker.recordSlice(playerIndex: 0)

        tracker.reset()

        XCTAssertEqual(tracker.count, 0)
        XCTAssertNil(tracker.lastPlayerIndex)
        XCTAssertFalse(tracker.hasRecordedSlice)
        XCTAssertEqual(tracker.recordSlice(playerIndex: 0), 1)
    }

    func testResetMatchingPlayerOnly() {
        var tracker = ChallengeConsecutiveTracker()
        tracker.recordSlice(playerIndex: 0)
        tracker.recordSlice(playerIndex: 0)

        tracker.reset(playerIndex: 1)
        XCTAssertEqual(tracker.count, 2, "非记录玩家不应清空计数")

        tracker.reset(playerIndex: 0)
        XCTAssertEqual(tracker.count, 0)
        XCTAssertNil(tracker.lastPlayerIndex)
    }

    // MARK: - 5. 特定水果类型透传

    func testSpecificTypeCarriesFruitNameVerbatim() {
        var tracker = ChallengeConsecutiveTracker()
        var types: [String?] = []

        emit(&tracker, name: "WaterMelon", noScore: false) { _, type in types.append(type) }
        emit(&tracker, name: "watermelon", noScore: false) { _, type in types.append(type) }

        // 大小写无关匹配由 ChallengeModeRuntime 负责；本层必须原样透传名称
        XCTAssertEqual(types, ["WaterMelon", "watermelon"])
        XCTAssertEqual(tracker.count, 2)
    }

    // MARK: - 6. 非 consecutive 不受影响 + 顺序契约

    func testNonConsecutiveConditionLeavesScoreAndComboUntouched() {
        var tracker = ChallengeConsecutiveTracker()
        let challengeIsConsecutive = false
        var score = 0
        var comboCount = 0

        emit(&tracker, name: "banana", noScore: false) { _, _ in
            // 非 CONSECUTIVE 激活时，runtime 的 recordConsecutive 为 no-op（不产出分数）
            if challengeIsConsecutive { score += 100 }
        }

        XCTAssertEqual(score, 0, "非 consecutive 条件不得加分")
        XCTAssertEqual(comboCount, 0, "不得触碰 combo 统计")
        XCTAssertEqual(tracker.count, 1, "计数与原版一样与激活条件类型无关，仍推进")
    }

    func testEventOrderScoreThenConsecutiveThenEvaluate() {
        var tracker = ChallengeConsecutiveTracker()
        var log: [String] = []

        // 模拟 FruitOnSliced.apply 内既有顺序
        log.append("score:7")                       // resolveCritical 成功后的计分
        emit(&tracker, name: "banana", noScore: false) { count, type in
            log.append("consecutive:\(count):\(type ?? "nil")")
        }
        log.append("evaluate")                      // GameWorld.step6 的 evaluateChallengeImmediate

        XCTAssertEqual(log, ["score:7", "consecutive:1:banana", "evaluate"])
    }

    func testMultipleFruitsInOneFrameKeepMonotonicCounts() {
        var tracker = ChallengeConsecutiveTracker()
        var counts: [Int32] = []

        for name in ["apple", "banana", "pear"] {
            emit(&tracker, name: name, noScore: false) { count, _ in counts.append(count) }
        }

        XCTAssertEqual(counts, [1, 2, 3])
    }

    // MARK: - Helpers

    @discardableResult
    private func emit(
        _ tracker: inout ChallengeConsecutiveTracker,
        name: String,
        noScore: Bool,
        send: (Int32, String?) -> Void
    ) -> Int32? {
        ChallengeConsecutiveDispatch.emit(
            tracker: &tracker,
            playerIndex: 0,
            fruitName: name,
            isNoScore: noScore,
            send: send
        )
    }
}

1. **patch 无法逐字校验**：未持有 `FruitOnSliced.swift` / `GameWorld.swift` / `GameState.swift` 生产源码，字段名（`fruit.name`、`fruit.noScore`、`services.challengeMode` 可选性、新局方法名）均为按任务卡描述推断，必须对照实码套用。

3. **单槽覆盖语义**：`recordSlice(playerIndex:)` 严格按原版单槽实现，多人同屏时后切者会覆盖前者记录。若产品上要求「多人各自连续计数」，则与原版不符——需 Kimi 决定是否背离原版。
4. **`evaluateChallengeImmediate` 的时序**：若 `GameWorld.step6` 是每个 step 只求值一次而非每次 `apply` 后求值，同帧多果时条件激活可能比原版晚一帧 [待验证]。本卡按任务卡要求不改 `step6`。

6. **playerIndex 固定 0**：原版 `DAT_005d4ce0` 记录玩家指针，本作单本地玩家映射为常量 0 是推断 [待验证]；若存在本地双人玩法，接线点时必须改为真实索引。

8. **测试 6 的强度**：`testNonConsecutiveConditionLeavesScoreAndComboUntouched` 是「顺序契约测试」，用本地变量模拟 runtime 的 no-op 行为，未接真实 `ChallengeModeRuntime` / `ComboTracker`。落盘后若能注入 spy，应替换为真实调用链断言。
