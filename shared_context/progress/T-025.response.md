## 设计说明

**任务卡缺陷（必须先指出）**

1. **交付物 1 与交付物 3 自相矛盾**：任务卡要求 `App/WaveSpawnBridge.swift`，但要求 `Tests/FNPBCoreTests/SpawnMotionTests.swift` 里“bridge.spawnFruit 后 store 有实体”。`FNPBCoreTests` 是纯逻辑测试 target，**不能 import App（application target）**。故本实现把 `WaveSpawnBridge` 放到 `Sources/FNPBCore/Entities/WaveSpawnBridge.swift`（它只依赖 Core 内类型，职责本就属实体生成层），App 层只做装配。若坚持 App 路径，则测试必须改为 App 测试 target（需 `TEST_HOST`）。
2. **“打通生成→运动”缺一环**：`GameHost.startGame` 目前传 `[WaveInfo()]`（`spawners == []`），`WaveDirector.runSpawnLoop` 遍历空集合 → `spawnSink` 永不被调用 → 水果永不生成。任务卡未提此事，但验收标准“水果按 FruitMotion.integrate 逐帧运动”无法达成。本实现补一个**显式标注的占位波次**（单 spawner / 单水果候选），待内容加载任务卡替换。
3. **实体数量失控防护**：`runSpawnLoop` 的 spawner 条目计时（`max(0, x + speed*dt)`）在 T-014 明确未实现 → 每帧每 spawner 抽一次 ≈ 60 果/秒。本实现在 bridge 里加**存活数上限**护栏（与帧率无关，测试不受影响），标注 `[待验证]`。

**实现要点**
- `FruitMotion.integrate(_:dt:tuning:)` 返回 `isAlive`，可直接作 `guard` 条件；`forEach` 的 `inout entity` 直传合法。
- 顺序遵循 §1.2“实体先推进后判定”：integrate 放在循环体首行，出界即回收（不再做切割判定）。
- 漏切只在 **水果**循环计入（炸弹出界仅回收，不计惩罚）。
- `GameWorld` 新增 `public let motionTuning` + 带默认值的 init 参数（**不改 `GameTicking` 协议签名**，`GameWorld()` 调用点保持兼容）；同一份 tuning 注入 GameWorld（积分）与 Bridge（发射），避免两处参数漂移。
- 新增常量按 `FruitMotion.Physics` 先例用 `extension Constants` 承载（含迁移提示）。

**兼容性风险**：本卡为纯 Core 逻辑 + App 装配，不触碰 OpenGL ES / UIWebView / 32 位相关 API。`UIApplication.shared.connectedScenes` 兜底路径仍为既有代码（iOS 27 可用）。

        _ = wave

        guard store.aliveCount < Constants.SpawnGeometry.maxLiveFruitCount else { return }

        store.spawn(makePrototype(kind: .bomb, typeID: Constants.EntityKindValue.bomb))
    }

    // MARK: Private

    /// 构造实体原型（水果 / 炸弹共用同一套抛射参数）。

        var fruit = FruitInfo()
        fruit.typeID = typeID
        // score / noCritical / onside 等一律保持 FruitInfo 默认（待 fruitlist 内容任务卡）。

        let spawnX = lerp(-Constants.SpawnGeometry.horizontalSpreadHalfWidth,
                          Constants.SpawnGeometry.horizontalSpreadHalfWidth,
                          rng.nextFloat())
        let position = SIMD3<Float>(spawnX, tuning.offscreenBottomMargin, 0)
        let velocity = FruitMotion.launchVelocity(rng: rng, tuning: tuning)
        let spin = FruitMotion.initialSpin(rng: rng, tuning: tuning)

        let transform = EntityTransform(position: position,
                                        velocity: velocity,
                                        rotation: .zero,
                                        angularVelocity: spin,
                                        scale: fruit.scale,
                                        radius: fruit.collision)

        return EntityPrototype(kind: kind, transform: transform, fruit: fruit)
    }

    private func lerp(_ a: Float, _ b: Float, _ t: Float) -> Float {
        a + (b - a) * t
    }
}

    /// [待验证] kind 语义待特效任务卡定名，0 为「普通命中」占位。
    private static let hitEffectKind: Int32 = 0

    /// - Parameter motionTuning: 抛物线积分参数。默认取 `FruitMotionTuning()` 的占位值，
    ///   故既有 `GameWorld()` 调用点无需改动；装配层应显式传入与 bridge 相同的一份。
    public init(motionTuning: FruitMotionTuning = FruitMotionTuning()) {
        self.motionTuning = motionTuning
    }

    /// R7：由 GameLoop 在 `FrameDriving.start()` 后调用一次，使首帧 dt==0 也能推进
    public func prime() {
        lastDelta = fallbackDelta
    }

    public func tick(dt: Float, state: inout GameState,
                     world: inout any EntityStore, services: GameServices) {
        let step = step1_resolveDelta(dt)
        guard step > 0 else { return }              // 首帧零步进（未 prime 时）

        step2_normalizeCollisionRadius(state: &state)
        step3_advanceSlashTrail(dt: step, state: &state)

        // 步骤 4 / 5 只门控「实体遍历」，不终止整帧（见文件头 T-023 控制流修正）。
        let levelGateOK = step4_applyLevelEffects(state: &state)
        let traverseOK = levelGateOK ? step5_shouldTraverseEntities(world: world) : false
        if traverseOK {
            step6_processFruits(dt: step, state: &state, world: &world, services: services)
            step7_processBombs(dt: step, state: &state, world: &world, services: services)
            step8_processSpecials(dt: step, state: &state, world: &world, services: services)
        }

        step9_settleCombo(dt: step, state: &state, services: services)
        step10_updateSubsystems(dt: step, state: &state, world: &world, services: services)
    }

    // MARK: 1 — dt 兜底 [both]
    /// §1.2 步骤 1：`dt==0` → 用上一帧缓存值（exe `param_1!=0` 分支 / macho fVar29）。
    /// 缓存初值 0.0（exe 自证），故首帧未 prime 时返回 0 → tick 直接返回（R7）。
    private func step1_resolveDelta(_ dt: Float) -> Float {
        let resolved = (dt == 0) ? lastDelta : dt
        lastDelta = resolved
        return resolved
    }

    // MARK: 2 — 碰撞半径归一化 [both]
    /// §1.2 步骤 2：d=|(x,y,z)|；d/15.0<=1.0 → w=0.5*(d/15.0)+0.5，否则 1.0。
    ///
    /// 原式输出到**全局最大值** `DAT_005573b8 = max(全局, w)`（↔ exe `DAT_019514c0`）；
    /// 重建无全局单例，直接写入 `state.sliceRadiusSquared`（= w²，供
    /// `SliceTester` 的二次确认半径² 使用，见 GameState.swift `sliceRadiusSquared` 注释）。
    /// **差异：原版是跨帧单调不减的最大值，重建是每帧覆盖**——[待验证] 若原版确实单调
    /// 递增，切刀半径会随时间膨胀；从 `0.0 < game+0xf0` 的门控看，更可能是每帧重算。
    ///
    /// ⚠️ 原版本步受 `game +0x50`（`collisionRadiusActive`）门控，且该字段默认 false；
    ///    照抄门控会让 `sliceRadiusSquared` 恒为 0 → 切割门控 `0.0 < radius²` 永远失败
    ///    → 游戏锁死。本卡按任务卡字面**无条件计算**，门控留待字段默认值闭合后再引入。
    private func step2_normalizeCollisionRadius(state: inout GameState) {
        let distance = simd_length(state.hitPoint)
        let ratio = distance / Constants.Simulation.collisionNormalizeDivisor

        let weight: Float
        if ratio <= Constants.Simulation.collisionNormalizeClamp {
            // 原式两个 0.5 同值：0.5 * (d/15) + 0.5，复用 collisionNormalizeWeight
            weight = Constants.Simulation.collisionNormalizeWeight * ratio
                   + Constants.Simulation.collisionNormalizeWeight
        } else {
            weight = Constants.Simulation.collisionNormalizeClamp
        }

        state.sliceRadiusSquared = weight * weight
    }

    // MARK: 3 — slash 轨迹 [both]
    /// §1.2 步骤 3：+0xa5!=0 → +0x17c += dt；超轨迹时长 → 槽位 (slot+1)%8 写入采样点、
    /// 清 +0xa5、+0x17c=0。容量 8 / 采样间隔 0.05 [macho H]。
    ///
    /// 实现：轨迹环形缓冲已收进 `GameState.slashTrail`（T-023），本步只做推进 +
    /// 镜像字段同步（`slashTrailActive` ↔ +0xa5，`slashTrailTimer` ↔ +0x17c），
    /// 保持与原偏移语义的可对账性。
    private func step3_advanceSlashTrail(dt: Float, state: inout GameState) {
        // 先取出采样点：`state.slashTrail.advance(... , point: state.hitPoint)`
        // 会同时对 state 做 mutating 与读访问 → Swift 独占检查报错，故拆开。
        let samplePoint = state.hitPoint
        state.slashTrail.advance(dt: dt, point: samplePoint)

        // 镜像字段同步（原 +0xa5 / +0x17c）
        state.slashTrailActive = state.slashTrail.isActive
        state.slashTrailTimer = state.slashTrail.elapsed
    }

    // MARK: 4 — 等级特效 [both]
    /// §1.2 步骤 4：`+0xb4 < 4` → `+0xf4 = 0` 并**跳过实体遍历**；返回 false 表示本帧不遍历。
    ///
    /// ⚠️ [待验证] `levelTier`(+0xb4) 默认 0 < 4 → 默认永远跳过遍历，会锁死游戏。
    ///    原版开局应把 +0xb4 写成 ≥4（或由关卡加载写入），但该写入点未从反编译定位。
    ///    本卡只实现判定本身，**不改 GameState 默认值**（避免掩盖未闭合事实）；
    ///    默认值问题列入 T-023 风险点交 Kimi 裁决。
    private func step4_applyLevelEffects(state: inout GameState) -> Bool {
        if state.levelTier < Constants.Simulation.lowLevelEffectCutoff {
            state.levelEffectParam = 0
            return false
        }
        return true
    }

    // MARK: 5 — 遍历门控 [macho]
    /// §1.2 步骤 5：爆炸/冻结计时 <=0，且全局位掩码 & 0x40 == 0，
    /// 且（单例 vtable+0x28 返回 1 或单例为空）。
    ///
    /// 三项门控来源均**未闭合**，本卡保持恒 true：
    ///  1. 爆炸/冻结计时：`PTR+0x24 <= 0.0` —— 对应单例（`BombBlast`？）的计时字段，
    ///     逻辑层经 `services` 无法访问 `WaveDirector.runtime`（协议无此入口）。
    ///  2. `DAT_005d4d74 & 0x40`：Debug 门控位域，exe 静态值 = 0（Constants Pending #7），
    ///     发售版正常路径恒关闭 → 重建**不迁移**（Constants 注释已注明仅保留 2 个 bool 位）。
    ///  3. 单例 `vtable + 0x28`：单例身份未定（疑 GameMode/暂停管理器），无协议入口。
    /// 结论：本步为纯守卫，不臆造字段；三项关闭条件待后续任务卡闭合。
    private func step5_shouldTraverseEntities(world: any EntityStore) -> Bool {
        _ = world
        return true
    }

    // MARK: 6 — 循环A（type=1 水果）[both]
    /// §1.2 步骤 6，每实体：
    ///  6.0 【T-025】先做抛物线积分（原「实体先推进再判定」），出界即回收 + 漏切计数
    ///  6.1 全局锁 / +0x2b>0 → 跳过
    ///  6.2 门控：+0x6f==0 && +0x41<=-1.0 && +0x34<=0 && (flags&1)==0 && +0x30!=0
    ///  6.3 isSliced==true →（a）+0xa5=1 开轨迹（b）(+0x33==0 || +0x32>3) && +0xc1==0 → +0x3c+=2
    ///      （c）清 +0x40/+0x44，写 +0xf8~+0x100（旧 slash 位）、+0x110=type、+0x104~+0x10c=位置
    ///  6.4 非炸弹：+0x21==2 或同实体 → 跳过窗口检查；否则 +0x180=entity、+0x174=0、
    ///      comboArr[+0x8c]=type、+0x90=(type!=0)、首发发 slashes/slashesScoring、+0x51=1、combo++；
    ///      combo>9 → +0x174=0.095；+0x54 -= combo*(0.5*rand/8+0.75)；vtable+0x44 回调
    ///  6.5 +0x1ba!=0 → +0x110 += specialBonus；+0x54 += -3.0*(0.5*rand/8+0.75)
    ///
    /// ── 实现边界（逐条与任务卡对齐）────────────────────────────────────────
    /// • 6.0 【T-025 新增】`FruitMotion.integrate` 放在循环体首行；返回 false（出界）→
    ///    `pendingDespawn` 收 id + `missedFruitCount += 1`，**不再进入切割判定**。
    /// • 6.1 全局锁 `DAT_00557390`：无协议入口 → 不实现 [待验证]。
    /// • 6.2 实现 `+0x6f`（isMarked，极性 [待验证]）、`+0x30`（isVisible）、`+0x2b`（fallDelay）
    ///    三项；**不实现** `+0x41 dropState <= -1.0` 与 `(flags&1)==0`
    ///   —— `Entity.dropState` 默认 0、`flags` 默认 0，照抄会让默认实体永不参与判定
    ///   （同 levelTier 的锁死模式），默认值闭合前不引入。
    /// • 6.4 的 `+0x21 == 2` 门控：该槽为实体字段（非 GameState），`Entity.stateSlot0`
    ///   语义 [M] 级未闭合 → 不臆造，改由 `ComboTracker.registerSlice` 的「同实体」分支
    ///   兜住主要重复计数路径。⚠️ 缺此门控时可能多计一次 combo（注释在 ComboTracker）。
    /// • 首发 `slashes` / `slashesScoring` 统计：**已由 ComboTracker 内部上报**
    ///   （其 `stats` 注入 = `services.stats`），本步不重复上报，避免双倍计数。
    /// • 计分基数：取 `entity.fruit?.score`（`FruitInfo.score ↔ +0x314`，§4.5 [macho H]）。
    ///   ⚠️ [待验证] `state.lastSliceScore`(+0x110) 被 `recordSliceHit` 写为「水果类型」，
    ///   与 blueprint 注释「水果类型 ID / 加成后的分值」存在双解；若后续确认 +0x110
    ///   才是计分基数，只需改此一处。
    /// • 6.5 特殊果：只读 `state.specialFruitBonusEnabled` 标记，实际加成由
    ///   `ComboTracker.registerSlice` 内部完成（`specialFruitBonus` 已在 App 注入 0），
    ///   本步不重复加分 [待验证]。
    ///
    /// ⚠️ [待验证] 6.3.2 的 `+0x3c += 2` 与 T-025 的漏切计数**共用同一字段**：
    ///    两处叠加会让 `missPenaltyCount` 在正常切中时也快速增长，
    ///    与 `GameHost` 新增的「>= 3 → finishGame」判定冲突（见 T-025 风险点 1）。
    private func step6_processFruits(dt: Float, state: inout GameState,
                                     world: inout any EntityStore, services: GameServices) {
        // T-025：值类型参数读一次到局部，避免闭包捕获 self。
        let tuning = motionTuning

        // 闭包内不访问 `world`（它是 inout，闭包内再访问会触发独占访问冲突），
        // 收集后统一 despawn。
        var pendingDespawn: [EntityID] = []
        var hitPositions: [SIMD3<Float>] = []
        var missedFruitCount: Int32 = 0

        world.forEach(kind: .fruit) { (entity: inout Entity) -> Bool in
            // 6.0【T-025】先运动、后判定（§1.2 循环A「实体先推进再参与判定」）。
            // 返回值 = 存活；false → 出界。
            guard FruitMotion.integrate(&entity, dt: dt, tuning: tuning) else {
                missedFruitCount &+= 1
                pendingDespawn.append(entity.id)
                return true
            }

            // 6.2 参与判定门控（子集，见方法注释）
            guard !entity.isMarked else { return true }      // +0x6f（极性 [待验证]）
            guard entity.isVisible else { return true }      // +0x30 != 0
            guard entity.fallDelay <= 0 else { return true } // +0x2b > 0 → 跳过

            // 6.3 两段式切割判定（SliceTesting 内部已做门控 + 形状相交 + 球体二次确认）
            guard services.slice.isSliced(game: state, entity: entity) else { return true }

            // 6.3.1 开轨迹（原 DAT_005d4d80 != 0 → +0xa5 = 1）
            state.slashTrail.begin()
            state.slashTrailActive = true

            let fruitType = entity.fruit?.typeID ?? Constants.EntityPool.emptySlot
            let entityPosition = entity.transform.position
            let previousSlashPosition = state.hitPoint

            // 6.3.3 清 +0x40/+0x44；+0xf8.. = 旧 slash 位；+0x110 = type；+0x104.. = 位置
            state.recordSliceHit(entityPosition: entityPosition,
                                 slashPosition: previousSlashPosition,
                                 fruitType: fruitType)

            // 6.3.2 (+0x33==0 || +0x32>3) && +0xc1==0 → +0x3c += 2
            //   +0x33 / +0x32 为实体状态槽（语义 [M] 未闭合），此处仅用 +0x33 的
            //   GameState 对应字段 stateSlot2 做「==0」判定的保守近似。
            if entity.stateSlot2 == 0 && !state.specialOrBombMarked {
                state.missPenaltyCount &+= 2
            }

            // 6.4 combo 计数（窗口/档位/节奏因子/首发统计均由 ComboTracker 内部处理）
            let previousVisualTier = ComboTier.visual(for: state.comboCount)
            let event = services.combo.registerSlice(fruitType: fruitType,
                                                     entity: entity.id,
                                                     state: &state,
                                                     rng: services.rng)

            if case let .incremented(_, visualTier, _)? = event {
                // 「档位升级」才播连击音效 / 特效（原 combo > 2 进入特效分支，§2.2）
                if visualTier > previousVisualTier {
                    services.audio.play("combo_\(visualTier)")      // 音效 key [待验证]
                    services.effects.emit(visualTier, at: entityPosition)
                }
                // 命中计分（基础分来源见方法注释）
                let baseScore = entity.fruit?.score ?? 0
                services.score.dispatch(score: baseScore,
                                        comboType: visualTier,
                                        isScoring: true,
                                        mode: Self.scoringDispatchMode)
            }

            // 6.5 特殊果加成：只做标记读取，实际加成在 ComboTracker 内部完成（不重复加分）。
            //     原 +0x1ba != 0 → +0x110 += DAT_004d90b4；+0x54 += -3.0*(...) [macho H]
            //     [待验证] 原版是否另有独立分支未被 ComboTracker 覆盖。

            pendingDespawn.append(entity.id)
            hitPositions.append(entityPosition)
            return true
        }

        // 6.3.4 / 6.3.5 回收 + 命中特效 + 音效（原 FUN_00033ca0/FUN_00035a08）
        for id in pendingDespawn {
            world.despawn(id)
        }

        // T-025：漏切惩罚（未切且出界的水果）。
        // ⚠️ [待验证] findings 无明确「漏切分支」直读；此处按「每漏一个水果 +1」近似，
        //    与 Classic 三振阈值（Constants.MissPenalty.threshold = 3）配套。
        if missedFruitCount > 0 {
            state.missPenaltyCount &+= missedFruitCount * Constants.MissPenalty.perMissedFruit
        }

        for position in hitPositions {
            services.effects.emit(Self.hitEffectKind, at: position)
            services.audio.play("splash")                            // 音效 key [待验证]
        }
    }

    // MARK: 7 — 循环B（type=2 炸弹）[macho]
    /// §1.2 循环B：炸弹遍历；近失 = isNearMiss（同一判定、球半径 ×1.5）仅在此循环调用，
    /// 结果写 +0xa4（isFaulted）并计入 near_miss 统计（L4：**不可**推广到水果）。
    ///
    /// T-025：炸弹同样逐帧做抛物线积分（与水果同一套 `FruitMotionTuning`）；
    /// 出界只回收、**不计**漏切（炸弹不是水果 [待验证]）。
    ///
    /// ⚠️ [待验证] 统计去重：原逻辑 `+0x7e && piVar13[0x26]==0 → 1` 的具体触发条件未闭合。
    ///    若每帧无条件 `stats.add(.nearMiss, 1)`，炸弹在空中停留期间会每帧累加（明显失真）。
    ///    本实现以 `isFaulted` 作为一次性门（首次近失才统计），语义待 Frida 核对。
    /// ⚠️ 炸弹「被切中」的爆炸结算在 §1.2 截断段，本卡**不实现**（见 step8 注释）。
    private func step7_processBombs(dt: Float, state: inout GameState,
                                    world: inout any EntityStore, services: GameServices) {
        let tuning = motionTuning
        var pendingDespawn: [EntityID] = []

        world.forEach(kind: .bomb) { (entity: inout Entity) -> Bool in
            // T-025：先运动、后判定。出界 → 回收（无漏切计数）。
            guard FruitMotion.integrate(&entity, dt: dt, tuning: tuning) else {
                pendingDespawn.append(entity.id)
                return true
            }

            guard services.slice.isNearMiss(game: state, entity: entity) else { return true }
            if !state.isFaulted {
                state.isFaulted = true                              // +0xa4
                services.stats.add(.nearMiss, 1)                    // L4：仅炸弹循环
            }
            return true
        }

        for id in pendingDespawn {
            world.despawn(id)
        }
    }

    // MARK: 8 — 循环C [待验证]
    /// §1.2 步骤 6 之后的循环C 在 findings 中截断，语义仅 [M] 级（L3 状态槽 0x21/0x26/0x33）。
    /// 本桩只占位，禁止在语义闭合前写入具体逻辑。
    ///
    /// [待验证] 三项已知但不可臆造的内容：
    ///  1. 循环C 的实体筛选 kind（原按 type 过滤，候选：coin=3 / slash=4 / blast=5）
    ///  2. 炸弹爆炸结算（`Bomb.explosionTimerSeconds` / `blastPushSpeedFactor` /
    ///     `Bomb.explodeAudioKey` 已闭合，但触发点与弹飞循环归属未定）
    ///  3. L3 状态槽 0x21/0x26/0x33 的状态机迁移
    /// 管线：待 T-01x 从 exe 侧 FUN_00bda510 的循环 C 段导出全量反编译后补齐。
    private func step8_processSpecials(dt: Float, state: inout GameState,
                                       world: inout any EntityStore, services: GameServices) {
        // 占位桩：语义未闭合，禁止写入逻辑。
    }

    // MARK: 9 — combo 结算 [both]
    /// §1.2 步骤 9：+0x174 推进（触发重置 0.1；>9 封顶 0.095）→ 到期结算 →
    /// clearComboTracking（comboArr memset 0xff）→ 分发 ComboEvent 到计分/统计/特效。
    ///
    /// 分工：窗口推进、统一清零、`combos` 统计均由 `ComboTracker.tickWindow` 内部完成
    /// （其 `stats` 注入 = `services.stats`），本步**只消费事件**做音效/特效分发，
    /// 避免统计重复计数。
    private func step9_settleCombo(dt: Float, state: inout GameState, services: GameServices) {
        let events = services.combo.tickWindow(dt: dt, state: &state)

        for event in events {
            switch event {
            case let .settled(count, _):
                // 档位对应 ComboTier（§2.3：visual = clamp(combo-3, 0, 2)）
                let tier = ComboTier.visual(for: count)
                services.audio.play("combo_settle_\(tier)")         // 音效 key [待验证]
                // 结算特效落点：最后一次命中点（原 +0x104）
                services.effects.emit(tier, at: state.hitPoint)
            case .started, .incremented, .specialBonusApplied:
                // tickWindow 按 §1.2 步骤 9 只产出 .settled；其余为防御性忽略
                break
            }
        }
    }

    // MARK: 10 — 子系统更新 [macho]
    /// §1.2 步骤 10：波次（services.waves.tick）→ 计分分发（services.score）→
    /// 统计（services.stats）→ 音效（services.audio）→ 特效（services.effects）。
    ///
    /// ⚠️ [待验证] 后四个子系统**没有逐帧 API**：
    ///  - `ScoreSystem.dispatch` 是事件驱动（由 step6/step9 调用），无帧级入口
    ///  - `StatsRecorder` 只有计数写入，原逐帧上报点（如 `all` / `score_without_beans`）
    ///    未从反编译定位
    ///  - `AudioPlaying` / `EffectEmitting` 由命中与结算事件驱动
    /// 故本步仅调用 `waves.tick`；其余调用点待各自任务卡闭合后回填。
    private func step10_updateSubsystems(dt: Float, state: inout GameState,
                                         world: inout any EntityStore, services: GameServices) {
        services.waves.tick(dt: dt, state: &state, world: world)
    }
}

final class SpawnMotionTests: XCTestCase {

    // MARK: - 1. 生成

    func testSpawnFruitCreatesEntityWithLaunchSpeedInRange() throws {
        let store = GenerationOrderStore()
        let rng = DeterministicTestRandom(seed: 0x5EED_1234)
        let tuning = FruitMotionTuning()
        let bridge = WaveSpawnBridge(store: store, rng: rng, tuning: tuning)

        bridge.spawnFruit(kind: Constants.EntityKindValue.fruit,
                          from: Spawner(),
                          wave: WaveInfo())

        XCTAssertEqual(store.aliveCount, 1)

        let id = try XCTUnwrap(firstID(in: store))
        let entity = try XCTUnwrap(store.entity(id))

        XCTAssertEqual(entity.kind, .fruit)
        XCTAssertEqual(entity.fruit?.typeID ?? -99, Constants.EntityKindValue.fruit)

        // 生成点在屏幕下缘
        XCTAssertEqual(entity.transform.position.y, tuning.offscreenBottomMargin, accuracy: 1e-5)
        // xy 平面内，z 恒 0（FruitMotion.launchVelocity 的约定）
        XCTAssertEqual(entity.transform.velocity.z, 0, accuracy: 1e-6)

        let speed = simd_length(SIMD2<Float>(entity.transform.velocity.x,
                                             entity.transform.velocity.y))
        XCTAssertGreaterThanOrEqual(speed, tuning.minLaunchSpeed - 1e-4)
        XCTAssertLessThanOrEqual(speed, tuning.maxLaunchSpeed + 1e-4)
    }

    func testTriggerExplosionSpawnsBombPlaceholder() throws {
        let store = GenerationOrderStore()
        let bridge = WaveSpawnBridge(store: store,
                                     rng: DeterministicTestRandom(seed: 0xB0MB_1),
                                     tuning: FruitMotionTuning())

        bridge.triggerExplosion(from: Spawner(), wave: WaveInfo())

        XCTAssertEqual(store.aliveCount, 1)
        let id = try XCTUnwrap(firstID(in: store))
        let entity = try XCTUnwrap(store.entity(id))
        XCTAssertEqual(entity.kind, .bomb)
    }

    // MARK: - 2. 抛物线积分（手算可复核）

    func testIntegrateFollowsParabolicMotion() throws {
        let store = GenerationOrderStore()
        let tuning = FruitMotionTuning()
        let bridge = WaveSpawnBridge(store: store,
                                     rng: DeterministicTestRandom(seed: 0x5EED_5678),
                                     tuning: tuning)

        bridge.spawnFruit(kind: Constants.EntityKindValue.fruit,
                          from: Spawner(),
                          wave: WaveInfo())

        let id = try XCTUnwrap(firstID(in: store))
        var entity = try XCTUnwrap(store.entity(id))

        let p0 = entity.transform.position
        let v0 = entity.transform.velocity
        let dt: Float = 1.0 / 60.0

        // 半隐式欧拉：v' = v + g*dt；p' = p + v'*dt
        XCTAssertTrue(FruitMotion.integrate(&entity, dt: dt, tuning: tuning))

        let v1 = entity.transform.velocity
        XCTAssertEqual(v1.y, v0.y + tuning.gravity * dt, accuracy: 1e-4)
        XCTAssertEqual(entity.transform.position.y, p0.y + v1.y * dt, accuracy: 1e-4)
        XCTAssertEqual(entity.transform.position.x, p0.x + v1.x * dt, accuracy: 1e-4)

        // 第二帧应在上升段继续上升（初速 14~22 / 60~120°，远大于单帧重力增量）
        let p1 = entity.transform.position
        XCTAssertTrue(FruitMotion.integrate(&entity, dt: dt, tuning: tuning))
        XCTAssertGreaterThan(entity.transform.position.y, p1.y)
    }

    // MARK: - 3. 出界回收

    func testOutOfBoundsFruitIsDespawnedByTick() throws {
        let store = GenerationOrderStore()
        let tuning = FruitMotionTuning()
        let bridge = WaveSpawnBridge(store: store,
                                     rng: DeterministicTestRandom(seed: 0x0FF_1),
                                     tuning: tuning)
        bridge.spawnFruit(kind: Constants.EntityKindValue.fruit,
                          from: Spawner(),
                          wave: WaveInfo())

        let id = try XCTUnwrap(firstID(in: store))

        // 推到底界外 + 向下速度 → 下一帧 integrate 必判死
        pushOutOfBounds(store, tuning: tuning)

        var world: any EntityStore = store
        var state = runnableState()
        let services = makeServices()
        let gameWorld = GameWorld(motionTuning: tuning)

        gameWorld.tick(dt: 1.0 / 60.0, state: &state, world: &world, services: services)

        XCTAssertNil(store.entity(id))
        XCTAssertEqual(store.aliveCount, 0)
    }

    // MARK: - 4. 漏切计数

    func testMissedFruitIncrementsPenaltyCount() throws {
        let store = GenerationOrderStore()
        let tuning = FruitMotionTuning()
        let bridge = WaveSpawnBridge(store: store,
                                     rng: DeterministicTestRandom(seed: 0x0FF_2),
                                     tuning: tuning)

        // 三个水果同时出界
        let spawnCount = 3
        for _ in 0..<spawnCount {
            bridge.spawnFruit(kind: Constants.EntityKindValue.fruit,
                              from: Spawner(),
                              wave: WaveInfo())
        }
        XCTAssertEqual(store.aliveCount, spawnCount)
        pushOutOfBounds(store, tuning: tuning)

        var world: any EntityStore = store
        var state = runnableState()
        let services = makeServices()
        let gameWorld = GameWorld(motionTuning: tuning)

        gameWorld.tick(dt: 1.0 / 60.0, state: &state, world: &world, services: services)

        XCTAssertEqual(state.missPenaltyCount,
                       Int32(spawnCount) * Constants.MissPenalty.perMissedFruit)
        XCTAssertEqual(store.aliveCount, 0)
    }

    func testBombOutOfBoundsDoesNotIncrementPenalty() throws {
        let store = GenerationOrderStore()
        let tuning = FruitMotionTuning()
        let bridge = WaveSpawnBridge(store: store,
                                     rng: DeterministicTestRandom(seed: 0x0FF_3),
                                     tuning: tuning)

        XCTAssertEqual(state.missPenaltyCount, 0)
        XCTAssertEqual(store.aliveCount, 0)
    }

    // MARK: - Helpers

    /// 取生成序第一个实体 id（forEach 返回 false 立即终止）。
    private func firstID(in store: GenerationOrderStore) -> EntityID? {
        var found: EntityID?
        store.forEach { entity in
            found = entity.id
            return false
        }
        return found
    }

    /// 把全部实体推到下界外并给向下速度（保证下一帧 integrate 返回 false）。
    private func pushOutOfBounds(_ store: GenerationOrderStore, tuning: FruitMotionTuning) {
        store.forEach { entity in
            entity.transform.position = SIMD3<Float>(0, tuning.offscreenBottomMargin - 1, 0)
            entity.transform.velocity = SIMD3<Float>(0, -10, 0)
            return true
        }
    }

    /// 可跑实体遍历的 GameState：levelTier 需 >= 4（否则 step4 门控跳过遍历）。
    private func runnableState() -> GameState {
        var state = GameState()
        state.levelTier = Constants.Simulation.lowLevelEffectCutoff
        state.sliceRadiusSquared = 225.0     // [待验证] 同 GameHost.startGame 的占位
        state.hitPoint = SIMD3<Float>(1, 1, 0)
        return state
    }

                     stats: StubStats(),
                     waves: StubWaves(),
                     audio: StubAudio(),

    }
}

/// 确定性 PRNG（SplitMix64）。见文件头「随机源说明」。
private final class DeterministicTestRandom: RandomSource {

    private var state: UInt64

    init(seed: UInt64) {
        self.state = seed == 0 ? 0x9E37_79B9_7F4A_7C15 : seed
    }

    func nextFloat() -> Float {
        Float(next() >> 40) * (1.0 / 16_777_216.0)
    }

    func nextInt(_ upperBound: Int32) -> Int32 {
        guard upperBound > 0 else { return 0 }
        return Int32((next() >> 33) % UInt64(upperBound))
    }

    func snapshot() -> UInt64 { state }

    func restore(_ s: UInt64) {
        state = s == 0 ? 0x9E37_79B9_7F4A_7C15 : s
    }

    private func next() -> UInt64 {
        state &+= 0x9E37_79B9_7F4A_7C15
        var z = state
        z = (z ^ (z >> 30)) &* 0xBF58_476D_1CE4_E5B9
        z = (z ^ (z >> 27)) &* 0x94D0_49BB_1331_11EB
        return z ^ (z >> 31)
    }
}

//
//  AppViewModel.swift
//  App
//
//  T-022：UI 会话状态机 + 每帧快照组装（FNPBUI 单向数据流的 App 侧实现）
//  T-025：快照补 lives（剩余机会，Classic 三振近似）
//
//  ┌────────────┐   handleFrame 末尾 refreshSnapshot()   ┌───────────────┐
//  │  GameHost  │ ────────────────────────────────────▶ │ AppViewModel  │
//  │  （逻辑层）│ ◀──────────────────────────────────── │  （UI 会话层）│
//  └────────────┘    send(UIIntent) → 流程方法           └───────────────┘
//                                                               │ snapshot / actions
//                                                               ▼
//                                                         FNPBUI 五屏
//
//  【持有关系（防环）】
//    GameHost ──强持有──▶ AppViewModel   （FNPBApp.swift: private(set) lazy var viewModel）
//    AppViewModel ──weak──▶ GameHost
//    ⇒ 两个方向只保留一条强引用。**不要**把 host 改成强引用（会与 lazy 持有形成保留环）。
//
//  【Swift 6 隔离】
//    全类 @MainActor。FNPBUI 的 UIActions 刻意不带隔离标注（见 GameSnapshot.swift 注释），
//    因此 actions 闭包内用 MainActor.assumeIsolated 回到主 actor —— 触发源必然是 SwiftUI
//    Button 的 action，运行在主线程。iOS 26 部署目标下该 API 可用。
//
//  【[待验证] 集中清单】见本回复末尾风险点 2/3/4/8/9；文件内每个猜测点都有同名标注。
//

import Foundation
import Observation

import FNPBCore
import FNPBUI

@Observable
@MainActor
final class AppViewModel: FNPBUIViewModel {

    // MARK: - 输出（视图唯一输入）

    private(set) var snapshot: GameSnapshot

    // MARK: - 会话状态

    /// 当前阶段。与 `snapshot.phase` 同步（refreshSnapshot 负责落盘）。
    private(set) var phase: UIPhase = .mainMenu

    /// 当前模式（结算摘要与最高分键都依赖它）。
    private(set) var mode: GameMode = .classic

    /// 音效开关（镜像 AudioEngine.masterVolume 的布尔语义，持久化见 GameHost.applySoundEnabled）。
    private(set) var soundEnabled: Bool

    /// 本局最大连击（ResultSummary 需要；ComboTracker 只保留当前值）。
    private(set) var maxComboSeen: Int = 0

    /// 本局结算摘要（仅在 phase == .results 时非 nil）。
    private(set) var result: ResultSummary?

    /// 逻辑层宿主。**weak**：宿主强持有本对象（FNPBApp.swift）。
    private(set) weak var host: GameHost?

    /// combo 窗口重置值（core_logic_notes §1.1：触发后重置 0.1，combo>9 时 0.095 封顶）。
    /// [待验证] Constants 中无对应命名常量，此处按笔记字面量落一个局部常量。
    private static let comboWindow: Float = 0.1

    // MARK: - 构造

    init(host: GameHost) {
        self.host = host
        self.soundEnabled = host.isSoundEnabled
        self.snapshot = GameSnapshot(
            phase: .mainMenu,
            mode: .classic,
            score: 0,
            bestScore: 0,
            // T-025：主菜单态先给满值，refreshSnapshot 起按 missPenaltyCount 实时计算。
            lives: Int(Constants.MissPenalty.threshold),
            elapsed: 0,      // [待验证] 本局计时未接线，v1 HUD 不显示
            pauseVisible: false,
            combo: ComboBanner(count: 0, text: Self.comboText(textTier: 1), visualTier: 0, opacity: 0),
            result: nil
        )
    }

    // MARK: - 意图入口（FNPBUIViewModel）

    /// 视图侧动作出口。
    ///
    /// [Kimi 修正 T-022 审查] UIActions 的 init 是无标签单闭包（见 GameSnapshot.swift）。
    var actions: UIActions {
        UIActions({ [weak self] intent in
            MainActor.assumeIsolated {
                self?.send(intent)
            }
        })
    }

    /// 全部 UIIntent 的落地实现（case 表以 GameSnapshot.swift 的 UIIntent 为准）。
    func send(_ intent: UIIntent) {
        switch intent {
        case .selectMode(let selected):
            startGame(mode: selected)

        case .showModeSelect:
            guard phase == .mainMenu else { return }
            phase = .modeSelect
            refreshSnapshot()

        case .pause:
            pauseGame()

        case .resume:
            resumeGame()

        case .restart:
            restartGame()

        case .quitToMenu:
            quitToMenu()

        case .toggleSound:
            toggleSound()

        case .openScreen:
            // v1 占位屏幕（extras/shop/about 等 15 个）无视图，静默忽略。
            break
        }
    }

    // MARK: - 流程（供 send 与 GameHost 调用）

    /// 开局。重置由 GameHost 负责（GameState/EntityStore/Score/Combo/Wave/DisplayLink）。
    func startGame(mode selected: GameMode) {
        guard let host else { return }
        self.mode = selected
        self.maxComboSeen = 0
        self.result = nil
        host.startGame(mode: selected)
        phase = .playing
        refreshSnapshot()
    }

    /// 暂停：停 DisplayLink（逻辑帧冻结），保留现场。
    func pauseGame() {
        guard phase == .playing, let host else { return }
        host.pauseGame()
        phase = .paused
        refreshSnapshot()
    }

    /// 继续：重启 DisplayLink。
    func resumeGame() {
        guard phase == .paused, let host else { return }
        phase = .playing
        host.resumeGame()
        refreshSnapshot()
    }

    /// 重开：全套状态重置（等价于以当前模式重新开局）。
    func restartGame() {
        guard let host else { return }
        maxComboSeen = 0
        result = nil
        host.restart()
        phase = .playing
        refreshSnapshot()
    }

    /// 回主菜单：停 DisplayLink 并持久化。
    func quitToMenu() {
        host?.quitToMenu()
        maxComboSeen = 0
        result = nil
        phase = .mainMenu
        refreshSnapshot()
    }

    /// 音效开关：切 AudioEngine.masterVolume 并写存档（收敛在 GameHost 内）。
    func toggleSound() {
        guard let host else { return }
        soundEnabled.toggle()
        host.applySoundEnabled(soundEnabled)
    }

    /// 结算入口。由 GameHost 在判定本局结束（core_logic_notes §1.2 步骤 7 的 `+0xa4` 失灵标志）
    /// 时调用。
    ///
    /// [待验证] 结束条件：`GameState.isFaulted`（炸弹近失置位）或
    /// `missPenaltyCount >= Constants.MissPenalty.threshold`（T-025 新增，Classic 三振近似）。
    /// 两条均未从反编译完全闭合，闭合后只需改 `GameHost.handleFrame` 的调用点。
    func finishGame() {
        guard phase == .playing || phase == .paused, let host else { return }

        let finalScore = Int(host.currentScore)
        let previousBest = Int(host.bestScore(for: mode))
        let best = Int(host.recordBestScore(host.currentScore, for: mode))

        let summary = ResultSummary(
            finalScore: finalScore,
            bestScore: best,
            isNewBest: finalScore > previousBest,
            maxCombo: maxComboSeen,
            fruitsSliced: Int(host.currentTotalFruitSliced),   // ← StatsRecorder 键 totalfruit
            mode: mode
        )

        result = summary
        phase = .results
        host.endGame()          // 停 DisplayLink：结算界面不再驱动逻辑
        refreshSnapshot()
    }

    // MARK: - 快照组装（GameHost.handleFrame 末尾调用）

    /// 用当前逻辑状态刷新 `snapshot`。
    ///
    /// 等值去抖：`GameSnapshot` 是 Equatable，值未变时不赋值 → 不触发 Observation 失效，
    /// 避免 120Hz 下 SwiftUI 每帧无条件重建视图树。
    func refreshSnapshot() {
        guard let host else { return }
        let state = host.currentState
        let comboCount = Int(state.comboCount)

        if comboCount > maxComboSeen {
            maxComboSeen = comboCount
        }

        var next = snapshot
        next.phase = phase
        next.mode = mode
        next.score = Int(host.currentScore)
        next.bestScore = Int(host.bestScore(for: mode))
        next.lives = lives(for: state)          // T-025
        next.combo = comboBanner(count: comboCount, windowTimer: state.comboWindowTimer)
        next.pauseVisible = (phase == .paused)
        if phase == .results {
            next.result = result
        }

        if next != snapshot {
            snapshot = next
        }
    }

    // MARK: - 私有

    /// T-025：剩余机会数。
    ///
    /// [待验证] 原版 Classic 的失败条件与 `+0x3c` 语义均未闭合：
    ///  - 本式按「阈值 − 惩罚计数」近似（3 − missPenaltyCount），负数钳到 0；
    ///  - ⚠️ 已知冲突：`GameWorld` 步骤 6.3.2 在**切中**分支里也执行 `+0x3c += 2`
    ///    （T-023 直译），与「漏切」共用同一字段 → 正常切中也会掉命。
    ///    若 Kimi 裁决 6.3.2 语义有误，只需改 GameWorld 一处，本式无需变动。
    private func lives(for state: GameState) -> Int {
        max(0, Int(Constants.MissPenalty.threshold) - Int(state.missPenaltyCount))
    }

    /// combo 横幅。
    ///
    /// 档位双源确认（DECISIONS T-004 / T-005）：
    ///   视觉档位 = clamp(combo − 3, 0, 2)；文本档位 = clamp(combo − 2, 1, 5)。
    private func comboBanner(count: Int, windowTimer: Float) -> ComboBanner {
        let visualTier = min(max(count - 3, 0), 2)
        let textTier = min(max(count - 2, 1), 5)

        // [待验证] 可见度方向：§1.2 步骤 9 记 `+0x174` 为「推进 → 到期结算」，
        // 故按「累加到 0.1 结算」取 1 − t/0.1。若为递减式，仅改此式。
        let opacity: Double = {
            guard count > 0 else { return 0 }
            let remaining = 1 - windowTimer / Self.comboWindow
            return Double(min(max(remaining, 0), 1))
        }()

        return ComboBanner(
            count: count,
            text: Self.comboText(textTier: textTier),
            visualTier: visualTier,
            opacity: opacity
        )
    }

    /// [待验证] 档位文案：原版 5 档文案在 `.uiscreen` 资源（本地化键）内，解析器尚未实现，
    /// 故统一占位 "COMBO"，不臆造原文。
    private static func comboText(textTier: Int) -> String {
        _ = textTier
        return "COMBO"
    }
}

import SwiftUI
import UIKit

import FNPBCore
import FNPBPlatform
import FNPBRender

// ============================================================================
// FNPBApp — SwiftUI 入口 + 装配层（蓝图 S1 收口 / S8 接线 / T-022 / T-024 / T-025）
//
// 数据流：
//   CADisplayLink(DisplayLinkDriver, 120Hz)
//     → FrameStamp
//     → FixedStepDriver(1/60 固定步长 + alpha 插值)
//     → GameWorld.tick(1/60)                ← 逻辑帧（含实体运动积分）
//     → AppViewModel.refreshSnapshot()      ← UI 帧（等值去抖后进 Observation）
//
//   渲染帧（独立于逻辑帧，T-024）：
//   MetalCanvasView.draw(in:) → GameRenderer.drawFrame()
//     → beginFrame → 读 currentState/currentEntities → submit → endFrame
//
//   实体生命周期（T-025）：
//   WaveDirector.runSpawnLoop → WaveSpawnSink.spawnFruit
//     → WaveSpawnBridge 建 EntityPrototype（抛物线初速）
//     → EntityStore.spawn
//     → GameWorld 步骤 6/7 FruitMotion.integrate 逐帧推进
//     → 出界回收；水果漏切 → state.missPenaltyCount
//
// 五屏闭环（T-022）：

// MARK: - 视图

struct ContentView: View {

    @ObservedObject var host: GameHost

    var body: some View {
        ZStack {
            Color.black.ignoresSafeArea()
            screenContent(viewModel: host.viewModel)
        }
        .onAppear { host.start() }
    }

    /// 顶层路由：只按 `snapshot.phase` 决定挂哪个屏幕。
    @ViewBuilder
    private func screenContent(viewModel vm: AppViewModel) -> some View {
        switch vm.snapshot.phase {

        case .mainMenu:
            MainMenuScreen(snapshot: vm.snapshot, actions: vm.actions)

        case .modeSelect:
            GameModeScreen(snapshot: vm.snapshot, actions: vm.actions)

        case .playing, .paused:
            gameplay(vm: vm)

        case .results:
            ResultsScreen(snapshot: vm.snapshot, actions: vm.actions)
        }
    }

    /// 局内层叠：渲染层 → 触摸层 → HUD →（暂停时）暂停浮层。
    @ViewBuilder
    private func gameplay(vm: AppViewModel) -> some View {
        ZStack {
            // T-024：真实 Metal 渲染层。
            // `MetalContext` 创建失败（无 GPU 环境）时降级回黑屏占位，App 不崩。
            if let renderer = host.renderDriver {
                renderer.canvasView
                    .ignoresSafeArea()
            } else {
                RenderAnchorView()
            }

            // 触摸层仅在真正游玩时挂载：paused 下若仍接收触摸，
            // began 会写入 GameState（slashTrailActive / hitPoint），逻辑态被污染。
            if vm.snapshot.phase == .playing {
                TouchSurfaceView(router: host.touchRouter)
                    .ignoresSafeArea()
            }

            InGameHUDScreen(snapshot: vm.snapshot, actions: vm.actions)

            if vm.snapshot.phase == .paused {
                PauseScreen(snapshot: vm.snapshot, actions: vm.actions)
            }
        }
    }
}

    func handleFrame(_ stamp: FrameStamp) {
        // R7：首帧 dt == 0，显式 prime 让 GameWorld 用兜底步长推进
        if stamp.isFirstFrame {
            gameWorld.prime()
        }

        // 固定步长推进；alpha 仅供渲染插值。
        // [Kimi 修正 T-022 审查] 签名以 TimeDriver.swift 为准：advance(delta:) /
        // GameWorld.tick(dt:state:world:services:) 且 world 为 inout。
        let alpha = fixedStep.advance(delta: stamp.clampedDelta()) { step in
            gameWorld.tick(dt: step, state: &state, world: &store, services: services)
        }
        renderInterpolationAlpha = alpha

        // 本局结束判定（T-025 并列两条）：
        //  ① +0xa4 失灵/结束标志（炸弹近失置位）[待验证 触发条件]
        //  ② 漏切惩罚计数达到阈值（Classic 三振近似）[待验证]
        // ⚠️ 与 T-023 遗留的「切中分支 +0x3c += 2」冲突：正常切两个水果即会触发 ②，
        //    见 T-025 风险点 1 —— 若 Kimi 裁决 6.3.2 语义有误，改 GameWorld 一处即可。
        if state.isFaulted || state.missPenaltyCount >= Constants.MissPenalty.threshold {
            viewModel.finishGame()
        }

        // T-024：渲染层不走「帧末通知」通路——`MTKView` 以 `isPaused = false` 自驱
        // （preferredFramesPerSecond = 120），每次 draw 时直接读
        // `currentState` / `currentEntities`。因此这里只刷新 UI 快照。
        // 选该通路的理由：注入文件里 `MetalCanvasView.onDraw` 是唯一帧入口，
        // 不存在「由逻辑层主动触发一次渲染」的 API。
        viewModel.refreshSnapshot()
    }

    func handleScenePhase(_ newPhase: ScenePhase) {
        switch newPhase {
        case .active:
            if shouldRunDisplayLink {
                displayLink.start()
            }
        case .inactive, .background:
            displayLink.stop()
            persist()
        @unknown default:
            break
        }
    }

        state.levelTier = Constants.Simulation.lowLevelEffectCutoff   // [待验证 写入点]

        store.clear()

        comboTracker.reset(state: &state)

        // 刻意留空

    // MARK: 存档

    /// 最高分读写。
    ///
    /// [待验证] `SaveData.highscores` 的容器类型与键名：任务卡只说明「沿用，键语义见

    func bestScore(for mode: GameMode) -> Int32 {
        saveData.highscores[mode.rawValue] ?? 0

    @discardableResult
    func recordBestScore(_ score: Int32, for mode: GameMode) -> Int32 {
        let key = mode.rawValue
        let previous = saveData.highscores[key] ?? 0
        if score > previous {
            saveData.highscores[key] = score
            persist()
        }
        return max(previous, score)
    }

    private func persist() {
        _ = saveStore.save(saveData)   // SaveStore.save(_:) -> Bool（Persistence.swift）
    }

    // MARK: 逻辑坐标

    /// 逻辑像素尺寸（兜底用）。TouchCoordinateMapper 要求 SIMD2<Float>。
    ///
    /// T-024：正常路径已改接 `RenderSurface.logicalSize`（逻辑点）；
    /// 本函数仅作为「渲染视图尚未布局」时的兜底（返回屏幕物理像素）。
    /// [待验证] 两套量纲混用是否符合原版归一化语义——闭合后应删除本兜底。
    static func currentLogicalSize() -> SIMD2<Float> {
        let scenes = UIApplication.shared.connectedScenes.compactMap { $0 as? UIWindowScene }
        let scene = scenes.first { $0.activationState == .foregroundActive } ?? scenes.first
        guard let screen = scene?.screen else {
            return SIMD2<Float>(1024, 768)          // 兜底（不应命中）
        }
        let bounds = screen.bounds
        return SIMD2<Float>(Float(bounds.width * screen.scale),
                            Float(bounds.height * screen.scale))
    }
}

// MARK: - 确定性 PRNG（T-016 原样保留）

/// 确定性 PRNG（SplitMix64）。回放一致性要求的 snapshot/restore 直接映射内部状态。
/// [待验证] 原版是 32 位定点随机（§5.3 DAT_005a7930 固定点随机），
/// 序列必然不同；此处只保证「可复现 + 均匀分布」，若需逐帧复刻原版请等 PRNG 任务卡。
final class SplitMix64Random: RandomSource {

    private var state: UInt64

    init(seed: UInt64) {
        self.state = seed == 0 ? 0x9E37_79B9_7F4A_7C15 : seed
    }

    func nextFloat() -> Float {
        // 取高 24 位 → [0, 1)
        Float(next() >> 40) * (1.0 / 16_777_216.0)
    }

    func nextInt(_ upperBound: Int32) -> Int32 {
        guard upperBound > 0 else { return 0 }
        return Int32((next() >> 33) % UInt64(upperBound))
    }

    func snapshot() -> UInt64 { state }

    func restore(_ s: UInt64) { state = s == 0 ? 0x9E37_79B9_7F4A_7C15 : s }

    private func next() -> UInt64 {
        state &+= 0x9E37_79B9_7F4A_7C15
        var z = state
        z = (z ^ (z >> 30)) &* 0xBF58_476D_1CE4_E5B9
        z = (z ^ (z >> 27)) &* 0x94D0_49BB_1331_11EB
        return z ^ (z >> 31)
    }
}
