## 1. 可加入 `Reader` 的辅助函数

```python
    def bytes_at(self, offset: int, length: int) -> bytes:
        if offset < 0 or length < 0 or offset + length > len(self.data):
            raise FormatError(
                f"bytes [{offset:#x}, {offset + length:#x}) outside file")
        return self.data[offset:offset + length]

    def f32s(self, offset: int, count: int, end: int) -> tuple[list[float], int]:
        if count < 0:
            raise FormatError(f"negative f32 count {count} at 0x{offset:x}")
        if offset < 0 or offset + count * 4 > end:
            raise FormatError(f"f32[{count}] outside range at 0x{offset:x}")
        values = [self.f32(offset + index * 4) for index in range(count)]
        return values, offset + count * 4
```

## 2. 静态属性解析相关完整函数

```python
_STATIC_PROPERTY_NAMES = frozenset({
    "texture",
    "text",
    "name",
    "parent_index",
    "position",
    "scale",
    "origin",
    "rotation",
    "width",
    "height",
    "colour",
    "color",
    "enabled",
    "inputEnabled",
})


def _read_static_string(reader: Reader, offset: int, end: int) -> tuple[str, int]:
    value, next_offset = reader.string(offset)
    if next_offset > end:
        raise FormatError(
            f"static string at 0x{offset:x} ends at 0x{next_offset:x}, beyond 0x{end:x}")
    return value, next_offset


def _read_static_u32(reader: Reader, offset: int, end: int) -> tuple[int, int]:
    if offset < 0 or offset + 4 > end:
        raise FormatError(f"static u32 outside range at 0x{offset:x}")
    return reader.u32(offset), offset + 4


def _read_static_i32(reader: Reader, offset: int, end: int) -> tuple[int, int]:
    if offset < 0 or offset + 4 > end:
        raise FormatError(f"static i32 outside range at 0x{offset:x}")
    return reader.i32(offset), offset + 4


def _looks_like_property_record_at(reader: Reader, offset: int, end: int) -> bool:
    """严格判断 offset 是否像一条静态属性记录头，不做值搜索。"""
    if offset < 0 or offset >= end:
        return False
    try:
        name, after_name = reader.string(offset)
    except FormatError:
        return False
    if not name or after_name + 8 > end:
        return False
    if any(ord(ch) < 0x20 or ord(ch) == 0x7f for ch in name):
        return False
    try:
        marker = reader.u32(after_name)
    except FormatError:
        return False
    # 已知记录/标记均满足高 16 位为 0x0007。
    return (marker >> 16) == 0x0007


def _looks_like_static_boundary(reader: Reader, offset: int, end: int) -> bool:
    """属性值之后必须是：UIAM 前边界、下一条属性记录，或一个分隔标记后接属性记录。"""
    if offset == end:
        return True
    if offset > end:
        return False
    if _looks_like_property_record_at(reader, offset, end):
        return True
    if offset + 4 <= end:
        try:
            marker = reader.u32(offset)
        except FormatError:
            return False
        if (marker >> 16) == 0x0007:
            return _looks_like_property_record_at(reader, offset + 4, end)
    return False


def _parse_static_value(
    reader: Reader,
    prop_name: str,
    offset: int,
    end: int,
) -> tuple[Any, int]:
    if prop_name in {"texture", "text", "name"}:
        return _read_static_string(reader, offset, end)

    if prop_name == "parent_index":
        return _read_static_i32(reader, offset, end)

    if prop_name in {"position", "scale", "origin"}:
        values, next_offset = reader.f32s(offset, 3, end)
        return values, next_offset

    if prop_name in {"width", "height"}:
        values, next_offset = reader.f32s(offset, 1, end)
        return values[0], next_offset

    if prop_name in {"enabled", "inputEnabled"}:
        return _read_static_u32(reader, offset, end)

    if prop_name == "rotation":
        # 证据不足以确定 rotation 是 vec3 还是单 float；用记录边界消歧，不搜索值。
        candidates = ((3, "vec3"), (1, "f32"))
    elif prop_name in {"colour", "color"}:
        # 同理，优先 RGBA vec4，再回退 RGB vec3。
        candidates = ((4, "vec4"), (3, "vec3"))
    else:
        raise FormatError(
            f"unsupported static property {prop_name!r} at value offset 0x{offset:x}")

    for count, label in candidates:
        try:
            values, next_offset = reader.f32s(offset, count, end)
        except FormatError:
            continue
        if _looks_like_static_boundary(reader, next_offset, end):
            return (values if count > 1 else values[0]), next_offset

    raise FormatError(
        f"cannot resolve static property {prop_name!r} at 0x{offset:x}: "
        f"none of candidate widths {[count for count, _ in candidates]} "
        f"lands on a static-record boundary before 0x{end:x}")


def parse_static_properties(
    reader: Reader,
    start: int,
    end: int,
    header_name: str,
    parent_index: int,
) -> tuple[dict[str, Any], list[dict[str, Any]]]:
    """
    解析 UICH 的 name/parent 之后、UIAM 之前的静态属性。
    name/parent_index 来自 UICH 头；若记录中重复出现，只允许与头部一致。
    """
    if start > end:
        raise FormatError(f"static property range start 0x{start:x} > end 0x{end:x}")

    properties: dict[str, Any] = {
        "name": header_name,
        "parent_index": parent_index,
    }
    records: list[dict[str, Any]] = []
    cursor = start

    while cursor < end:
        # 记录之间的 4 字节标记按原始信息保留；仅当其后确实跟随属性记录时才消费。
        if cursor + 4 <= end:
            maybe_marker = reader.u32(cursor)
            if (maybe_marker >> 16) == 0x0007:
                if _looks_like_property_record_at(reader, cursor + 4, end):
                    records.append({
                        "kind": "separator",
                        "offset": cursor,
                        "marker": maybe_marker,
                        "marker_hex": f"0x{maybe_marker:08x}",
                        "end": cursor + 4,
                    })
                    cursor += 4
                    continue

        record_start = cursor
        prop_name, after_name = reader.string(cursor)
        if after_name + 8 > end:
            raise FormatError(
                f"truncated static property header for {prop_name!r} at 0x{record_start:x}")

        marker = reader.u32(after_name)
        if (marker >> 16) != 0x0007:
            raise FormatError(
                f"unexpected static property marker 0x{marker:08x} "
                f"for {prop_name!r} at 0x{after_name:x}")

        # 这 4 字节语义仍未知，保留原始字节，不解释。
        flags_raw = reader.bytes_at(after_name + 4, 4)
        payload_offset = after_name + 8

        if prop_name not in _STATIC_PROPERTY_NAMES:
            raise FormatError(
                f"unsupported static property {prop_name!r} at 0x{record_start:x}; "
                f"marker=0x{marker:08x}, payload=0x{payload_offset:x}")

        value, next_cursor = _parse_static_value(
            reader, prop_name, payload_offset, end)

        if next_cursor > end:
            raise FormatError(
                f"static property {prop_name!r} at 0x{record_start:x} "
                f"ends at 0x{next_cursor:x}, beyond 0x{end:x}")

        if not _looks_like_static_boundary(reader, next_cursor, end):
            raise FormatError(
                f"static property {prop_name!r} at 0x{record_start:x} "
                f"ends at 0x{next_cursor:x}, not at a static-record boundary "
                f"before 0x{end:x}")

        if prop_name in properties:
            if prop_name == "name":
                if value != properties["name"]:
                    raise FormatError(
                        f"duplicate name property at 0x{record_start:x} "
                        f"differs from UICH header: {value!r} != {properties['name']!r}")
            elif prop_name == "parent_index":
                if value != properties["parent_index"]:
                    raise FormatError(
                        f"duplicate parent_index property at 0x{record_start:x} "
                        f"differs from UICH header: {value!r} != {properties['parent_index']!r}")
            else:
                raise FormatError(
                    f"duplicate static property {prop_name!r} at 0x{record_start:x}")
        else:
            properties[prop_name] = value

        records.append({
            "kind": "property",
            "offset": record_start,
            "name": prop_name,
            "record_marker": marker,
            "record_marker_hex": f"0x{marker:08x}",
            "record_flags_raw": flags_raw.hex(),
            "payload_offset": payload_offset,
            "value_end": next_cursor,
            "end": next_cursor,
            "value": value,
        })
        cursor = next_cursor

    if cursor != end:
        raise FormatError(
            f"static property parse ended at 0x{cursor:x}, expected 0x{end:x}")
    return properties, records
```

## 3. `parse_file` 完整替换版本

```python
def parse_file(path: Path) -> dict[str, Any]:
    data = path.read_bytes()
    reader = Reader(data)
    reader.require(0, b"UISF")
    if len(reader.positions(b"UISF")) != 1 or len(reader.positions(b"UISB")) != 1:
        raise FormatError("expected exactly one UISF and UISB")

    component_positions = reader.positions(b"UICH")
    if not component_positions:
        raise FormatError("no UICH components")

    components = []
    for index, component_offset in enumerate(component_positions):
        component_end = (
            component_positions[index + 1]
            if index + 1 < len(component_positions)
            else len(data)
        )

        reader.require(component_offset, b"UICH")
        name, after_name = reader.string(component_offset + 20)
        parent = reader.i32(after_name)
        static_start = after_name + 4
        if static_start > component_end:
            raise FormatError(
                f"UICH {name!r}: static_start 0x{static_start:x} "
                f"beyond component_end 0x{component_end:x}")

        animation_maps = reader.positions(b"UIAM", static_start, component_end)
        component_ends = reader.positions(b"UIME", static_start, component_end)
        if len(animation_maps) != 1 or len(component_ends) != 1:
            raise FormatError(f"UICH {name!r}: UIAM/UIME count mismatch")

        map_offset = animation_maps[0]
        uime_offset = component_ends[0]
        if not (static_start <= map_offset < uime_offset <= component_end):
            raise FormatError(
                f"UICH {name!r}: invalid ranges "
                f"static=0x{static_start:x} UIAM=0x{map_offset:x} "
                f"UIME=0x{uime_offset:x} component_end=0x{component_end:x}")

        properties, property_records = parse_static_properties(
            reader, static_start, map_offset, name, parent)

        reader.require(map_offset + 4, b"UIAB")
        animation_count = reader.u32(map_offset + 8)
        animation_positions = reader.positions(b"USAM", map_offset + 12, uime_offset)
        if len(animation_positions) != animation_count:
            raise FormatError(
                f"UICH {name!r}: UIAB={animation_count}, "
                f"USAM={len(animation_positions)}")

        animations = []
        for animation_index, animation_offset in enumerate(animation_positions):
            animation_end = (
                animation_positions[animation_index + 1]
                if animation_index + 1 < len(animation_positions)
                else uime_offset
            )
            animations.append(parse_animation(reader, animation_offset, animation_end))

        components.append({
            "name": name,
            "parent_index": parent,
            "component_offset": component_offset,
            "animation_map_offset": map_offset,
            "property_range": [static_start, map_offset],
            "properties": properties,
            "property_records": property_records,
            "animations": [
                {**asdict(animation), "end_time_ms": animation.end_time_ms}
                for animation in animations
            ],
        })

    for index, component in enumerate(components):
        parent = component["parent_index"]
        if parent < -1 or parent >= len(components) or parent == index:
            raise FormatError(
                f"invalid parent index {parent} for {component['name']!r}")

    return {
        "file": path.name,
        "bytes": len(data),
        "component_count": len(components),
        "animation_count": sum(
            len(component["animations"]) for component in components),
        "components": components,
    }
```

## 4. 验证函数与 `main` 建议代码

```python
def verify_layout(path: Path) -> None:
    report = parse_file(path)
    components = report["components"]

    icon = next((c for c in components if c["name"] == "iconSlot1"), None)
    if icon is None:
        raise AssertionError("iconSlot1 component not found")

    props = icon["properties"]

    expected_scalars = {
        "texture": "textures/adventureProgress/iconSlot.tex",
        "width": 83,
        "height": 83,
    }
    expected_vectors = {
        "scale": [1.0, 1.0, 1.0],
        "origin": [83.5, 0.0, 0.0],
        "position": [126.0, 42.5, 0.0],
    }

    for key, expected in expected_scalars.items():
        actual = props.get(key)
        assert actual is not None, (key, actual, expected)
        if isinstance(expected, str):
            assert actual == expected, (key, actual, expected)
        else:
            assert float(actual) == float(expected), (key, actual, expected)

    for key, expected in expected_vectors.items():
        actual = props.get(key)
        assert isinstance(actual, list), (key, actual, expected)
        assert len(actual) == len(expected), (key, actual, expected)
        assert all(
            float(a) == float(b)
            for a, b in zip(actual, expected)
        ), (key, actual, expected)

    # 每个 component 的静态属性记录必须严格位于 UIAM 之前。
    for component in components:
        map_offset = component["animation_map_offset"]
        prop_start, prop_end = component["property_range"]
        assert prop_end == map_offset, (
            component["name"], component["property_range"], map_offset)
        assert prop_start <= prop_end, (
            component["name"], component["property_range"], map_offset)

        for record in component.get("property_records", []):
            if record.get("kind") == "separator":
                assert record["end"] <= map_offset, (
                    component["name"], record, map_offset)
                continue
            assert record["offset"] < map_offset, (
                component["name"], record, map_offset)
            assert record["end"] <= map_offset, (
                component["name"], record, map_offset)

    print(f"verify-layout OK: {path} ({len(components)} components)")
```
