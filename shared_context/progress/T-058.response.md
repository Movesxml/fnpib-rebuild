import Foundation

public struct ChallengeMapRoundState: Equatable, Sendable {
    public let challengeName: String
    public let outcome: ChallengeOutcome?

    public init(challengeName: String, outcome: ChallengeOutcome? = nil) {
        self.challengeName = challengeName
        self.outcome = outcome
    }
}

public struct ChallengeMapInput: Equatable, Sendable {
    public let tierIndex: Int32
    public let isBonus: Bool
    public let currentRoundIndex: Int
    public let rounds: [ChallengeMapRoundState]
    public let actChange: Bool

    public init(tierIndex: Int32,
                isBonus: Bool,
                currentRoundIndex: Int,
                rounds: [ChallengeMapRoundState],
                actChange: Bool) {
        self.tierIndex = tierIndex
        self.isBonus = isBonus
        self.currentRoundIndex = currentRoundIndex
        self.rounds = rounds
        self.actChange = actChange
    }
}

public enum ChallengeMapCommand: Equatable, Sendable {
    case setVisible(component: String, visible: Bool)
    case playSound(String)
    case prepareRound(Int32)
    case startRound(Int32)
}

public struct ChallengeMapOutput: Equatable, Sendable {
    public var animations: [UIScreenAnimationOutput]
    public var commands: [ChallengeMapCommand]

    public init(animations: [UIScreenAnimationOutput],
                commands: [ChallengeMapCommand]) {
        self.animations = animations
        self.commands = commands
    }

    public static let empty = ChallengeMapOutput(animations: [], commands: [])
}

public enum ChallengeMapControllerError: Error, Equatable, Sendable {
    case alreadyBegan
    case notBegun
    case invalidRoundIndex(current: Int, count: Int)
    case invalidDelta(Float)
    case unsupportedActChange
    case unsupportedBonus
}

extension ChallengeMapControllerError: LocalizedError {
    public var errorDescription: String? {
        switch self {
        case .alreadyBegan:
            return "ChallengeMapController 已 begin"
        case .notBegun:
            return "ChallengeMapController 尚未 begin"
        case let .invalidRoundIndex(current, count):
            return "currentRoundIndex 越界：current=\(current), count=\(count)"
        case let .invalidDelta(delta):
            return "无效 deltaSeconds：\(delta)"
        case .unsupportedActChange:
            return "暂不支持 actChange Challenge 地图分支"
        case .unsupportedBonus:
            return "暂不支持 bonus Challenge 地图分支"
        }
    }
}

/// Release normal Challenge map controller for the first closed path:
/// state 0 start_game -> state 2 animate_in -> state 7 result display/selection
/// -> state 8 animate_out -> state 9 startRound request.
public struct ChallengeMapController: Sendable, Equatable {
    public private(set) var state: Int32 = -1
    public private(set) var totalElapsed: Float = 0
    public private(set) var stateElapsed: Float = 0
    public private(set) var selectedRoundIndex: Int?
    public private(set) var snapshot: ChallengeMapInput?

    private var startRoundIssued = false

    public init() {}

    @discardableResult
    public mutating func begin(input: ChallengeMapInput,
                               player: inout UIScreenAnimationPlayer) throws -> ChallengeMapOutput {
        guard snapshot == nil, state == -1 else {
            throw ChallengeMapControllerError.alreadyBegan
        }
        try Self.validate(input)

        player.reset()
        let animations = try player.playRecursively(component: "", animation: "start_game")

        snapshot = input
        state = 0
        totalElapsed = 0
        stateElapsed = 0
        selectedRoundIndex = nil
        startRoundIssued = false

        return ChallengeMapOutput(animations: animations, commands: [])
    }

    @discardableResult
    public mutating func update(deltaSeconds: Float,
                                input: ChallengeMapInput,
                                player: inout UIScreenAnimationPlayer) throws -> ChallengeMapOutput {
        guard snapshot != nil, state >= 0 else {
            throw ChallengeMapControllerError.notBegun
        }
        try Self.validate(input)
        guard deltaSeconds.isFinite, deltaSeconds >= 0 else {
            throw ChallengeMapControllerError.invalidDelta(deltaSeconds)
        }

        snapshot = input
        totalElapsed += deltaSeconds
        stateElapsed += deltaSeconds

        var animations = try player.advance(deltaSeconds: deltaSeconds)
        var commands: [ChallengeMapCommand] = []

        switch state {
        case 0:
            if try !player.isPlayingRecursively(component: "", animation: "start_game") {
                stateElapsed = 0
                state = 2
                animations.append(contentsOf: try player.playRecursively(component: "",
                                                                         animation: "animate_in"))
            }

        case 2:
            let animateInPlaying = try player.isPlayingRecursively(component: "",
                                                                   animation: "animate_in")
            if !animateInPlaying {
                let startGamePlaying = try player.isPlayingRecursively(component: "",
                                                                       animation: "start_game")
                if !startGamePlaying {
                    let actChangePlaying = try player.isPlayingRecursively(component: "",
                                                                           animation: "act_change")
                    if !actChangePlaying {
                        if input.actChange {
                            throw ChallengeMapControllerError.unsupportedActChange
                        }

                        totalElapsed = 0
                        let marked = try markResults(input: input, player: &player)
                        animations.append(contentsOf: marked.animations)
                        commands.append(contentsOf: marked.commands)
                        totalElapsed = 0
                        stateElapsed = 0
                        state = 7
                    }
                }
            }

        case 7:
            break

        case 8:
            if input.isBonus {
                throw ChallengeMapControllerError.unsupportedBonus
            }

            let outroPlaying = try player.isPlayingRecursively(component: "progressUIContainer",
                                                               animation: "animate_out")
            if !outroPlaying {
                state = 9
                if !startRoundIssued {
                    startRoundIssued = true
                    commands.append(.startRound(input.tierIndex))
                }
            }

        case 9:
            break

        default:
            break
        }

        return ChallengeMapOutput(animations: animations, commands: commands)
    }

    @discardableResult
    public mutating func selectRound(index: Int,
                                     player: inout UIScreenAnimationPlayer) throws -> ChallengeMapOutput {
        guard let snapshot, state == 7 else {
            return .empty
        }
        guard index >= 0, index < snapshot.rounds.count else {
            return .empty
        }
        guard snapshot.rounds[index].outcome == nil else {
            return .empty
        }

        var commands: [ChallengeMapCommand] = [.playSound("button-press")]
        let animations = try player.playRecursively(component: "", animation: "animate_out")
        commands.append(.prepareRound(Int32(index)))

        selectedRoundIndex = index
        totalElapsed = 0
        stateElapsed = 0
        state = 8

        return ChallengeMapOutput(animations: animations, commands: commands)
    }

    private func markResults(input: ChallengeMapInput,
                             player: inout UIScreenAnimationPlayer) throws -> ChallengeMapOutput {
        let currentIndex = input.currentRoundIndex
        var animations: [UIScreenAnimationOutput] = []
        var commands: [ChallengeMapCommand] = []

        let currentNode = "challengeContainer.spinner.adventureNode_\(currentIndex + 1)"
        animations.append(contentsOf: try player.play(component: currentNode,
                                                      animation: "animate_trigger"))

        if let outcome = input.rounds[currentIndex].outcome {
            if outcome == .pass || outcome == .fail {
                let isPass = outcome == .pass
                let sound = isPass ? "challenge-tick" : "gank"
                commands.append(.playSound(sound))

                let suffix = isPass ? "Pass" : "Fail"
                let resultNode = "challengeContainer.spinner.adventureNode\(suffix)_\(currentIndex + 1)"
                commands.append(.setVisible(component: resultNode, visible: true))
                animations.append(contentsOf: try player.play(component: resultNode,
                                                              animation: "animate_trigger"))

                let progressSuffix = isPass ? "pass" : "fail"
                let progressNode = "progressUIContainer.map.challenge_\(input.tierIndex + 1)_\(currentIndex + 1).\(progressSuffix)"
                commands.append(.setVisible(component: progressNode, visible: true))
                animations.append(contentsOf: try player.play(component: progressNode,
                                                              animation: "animate_trigger"))
            }
        }

        for (index, round) in input.rounds.enumerated() where round.outcome == nil {
            let node = "challengeContainer.spinner.adventureNode_\(index + 1)"
            commands.append(.setVisible(component: node, visible: true))
        }

        return ChallengeMapOutput(animations: animations, commands: commands)
    }

    private static func validate(_ input: ChallengeMapInput) throws {
        guard !input.rounds.isEmpty,
              input.currentRoundIndex >= 0,
              input.currentRoundIndex < input.rounds.count else {
            throw ChallengeMapControllerError.invalidRoundIndex(
                current: input.currentRoundIndex,
                count: input.rounds.count)
        }
    }
}

import Foundation
import XCTest
@testable import FNPBCore

final class ChallengeMapControllerTests: XCTestCase {
    func testStartupSequenceMarksPassAndEntersState7() throws {
        var player = try makeMapPlayer()
        let input = makeInput(currentRoundIndex: 0,
                              outcomes: [.pass, nil, nil, nil])
        var controller = ChallengeMapController()

        _ = try controller.begin(input: input, player: &player)
        XCTAssertEqual(controller.state, 0)
        XCTAssertTrue(try player.isPlayingRecursively(component: "",
                                                      animation: "start_game"))

        do {
            _ = try controller.begin(input: input, player: &player)
            XCTFail("begin 不应允许重复调用")
        } catch let error as ChallengeMapControllerError {
            XCTAssertEqual(error, .alreadyBegan)
        } catch {
            XCTFail("unexpected error: \(error)")
        }

        let outputs = try driveToState7(&controller, player: &player, input: input)
        XCTAssertEqual(controller.state, 7)

        let commands = outputs.flatMap { $0.commands }
        XCTAssertTrue(commands.contains(.playSound("challenge-tick")))
        XCTAssertTrue(commands.contains(
            .setVisible(component: "challengeContainer.spinner.adventureNodePass_1",
                        visible: true)))
        XCTAssertTrue(commands.contains(
            .setVisible(component: "progressUIContainer.map.challenge_1_1.pass",
                        visible: true)))
        XCTAssertTrue(commands.contains(
            .setVisible(component: "challengeContainer.spinner.adventureNode_2",
                        visible: true)))
        XCTAssertFalse(commands.contains(
            .setVisible(component: "challengeContainer.spinner.adventureNode_1",
                        visible: true)))
    }

    func testFailResultMarkingUsesFailResources() throws {
        var player = try makeMapPlayer()
        let input = makeInput(currentRoundIndex: 0,
                              outcomes: [.fail, nil, nil, nil])
        var controller = ChallengeMapController()

        _ = try controller.begin(input: input, player: &player)
        let outputs = try driveToState7(&controller, player: &player, input: input)
        XCTAssertEqual(controller.state, 7)

        let commands = outputs.flatMap { $0.commands }
        XCTAssertTrue(commands.contains(.playSound("gank")))
        XCTAssertTrue(commands.contains(
            .setVisible(component: "challengeContainer.spinner.adventureNodeFail_1",
                        visible: true)))
        XCTAssertTrue(commands.contains(
            .setVisible(component: "progressUIContainer.map.challenge_1_1.fail",
                        visible: true)))
        XCTAssertFalse(commands.contains(
            .setVisible(component: "challengeContainer.spinner.adventureNode_1",
                        visible: true)))
    }

    func testIllegalSelectionsAreIdempotentAndPendingIsNotNil() throws {
        var player = try makeMapPlayer()
        let input = makeInput(currentRoundIndex: 0,
                              outcomes: [.pending, nil, nil, nil])
        var controller = ChallengeMapController()

        _ = try controller.begin(input: input, player: &player)
        let outputs = try driveToState7(&controller, player: &player, input: input)
        XCTAssertEqual(controller.state, 7)

        let commands = outputs.flatMap { $0.commands }
        XCTAssertFalse(commands.contains(.playSound("challenge-tick")))
        XCTAssertFalse(commands.contains(.playSound("gank")))
        XCTAssertFalse(commands.contains(
            .setVisible(component: "challengeContainer.spinner.adventureNode_1",
                        visible: true)))

        XCTAssertEqual(try controller.selectRound(index: -1, player: &player), .empty)
        XCTAssertEqual(try controller.selectRound(index: 99, player: &player), .empty)
        XCTAssertEqual(try controller.selectRound(index: 0, player: &player), .empty)
        XCTAssertEqual(controller.state, 7)
    }

    func testLegalSelectionPreparesRoundAndOutroEmitsStartRoundOnce() throws {
        var player = try makeMapPlayer()
        let input = makeInput(currentRoundIndex: 0,
                              outcomes: [nil, nil, nil, nil],
                              tierIndex: 2)
        var controller = ChallengeMapController()

        _ = try controller.begin(input: input, player: &player)
        _ = try driveToState7(&controller, player: &player, input: input)
        XCTAssertEqual(controller.state, 7)

        let selected = try controller.selectRound(index: 1, player: &player)
        XCTAssertEqual(controller.state, 8)
        XCTAssertEqual(controller.selectedRoundIndex, 1)
        XCTAssertTrue(selected.commands.contains(.playSound("button-press")))
        XCTAssertTrue(selected.commands.contains(.prepareRound(1)))

        XCTAssertEqual(try controller.selectRound(index: 2, player: &player), .empty)
        XCTAssertEqual(controller.state, 8)

        var transitionOutputs: [ChallengeMapOutput] = []
        for _ in 0..<8 {
            let output = try controller.update(deltaSeconds: 10,
                                              input: input,
                                              player: &player)
            transitionOutputs.append(output)
            if controller.state == 9 {
                break
            }
        }

        XCTAssertEqual(controller.state, 9)

        let transitionCommands = transitionOutputs.flatMap { $0.commands }
        let startRoundCommands = transitionCommands.filter { command in
            if case .startRound = command {
                return true
            }
            return false
        }
        XCTAssertEqual(startRoundCommands, [.startRound(2)])

        let afterState9 = try controller.update(deltaSeconds: 10,
                                                input: input,
                                                player: &player)
        XCTAssertFalse(afterState9.commands.contains { command in
            if case .startRound = command {
                return true
            }
            return false
        })
        XCTAssertEqual(controller.state, 9)
    }

    func testActChangeIsExplicitlyUnsupported() throws {
        var player = try makeMapPlayer()
        let input = makeInput(actChange: true)
        var controller = ChallengeMapController()

        _ = try controller.begin(input: input, player: &player)

        do {
            for _ in 0..<8 {
                _ = try controller.update(deltaSeconds: 10,
                                          input: input,
                                          player: &player)
                if controller.state == 7 {
                    break
                }
            }
            XCTFail("actChange=true 应显式失败")
        } catch let error as ChallengeMapControllerError {
            XCTAssertEqual(error, .unsupportedActChange)
        } catch {
            XCTFail("unexpected error: \(error)")
        }
    }

    func testBonusIsExplicitlyUnsupportedAtState8() throws {
        var player = try makeMapPlayer()
        let input = makeInput(isBonus: true)
        var controller = ChallengeMapController()

        _ = try controller.begin(input: input, player: &player)
        _ = try driveToState7(&controller, player: &player, input: input)
        XCTAssertEqual(controller.state, 7)

        _ = try controller.selectRound(index: 0, player: &player)
        XCTAssertEqual(controller.state, 8)

        do {
            _ = try controller.update(deltaSeconds: 10,
                                      input: input,
                                      player: &player)
            XCTFail("isBonus=true 应显式失败")
        } catch let error as ChallengeMapControllerError {
            XCTAssertEqual(error, .unsupportedBonus)
        } catch {
            XCTFail("unexpected error: \(error)")
        }
    }

    @discardableResult
    private func driveToState7(_ controller: inout ChallengeMapController,
                               player: inout UIScreenAnimationPlayer,
                               input: ChallengeMapInput,
                               file: StaticString = #filePath,
                               line: UInt = #line) throws -> [ChallengeMapOutput] {
        var outputs: [ChallengeMapOutput] = []
        for _ in 0..<8 {
            if controller.state == 7 {
                return outputs
            }
            outputs.append(try controller.update(deltaSeconds: 10,
                                                 input: input,
                                                 player: &player))
        }
        XCTAssertEqual(controller.state, 7, file: file, line: line)
        return outputs
    }

    private func makeInput(currentRoundIndex: Int = 0,
                           outcomes: [ChallengeOutcome?] = [nil, nil, nil, nil],
                           tierIndex: Int32 = 0,
                           isBonus: Bool = false,
                           actChange: Bool = false) -> ChallengeMapInput {
        let rounds = outcomes.enumerated().map {
            ChallengeMapRoundState(challengeName: "round\($0.offset + 1)",
                                   outcome: $0.element)
        }
        return ChallengeMapInput(tierIndex: tierIndex,
                                 isBonus: isBonus,
                                 currentRoundIndex: currentRoundIndex,
                                 rounds: rounds,
                                 actChange: actChange)
    }

    private func makeMapPlayer() throws -> UIScreenAnimationPlayer {
        let url = try challengeMapURL()
        let data = try Data(contentsOf: url)
        return try UIScreenAnimationPlayer(data: data)
    }

    private func challengeMapURL() throws -> URL {
        var candidates: [URL] = []

        if let bundled = Bundle(for: ChallengeMapControllerTests.self)
            .url(forResource: "challengeMapAnimations", withExtension: "json") {
            candidates.append(bundled)
        }

        let testFile = URL(fileURLWithPath: #filePath)
        let testDirectory = testFile.deletingLastPathComponent()
        let repositoryRoot = testDirectory
            .deletingLastPathComponent()
            .deletingLastPathComponent()

        candidates.append(repositoryRoot
            .appendingPathComponent("Content/challengeMapAnimations.json"))
        candidates.append(repositoryRoot
            .appendingPathComponent("Sources/FNPBCore/Content/challengeMapAnimations.json"))
        candidates.append(repositoryRoot
            .appendingPathComponent("Sources/FNPBCore/Content/Challenge/challengeMapAnimations.json"))
        candidates.append(repositoryRoot
            .appendingPathComponent("Sources/FNPBCore/Resources/challengeMapAnimations.json"))

        let currentDirectory = URL(fileURLWithPath: FileManager.default.currentDirectoryPath)
        candidates.append(currentDirectory
            .appendingPathComponent("Content/challengeMapAnimations.json"))
        candidates.append(currentDirectory
            .appendingPathComponent("Sources/FNPBCore/Content/challengeMapAnimations.json"))

        for candidate in candidates where FileManager.default.fileExists(atPath: candidate.path) {
            return candidate
        }

        throw NSError(domain: "ChallengeMapControllerTests",
                      code: 1,
                      userInfo: [
                        NSLocalizedDescriptionKey:
                            "challengeMapAnimations.json not found; tried \(candidates.map(\.path))"
                      ])
    }
}
