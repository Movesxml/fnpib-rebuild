## 1. 新增文件：`Sources/FNPBUI/ChallengeMapLayoutSnapshot.swift`

```swift
//
//  ChallengeMapLayoutSnapshot.swift
//  FNPBUI
//
//  挑战地图 UIScreen 组件的只读值投影。
//
//  本文件不 import FNPBCore。App 层负责把 Core 的
//  UIScreenLayoutProjection 映射成这里的纯值类型；UI 只消费字段，
//  不自行推导布局、不猜纹理方向、不计算根缩放。
//

import Foundation

/// 逻辑空间三维点。
public struct ChallengeMapLayoutPoint: Sendable, Equatable {
    public var x: Float
    public var y: Float
    public var z: Float

    public init(x: Float, y: Float, z: Float) {
        self.x = x
        self.y = y
        self.z = z
    }
}

/// 组件四个世界角点，命名与原始投影的 (minX,minY) … (minX,maxY) 对应。
public struct ChallengeMapLayoutCorners: Sendable, Equatable {
    public var minimumXMinimumY: ChallengeMapLayoutPoint
    public var maximumXMinimumY: ChallengeMapLayoutPoint
    public var maximumXMaximumY: ChallengeMapLayoutPoint
    public var minimumXMaximumY: ChallengeMapLayoutPoint

    public init(minimumXMinimumY: ChallengeMapLayoutPoint,
                maximumXMinimumY: ChallengeMapLayoutPoint,
                maximumXMaximumY: ChallengeMapLayoutPoint,
                minimumXMaximumY: ChallengeMapLayoutPoint) {
        self.minimumXMinimumY = minimumXMinimumY
        self.maximumXMinimumY = maximumXMinimumY
        self.maximumXMaximumY = maximumXMaximumY
        self.minimumXMaximumY = minimumXMaximumY
    }
}

/// RGBA 四字节，保持原始字节序；不预乘、不归一化。
public struct ChallengeMapLayoutColor: Sendable, Equatable {
    public var red: UInt8
    public var green: UInt8
    public var blue: UInt8
    public var alpha: UInt8

    public init(red: UInt8, green: UInt8, blue: UInt8, alpha: UInt8) {
        self.red = red
        self.green = green
        self.blue = blue
        self.alpha = alpha
    }
}

/// 4×4 列主序矩阵。
///
/// 固定保存四个 `SIMD4<Float>` 列，因此恰好包含 16 个 Float；
/// `elements` 计算属性按列主序把这 16 个 Float 暴露给 UI。
/// 这样不需要在 UI 层对 `[Float]` 做越界下标操作。
public struct ChallengeMapLayoutMatrix4: Sendable, Equatable {
    public var column0: SIMD4<Float>
    public var column1: SIMD4<Float>
    public var column2: SIMD4<Float>
    public var column3: SIMD4<Float>

    public init(column0: SIMD4<Float>,
                column1: SIMD4<Float>,
                column2: SIMD4<Float>,
                column3: SIMD4<Float>) {
        self.column0 = column0
        self.column1 = column1
        self.column2 = column2
        self.column3 = column3
    }

    /// 从 Core 投影得到的 16 元素数组构造。
    /// 长度不匹配时返回 nil，由 App 层转为明确诊断，而不是崩溃。
    public init?(elements: [Float]) {
        guard elements.count == 16 else { return nil }
        self.init(
            column0: SIMD4(elements[0], elements[1], elements[2], elements[3]),
            column1: SIMD4(elements[4], elements[5], elements[6], elements[7]),
            column2: SIMD4(elements[8], elements[9], elements[10], elements[11]),
            column3: SIMD4(elements[12], elements[13], elements[14], elements[15]))
    }

    /// 列主序 16 个 Float。
    public var elements: [Float] {
        [
            column0.x, column0.y, column0.z, column0.w,
            column1.x, column1.y, column1.z, column1.w,
            column2.x, column2.y, column2.z, column2.w,
            column3.x, column3.y, column3.z, column3.w,
        ]
    }
}

/// 挑战地图单个 UIScreen 组件的 UI 侧投影。
///
/// `texCoordPos` / `texCoordSize` 原样保留 Core 的 UV 位置与尺寸，
/// 包括可能为负的尺寸；UI 不在本层反转纹理方向。
public struct ChallengeMapLayoutComponent: Sendable, Equatable, Identifiable {
    public var name: String
    public var index: Int
    public var parentIndex: Int?
    /// 16 个 Float 的列主序世界矩阵。
    public var worldMatrix: ChallengeMapLayoutMatrix4
    public var worldCorners: ChallengeMapLayoutCorners
    public var width: Float
    public var height: Float
    public var effectiveOrigin: ChallengeMapLayoutPoint
    public var localRGBA: ChallengeMapLayoutColor
    public var absoluteRGBA: ChallengeMapLayoutColor
    public var enabled: Bool
    public var screenVisibility: Bool
    /// 原始资源路径，不在此层解析或改写。
    public var texturePath: String?
    public var texCoordPos: SIMD2<Float>
    public var texCoordSize: SIMD2<Float>
    /// ChallengeMapController 下发的图标索引；没有对应图标时为 nil。
    public var iconIndex: Int32?

    public var id: Int { index }

    public init(name: String,
                index: Int,
                parentIndex: Int?,
                worldMatrix: ChallengeMapLayoutMatrix4,
                worldCorners: ChallengeMapLayoutCorners,
                width: Float,
                height: Float,
                effectiveOrigin: ChallengeMapLayoutPoint,
                localRGBA: ChallengeMapLayoutColor,
                absoluteRGBA: ChallengeMapLayoutColor,
                enabled: Bool,
                screenVisibility: Bool,
                texturePath: String?,
                texCoordPos: SIMD2<Float>,
                texCoordSize: SIMD2<Float>,
                iconIndex: Int32?) {
        self.name = name
        self.index = index
        self.parentIndex = parentIndex
        self.worldMatrix = worldMatrix
        self.worldCorners = worldCorners
        self.width = width
        self.height = height
        self.effectiveOrigin = effectiveOrigin
        self.localRGBA = localRGBA
        self.absoluteRGBA = absoluteRGBA
        self.enabled = enabled
        self.screenVisibility = screenVisibility
        self.texturePath = texturePath
        self.texCoordPos = texCoordPos
        self.texCoordSize = texCoordSize
        self.iconIndex = iconIndex
    }
}
```

```swift
public struct ChallengeMapHUDState: Sendable, Equatable {
    public var controllerState: Int
    public var rounds: [ChallengeMapRoundHUDState]
    /// 当前挑战地图 UIScreen 的组件投影。
    ///
    /// - 无地图或地图播放器不可用时由 App 层传 []；

    /// - 不复用上一帧成功组件。
    public var components: [ChallengeMapLayoutComponent]

    public init(controllerState: Int,
                rounds: [ChallengeMapRoundHUDState],
                components: [ChallengeMapLayoutComponent] = []) {
        self.controllerState = controllerState
        self.rounds = rounds
        self.components = components
    }

    public var canSelect: Bool { controllerState == 7 }
}
```

`components` 带默认值 `[]`，现有 `ChallengeMapHUDState(controllerState:rounds:)` 调用保持源码兼容。

```swift
    var challengeHUD: ChallengeHUDState? {
        guard currentMode == .challenge, let round = currentChallengeRound else { return nil }
        let runtime = gameWorld.challenge
        let controller = challengeSettlement

        var mapState: ChallengeMapHUDState?

            bonusState = ChallengeBonusHUDState(

        let errorMessage: String?

        return ChallengeHUDState(
            challengeName: round.challenge.name,
            roundNumber: round.roundIndex + 1,

            resultVisible: controller?.normalResultVisible ?? false,
            resultFinished: controller?.isNormalResultFinished ?? false,
            canAdvance: challengeAssembler?.plan.outcomes.contains(where: {
                $0 == nil
            }) ?? false,
            map: mapState,
            bonus: bonusState,
            errorMessage: errorMessage)
```

同一 `GameHost` 文件中新增以下私有静态 helper（放在 `challengeHUD` 附近即可）：

```swift
    /// 把 Core 的 UIScreenLayoutProjection 映射为 FNPBUI 的纯值组件。
    ///
    /// Core 的 UIScreenMatrix4 与 UIScreenTexCoord 构造时已保证元素数量，
    /// 这里仍显式校验，避免类型不变量被破坏时发生下标越界或静默错误。
    private static func makeChallengeMapComponents(
        from projection: UIScreenLayoutProjection,
        iconTargets: [String: Int32]
    ) throws -> [ChallengeMapLayoutComponent] {
        var result: [ChallengeMapLayoutComponent] = []
        result.reserveCapacity(projection.components.count)

        for component in projection.components {
            guard let worldMatrix = ChallengeMapLayoutMatrix4(
                elements: component.worldMatrix.elements) else {
                throw ChallengeHostError(
                    "组件 \(component.name) 的 worldMatrix 不是 16 个元素")
            }
            guard component.texCoord.position.count == 2 else {
                throw ChallengeHostError(
                    "组件 \(component.name) 的 texCoordPos 不是 2 个元素")
            }
            guard component.texCoord.size.count == 2 else {
                throw ChallengeHostError(
                    "组件 \(component.name) 的 texCoordSize 不是 2 个元素")
            }

            let corners = ChallengeMapLayoutCorners(
                minimumXMinimumY: ChallengeMapLayoutPoint(
                    x: component.worldCorners.minimumXMinimumY.x,
                    y: component.worldCorners.minimumXMinimumY.y,
                    z: component.worldCorners.minimumXMinimumY.z),
                maximumXMinimumY: ChallengeMapLayoutPoint(
                    x: component.worldCorners.maximumXMinimumY.x,
                    y: component.worldCorners.maximumXMinimumY.y,
                    z: component.worldCorners.maximumXMinimumY.z),
                maximumXMaximumY: ChallengeMapLayoutPoint(
                    x: component.worldCorners.maximumXMaximumY.x,
                    y: component.worldCorners.maximumXMaximumY.y,
                    z: component.worldCorners.maximumXMaximumY.z),
                minimumXMaximumY: ChallengeMapLayoutPoint(
                    x: component.worldCorners.minimumXMaximumY.x,
                    y: component.worldCorners.minimumXMaximumY.y,
                    z: component.worldCorners.minimumXMaximumY.z))

            result.append(ChallengeMapLayoutComponent(
                name: component.name,
                index: component.index,
                parentIndex: component.parentIndex,
                worldMatrix: worldMatrix,
                worldCorners: corners,
                width: component.width,
                height: component.height,
                effectiveOrigin: ChallengeMapLayoutPoint(
                    x: component.effectiveOrigin.x,
                    y: component.effectiveOrigin.y,
                    z: component.effectiveOrigin.z),
                localRGBA: ChallengeMapLayoutColor(
                    red: component.localRGBA.red,
                    green: component.localRGBA.green,
                    blue: component.localRGBA.blue,
                    alpha: component.localRGBA.alpha),
                absoluteRGBA: ChallengeMapLayoutColor(
                    red: component.absoluteRGBA.red,
                    green: component.absoluteRGBA.green,
                    blue: component.absoluteRGBA.blue,
                    alpha: component.absoluteRGBA.alpha),
                enabled: component.enabled,
                screenVisibility: component.screenVisibility,
                texturePath: component.texturePath,
                texCoordPos: SIMD2<Float>(
                    component.texCoord.position[0],
                    component.texCoord.position[1]),
                texCoordSize: SIMD2<Float>(
                    component.texCoord.size[0],
                    component.texCoord.size[1]),
                iconIndex: iconTargets[component.name]))
        }

        return result
    }
```

## 5. 诊断与失败行为说明

- 投影成功：`UIScreenLayoutProjection.project(challengeMapAnimationPlayer)` 使用其固定的逻辑视口 `480×320`；根缩放仍由 UI 层以后处理，不在本次投影中应用。
- 无地图（`challengeMap == nil` 或 `map.snapshot == nil`）：`mapState` 为 `nil`。
- 有地图但无播放器：`components` 保持空数组 `[]`，不会返回陈旧组件。
- 投影抛出错误，或 Core 类型不变量异常（矩阵不是 16 元素、UV 不是 2 元素）：`challengeHUD` 捕获并构造 `mapProjectionError`，最终写入 `ChallengeHUDState.errorMessage`；若同时存在 `challengeConfigurationError`，两者用分号拼接，不遮蔽原状态机错误。
- 投影失败不调用 `rejectChallenge`、不结束游戏、不改变地图 `state`、回合选择或任何动画推进逻辑；它只影响本帧 `ChallengeHUDState.map.components` 与错误文本。
- 失败时不会索引越界：矩阵与 UV 在映射 helper 中先校验，所有 Core 数组访问都先确认元素数量。
