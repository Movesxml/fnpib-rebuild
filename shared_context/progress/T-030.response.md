## 设计说明

按 T-030 任务卡，仅改动四份文件 + 新增一份 App 侧资源加载辅助：

1. **新增 `App/OriginalContentBundle.swift`**：封装「Bundle.main 读取 `fruitList.json` / `originalWaveList.json` → `OriginalContentLoader.load`」，返回 `Result<OriginalContent, LoadFailure>`（全部 Sendable），错误带资源名/解析原文，方便诊断。选择新增文件而非塞进 `FNPBApp.swift`，是因为资源定位/解码与装配宿主解耦，且任务卡允许“新增 App 侧资源加载辅助”。
2. **`FNPBApp.swift`**：`GameHost` 增加 `originalContent` / `contentLoadError` 存储；init 期加载内容（失败不崩、不占位），仅成功时挂 `spawnSink`；`startGame` / `restart` 改 `@discardableResult -> Bool`，用真实 `content.waves.waves` + `content.fruits.fruits` 调 `beginGame(waves:fruits:)`，删除 `PlaceholderWave` 波次与常量引用；`ensureContent()` 支持失败后重试加载但绝不进入假开局。
3. **`AppViewModel.swift`**：`startGame` / `restartGame` 先调宿主，仅宿主返回 true 才切 `.playing`；失败保持原界面（含 `results` 重开失败时不清空摘要），并暴露 `contentErrorDescription` 供诊断。
4. **`WaveSpawnBridge.swift`**：init 接收 `fruits: [FruitInfo]` + `bomb: BombInfo` 并强持有（含 typeID 索引字典）；`spawnFruit` 按 typeID 精确取元数据，未知 ID 直接不生成（无水果 0 兜底）；`spawnBomb` 用加载后的 `BombInfo` 构造 scale/radius；删除 `maxLiveFruitCount` 与两处 24 上限；生成几何（随机 X、初速、自旋、重力）原样保留并标 `[待验证]`。
5. **`SpawnMotionTests.swift`**：改用真实 apple（typeID 0 / scale 60 / collision 5）与 watermelon（typeID 3）表、BombInfo(55/35)；新增元数据精确匹配、未知 ID 不生成、bomb 尺寸、>24 不截断等用例；原有运动/出界/漏切用例保留。

import Foundation

import FNPBCore

// ============================================================================
// OriginalContentBundle（T-030）
//
// 定位：把「App Bundle 中的原始内容资源 → OriginalContent」这一段收口。
//   - Content/fruitList.json、Content/originalWaveList.json 由 project.yml 的
//     `Content` resources 构建阶段打进 Bundle（T-026 转换器产物，只读）。
//   - 加载失败必须可诊断：区分「资源缺失 / 读取失败 / 解析失败」三种原因，
//     全部转成 Sendable 的 LoadFailure，绝不 try!、绝不在 App 层兜底假数据。
//
// [待验证] 资源查找方式：XcodeGen 对 `- path: Content`（无 type: folder）默认按
//   group 展开，JSON 落在 Bundle 根目录，故 `url(forResource:withExtension:)` 可直接命中。
//   若将来改为 folder reference，需要改成子目录查找（只改本文件的 readResource）。
// ============================================================================

enum OriginalContentBundle {

    /// Bundle 内的资源名（不含扩展名）。
    static let fruitListResourceName = "fruitList"
    static let waveListResourceName = "originalWaveList"

    /// 展示用文件名（错误信息里出现）。
    static let fruitListFileName = "fruitList.json"
    static let waveListFileName = "originalWaveList.json"

    /// 加载失败原因。刻意做成 Sendable 值类型：调用方（GameHost）是 @MainActor，
    /// 不把非 Sendable 的 `any Error` 跨隔离搬运。
    enum LoadFailure: Error, LocalizedError, Sendable, Equatable {
        case missingResource(name: String)
        case unreadableResource(name: String, reason: String)
        case invalidContent(reason: String)

        var errorDescription: String? {
            switch self {
            case .missingResource(let name):
                return "App Bundle 缺少原始内容资源 \(name)"
            case .unreadableResource(let name, let reason):
                return "读取 \(name) 失败：\(reason)"
            case .invalidContent(let reason):
                return "原始内容解析失败：\(reason)"
            }
        }
    }

    /// 读取并解析两份原始内容。
    ///
    /// - 成功：返回 `OriginalContent`（waves.fruits 的 typeID 已按文档零基序号填好）。
    /// - 失败：返回带诊断信息的 `LoadFailure`，**不产生任何占位内容**。
    @MainActor
    static func load() -> Result<OriginalContent, LoadFailure> {
        let bundle = Bundle.main
        do {
            let fruitsData = try readResource(named: fruitListResourceName,
                                              displayName: fruitListFileName,
                                              bundle: bundle)
            let wavesData = try readResource(named: waveListResourceName,
                                             displayName: waveListFileName,
                                             bundle: bundle)
            let content = try OriginalContentLoader.load(fruitsData: fruitsData,
                                                         wavesData: wavesData)
            return .success(content)
        } catch let failure as LoadFailure {
            return .failure(failure)
        } catch {
            // OriginalContentError 实现了 LocalizedError（带节点路径），优先用它。
            let reason = (error as? LocalizedError)?.errorDescription
                ?? String(describing: error)
            return .failure(.invalidContent(reason: reason))
        }
    }

    private static func readResource(named name: String,
                                     displayName: String,
                                     bundle: Bundle) throws -> Data {
        guard let url = bundle.url(forResource: name, withExtension: "json") else {
            throw LoadFailure.missingResource(name: displayName)
        }
        do {
            return try Data(contentsOf: url)
        } catch {
            throw LoadFailure.unreadableResource(name: displayName,
                                                 reason: error.localizedDescription)
        }
    }
}

    func handleFrame(_ stamp: FrameStamp) {
        // R7：首帧 dt == 0，显式 prime 让 GameWorld 用兜底步长推进
        if stamp.isFirstFrame {
            gameWorld.prime()
        }

        // 固定步长推进；alpha 仅供渲染插值。
        // [Kimi 修正 T-022 审查] 签名以 TimeDriver.swift 为准：advance(delta:) /
        // GameWorld.tick(dt:state:world:services:) 且 world 为 inout。
        let alpha = fixedStep.advance(delta: stamp.clampedDelta()) { step in
            gameWorld.tick(dt: step, state: &state, world: &store, services: services)
        }
        renderInterpolationAlpha = alpha

        // 本局结束判定均为 [待验证]：原 +0xa4 标志；Classic 三次漏切近似。
        // missedFruitCount 是重建辅助字段，不能与原 +0x3c 的切中分支计数混用。
        if state.isFaulted || state.missedFruitCount >= Constants.MissPenalty.threshold {
            viewModel.finishGame()
        }

        // T-024：渲染层不走「帧末通知」通路——`MTKView` 以 `isPaused = false` 自驱
        // （preferredFramesPerSecond = 120），每次 draw 时直接读
        // `currentState` / `currentEntities`。因此这里只刷新 UI 快照。
        reviewSnapshotIfNeeded()
    }

    private func reviewSnapshotIfNeeded() {
        viewModel.refreshSnapshot()
    }

            persist()
        @unknown default:
            break
        }

    ///
    /// - Returns: `true` 表示已用原始 38 波次 / 24 水果表开局；
    ///   `false` 表示内容不可用（未加载/解析失败），调用方**必须**停留在原界面，
    ///   不得显示游玩态。失败原因见 `contentLoadError`。
    ///
    /// T-030：内容缺失时不重置任何状态（避免留下半初始化局面）。
    @discardableResult
    func startGame(mode: GameMode) -> Bool {
        guard let content = ensureContent() else {
            Self.contentLogger.error(
                "开局被拒绝：原始内容不可用 — \(self.contentLoadError ?? "unknown", privacy: .public)")

        }

/// 确定性 PRNG（SplitMix64）。回放一致性要求的 snapshot/restore 直接映射内部状态。
/// [待验证] 原版是 32 位定点随机（§5.3 DAT_005a7930 固定点随机），
/// 序列必然不同；此处只保证「可复现 + 均匀分布」，若需逐帧复刻原版请等 PRNG 任务卡。
final class SplitMix64Random: RandomSource {

    private var state: UInt64

    init(seed: UInt64) {
        self.state = seed == 0 ? 0x9E37_79B9_7F4A_7C15 : seed
    }

    func nextFloat() -> Float {
        // 取高 24 位 → [0, 1)
        Float(next() >> 40) * (1.0 / 16_777_216.0)
    }

    func nextInt(_ upperBound: Int32) -> Int32 {
        guard upperBound > 0 else { return 0 }
        return Int32((next() >> 33) % UInt64(upperBound))
    }

    func snapshot() -> UInt64 { state }

    func restore(_ s: UInt64) { state = s == 0 ? 0x9E37_79B9_7F4A_7C15 : s }

    private func next() -> UInt64 {
        state &+= 0x9E37_79B9_7F4A_7C15
        var z = state
        z = (z ^ (z >> 30)) &* 0xBF58_476D_1CE4_E5B9
        z = (z ^ (z >> 27)) &* 0x94D0_49BB_1331_11EB
        return z ^ (z >> 31)
    }
}

## Kimi 审查应注意的风险点

1. **[高] `WaveDirector` 不在本卡注入范围，`beginGame(waves:fruits:)` 签名未验证**：任务卡明确要求该调用，但注入文件里只有 `beginGame(waves:)` 的旧调用痕迹。若 `WaveDirector` 尚未提供 `fruits:` 参数，App 编译会失败。需要确认 `Sources/FNPBCore/Wave/…WaveDirector.swift` 是否已在该分支落盘（若没有，需要一张 WaveDirector 任务卡补齐 `fruits:` 入参与内部水果池构建）。同理 `spawnSink` 是否为可写 `weak var`。

2. **[高] 资源打包方式未验证**：`Content/fruitList.json` 与 `Content/originalWaveList.json` 是否真的落在 Bundle 根目录，取决于 XcodeGen 对 `- path: Content`（`buildPhase: resources`）的处理（group 展开 vs folder reference）。若是 folder reference，`url(forResource:withExtension:)` 会返回 nil，App 将永远无法开局。已在 `OriginalContentBundle` 标 `[待验证]`，建议 CI 加一条 `Bundle.main.path` 断言型测试或改用子目录查找。

5. **[中] 炸弹实体带 `fruit` 元数据**：`spawnBomb` 构造了一个 `typeID = Constants.EntityKindValue.bomb(2)` 的 `FruitInfo`（scale/collision/bomb 已填），以保证渲染层能读到尺寸。这与“水果表 typeID 是零基序号”存在语义混用风险；渲染层若按 typeID 查图集，可能命中错误条目。需确认渲染层消费方式（若渲染只看 `kind`/`transform`，可改为 `fruit = nil`）。
