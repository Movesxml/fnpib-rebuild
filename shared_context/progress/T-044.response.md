import Foundation

// T-044：纯 FNPBCore 挑战回合装配器。
//
// 证据边界：
// - FUN_00061128(param_2 == 0)：新建挑战时先做默认水果初选，
//   再按 challenge.challengeFruitSize 做非 normal 覆盖选择；
//   最终选中水果写入 controller+0x88，之后才激活 challenge 条件。
// - FUN_000619d0：恢复路径同样先恢复 controller+0x88，再激活条件。
// - 本装配器只负责：
//   1) 持有 ChallengeTierSelector / ChallengeFruitSelector / ChallengeRoundPlan；
//   2) 按显式 tierIndex/difficulty/predicateP 生成首回合计划；
//   3) 首回合执行 selectNewGameChallengeFruit（默认 + 非 normal 覆盖）；
//   4) 后续回合仅对非 normal 做覆盖选择，normal 保留默认水果；
//   5) 先定水果，再创建 ChallengeModeRuntime（其 init 内激活条件）。
//
// 不实现：UI、存档格式、波次、结算控制器、tier progression。
// 不虚构：ChallengeCatalog.Entry、selectNext、inout OriginalLCGRandom。
// 随机源使用现有 RandomSource；生产可注入 OriginalLCGRandom（final class）。

public enum ChallengeRoundAssemblerError: Error, Equatable, Sendable {
    case tierIndexOutOfBounds(index: Int32, count: Int)
    case emptyPlan(tierIndex: Int32)
    case missingChallengeAtSlot(index: Int)
    case planExhausted(currentIndex: Int, count: Int)
    case missingFruitName(fruitID: Int32)
    case duplicateFruitTypeID(Int32)
    case firstRoundAlreadyStarted(currentIndex: Int)
    case runtimeNotAssembled
    case missingCandidates(size: ChallengeFruitSize)
}

extension ChallengeRoundAssemblerError: LocalizedError {
    public var errorDescription: String? {
        switch self {
        case let .tierIndexOutOfBounds(index, count):
            return "tierIndex 越界：index=\(index), count=\(count)"
        case let .emptyPlan(tierIndex):
            return "tierIndex=\(tierIndex) 生成的回合计划为空"
        case let .missingChallengeAtSlot(index):
            return "回合计划槽位 \(index) 没有 challenge"
        case let .planExhausted(currentIndex, count):
            return "回合计划已结束：currentIndex=\(currentIndex), count=\(count)"
        case let .missingFruitName(fruitID):
            return "找不到 FruitInfo.typeID=\(fruitID) 对应的水果名称"
        case let .duplicateFruitTypeID(fruitID):
            return "FruitInfo.typeID 重复：\(fruitID)"
        case let .firstRoundAlreadyStarted(currentIndex):
            return "首回合已不可用：currentIndex=\(currentIndex)"
        case .runtimeNotAssembled:
            return "ChallengeModeRuntime 尚未装配"
        case let .missingCandidates(size):
            return "非 normal 水果模式缺少候选：size=\(size)"
        }
    }
}

public struct ChallengeRound: Sendable, Equatable {
    public let roundIndex: Int
    public let challenge: ChallengeDefinition
    public let fruitID: Int32
    public let fruitName: String
    public let runtime: ChallengeModeRuntime

    public init(roundIndex: Int,
                challenge: ChallengeDefinition,
                fruitID: Int32,
                fruitName: String,
                runtime: ChallengeModeRuntime) {
        self.roundIndex = roundIndex
        self.challenge = challenge
        self.fruitID = fruitID
        self.fruitName = fruitName
        self.runtime = runtime
    }
}

public struct ChallengeRoundAssembler {
    public let tier: ChallengeTierDefinition
    public let tierIndex: Int32
    public let difficulty: Int32
    public let predicateP: Int32
    public let fruits: [FruitInfo]
    public let conditionDefinitions: [ChallengeConditionDefinition]
    public let persistence: any ChallengeSelectionPersisting
    public let random: any RandomSource

    public private(set) var tierSelector: ChallengeTierSelector
    public private(set) var fruitSelector: ChallengeFruitSelector
    public private(set) var plan: ChallengeRoundPlan
    public private(set) var currentRoundIndex: Int
    public private(set) var currentFruitID: Int32
    public private(set) var defaultFruitID: Int32
    public private(set) var runtime: ChallengeModeRuntime?

    private let weights: [Int32: ChallengeFruitWeightEntry]
    private let namesByFruitID: [Int32: String]

    public init(catalog: ChallengeCatalog,
                fruits: [FruitInfo],
                conditionDefinitions: [ChallengeConditionDefinition],
                persistence: any ChallengeSelectionPersisting,
                random: any RandomSource,
                tierIndex: Int32,
                difficulty: Int32 = 0,
                predicateP: Int32 = 0) throws {
        guard tierIndex >= 0, Int(tierIndex) < catalog.tiers.count else {
            throw ChallengeRoundAssemblerError.tierIndexOutOfBounds(
                index: tierIndex, count: catalog.tiers.count)
        }

        let tier = catalog.tiers[Int(tierIndex)]

        self.tier = tier
        self.tierIndex = tierIndex
        self.difficulty = difficulty
        self.predicateP = predicateP
        self.fruits = fruits
        self.conditionDefinitions = conditionDefinitions
        self.persistence = persistence
        self.random = random

        self.tierSelector = ChallengeTierSelector(
            tier: tier,
            persistence: persistence,
            specialChallenges: catalog.specialChallenges)
        self.fruitSelector = ChallengeFruitSelector()
        self.plan = ChallengeRoundPlan()
        self.currentRoundIndex = 0
        self.currentFruitID = 0
        self.defaultFruitID = 0
        self.runtime = nil

        var names: [Int32: String] = [:]
        for fruit in fruits {
            guard names[fruit.typeID] == nil else {
                throw ChallengeRoundAssemblerError.duplicateFruitTypeID(fruit.typeID)
            }
            names[fruit.typeID] = fruit.name
        }
        self.namesByFruitID = names
        self.weights = ChallengeFruitWeightMap.make(from: fruits)

        try assembleInitialRound()
    }

    /// 首回合装配：FUN_00061128(param_2 == 0) 的核心顺序。
    /// 1. 生成计划；
    /// 2. 默认水果初选；
    /// 3. 非 normal 覆盖选择；
    /// 4. 最后创建 ChallengeModeRuntime（条件激活发生在其 init 内）。
    private mutating func assembleInitialRound() throws {
        plan = tierSelector.planRound(
            persistence: persistence,
            rng: random,
            difficulty: difficulty,
            tierIndex: tierIndex)

        guard plan.count > 0 else {
            throw ChallengeRoundAssemblerError.emptyPlan(tierIndex: tierIndex)
        }
        guard let firstChallenge = plan.challenge(at: 0) else {
            throw ChallengeRoundAssemblerError.missingChallengeAtSlot(index: 0)
        }

        let newGameSelection = try fruitSelector.selectNewGameChallengeFruit(
            size: firstChallenge.challengeFruitSize,
            catalog: fruits,
            weights: weights,
            predicateP: predicateP,
            random: random)

        defaultFruitID = newGameSelection.defaultSelection
        currentFruitID = newGameSelection.finalSelection

        let name = try fruitName(for: currentFruitID)
        runtime = ChallengeModeRuntime(
            tier: tier,
            challenge: firstChallenge,
            conditionDefinitions: conditionDefinitions,
            selectedFruitName: name)
        currentRoundIndex = 0
    }

    /// 创建/读取首回合。初始化时已经完成水果选择与 runtime 创建。
    public func startFirstRound() throws -> ChallengeRound {
        guard currentRoundIndex == 0 else {
            throw ChallengeRoundAssemblerError.firstRoundAlreadyStarted(
                currentIndex: currentRoundIndex)
        }
        guard let challenge = plan.challenge(at: 0) else {
            throw ChallengeRoundAssemblerError.missingChallengeAtSlot(index: 0)
        }
        guard let runtime else {
            throw ChallengeRoundAssemblerError.runtimeNotAssembled
        }
        let name = try fruitName(for: currentFruitID)
        return ChallengeRound(
            roundIndex: 0,
            challenge: challenge,
            fruitID: currentFruitID,
            fruitName: name,
            runtime: runtime)
    }

    /// 推进到下一回合。
    /// 后续回合只对非 normal 做覆盖选择；normal 保留初始化时的默认水果。
    /// 水果先确定，随后才创建新的 ChallengeModeRuntime。
    public mutating func advanceToNextRound() throws -> ChallengeRound {
        let nextIndex = currentRoundIndex + 1
        guard nextIndex < plan.count else {
            throw ChallengeRoundAssemblerError.planExhausted(
                currentIndex: currentRoundIndex, count: plan.count)
        }
        guard let nextChallenge = plan.challenge(at: nextIndex) else {
            throw ChallengeRoundAssemblerError.missingChallengeAtSlot(index: nextIndex)
        }

        let nextFruitID: Int32
        if nextChallenge.challengeFruitSize == .normal {
            nextFruitID = defaultFruitID
        } else {
            guard let candidates = ChallengeFruitCandidates.candidates(
                for: nextChallenge.challengeFruitSize, from: fruits) else {
                throw ChallengeRoundAssemblerError.missingCandidates(
                    size: nextChallenge.challengeFruitSize)
            }
            nextFruitID = try fruitSelector.select(
                candidates: candidates,
                weights: weights,
                param1: 0,
                predicateP: predicateP,
                random: random)
        }

        currentFruitID = nextFruitID
        let name = try fruitName(for: nextFruitID)
        let nextRuntime = ChallengeModeRuntime(
            tier: tier,
            challenge: nextChallenge,
            conditionDefinitions: conditionDefinitions,
            selectedFruitName: name)

        runtime = nextRuntime
        currentRoundIndex = nextIndex

        return ChallengeRound(
            roundIndex: nextIndex,
            challenge: nextChallenge,
            fruitID: nextFruitID,
            fruitName: name,
            runtime: nextRuntime)
    }

    /// 按 FruitInfo.typeID 查名称；缺失时明确抛错，不回退、不猜测。
    public func fruitName(for fruitID: Int32) throws -> String {
        guard let name = namesByFruitID[fruitID] else {
            throw ChallengeRoundAssemblerError.missingFruitName(fruitID: fruitID)
        }
        return name
    }
}

import XCTest
@testable import FNPBCore

final class ChallengeRoundAssemblerTests: XCTestCase {
    private final class ScriptedRandomSource: RandomSource {
        private(set) var zeroBoundCalls = 0

        func nextInt(_ bound: Int32) -> Int32 {
            if bound == 0 {
                zeroBoundCalls += 1
            }
            return 0
        }

        func nextFloat() -> Float {
            0
        }
    }

    private func makeFruit(id: Int32, name: String, chance: Int32 = 100) -> FruitInfo {
        var info = FruitInfo()
        info.typeID = id
        info.name = name
        info.chance = chance
        info.noCritical = false
        return info
    }

    private func makeFruits() -> [FruitInfo] {
        [
            makeFruit(id: 0, name: "banana"),
            makeFruit(id: 1, name: "apple"),
            makeFruit(id: 2, name: "strawberry"),
            makeFruit(id: 3, name: "kiwifruit"),
            makeFruit(id: 4, name: "lime"),
        ]
    }

    private func makeChallenge(name: String,
                               size: ChallengeFruitSize) -> ChallengeDefinition {
        ChallengeDefinition(
            name: name,
            challengeFruitSize: size,
            conditionNames: [],
            numWaves: 1)
    }

    private func makeCatalog(challenges: [ChallengeDefinition],
                             numRounds: Int32) -> ChallengeCatalog {
        let group = ChallengeGroupDefinition(
            name: "g",
            mustSelect: numRounds,
            challenges: challenges)
        let tier = ChallengeTierDefinition(
            name: "t",
            numRounds: numRounds,
            groups: [group])
        return ChallengeCatalog(tiers: [tier])
    }

    private func makeAssembler(catalog: ChallengeCatalog,
                               fruits: [FruitInfo],
                               random: ScriptedRandomSource,
                               tierIndex: Int32 = 0) throws -> ChallengeRoundAssembler {
        try ChallengeRoundAssembler(
            catalog: catalog,
            fruits: fruits,
            conditionDefinitions: [],
            persistence: ChallengeSelectionMemoryStore(),
            random: random,
            tierIndex: tierIndex,
            difficulty: 0,
            predicateP: 0)
    }

    private func assertAssemblerError<T>(
        _ expression: @autoclosure () throws -> T,
        _ expected: ChallengeRoundAssemblerError,
        file: StaticString = #filePath,
        line: UInt = #line
    ) {
        XCTAssertThrowsError(try expression(), file: file, line: line) { error in
            guard let actual = error as? ChallengeRoundAssemblerError else {
                return XCTFail("unexpected error: \(error)", file: file, line: line)
            }
            XCTAssertEqual(actual, expected, file: file, line: line)
        }
    }

    func testFirstRoundNormalConsumesOneFruitSelection() throws {
        let fruits = makeFruits()
        let catalog = makeCatalog(
            challenges: [makeChallenge(name: "c1", size: .normal)],
            numRounds: 1)
        let random = ScriptedRandomSource()
        let assembler = try makeAssembler(catalog: catalog, fruits: fruits, random: random)

        XCTAssertEqual(random.zeroBoundCalls, 1)

        let round = try assembler.startFirstRound()
        XCTAssertEqual(round.challenge.name, "c1")
        XCTAssertEqual(round.fruitID, 1) // 默认候选排除 banana 后的首项 apple
        XCTAssertEqual(round.fruitName, "apple")
        XCTAssertEqual(round.runtime.selectedFruitName, "apple")
    }

    func testFirstRoundSmallDoesDefaultThenOverride() throws {
        let fruits = makeFruits()
        let catalog = makeCatalog(
            challenges: [makeChallenge(name: "small", size: .small)],
            numRounds: 1)
        let random = ScriptedRandomSource()
        let assembler = try makeAssembler(catalog: catalog, fruits: fruits, random: random)

        XCTAssertEqual(random.zeroBoundCalls, 2)
        XCTAssertEqual(assembler.defaultFruitID, 1) // apple

        let round = try assembler.startFirstRound()
        XCTAssertEqual(round.fruitID, 2) // small 覆盖为 strawberry
        XCTAssertEqual(round.fruitName, "strawberry")
        XCTAssertEqual(round.runtime.selectedFruitName, "strawberry")
    }

    func testSubsequentNormalKeepsDefaultFruit() throws {
        let fruits = makeFruits()
        let catalog = makeCatalog(
            challenges: [
                makeChallenge(name: "small", size: .small),
                makeChallenge(name: "normal", size: .normal),
            ],
            numRounds: 2)
        let random = ScriptedRandomSource()
        let assembler = try makeAssembler(catalog: catalog, fruits: fruits, random: random)

        let first = try assembler.startFirstRound()
        XCTAssertEqual(first.challenge.name, "small")
        XCTAssertEqual(first.fruitID, 2) // strawberry
        XCTAssertEqual(random.zeroBoundCalls, 2)

        let second = try assembler.advanceToNextRound()
        XCTAssertEqual(second.challenge.name, "normal")
        XCTAssertEqual(second.fruitID, assembler.defaultFruitID)
        XCTAssertEqual(second.fruitID, 1) // apple
        XCTAssertEqual(second.fruitName, "apple")
        XCTAssertEqual(second.runtime.selectedFruitName, "apple")
        XCTAssertEqual(random.zeroBoundCalls, 2) // normal 不重新选水果
    }

    func testSubsequentSmallOverridesFruit() throws {
        let fruits = makeFruits()
        let catalog = makeCatalog(
            challenges: [
                makeChallenge(name: "normal", size: .normal),
                makeChallenge(name: "small", size: .small),
            ],
            numRounds: 2)
        let random = ScriptedRandomSource()
        let assembler = try makeAssembler(catalog: catalog, fruits: fruits, random: random)

        let first = try assembler.startFirstRound()
        XCTAssertEqual(first.challenge.name, "normal")
        XCTAssertEqual(first.fruitID, 1) // apple
        XCTAssertEqual(random.zeroBoundCalls, 1)

        let second = try assembler.advanceToNextRound()
        XCTAssertEqual(second.challenge.name, "small")
        XCTAssertEqual(second.fruitID, 2) // strawberry
        XCTAssertEqual(second.fruitName, "strawberry")
        XCTAssertEqual(second.runtime.selectedFruitName, "strawberry")
        XCTAssertEqual(random.zeroBoundCalls, 2)
    }

    func testFruitBindingMatchesRuntimeSelectedFruitName() throws {
        let fruits = makeFruits()
        let catalog = makeCatalog(
            challenges: [makeChallenge(name: "c1", size: .normal)],
            numRounds: 1)
        let random = ScriptedRandomSource()
        let assembler = try makeAssembler(catalog: catalog, fruits: fruits, random: random)

        let round = try assembler.startFirstRound()
        XCTAssertEqual(round.runtime.selectedFruitName, round.fruitName)
        XCTAssertEqual(round.runtime.selectedFruitName, "apple")
    }

    func testPlanExhaustedThrows() throws {
        let fruits = makeFruits()
        let catalog = makeCatalog(
            challenges: [makeChallenge(name: "c1", size: .normal)],
            numRounds: 1)
        let random = ScriptedRandomSource()
        let assembler = try makeAssembler(catalog: catalog, fruits: fruits, random: random)

        _ = try assembler.startFirstRound()

        XCTAssertThrowsError(try assembler.advanceToNextRound()) { error in
            guard let actual = error as? ChallengeRoundAssemblerError else {
                return XCTFail("unexpected error: \(error)")
            }
            XCTAssertEqual(actual, .planExhausted(currentIndex: 0, count: 1))
        }
    }

    func testMissingFruitNameThrows() throws {
        let fruits = makeFruits()
        let catalog = makeCatalog(
            challenges: [makeChallenge(name: "c1", size: .normal)],
            numRounds: 1)
        let random = ScriptedRandomSource()
        let assembler = try makeAssembler(catalog: catalog, fruits: fruits, random: random)

        XCTAssertThrowsError(try assembler.fruitName(for: 999)) { error in
            guard let actual = error as? ChallengeRoundAssemblerError else {
                return XCTFail("unexpected error: \(error)")
            }
            XCTAssertEqual(actual, .missingFruitName(fruitID: 999))
        }
    }

    func testEmptyPlanThrows() {
        let fruits = makeFruits()
        let catalog = makeCatalog(challenges: [], numRounds: 0)
        let random = ScriptedRandomSource()

        assertAssemblerError(
            try makeAssembler(catalog: catalog, fruits: fruits, random: random),
            .emptyPlan(tierIndex: 0))
    }

    func testTierIndexOutOfBoundsThrows() {
        let fruits = makeFruits()
        let catalog = ChallengeCatalog(tiers: [])
        let random = ScriptedRandomSource()

        assertAssemblerError(
            try makeAssembler(catalog: catalog, fruits: fruits, random: random, tierIndex: 0),
            .tierIndexOutOfBounds(index: 0, count: 0))
    }
}
