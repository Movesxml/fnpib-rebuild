import Foundation

extension UIScreenLayoutProjection {
    /// Returns the original ordered visible draw list. The screen root is a
    /// traversal container only; it is never included in the result.
    /// screenVisibility is intentionally not consulted here.
    public func orderedVisibleComponents(
        visibleTargets: [String: Bool] = [:]
    ) -> [UIScreenProjectedComponent] {
        var childrenByParent: [Int: [Int]] = [:]
        var roots: [Int] = []

        for (index, component) in components.enumerated() {
            if let parentIndex = component.parentIndex, parentIndex != -1 {
                childrenByParent[parentIndex, default: []].append(index)
            } else {
                roots.append(index)
            }
        }

        struct Entry {
            let component: UIScreenProjectedComponent
            let traversalOrder: Int
        }

        var entries: [Entry] = []
        entries.reserveCapacity(components.count)
        var traversalOrder = 0

        func visit(_ index: Int) {
            guard index >= 0, index < components.count else { return }
            let component = components[index]
            let visible = visibleTargets[component.name] ?? component.enabled
            guard visible else { return }

            entries.append(Entry(component: component,
                                 traversalOrder: traversalOrder))
            traversalOrder += 1

            for childIndex in childrenByParent[index] ?? [] {
                visit(childIndex)
            }
        }

        for rootIndex in roots {
            for childIndex in childrenByParent[rootIndex] ?? [] {
                visit(childIndex)
            }
        }

        entries.sort { lhs, rhs in
            if lhs.component.effectiveLayer != rhs.component.effectiveLayer {
                return lhs.component.effectiveLayer < rhs.component.effectiveLayer
            }
            return lhs.traversalOrder < rhs.traversalOrder
        }

        return entries.map(\.component)
    }
}

import Foundation
import XCTest
@testable import FNPBCore

final class UIScreenDrawListTests: XCTestCase {
    func testRootDisabledDoesNotAppearAndDoesNotHideChildren() throws {
        let projection = try project([
            component("", parent: -1, layout: .init(enabled: false)),
            component("childA", parent: 0),
            component("childB", parent: 0),
        ])

        let ordered = projection.orderedVisibleComponents()
        XCTAssertEqual(ordered.map(\.name), ["childA", "childB"])
        XCTAssertFalse(ordered.contains { $0.name.isEmpty })
    }

    func testResourcePreorderAndCrossLayerSorting() throws {
        let projection = try project([
            component("", parent: -1),
            component("a", parent: 0, layout: .init(layer: 5)),
            component("aChild", parent: 1, layout: .init(layer: 0)),
            component("b", parent: 0, layout: .init(layer: 5)),
            component("bChild", parent: 3, layout: .init(layer: 0)),
            component("c", parent: 0, layout: .init(layer: 10)),
        ])

        XCTAssertEqual(
            projection.orderedVisibleComponents().map(\.name),
            ["aChild", "bChild", "a", "b", "c"])
    }

    func testSameLayerUsesResourceOrderEvenWhenNamesAreReverseSorted() throws {
        let projection = try project([
            component("", parent: -1),
            component("zeta", parent: 0, layout: .init(layer: 0)),
            component("alpha", parent: 0, layout: .init(layer: 0)),
            component("middle", parent: 0, layout: .init(layer: 0)),
        ])

        XCTAssertEqual(
            projection.orderedVisibleComponents().map(\.name),
            ["zeta", "alpha", "middle"])
    }

    func testHiddenParentSkipsEntireSubtreeEvenWithChildOverrides() throws {
        let disabledParentProjection = try project([
            component("", parent: -1),
            component("hiddenParent", parent: 0, layout: .init(enabled: false)),
            component("child", parent: 1),
            component("grandchild", parent: 2),
            component("visibleSibling", parent: 0),
        ])

        XCTAssertEqual(
            disabledParentProjection.orderedVisibleComponents(
                visibleTargets: ["child": true, "grandchild": true]).map(\.name),
            ["visibleSibling"])

        let overriddenParentProjection = try project([
            component("", parent: -1),
            component("parent", parent: 0),
            component("child", parent: 1),
            component("grandchild", parent: 2),
        ])

        XCTAssertTrue(
            overriddenParentProjection.orderedVisibleComponents(
                visibleTargets: [
                    "parent": false,
                    "child": true,
                    "grandchild": true,
                ]).isEmpty)
    }

    func testVisibleTargetsShowHideAndUnknownNamesAreIgnored() throws {
        let projection = try project([
            component("", parent: -1),
            component("offByDefault", parent: 0, layout: .init(enabled: false)),
            component("onByDefault", parent: 0),
        ])

        XCTAssertEqual(
            projection.orderedVisibleComponents().map(\.name),
            ["onByDefault"])

        XCTAssertEqual(
            projection.orderedVisibleComponents(
                visibleTargets: ["offByDefault": true]).map(\.name),
            ["offByDefault", "onByDefault"])

        XCTAssertEqual(
            projection.orderedVisibleComponents(
                visibleTargets: ["onByDefault": false]).map(\.name),
            [])

        XCTAssertEqual(
            projection.orderedVisibleComponents(
                visibleTargets: ["unknown": true]).map(\.name),
            ["onByDefault"])
    }

    func testScreenVisibilityFalseDoesNotFilterDrawList() throws {
        let projection = try project([
            component("", parent: -1),
            component("offscreen", parent: 0, layout: .init(
                position: [1000, 1000, 0], width: 10, height: 10)),
            component("onscreen", parent: 0, layout: .init(
                position: [10, 10, 0], width: 10, height: 10)),
        ])

        let offscreen = try XCTUnwrap(
            projection.components.first(where: { $0.name == "offscreen" }))
        XCTAssertFalse(offscreen.screenVisibility)

        XCTAssertEqual(
            projection.orderedVisibleComponents().map(\.name),
            ["offscreen", "onscreen"])
    }

    func testProductionChallengeMapKeepsNormalContainerButHidesBonusContainerAndRoot() throws {
        let player = try productionChallengeMapPlayer()
        let projection = try UIScreenLayoutProjection.project(player)
        XCTAssertEqual(projection.components.count, 95)

        let normal = try XCTUnwrap(
            projection.components.first(where: { $0.name == "challengeContainer" }))

        let bonus = try XCTUnwrap(
            projection.components.first(where: {
                $0.name == "bonusChallengeContainer"
            })
            ?? projection.components.first(where: {
                $0.name != normal.name &&
                $0.parentIndex == normal.parentIndex &&
                $0.name.lowercased().contains("bonus")
            })
            ?? projection.components.first(where: {
                $0.name != normal.name &&
                $0.parentIndex == normal.parentIndex &&
                $0.name.lowercased().contains("boss")
            }))

        let ordered = projection.orderedVisibleComponents(visibleTargets: [
            normal.name: true,
            bonus.name: false,
        ])
        let names = Set(ordered.map(\.name))

        XCTAssertTrue(names.contains(normal.name))
        XCTAssertFalse(names.contains(bonus.name))
        XCTAssertFalse(names.contains(""))
        XCTAssertFalse(names.contains { $0.hasPrefix(bonus.name + ".") })
    }

    private func component(
        _ name: String,
        parent: Int? = -1,
        layout: UIScreenAnimationContent.Component.Layout = .init()
    ) -> UIScreenAnimationContent.Component {
        UIScreenAnimationContent.Component(
            name: name, animations: [], parentIndex: parent, layout: layout)
    }

    private func project(_ components: [UIScreenAnimationContent.Component])
        throws -> UIScreenLayoutProjection {
        let player = try UIScreenAnimationPlayer(content: makeContent(components))
        return try UIScreenLayoutProjection.project(player)
    }

    private func makeContent(_ components: [UIScreenAnimationContent.Component])
        -> UIScreenAnimationContent {
        UIScreenAnimationContent(
            formatVersion: 1,
            source: UIScreenAnimationContent.Source(
                file: "test.uiscreen",
                sha256: String(repeating: "0", count: 64)),
            components: components)
    }

    private func productionChallengeMapPlayer() throws -> UIScreenAnimationPlayer {
        let data = try Data(contentsOf: repositoryRoot()
            .appendingPathComponent("Content/challengeMapAnimations.json"))
        return try UIScreenAnimationPlayer(data: data)
    }

    private func repositoryRoot() -> URL {
        URL(fileURLWithPath: #filePath)
            .deletingLastPathComponent()
            .deletingLastPathComponent()
            .deletingLastPathComponent()
    }
}
