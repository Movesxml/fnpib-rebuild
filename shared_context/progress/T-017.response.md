## 设计说明

任务卡要求 4 个文件块，都是构建期离线工具，不进入 iOS 运行时。我的取舍：

1. **extract.py**：只用标准库（不硬依赖 Pillow/numpy，便于只解包不转换的场景），负责 ipa→assets.zip→Raw/ 与清点；对 .tex/.mmd/.xml 只做**头部探针**（12 字节头、HBR0 magic、ET 可解析性），权威解码留给 tex_convert，避免两处格式定义漂移。zip-slip 防护 + 拒绝 `__MACOSX/.DS_Store` 噪音。
2. **tex_convert.py**：核心是**数据驱动的格式判定**而非硬编码 LA88——对 16bpp 的三个候选（la88 / rgba4444 / abgr4444）各算「alpha 通道取满值比例」，取最高分；平局按任务卡侦察倾向 la88 优先；低熵（纯色）贴图降置信。判定结果+三个分数+type 字节全部进 manifest，供批量统计反推 type 字节语义。尺寸校验按任务卡要求做：精确匹配 → 1B/4B per-pixel（8bpp/32bpp）→ 整数倍冗余（疑似 mip）→ 4bpp 色板（不支持）→ 记 warning 跳过。行序（GL 左下原点）作为显式开关 `--flip-y`，默认不翻转并记录在案。
3. **xml_convert.py**：只搬结构不做类型强转；编码兜底 gb18030（老中文资源）；fruitlist/wave 的**人工抽检样例我不敢臆造**，在文件头留 TODO 占位 + 抽检清单 + 复跑命令。
4. **MMD_FORMAT.md**：严格执行「已确认/推测」标注。我手上只有侦察描述而非真实字节，所以报告把 chunk 头布局列为**候选假设 A/B/C**，给出可编程的自洽判据与验证脚本，并把「顶层声明长度 == 文件长 还是 +12」作为 T-018 第一个要闭合的问题。

兼容性/风险见末尾清单。

#!/usr/bin/env python3
# -*- coding: utf-8 -*-
"""T-017 / tools/AssetPipeline/extract.py — IPA → assets.zip → Raw/ 解包 + 全量清点

用法（Windows PowerShell / cmd 均可）::

    # 默认输入 ipa/FruitNinjaPiB-v1.0412.ipa
    python tools/AssetPipeline/extract.py
    python tools/AssetPipeline/extract.py ipa/FruitNinjaPiB-v1.0412.ipa \
        --out-dir build/AssetPipeline/Raw --report-dir build/AssetPipeline/reports

    # 直接给 assets.zip
    python tools/AssetPipeline/extract.py assets.zip --out-dir Raw

    # 只清点已解包目录，不动文件
    python tools/AssetPipeline/extract.py Raw --list-only

产出::

    <out-dir>/                      原始资产（保留 assets.zip 内部目录结构）
    <report-dir>/inventory.json     机器可读清点结果
    <report-dir>/inventory.md       人读报告

设计说明：
- **只用标准库**：清点阶段只需读文件头，不 import Pillow/numpy，
  因此「只想解包不想装依赖」的场景也能跑。
- .tex 只读 12 字节头做宽高/bpp 粗估；**权威解析在 tex_convert.py**
  （TEX_HEADER_SIZE 两边必须一致，改动时同步）。
- zip 路径安全：拒绝绝对路径 / 盘符 / `..` 段（zip-slip），被拒条目记入报告。
- 解包是幂等的：已存在文件默认不覆盖（--overwrite 才覆盖）。
- 不做任何格式转换，保证「Raw 目录 = 原始字节」这一可审计前提。
"""

# --- 与 tex_convert.py 保持一致的常量（改动需同步）---
TEX_HEADER_SIZE = 12

MMD_MAGIC = b"HBR0"
DEFAULT_ASSETS_NAME = "assets.zip"
DEFAULT_INPUT = "ipa/FruitNinjaPiB-v1.0412.ipa"
DEFAULT_OUT_DIR = "build/AssetPipeline/Raw"
DEFAULT_REPORT_DIR = "build/AssetPipeline/reports"

JUNK_NAMES = {".DS_Store", "Thumbs.db"}
JUNK_PREFIXES = ("__MACOSX/",)

def _setup_console() -> None:
    """Windows 控制台默认 cp936，打印中文可能 UnicodeEncodeError。"""
    for stream in (sys.stdout, sys.stderr):
        try:
            stream.reconfigure(encoding="utf-8", errors="replace")  # py>=3.7
        except Exception:
            pass


def _now() -> str:
    return datetime.now().astimezone().isoformat(timespec="seconds")


def _is_unsafe_zip_name(name: str) -> bool:
    n = name.replace("\\", "/")
    if n.startswith("/"):
        return True
    if len(n) >= 2 and n[1] == ":":          # "C:/..." 之类
        return True
    return any(part == ".." for part in n.split("/"))


def _is_junk_name(name: str) -> bool:
    n = name.replace("\\", "/")
    if any(n.startswith(p) for p in JUNK_PREFIXES):
        return True
    return Path(n).name in JUNK_NAMES

def _extract_zip(zf: zipfile.ZipFile, dest: Path, *, overwrite: bool) -> dict:
    stats = {
        "extracted": 0,
        "unsafe_skipped": [],
        "junk_skipped": [],
        "exists_skipped": [],
        "errors": [],
    }
    for info in zf.infolist():
        if info.is_dir():
            continue
        name = info.filename
        if _is_unsafe_zip_name(name):
            stats["unsafe_skipped"].append(name)
            continue
        if _is_junk_name(name):
            stats["junk_skipped"].append(name)
            continue
        target = dest / Path(name.replace("\\", "/"))
        if target.exists() and not overwrite:
            stats["exists_skipped"].append(name)
            continue
        try:
            target.parent.mkdir(parents=True, exist_ok=True)
            with zf.open(info) as src, open(target, "wb") as dst:
                shutil.copyfileobj(src, dst)
            stats["extracted"] += 1
        except OSError as exc:
            stats["errors"].append(f"{name}: {exc}")
    return stats


def load_assets_zip(path: Path, assets_name: str = DEFAULT_ASSETS_NAME):
    """打开输入，返回 (内层 assets ZipFile, 源描述 dict)。

    支持三种输入：.ipa / 裸 assets.zip / 任何含 assets.zip 的容器 zip。
    内层 zip 走内存 BytesIO，调用方负责 close。
    """
    if not path.exists():
        raise SystemExit(f"[extract] 输入不存在: {path}")

    try:
        outer = zipfile.ZipFile(path)
    except zipfile.BadZipFile as exc:
        raise SystemExit(f"[extract] 不是有效 zip: {path} ({exc})")

    try:
        names = [i.filename for i in outer.infolist() if not i.is_dir()]
    except zipfile.BadZipFile as exc:
        outer.close()
        raise SystemExit(f"[extract] zip 目录损坏: {path} ({exc})")

    is_ipa = any(n.startswith("Payload/") for n in names)
    if not is_ipa:
        # 输入本身就是资产包
        return outer, {
            "kind": "assets_zip",
            "container": str(path),
            "assets_zip": str(path),
            "members": len(names),
        }

    cands = [n for n in names
             if n.replace("\\", "/").lower().endswith(assets_name.lower())]
    # Payload/ 下的优先，路径层级浅的优先
    cands.sort(key=lambda n: (0 if n.startswith("Payload/") else 1,
                              n.count("/"), len(n)))
    if not cands:
        outer.close()
        raise SystemExit(f"[extract] IPA 内未找到 {assets_name}（共 {len(names)} 个条目）")

    chosen = cands[0]
    try:
        blob = outer.read(chosen)
    except KeyError as exc:
        outer.close()
        raise SystemExit(f"[extract] 读取 {chosen} 失败: {exc}")
    finally:
        outer.close()

    try:
        inner = zipfile.ZipFile(io.BytesIO(blob))
    except zipfile.BadZipFile as exc:
        raise SystemExit(f"[extract] 内层 {chosen} 不是有效 zip: {exc}")

    return inner, {
        "kind": "ipa",
        "container": str(path),
        "ipa_members": len(names),
        "assets_zip": chosen,
        "assets_zip_candidates": cands,
        "assets_zip_bytes": len(blob),
    }

def _probe_tex(path: Path, size: int) -> dict:
    """只读 12 字节头。字段语义未全部证实，仅做尺寸/bpp 粗估。"""
    with open(path, "rb") as fh:
        head = fh.read(TEX_HEADER_SIZE)
    if len(head) < TEX_HEADER_SIZE:
        return {"error": "header_truncated", "size": size}

    type_id, flags, unk, width, height = struct.unpack_from("<BBHHH", head, 0)
    tail = head[8:12]
    pixels = width * height
    payload = size - TEX_HEADER_SIZE
    bpp_exact = (payload / pixels) if pixels else None
    unit_guess = None
    if bpp_exact is not None:
        for unit in (1, 2, 4):
            if payload == pixels * unit:
                unit_guess = unit
                break

    return {
        "type": type_id,
        "flags": flags,
        "unk": unk,
        "width": width,
        "height": height,
        "pixels": pixels,
        "payload_bytes": payload,
        "bpp_exact": round(bpp_exact, 4) if bpp_exact is not None else None,
        "unit_guess": unit_guess,
        "tail_ok": tail == b"\xff\xff\xff\xff",
    }


def _probe_mmd(path: Path, size: int) -> dict:
    with open(path, "rb") as fh:
        head = fh.read(16)
    magic_ok = head[:4] == MMD_MAGIC
    version = int.from_bytes(head[4:8], "little") if len(head) >= 8 else None
    declared = int.from_bytes(head[8:12], "little") if len(head) >= 12 else None
    matches = None
    if declared is not None:
        matches = {
            "declared_equals_filesize": declared == size,
            "declared_plus_12_equals_filesize": declared + 12 == size,
            "declared_plus_8_equals_filesize": declared + 8 == size,
        }
    return {
        "magic_ok": magic_ok,
        "version": version,
        "declared_len": declared,
        "filesize": size,
        "len_hypotheses": matches,
    }


def _probe_xml(path: Path) -> dict:
    try:
        ET.parse(path)
        return {"ok": True}
    except ET.ParseError as exc:
        return {"ok": False, "error": f"ParseError: {exc}"}
    except OSError as exc:
        return {"ok": False, "error": f"OSError: {exc}"}

def _iter_assets(root: Path):
    for path in sorted(root.rglob("*")):
        if not path.is_file():
            continue
        rel = path.relative_to(root).as_posix()
        if _is_junk_name(rel):
            continue
        yield path, rel


def build_inventory(root: Path) -> dict:
    by_ext: dict[str, dict] = {}
    tex_items: list[tuple[str, int, dict]] = []
    mmd_items: list[tuple[str, int, dict]] = []
    xml_items: list[tuple[str, int, dict]] = []
    total_files = 0
    total_bytes = 0

    for path, rel in _iter_assets(root):
        try:
            size = path.stat().st_size
        except OSError:
            continue
        total_files += 1
        total_bytes += size
        ext = path.suffix.lower() or "<none>"

        bucket = by_ext.get(ext)
        if bucket is None:
            by_ext[ext] = {"count": 0, "bytes": 0, "size_min": size, "size_max": size}
            bucket = by_ext[ext]
        bucket["count"] += 1
        bucket["bytes"] += size
        bucket["size_min"] = min(bucket["size_min"], size)
        bucket["size_max"] = max(bucket["size_max"], size)

        if ext == ".tex":
            tex_items.append((rel, size, _probe_tex(path, size)))
        elif ext == ".mmd":
            mmd_items.append((rel, size, _probe_mmd(path, size)))
        elif ext == ".xml":
            xml_items.append((rel, size, _probe_xml(path)))

    warnings: list[str] = []

    # ---- tex ----
    tex_ok = [it for it in tex_items if "error" not in it[2]]
    tex_bad = [it for it in tex_items if "error" in it[2]]
    tex_summary = {
        "count": len(tex_items),
        "header_ok": len(tex_ok),
        "errors": [{"path": rel, "error": probe["error"]}
                   for rel, _size, probe in tex_bad][:32],
    }
    if tex_ok:
        widths = [it[2]["width"] for it in tex_ok]
        heights = [it[2]["height"] for it in tex_ok]
        units = Counter(it[2]["unit_guess"] if it[2]["unit_guess"] is not None else 0
                        for it in tex_ok)
        tex_summary.update({
            "width_range": [min(widths), max(widths)],
            "height_range": [min(heights), max(heights)],
            "unit_hist": {str(k): v for k, v in sorted(units.items())},
            "type_hist": {str(k): v for k, v in sorted(
                Counter(it[2]["type"] for it in tex_ok).items())},
            "tail_not_ff": [rel for rel, _s, p in tex_ok if not p["tail_ok"]][:32],
            "size_not_canonical": [
                {"path": rel, "pixels": p["pixels"], "payload": p["payload_bytes"],
                 "bpp": p["bpp_exact"]}
                for rel, _s, p in tex_ok if p["unit_guess"] is None
            ][:32],
        })
        zero_unit = tex_summary["unit_hist"].get("0", 0)
        if zero_unit:
            warnings.append(
                f"{zero_unit} 个 .tex 的 payload 长度不是 w*h*(1|2|4) 的整数倍，"
                f"tex_convert 会记 warning 并跳过（见 tex_summary.size_not_canonical）")
    if tex_bad:
        warnings.append(f"{len(tex_bad)} 个 .tex 头部不足 12 字节，无法解析")
    if not tex_items:
        warnings.append("Raw/ 内没有 .tex 文件 —— 解包是否成功？")

    # ---- mmd ----
    mmd_magic_bad = [rel for rel, _s, p in mmd_items if not p["magic_ok"]]
    hyp = Counter()
    for rel, _s, p in mmd_items:
        h = p.get("len_hypotheses") or {}
        for key, ok in h.items():
            if ok:
                hyp[key] += 1
    mmd_summary = {
        "count": len(mmd_items),
        "magic_ok": len(mmd_items) - len(mmd_magic_bad),
        "magic_bad": mmd_magic_bad[:32],
        "versions": {str(k): v for k, v in sorted(
            Counter(p.get("version") for _r, _s, p in mmd_items if p["magic_ok"]).items(),
            key=lambda kv: str(kv[0]))},
        "len_hypothesis_hits": dict(hyp),
    }
    if mmd_magic_bad:
        warnings.append(f"{len(mmd_magic_bad)} 个 .mmd 不以 HBR0 开头")

    # ---- xml ----
    xml_bad = [(rel, p["error"]) for rel, _s, p in xml_items if not p.get("ok")]
    xml_summary = {
        "count": len(xml_items),
        "parsed_ok": len(xml_items) - len(xml_bad),
        "errors": [{"path": rel, "error": err} for rel, err in xml_bad[:32]],
    }
    if xml_bad:
        warnings.append(
            f"{len(xml_bad)} 个 .xml 用 ElementTree 默认解析失败"
            f"（xml_convert.py 会尝试 gb18030 兜底）")

    return {
        "generated_at": _now(),
        "raw_dir": str(root),
        "total_files": total_files,
        "total_bytes": total_bytes,
        "by_ext": by_ext,
        "tex": tex_summary,
        "mmd": mmd_summary,
        "xml": xml_summary,
        "warnings": warnings,
    }


def render_markdown(inv: dict) -> str:
    lines: list[str] = []
    src = inv.get("source", {})
    lines.append("# 资产清点报告（T-017 · extract.py）")
    lines.append("")
    lines.append(f"- 生成时间：`{inv['generated_at']}`")
    lines.append(f"- 输入：`{src.get('container', '?')}`（kind=`{src.get('kind', 'dir')}`）")
    if src.get("assets_zip"):
        lines.append(f"- 内层资产包：`{src['assets_zip']}` "
                     f"（{src.get('assets_zip_bytes', 0)} 字节）")
    lines.append(f"- Raw 目录：`{inv['raw_dir']}`")
    lines.append(f"- 文件总数：**{inv['total_files']}**，总字节：**{inv['total_bytes']}**")
    lines.append("")

    lines.append("## 按扩展名")
    lines.append("")
    lines.append("| 扩展名 | 数量 | 总字节 | 最小 | 最大 |")
    lines.append("|---|---:|---:|---:|---:|")
    for ext, s in sorted(inv["by_ext"].items(), key=lambda kv: (-kv[1]["count"], kv[0])):
        lines.append(f"| `{ext}` | {s['count']} | {s['bytes']} | {s['size_min']} | {s['size_max']} |")
    lines.append("")

    tex = inv.get("tex", {})
    if tex.get("count"):
        lines.append("## .tex 头部统计（粗估，权威解码见 tex_convert.py）")
        lines.append("")
        lines.append(f"- 数量：{tex['count']}（头部可解析 {tex['header_ok']}）")
        lines.append(f"- 宽范围：{tex.get('width_range')}，高范围：{tex.get('height_range')}")
        lines.append(f"- 每像素字节数分布：`{tex.get('unit_hist')}`")
        lines.append(f"- 头字节 type 分布：`{tex.get('type_hist')}`")
        lines.append(f"- 尾部非 `ffffffff` 的文件：{len(tex.get('tail_not_ff', []))} 个")
        lines.append(f"- 尺寸不规范的 `.tex`：{len(tex.get('size_not_canonical', []))} 个")
        lines.append("")

    mmd = inv.get("mmd", {})
    if mmd.get("count"):
        lines.append("## .mmd 头部统计（HBR0 容器）")
        lines.append("")
        lines.append(f"- 数量：{mmd['count']}，magic 正确 {mmd['magic_ok']}")
        lines.append(f"- 版本号分布：`{mmd.get('versions')}`")
        lines.append(f"- 长度字段假设命中：`{mmd.get('len_hypothesis_hits')}` "
                     f"（用于判定「声明长度是否含 12 字节头」，见 MMD_FORMAT.md）")
        lines.append("")

    xml = inv.get("xml", {})
    if xml.get("count"):
        lines.append("## .xml 可解析性")
        lines.append("")
        lines.append(f"- 数量：{xml['count']}，默认解析成功 {xml['parsed_ok']}")
        if xml["errors"]:
            lines.append("- 失败清单（前 32）：")
            for item in xml["errors"]:
                lines.append(f"  - `{item['path']}`：{item['error']}")
        lines.append("")

    if inv.get("warnings"):
        lines.append("## 警告")
        lines.append("")
        for w in inv["warnings"]:
            lines.append(f"- {w}")
        lines.append("")

    return "\n".join(lines)

def main(argv=None) -> int:
    _setup_console()
    ap = argparse.ArgumentParser(
        description="IPA → assets.zip → Raw/ 解包 + 全量清点",
        formatter_class=argparse.RawDescriptionHelpFormatter,
        epilog=__doc__,
    )
    ap.add_argument("input", nargs="?", default=None,
                    help=f"ipa / assets.zip / 已解包目录（默认 {DEFAULT_INPUT}）")
    ap.add_argument("--out-dir", default=DEFAULT_OUT_DIR, help="Raw 输出目录")
    ap.add_argument("--report-dir", default=DEFAULT_REPORT_DIR, help="报告输出目录")
    ap.add_argument("--assets-name", default=DEFAULT_ASSETS_NAME,
                    help="ipa 内资产包文件名（默认 assets.zip）")
    ap.add_argument("--overwrite", action="store_true", help="覆盖已存在文件")
    ap.add_argument("--list-only", action="store_true",
                    help="输入按已解包目录处理，只清点不解包")
    args = ap.parse_args(argv)

    in_path = Path(args.input or DEFAULT_INPUT)
    out_dir = Path(args.out_dir)
    report_dir = Path(args.report_dir)

    source: dict
    if args.list_only or in_path.is_dir():
        if not in_path.is_dir():
            raise SystemExit(f"[extract] --list-only 需要目录输入: {in_path}")
        raw_dir = in_path
        source = {"kind": "dir", "container": str(in_path)}
        print(f"[extract] 只清点，不修改: {raw_dir}")
    else:
        raw_dir = out_dir
        print(f"[extract] 打开 {in_path}")
        zf, source = load_assets_zip(in_path, args.assets_name)
        try:
            print(f"[extract] 资产包: {source.get('assets_zip')} "
                  f"({len(zf.infolist())} 条目) → {raw_dir}")
            stats = _extract_zip(zf, raw_dir, overwrite=args.overwrite)
        finally:
            zf.close()
        source["extract_stats"] = {
            "extracted": stats["extracted"],
            "unsafe_skipped": stats["unsafe_skipped"][:32],
            "junk_skipped": len(stats["junk_skipped"]),
            "exists_skipped": len(stats["exists_skipped"]),
            "errors": stats["errors"][:32],
        }
        print(f"[extract] 解包 {stats['extracted']} 个文件"
              f"（已存在跳过 {len(stats['exists_skipped'])}，"
              f"不安全路径跳过 {len(stats['unsafe_skipped'])}，"
              f"错误 {len(stats['errors'])}）")

    if not raw_dir.is_dir():
        raise SystemExit(f"[extract] Raw 目录不存在: {raw_dir}")

    inv = build_inventory(raw_dir)
    inv["source"] = source

    report_dir.mkdir(parents=True, exist_ok=True)
    json_path = report_dir / "inventory.json"
    md_path = report_dir / "inventory.md"
    json_path.write_text(json.dumps(inv, ensure_ascii=False, indent=2), encoding="utf-8")
    md_path.write_text(render_markdown(inv), encoding="utf-8")

    print("")
    print(f"[extract] 文件总数 {inv['total_files']}，总字节 {inv['total_bytes']}")
    for ext, s in sorted(inv["by_ext"].items(), key=lambda kv: -kv[1]["count"])[:12]:
        print(f"           {ext:>10s}  x{s['count']:<5d} {s['bytes']:>10d} B")
    if inv["warnings"]:
        print("")

# -*- coding: utf-8 -*-
"""T-017 / tools/AssetPipeline/tex_convert.py — .tex → PNG + manifest.json

【格式依据：Kimi T-017 前置侦察结论】
::

    offset 0    1B   type    可疑的像素格式/资源类型 ID —— 语义未证实，仅记录
    offset 1    1B   flags   未证实
    offset 2    2B   unk     未证实
    offset 4    2B   width   LE
    offset 6    2B   height  LE
    offset 8    4B   ff ff ff ff   哨兵/调色板占位
    offset 12   ...  像素数据

样例：64x32 贴图文件 4108 字节 = 12 + 64*32*2 → **16bpp**。

16bpp 的两个主候选：
  * ``la88``     —— 字节序 [L, A]，A 通常 0xFF；侦察样例 '40 ff' 即 L=0x40, A=0xFF
  * ``rgba4444`` —— u16 = (R<<12)|(G<<8)|(B<<4)|A
  * ``abgr4444`` —— u16 = (A<<12)|(B<<8)|(G<<4)|R（通道序对调的变体）

【判定策略：数据驱动，不做单一硬编码假设】
对每个候选统计「alpha 通道取满值」的比例：
    la88     → 高字节 == 0xFF 的比例
    rgba4444 → 低 nibble == 0xF 的比例
    abgr4444 → 高 nibble == 0xF 的比例
取分数最高者；平局按 la88 > rgba4444 > abgr4444 的优先级（侦察样例倾向 LA88）。
判定结果 + 三个原始分数 + type 字节全部写入 manifest，可离线复核/批量修正。
纯色（低熵）贴图无法判定，显式降为 low confidence 并打 ``low_entropy`` warning。

【尺寸校验（任务卡硬要求）】
``len(data)-12 == w*h*2`` 不成立时：
  1) 依次尝试 1B / 4B per-pixel（8bpp / 32bpp）
  2) 尝试整数倍冗余（疑似 mip 链或尾部填充）→ 取第一层并记 ``extra_tail_data``
  3) 若 ``(len-12)*2 == w*h`` → 4bpp 色板，未支持，记 ``4bpp_paletted_unsupported``
  4) 都不成立 → 记 ``size_mismatch``，**不产出 PNG**，但 manifest 保留该条目

【已知不确定项（须在真实资源上复核）】
  * 像素行序：GL 纹理原点在左下。若打包器未做翻转，输出 PNG 会上下颠倒。
    本工具默认**不翻转**，用 ``--flip-y`` 切换；manifest 记录 flip_y 以便批量回归。
  * type 字节与像素格式的对应关系未证实 [待验证]；manifest 的
    ``summary.cross_type_format`` 给出 type×format 交叉表，用于反推。
  * RGBA4444 的通道排布存在两种写法，已同时实现并靠分数择一。

【用法】
::

    python tools/AssetPipeline/tex_convert.py Raw build/AssetPipeline/Textures
    python tools/AssetPipeline/tex_convert.py Raw out --format la88 --flip-y
    python tools/AssetPipeline/tex_convert.py Raw out --only sub/hud.tex   # 单文件调试

FORMATS_ALL = ("la88", "rgba4444", "abgr4444", "l8", "a8", "rgba8888", "bgra8888")

def parse_header(data: bytes) -> TexHeader:
    if len(data) < TEX_HEADER_SIZE:
        raise TexError(f"header_truncated({len(data)}B)")
    type_id, flags, unk, width, height = struct.unpack_from("<BBHHH", data, 0)
    return TexHeader(type_id, flags, unk, width, height, bytes(data[8:12]))


def infer_layout(payload_len: int, pixels: int) -> tuple[int, int, str | None]:
    """返回 (unit_bytes, levels, warning)。unit_bytes==0 表示无法处理。"""
    if pixels <= 0:
        return 0, 0, "zero_pixels"

    for unit in (2, 4, 1):                       # 先 16bpp（最常见）
        if payload_len == pixels * unit:
            return unit, 1, None

    if payload_len * 2 == pixels:
        return 0, 0, "4bpp_paletted_unsupported"

    for unit in (2, 4, 1):                       # 整数倍冗余：mip / padding
        base = pixels * unit
        if payload_len > base and payload_len % base == 0:
            return unit, payload_len // base, "extra_tail_data"

    return 0, 0, "size_mismatch"


def _candidate_scores(unit: int, payload: bytes, pixels: int) -> dict[str, float]:
    """候选 → 「alpha 取满值」比例，1.0 表示该假设下 alpha 完全合理。"""
    if unit == 2:
        v = np.frombuffer(payload, dtype="<u2", count=pixels)
        hi = (v >> 8) & 0xFF
        scores = {
            "la88": float(np.mean(hi == 0xFF)),
            "rgba4444": float(np.mean((v & 0xF) == 0xF)),
            "abgr4444": float(np.mean(((v >> 12) & 0xF) == 0xF)),
        }
        return scores
    if unit == 4:
        arr = np.frombuffer(payload, dtype=np.uint8)
        alpha_ok = float(np.mean(arr[3::4] == 0xFF))
        # 32bpp 只有 alpha 字节位置这一条判据，R/B 顺序无法从数据判定
        return {"rgba8888": alpha_ok, "bgra8888": alpha_ok}
    if unit == 1:
        return {"l8": 1.0, "a8": 0.5}
    return {}


def _pick_format(scores: dict[str, float], unit: int) -> str:
    if unit == 2:
        order = FORMATS_16BPP
    elif unit == 4:
        order = ("rgba8888", "bgra8888")
    else:
        order = ("l8", "a8")

    best = None
    for name in PREFERENCE:                      # 严格大于才替换 → 平局保留高优先级
        if name not in scores or name not in order:
            continue
        if best is None or scores[name] > scores[best]:
            best = name
    if best is None:
        raise TexError(f"no_candidate_for_unit{unit}")
    return best


def _confidence(scores: dict[str, float], chosen: str, warnings: list[str]) -> str:
    if "low_entropy" in warnings:
        return "low"
    if not scores:
        return "low"
    best = scores.get(chosen, 0.0)
    second = max((v for k, v in scores.items() if k != chosen), default=0.0)
    if best >= 0.995 and (best - second) >= 0.05:
        return "high"
    if best >= 0.90 and (best - second) >= 0.01:
        return "medium"
    return "low"

def _nib(v: np.ndarray, shift: int) -> np.ndarray:
    """取 4bit 分量并扩展到 0..255（x * 17）。"""
    return (((v >> shift) & 0xF).astype(np.uint8)) * 17


def _build_image(fmt: str, payload: bytes, width: int, height: int) -> Image.Image:
    pixels = width * height
    if fmt == "la88":
        arr = np.frombuffer(payload, dtype=np.uint8, count=pixels * 2).reshape(height, width, 2)
        return Image.fromarray(np.ascontiguousarray(arr), "LA")

    if fmt in ("rgba4444", "abgr4444"):
        v = np.frombuffer(payload, dtype="<u2", count=pixels).reshape(height, width)
        if fmt == "rgba4444":
            chans = [_nib(v, 12), _nib(v, 8), _nib(v, 4), _nib(v, 0)]
        else:                                     # abgr4444
            chans = [_nib(v, 0), _nib(v, 4), _nib(v, 8), _nib(v, 12)]
        arr = np.dstack(chans)
        return Image.fromarray(np.ascontiguousarray(arr), "RGBA")

    if fmt in ("l8", "a8"):
        arr = np.frombuffer(payload, dtype=np.uint8, count=pixels).reshape(height, width)
        return Image.fromarray(np.ascontiguousarray(arr), "L")

    if fmt in ("rgba8888", "bgra8888"):
        arr = np.frombuffer(payload, dtype=np.uint8, count=pixels * 4).reshape(height, width, 4)
        if fmt == "bgra8888":
            arr = arr[..., [2, 1, 0, 3]]
        return Image.fromarray(np.ascontiguousarray(arr), "RGBA")

    raise TexError(f"unknown_format:{fmt}")


def decode(data: bytes, *, fmt: str = "auto", flip_y: bool = False) -> tuple[Image.Image, dict]:
    """把 .tex 字节解码为 PIL Image，返回 (image, info)。失败抛 TexError。"""
    if len(data) < TEX_HEADER_SIZE:
        raise TexError(f"header_truncated({len(data)}B)")

    hdr = parse_header(data)
    width, height = hdr.width, hdr.height
    pixels = width * height
    if pixels <= 0:
        raise TexError(f"bad_dimensions {width}x{height}")

    payload = data[TEX_HEADER_SIZE:]
    unit, levels, layout_warn = infer_layout(len(payload), pixels)
    if unit == 0:
        raise TexError(layout_warn or "unknown_layout")

    payload = payload[: pixels * unit]            # 丢弃 mip / padding

    warnings: list[str] = []
    if layout_warn:
        warnings.append(layout_warn)

    if unit == 2:
        v = np.frombuffer(payload, dtype="<u2", count=pixels)
        if int(np.unique(v).size) <= 4:
            warnings.append("low_entropy")

    scores = _candidate_scores(unit, payload, pixels)

    if fmt != "auto":
        if fmt not in UNIT_OF_FORMAT:
            raise TexError(f"unknown_requested_format:{fmt}")
        if UNIT_OF_FORMAT[fmt] != unit:
            raise TexError(f"format_mismatch unit={unit} fmt={fmt}")
        chosen = fmt
    else:
        chosen = _pick_format(scores, unit)

    confidence = _confidence(scores, chosen, warnings)

    if hdr.tail != SENTINEL:
        warnings.append("header_tail_not_ff")

    image = _build_image(chosen, payload, width, height)
    if flip_y:
        image = image.transpose(FLIP_TOP_BOTTOM)

    info = {
        "format": chosen,
        "confidence": confidence,
        "scores": {k: round(v, 4) for k, v in scores.items()},
        "unit": unit,
        "levels": levels,
        "warning": " ".join(warnings) if warnings else None,
        "header": hdr.as_dict(),
        "flip_y": flip_y,
    }
    return image, info

def _rel_of(path: Path, in_path: Path) -> Path:
    try:
        return path.relative_to(in_path)
    except ValueError:
        return Path(path.name)


def collect_tex(in_path: Path) -> list[Path]:
    if in_path.is_file():
        return [in_path]
    files = [p for p in in_path.rglob("*") if p.is_file() and p.suffix.lower() == ".tex"]
    files.sort()
    return files


def convert_one(src: Path, in_path: Path, out_root: Path, *,
                default_fmt: str, fmt_map: dict[str, str],
                flip_y: bool, force: bool) -> dict:
    rel = _rel_of(src, in_path)
    rel_posix = rel.as_posix()
    out_rel = rel.with_suffix(".png")
    entry: dict = {
        "source": rel_posix,
        "output": out_rel.as_posix(),
        "requested_format": fmt_map.get(rel_posix, default_fmt),
    }

    try:
        data = src.read_bytes()
    except OSError as exc:
        entry.update({"status": "error", "error": f"read: {exc}"})
        return entry
    entry["bytes"] = len(data)

    try:
        image, info = decode(data, fmt=entry["requested_format"], flip_y=flip_y)
    except TexError as exc:
        entry.update({"status": "error", "error": str(exc)})
        return entry
    except Exception as exc:                       # noqa: BLE001 - 批量容错
        entry.update({"status": "error", "error": f"{type(exc).__name__}: {exc}"})
        return entry

    entry.update({k: info[k] for k in
                  ("format", "confidence", "scores", "unit", "levels",
                   "warning", "header", "flip_y")})

    dst = out_root / out_rel
    if dst.exists() and not force:
        entry.update({"status": "skipped", "reason": "exists"})
        return entry

    try:
        dst.parent.mkdir(parents=True, exist_ok=True)
        image.save(dst, format="PNG", optimize=True)
    except OSError as exc:
        entry.update({"status": "error", "error": f"write: {exc}"})
        return entry

    entry["status"] = "ok"
    entry["png_bytes"] = dst.stat().st_size
    return entry


def _summarize(entries: list[dict], expect_count: int | None) -> dict:
    by_status = Counter(e["status"] for e in entries)
    by_format = Counter(e.get("format", "<none>") for e in entries)
    by_conf = Counter(e.get("confidence", "<none>") for e in entries)
    warnings = Counter()
    for e in entries:
        w = e.get("warning")
        if w:
            for token in str(w).split():
                warnings[token] += 1

    cross: dict[str, Counter] = defaultdict(Counter)
    for e in entries:
        header = e.get("header") or {}
        cross[str(header.get("type", "?"))][e.get("format", "<none>")] += 1

    summary = {
        "total": len(entries),
        "by_status": dict(by_status),
        "by_format": dict(by_format),
        "by_confidence": dict(by_conf),
        "warnings": dict(warnings),
        "cross_type_format": {k: dict(v) for k, v in sorted(cross.items())},
    }
    if expect_count is not None:
        summary["expected_total"] = expect_count
        summary["matches_expected"] = len(entries) == expect_count
        if len(entries) != expect_count:
            summary.setdefault("notes", []).append(
                f"条目数 {len(entries)} != 期望 {expect_count}，需核对 Raw/ 是否全量解包")
    return summary

def convert_tree(in_path: Path, out_root: Path, *,

def _make_tex(type_id: int, flags: int, unk: int, width: int, height: int,

def selftest(verbose: bool = True) -> int:
    failures: list[str] = []
    checks = 0

    def check(cond: bool, msg: str) -> None:
        nonlocal checks
        checks += 1
        if cond:
            if verbose:
                print(f"[selftest] ok   {msg}")
        else:
            failures.append(msg)
            print(f"[selftest] FAIL {msg}")

    tmp = Path(tempfile.mkdtemp(prefix="fnpib_tex_selftest_"))
    if verbose:
        print(f"[selftest] 临时目录 {tmp}")

    # ---- 样例 1：LA88（alpha 全 0xFF，L 为渐变）----
    w, h = 64, 32
    lum = ((np.tile(np.arange(w, dtype=np.uint16), (h, 1)) * 3) % 256).astype(np.uint8)
    alpha = np.full((h, w), 0xFF, np.uint8)
    la_src = np.dstack([lum, alpha])
    data_la = _make_tex(0x01, 0, 0, w, h, la_src.tobytes())

    img, info = decode(data_la)
    check(info["header"]["width"] == w and info["header"]["height"] == h,
          "la88 头部宽高正确")
    check(info["format"] == "la88",
          f"la88 判定（got {info['format']}, scores={info['scores']}）")
    check(np.array_equal(np.array(img), la_src), "la88 像素往返一致")
    check(info["unit"] == 2 and info["levels"] == 1, "la88 unit/levels 正确")

    # ---- 样例 2：RGBA4444（alpha nibble 满，高字节非 0xFF → 不应判成 la88）----
    w2, h2 = 32, 16
    r = np.full((h2, w2), 2, np.uint16)   # 2  → 34
    g = np.full((h2, w2), 4, np.uint16)   # 4  → 68
    b = np.full((h2, w2), 8, np.uint16)   # 8  → 136
    a = np.full((h2, w2), 15, np.uint16)  # 15 → 255
    packed = ((r << 12) | (g << 8) | (b << 4) | a).astype("<u2")
    data_rgba = _make_tex(0x02, 0, 0, w2, h2, packed.tobytes())

    img2, info2 = decode(data_rgba)
    check(info2["format"] == "rgba4444",
          f"rgba4444 判定（got {info2['format']}, scores={info2['scores']}）")
    expect2 = np.dstack([
        np.full((h2, w2), 34, np.uint8),
        np.full((h2, w2), 68, np.uint8),
        np.full((h2, w2), 136, np.uint8),
        np.full((h2, w2), 255, np.uint8),
    ])
    check(np.array_equal(np.array(img2), expect2), "rgba4444 像素往返一致")

    # ---- 样例 3：8bpp 灰度 ----
    w3, h3 = 16, 16
    gray = np.arange(w3 * h3, dtype=np.uint8).reshape(h3, w3)
    data_l8 = _make_tex(0x03, 0, 0, w3, h3, gray.tobytes())
    img3, info3 = decode(data_l8)
    check(info3["format"] == "l8" and np.array_equal(np.array(img3), gray),
          "8bpp 灰度 判定 + 往返一致")

    # ---- 样例 4：尺寸校验失败（声明 64x32 但只给 100 字节）----
    bad = _make_tex(0, 0, 0, 64, 32, b"\x00" * 100)
    try:
        decode(bad)
        check(False, "size_mismatch 应抛 TexError")
    except TexError as exc:
        check("size_mismatch" in str(exc), f"size_mismatch 报错（{exc}）")

          f"冗余数据被识别（levels={info5['levels']}, warning={info5['warning']}）")
    check(np.array(img5).shape == (8, 8, 2), "冗余数据只取第一层")

    # ---- 样例 6：flip_y ----
    img_f, _ = decode(data_la, flip_y=True)
    check(np.array_equal(np.array(img_f)[0], la_src[-1]), "flip_y 生效（首行=原末行）")

    # ---- 样例 7：强制格式 ----
    img_forced, info_forced = decode(data_la, fmt="la88")
    check(info_forced["format"] == "la88", "强制 --format la88 生效")
    try:
        decode(data_la, fmt="rgba8888")
        check(False, "格式与 bpp 不符应抛 TexError")
    except TexError as exc:
        check("format_mismatch" in str(exc), f"format_mismatch 报错（{exc}）")

    out = tmp / "Out"

    (raw / "sub" / "b.tex").write_bytes(data_rgba)

def parse_xml(data: bytes) -> tuple[ET.Element | None, str | None, str | None]:
    """返回 (root, error, encoding_tag)。"""
    try:
        return ET.fromstring(data), None, "as-declared"
    except ET.ParseError as first_err:
        try:
            text = data.decode("gb18030")
        except UnicodeDecodeError as dec_err:
            return None, (f"ParseError(as-declared): {first_err} | "
                          f"gb18030 decode failed: {dec_err}"), None
        try:
            root = ET.fromstring(text)
            return root, None, "gb18030-fallback"
        except ET.ParseError as second_err:
            return None, (f"ParseError(as-declared): {first_err} | "
                          f"ParseError(gb18030): {second_err}"), None


def elem_to_obj(el: ET.Element, depth_limit: int | None, depth: int = 0) -> dict:
    obj: dict = {"tag": el.tag}
    if el.attrib:
        obj["attrs"] = dict(el.attrib)

    text = (el.text or "").strip()
    if text:
        obj["text"] = text

    kids = list(el)
    if kids:
        if depth_limit is not None and depth >= depth_limit:
            obj["children_omitted"] = len(kids)
        else:
            obj["children"] = [elem_to_obj(k, depth_limit, depth + 1) for k in kids]
    return obj


def walk_stats(root: ET.Element) -> tuple[int, int, set[str]]:
    nodes = 0
    max_depth = 0
    attrs: set[str] = set()
    stack: list[tuple[ET.Element, int]] = [(root, 0)]
    while stack:
        node, depth = stack.pop()
        nodes += 1
        max_depth = max(max_depth, depth)
        attrs.update(node.attrib.keys())
        stack.extend((child, depth + 1) for child in node)
    return nodes, max_depth, attrs

                       encoding="utf-8")
    except OSError as exc:
        entry.update({"status": "error", "error": f"write: {exc}"})
        return entry

    entry["status"] = "ok"
    entry["json_bytes"] = dst.stat().st_size
    return entry

## 2. 容器布局：候选假设

### 假设 A —— 通用 chunk 流（推荐优先验证）

```
offset  size  field
0x00    4     magic  = "HBR0"
0x04    4     version        (u32 LE)
0x08    4     declared_len   (u32 LE)   ← 语义待定，见 §3
0x0C    ...   chunk stream
```

chunk 头候选：

```
[id: u8][len: u32 LE][payload: len bytes]
```

### 假设 B —— 对齐的 chunk 头

```
[id: u8][pad: 3][len: u32 LE][payload: len bytes]        // 8 字节头，4 对齐
```

很多自研引擎为了让 payload 4 字节对齐会插 3 字节 padding。
判别方法：按 A 走一遍，若 `pos` 落点普遍是 `4k+3` 且后续字节恒为 `0x00`，则 B 更可能。

### 假设 C —— 带计数的 chunk 头

```
[id: u8][len: u32 LE][count: u32 LE][payload]
```

判别方法：若某 chunk 的 payload 长度恰好能被 `count` 整除，且 `count` 与
「子块数量 / 顶点数」强相关，则 C 更可能。

> 侦察描述里 `02/01/00` 被记为「层级样例」，存在两种互斥解释：
> **(a) 块类型 ID**（同一 id 在不同文件中反复出现、内容类型稳定）；
> **(b) 嵌套层级深度**（id 沿树严格单调）。
> 判别方法见 §4 脚本的 `id_hist` 统计：若 `id` 取值只有 0/1/2 三种且分布极偏，支持 (b)；
> 若出现更多取值或每种 id 后面接的结构长度分布完全不同，支持 (a)。

## 3. 长度字段语义（必须先闭合的问题）

`0x08` 的 4 字节到底是：

| 假设 | 含义 | 自洽判据 |
|---|---|---|
| L1 | 整个文件剩余字节数 | `declared + 12 == filesize` |
| L2 | 整个文件字节数 | `declared == filesize` |
| L3 | 第一个顶层 chunk 的 payload 长度 | `0x0C + 5 + declared == filesize`（假设 A） |
| L4 | 顶层 chunk 数量 | 值应很小（个位数），且远小于 filesize |

`extract.py` 的清点报告已经把这个判定做成了统计：

```python
import struct
from collections import Counter
from pathlib import Path

MAGIC = b"HBR0"

def probe(path: Path):
    d = path.read_bytes()
    if d[:4] != MAGIC:
        return {"file": path.name, "error": "bad_magic"}
    ver, declared = struct.unpack_from("<II", d, 4)

    result = {
        "file": path.name,
        "size": len(d),
        "version": ver,
        "declared": declared,
        "L1_declared_plus_12_eq_size": declared + 12 == len(d),
        "L2_declared_eq_size": declared == len(d),
        "L3_first_chunk": None,
        "chunks": [],
    }

    # 按假设 A 走一遍 chunk 流
    pos, ids = 0x0C, Counter()
    while pos + 5 <= len(d):
        cid = d[pos]
        ln = struct.unpack_from("<I", d, pos + 1)[0]
        if pos + 5 + ln > len(d):
            result["chunks"].append({"offset": pos, "id": cid, "len": ln,
                                     "error": "overruns_file"})
            break
        result["chunks"].append({"offset": pos, "id": cid, "len": ln})
        ids[cid] += 1
        pos += 5 + ln

    result["L3_first_chunk"] = (0x0C + 5 + result["chunks"][0]["len"] == len(d)
                               if result["chunks"] else None)
    result["tail_pos"] = pos
    result["tail_slack"] = len(d) - pos
    result["id_hist"] = dict(ids)
    return result

root = Path("build/AssetPipeline/Raw")
rows = [probe(p) for p in sorted(root.rglob("*.mmd"))]
ok = [r for r in rows if "error" not in r]
print(f"probed {len(ok)}/{len(rows)}")
print("version:", Counter(r["version"] for r in ok))
for key in ("L1_declared_plus_12_eq_size", "L2_declared_eq_size", "L3_first_chunk"):
    print(key, Counter(r[key] for r in ok))
print("tail_slack:", Counter(r["tail_slack"] for r in ok))
all_ids = Counter()
for r in ok:
    all_ids.update(r["id_hist"])
print("global id_hist:", dict(sorted(all_ids.items())))
```

- `tail_slack == 0` 的文件占多数 → 假设 A 的 chunk 头 `[u8 id][u32 len]` 正确；
- `tail_slack` 恒为某个小常数（1..3）→ 假设 B 的对齐 padding 成立；
- `L1/L2/L3` 三选一几乎全中 → §3 的长度语义闭合；

设计原则：
1. **解析器只负责切块，不解释语义**。语义（顶点布局、材质字段）另立 `mmd_semantic.py`，
   避免格式一改全盘推翻。
2. **`strict` 开关**：构建期 `strict=True`，遇到越界立刻抛错；实验期 `strict=False`
   尽力解析并收集 `validate()` 问题，便于对着 60 个文件做统计归纳。
3. **不猜版本**：`version` 只记录不分支；等确认是否会变（当前 1.0.4 版本大概率全同）。
4. 输出中间态用 JSON（`{"file": ..., "chunks": [...]}`）便于人工 diff，
   二进制 payload 不 dump。
