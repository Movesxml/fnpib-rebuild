## 设计说明

**核心结论**：原版并非“每个挑战水果维护独立计数器”，而是每个条件对象有两个字段：

- `condition+0x4c`：XML `specific_type` 的静态类型哈希；
- `condition+0x50`：运行期水果过滤值。条件激活时（`FUN_0005ef64`），若 `+0x4c == DAT_005f949c`（challengeFruit 占位符），则用 `controller+0x88` 对应的 `FruitInfo+0x250` 名称哈希覆盖 `+0x50`；否则保持 XML 原值。`+0x50 == 0` 对 CONSECUTIVE 是通配，不是禁用。

因此重建方案：

1. `ChallengeCondition` 增加运行期字段 `selectedFruitName`（代替 `+0x50` 的只读投影），**不修改 `definition`**；`activate(selectedFruitName:)` 每次选择/恢复时重设，保证下一回合可重绑。
2. 新增 `resolvedStatisticKey(_:)`：
   - `challengeFruit` 占位符 → 返回 `selectedFruitName`（小写，nil/空为通配）；
   - 其他 `specific_type` → 原样小写返回。
   STATS 与 STATS_COMPARE 两个键均经过它解析，从而读取 `StatsRecorder` 中真实水果名键，不写 `challengefruit` 虚构计数器。
3. CONSECUTIVE 的 `recordConsecutive` 改用 `resolvedStatisticKey(definition.specificType)` 作为过滤值：占位符换选中水果名，nil/空不过滤；非占位保持原类型过滤。
4. `ChallengeConditionsRuntime.select` 增加默认参数 `selectedFruitName: String? = nil`，对每个选中条件调用 `activate`。
5. `ChallengeModeRuntime` 初始化与 `updateSettlement` 的下一回合入口显式接收 `selectedFruitName`，并在 `prepareRound`/`beginRound` 中传给条件表。
6. `GameWorld` **无需改动**：它已用 `services.stats` 真实 `StatsRecorder` 构造 `ChallengeEvaluationContext`；“challengeFruit → 真实水果名”的解析发生在条件求值层，不另建计数副本，也不接 App 的水果选择逻辑。

以下为草案，仅写入 `build/staging/T-040/`。

public struct ChallengeCondition: Sendable, Equatable {
    public var definition: ChallengeConditionDefinition
    public private(set) var current: ChallengeOutcome
    public private(set) var timeRemaining: Float
    /// 原 condition+0x50 的只读投影：当前条件绑定的水果过滤名。
    /// XML 解析后等于 `specific_type`；激活 challengeFruit 条件时改写为选中水果名。
    /// nil/空表示通配（原 +0x50 == 0），STATS 读到空键返回 0，CONSECUTIVE 不过滤。
    public private(set) var selectedFruitName: String?

    public init(_ definition: ChallengeConditionDefinition) {
        self.definition = definition
        current = .pending
        timeRemaining = definition.time
        selectedFruitName = nil
    }

    /// 所有类型先清当前结果；TIME 的 0x5f624 另外恢复完整倒计时。
    public mutating func reset() {
        current = .pending
        if definition.kind == .time {
            timeRemaining = definition.time
        }
    }

    /// FUN_0005ef64 的激活语义：先清当前结果，再按 XML 类型与当前选中水果
    /// 重设 condition+0x50 等价过滤值。**不修改 definition**，因此同一份原始
    /// 条件表可在下一回合重新绑定。
    public mutating func activate(selectedFruitName: String?) {
        reset()
        let normalized = selectedFruitName?.lowercased()
        self.selectedFruitName = (normalized?.isEmpty == true) ? nil : normalized
    }

    /// 虚表 +0x44。SCORE / STATS 等只返回 target <= current；
    /// STATS_COMPARE 的 0x5f0b8 则在比较后自行更新 current 并返回是否变化。
    @discardableResult
    public mutating func check(context: ChallengeEvaluationContext) -> Bool {
        if definition.kind == .statsCompare {
            let old = current
            let lhs = context.statistic(resolvedStatisticKey(definition.specificType))
            let rhs = context.statistic(resolvedStatisticKey(definition.specificType2))
            current = definition.comparison.evaluate(lhs, rhs) ? definition.result : .pending
            return current != old
        }
        return currentValue(context: context) >= definition.total
    }

    /// 虚表 +0x48。基类、SCORE、STATS、CONSECUTIVE、TIME 使用 0x5f5c0；
    /// STATS_COMPARE 覆盖为 0x5f0b8，比较后同样更新 current。
    @discardableResult
    public mutating func updateResult(context: ChallengeEvaluationContext) -> Bool {
        if definition.kind == .statsCompare {
            return check(context: context)
        }
        let old = current
        current = check(context: context) ? definition.result : .pending
        return current != old
    }

    /// 虚表 +0x14：回合结束求值。
    @discardableResult
    public mutating func evaluateEnd(context: ChallengeEvaluationContext) -> Bool {
        switch definition.kind {
        case .base, .score, .stats:
            return updateResult(context: context)
        case .statsCompare:
            return definition.evaluateOnEnd ? updateResult(context: context) : false
        case .consecutive, .time:
            return false
        }
    }

    /// 虚表 +0x18：即时求值入口。
    @discardableResult
    public mutating func evaluateImmediate(context: ChallengeEvaluationContext) -> Bool {
        switch definition.kind {
        case .base, .score, .stats:
            return updateResult(context: context)
        case .statsCompare:
            return definition.evaluateOnEnd ? false : updateResult(context: context)
        case .consecutive, .time:
            return false
        }
    }

    /// CONSECUTIVE 的 0x5f21c：可选类型匹配后，按本刀数量与 total 比较。
    /// `condition+0x50` 为零/空表示通配；challengeFruit 占位符已由 activate
    /// 绑定为选中水果名，非占位条件保持 XML 原类型。
    @discardableResult
    public mutating func recordConsecutive(count: Int32,
                                           specificType: String?) -> Bool {
        guard definition.kind == .consecutive else { return false }
        let required = resolvedStatisticKey(definition.specificType)
        if let required, required != specificType?.lowercased() {
            return false
        }
        let old = current
        current = count >= definition.total ? definition.result : .pending
        return current != old
    }

    /// 虚表 +0x28；只有 TIME 的 0x5f1cc 消耗 dt。
    @discardableResult
    public mutating func update(dt: Float,
                                context: ChallengeEvaluationContext) -> Bool {
        guard definition.kind == .time else { return false }
        let old = current
        timeRemaining -= dt
        if timeRemaining <= 0, context.timerCompletionAllowed {
            current = definition.result
        } else {
            current = .pending
        }
        return current != old
    }

    /// 原 condition+0x50 的取值投影：challengeFruit 占位符取激活时绑定的
    /// 真实水果名，其他类型保持 XML `specific_type`（统一小写以匹配
    /// StatsRecorder.challengeValues() 的键）。nil 表示通配/空键。
    private func resolvedStatisticKey(_ raw: String?) -> String? {
        guard let raw, !raw.isEmpty else { return nil }
        if raw.lowercased() == ChallengeConditionDefinition.challengeFruitPlaceholder {
            return selectedFruitName
        }
        return raw.lowercased()
    }

    private func currentValue(context: ChallengeEvaluationContext) -> Int32 {
        switch definition.kind {
        case .score:
            return context.score
        case .stats:
            return context.statistic(resolvedStatisticKey(definition.specificType))
        case .base, .consecutive, .statsCompare, .time:
            return 0
        }
    }
}

            switch node.attrs["comparison"]?.uppercased() {
            case nil, "EQUAL": comparison = .equal
            case "NOT_EQUAL": comparison = .notEqual
            case "GREATER": comparison = .greater
            default: comparison = .unsupported
            }

                name: name,

/// DAT_005f94b4 的值语义重建：完整条件表 + 当前挑战的保序激活列表。
public struct ChallengeConditionsRuntime: Sendable, Equatable {
    private var conditions: [String: ChallengeCondition]
    private var activeNames: [String] = []

    public private(set) var result: ChallengeOutcome = .pending
    /// 原聚合器 +0x24；选择多个 TIME 时后项覆盖前项，缺省为 0。
    public private(set) var timerDuration: Float = 0

    public init(definitions: [ChallengeConditionDefinition]) {
        var conditions: [String: ChallengeCondition] = [:]
        for definition in definitions {
            // 原加载表按名称哈希索引；重复名称由后项替换，原始文档仍可由 Loader 全量审计。
            conditions[definition.name.lowercased()] = ChallengeCondition(definition)
        }
        self.conditions = conditions
    }

    public var activeConditions: [ChallengeCondition] {
        activeNames.compactMap { conditions[$0] }
    }

    public func condition(named name: String) -> ChallengeCondition? {
        conditions[name.lowercased()]
    }

    /// FUN_0005ed54：只重置并入列真实存在的条件，未知名称静默跳过。
    /// `selectedFruitName` 对应 controller+0x88 的当前选择；在 activate 阶段
    /// 写入 challengeFruit 条件的过滤值，而不是永久改写共享 definition。
    public mutating func select(conditionNames names: [String],
                                selectedFruitName: String? = nil) {
        result = .pending
        timerDuration = 0
        activeNames.removeAll(keepingCapacity: false)
        for name in names {
            let key = name.lowercased()
            guard var condition = conditions[key] else { continue }
            condition.activate(selectedFruitName: selectedFruitName)
            if condition.definition.kind == .time {
                timerDuration = condition.timeRemaining
            }
            conditions[key] = condition
            activeNames.append(key)
        }
    }

    public mutating func update(dt: Float, context: ChallengeEvaluationContext) {
        guard result == .pending else { return }
        for name in activeNames {
            conditions[name]?.update(dt: dt, context: context)
        }
        // 0x5ec34 随后以 end=false 调 0x5ec7c；有 PASS 条件时结果保持 pending，
        // 不在这里把即时 PASS 错当成回合完成。
    }

    public mutating func evaluateImmediate(context: ChallengeEvaluationContext) {
        guard result == .pending else { return }
        for name in activeNames {
            conditions[name]?.evaluateImmediate(context: context)
        }
    }

    public mutating func recordConsecutive(count: Int32, specificType: String?) {
        guard result == .pending else { return }
        for name in activeNames {
            conditions[name]?.recordConsecutive(count: count, specificType: specificType)
        }
    }

    /// FUN_0005ecf4 先筛 definition.result == PASS，再调虚表 +0x30；
    /// 只有 STATS_COMPARE 在该槽执行比较，FAIL 定义不会走这个分发入口。
    public mutating func checkComparisons(context: ChallengeEvaluationContext) {
        guard result == .pending else { return }
        for name in activeNames {
            guard let condition = conditions[name],
                  condition.definition.result == .pass,
                  condition.definition.kind == .statsCompare else { continue }
            conditions[name]?.check(context: context)
        }
    }

    /// FUN_0005edc8 + FUN_0005ec7c(end=true)：结束求值只执行一次。
    @discardableResult
    public mutating func finish(context: ChallengeEvaluationContext) -> ChallengeOutcome {
        guard result == .pending else { return result }
        for name in activeNames {
            conditions[name]?.evaluateEnd(context: context)
        }

        // 第一趟：结束型 FAIL 优先，顺序与原链表一致。
        for name in activeNames {
            guard let condition = conditions[name] else { continue }
            if condition.definition.evaluateOnEnd, condition.current == .fail {
                result = .fail
                return result
            }
        }

        // 第二趟：只有结束型 PASS 参与通过判定；即时 PASS 不直接结束回合。
        var aggregate: ChallengeOutcome = .pending
        for name in activeNames {
            guard let condition = conditions[name] else { continue }
            guard condition.definition.result == .pass else { continue }
            guard condition.definition.evaluateOnEnd, condition.current == .pass else {
                aggregate = .pending
                break
            }
            aggregate = .pass
        }

        // 0x5edc8 的空结果兜底：先置 PASS；存在任一 FAIL 定义则改置 FAIL。
        if aggregate == .pending {
            aggregate = .pass
            for name in activeNames {
                if conditions[name]?.definition.result == .fail {
                    aggregate = .fail
                    break
                }
            }
        }
        result = aggregate
        return result
    }
}

        // FUN_0005fd7c 在 TIME 更新后同帧调用 FUN_0005ecf4。
        conditions.checkComparisons(context: frame.evaluation)

        let timedOrConditionEnd = (endsOnTimer && countdown <= 0)
            || conditions.result != .pending
        let shouldFinish: Bool
        if !wavesEnded {
            shouldFinish = timedOrConditionEnd
        } else if isBonusRound
                    || frame.waveCount < challenge.numWaves
                    || frame.fruitEntityCount != 0 {
            shouldFinish = timedOrConditionEnd
        } else {
            shouldFinish = true
        }
        if shouldFinish {
            finishRound(context: frame.evaluation)
            wavesEnded = false
        }
    }

    /// FUN_0005fd7c 状态 3：只响应结算控制器 state==7 的真实子状态。
    /// 新回合配置与水果选择均由外部选择器显式提供；缺失时不推进、不虚构选择。
    /// bonusIntro 之前同样先完成外部回合 setup，避免沿用上一轮已冻结的聚合结果。
    public mutating func updateSettlement(controllerState: Int32,
                                          substate: Int32,
                                          nextChallenge: ChallengeDefinition? = nil,
                                          nextSelectedFruitName: String? = nil) {
        guard phase == .settling, controllerState == 7 else { return }
        switch ChallengeSettlementSubstate(rawValue: substate) ?? .other {
        case .bonus:
            guard let nextChallenge else { return }
            prepareRound(challenge: nextChallenge,
                         selectedFruitName: nextSelectedFruitName)
            startBonusIntro()
        case .nextRound:
            guard let nextChallenge else { return }
            beginRound(challenge: nextChallenge,
                       selectedFruitName: nextSelectedFruitName)
            events.append(.nextRound)
        case .finished:
            cleanup()
        case .other:
            phase = .suspended
        }
    }

    /// FUN_0005ffa8：状态 1 只由 bonusIntro 的真实播放完成回调推进；
    /// 条件与倒计时已在此前的外部回合 setup 中复位，这里不再次结算或选择。
    public mutating func completeBonusIntro() {
        guard phase == .bonusIntro else { return }
        phase = .running
        events.append(.playAnimation(target: "Timer", name: "animate_in"))
    }

    private mutating func prepareRound(challenge nextChallenge: ChallengeDefinition,
                                       selectedFruitName: String?) {
        challenge = nextChallenge
        self.selectedFruitName = selectedFruitName
        countdown = 0
        wavesEnded = false
        conditions.select(conditionNames: challenge.conditionNames,
                          selectedFruitName: selectedFruitName)
        countdown = conditions.timerDuration != 0
            ? conditions.timerDuration : challenge.timeLimit
    }

    private mutating func beginRound(challenge nextChallenge: ChallengeDefinition,
                                     selectedFruitName: String?) {
        prepareRound(challenge: nextChallenge, selectedFruitName: selectedFruitName)
        phase = .running
        events.append(.playAnimation(target: "Timer", name: "animate_in"))
    }

    private mutating func startBonusIntro() {
        phase = .bonusIntro
        events.append(.playAnimation(target: "bonusIntro", name: "play"))
        events.append(.playMusic("Music-bonus"))
    }

    private mutating func cleanup() {
        phase = .inactive
        events.append(.setVisible(target: "inGameChallengeHud", false))
        events.append(.cleanedUp)
    }

    private mutating func finishRound(context: ChallengeEvaluationContext) {
        events.append(.playAnimation(target: "inGameChallengeHud", name: "animate_out"))
        let outcome = conditions.finish(context: context)
        guard outcome != .pending else { return }

        roundsPlayed += 1
        if outcome == .pass { roundsPassed += 1 }
        roundResults.append(outcome)
        events.append(.roundFinished(outcome))
        if outcome == .fail {
            events.append(.playSound("gank"))
        }
        if isBonusRound {
            events.append(.playMusic("Music-bonus-end"))
        }
        phase = .settling
        events.append(.playAnimation(target: "instructionsText", name: "instructions_out"))
        events.append(.setVisible(target: "Timer", false))
        events.append(.setVisible(target: "challengeMods", false))
    }
}
