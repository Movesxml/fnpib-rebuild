## 设计说明

**本卡的核心约束是“不猜测”**：任务卡要求接入的是**已存在的** `ChallengeCatalogLoader` / `ChallengeConditionLoader` / `OriginalContent` / `OriginalContentBundle`，但这四者的源代码没有出现在我的上下文里（DECISIONS 只记录到 T-020，挑战模块是 T-021~T-040 之间落盘的）。因此草案采取三个策略：

1. **把“对接既有类型”压缩到极少数点位**，集中在一处并逐条标注 `[需核对]`，Kimi 核对后只需改这几行；
2. **把“不猜测”落实到测试**：条件数量、名称、tier/group/challenge 顺序都不硬编码，而是**从真实 JSON 文档现场派生期望值**再比对——这样既覆盖验收项，又不会把猜测写成黄金值；
3. **`coinsMax = 0` 显式成常量并写明证据链**；`chance` / `noCritical` 直接透传 `FruitInfo` 字段，缺省 chance 保持 0（附测试守住“不得改成非零”）。

不接 `GameWorld`、不选 tier/challenge、不替换随机源、不改任何生产文件。

### 3.1 `coinsMax` 为什么是 0

- 任务卡 T-041 已核实：`coinsMax` 取 **`FruitInfo` 有效默认值 0**。
- 依据：`FruitInfo` 由 `fruitlist.xml` 解码，原始 XML 无 coin 相关字段，因此有效运行时值就是解码默认值 `0`。

- 落点：`ChallengeFruitWeightMap.coinsMaxDefault = 0`，测试 `testWeightMapUsesRuntimeFruitFields` 守住。
- `[待验证]` findings 未收录 T-014 的 `FruitInfo` 默认值表原文；若该表存在 coin 字段，须先登记 findings 再改此常量。

//
//  ChallengeResources.swift
//  FNPBCore (T-041 草案)
//
//  把 Content/ch1challengeList.json 与 Content/challengeConditions.json
//  通过既有 ChallengeCatalogLoader / ChallengeConditionLoader 接入 OriginalContent。
//
//  范围：只做「读取 + 解码 + 组装 + 错误」。不接 GameWorld，不选择 tier/challenge，不碰随机源。
//
//  [需核对] 本文件对既有 API 的假设（签名不同时只改 load(locator:) 的两个默认闭包）：
//    1. ChallengeCatalogLoader.decode(_ data: Data) throws -> ChallengeCatalog
//    2. ChallengeConditionLoader.decode(_ data: Data) throws -> ChallengeConditions
//

import Foundation

// MARK: - 错误

/// 挑战资源加载错误。资源缺失必须是**显式错误**，不允许静默降级为空内容。
public enum ChallengeResourceError: LocalizedError, Equatable {
    /// 在给定位置找不到资源文件
    case missingResource(name: String, searchedIn: String)
    /// 文件存在但不可读 / 不可解码为 Data
    case unreadableResource(name: String, reason: String)

    public var errorDescription: String? {
        switch self {
        case let .missingResource(name, searchedIn):
            return "缺少挑战资源 \(name)：未在 \(searchedIn) 中找到。"
                + "请确认 Content 目录已加入 App Bundle 的 Copy Bundle Resources。"
        case let .unreadableResource(name, reason):
            return "无法读取挑战资源 \(name)：\(reason)"
        }
    }
}

// MARK: - 资源定位

/// 挑战资源定位器。
///
/// 支持两种来源：
/// - `.bundle(Bundle)`：App 运行时
/// - `.directory(URL)`：XCTest 直接读仓库内 `Content/`，无需构造成 Bundle
public struct ChallengeResourceLocator {

    public enum Location {
        case bundle(Bundle)
        case directory(URL)
    }

    /// 任务卡给定的真实资源文件名（不含扩展名）
    public static let catalogResourceName = "ch1challengeList"
    public static let conditionsResourceName = "challengeConditions"
    public static let resourceExtension = "json"

    public let location: Location

    public init(location: Location) {
        self.location = location
    }

    public init(bundle: Bundle) {
        self.init(location: .bundle(bundle))
    }

    public init(directory: URL) {
        self.init(location: .directory(directory))
    }

    public func catalogData() throws -> Data {
        try data(named: Self.catalogResourceName)
    }

    public func conditionsData() throws -> Data {
        try data(named: Self.conditionsResourceName)
    }

    public func data(named name: String) throws -> Data {
        let fileName = "\(name).\(Self.resourceExtension)"
        let url: URL

        switch location {
        case let .bundle(bundle):
            guard let found = bundle.url(forResource: name, withExtension: Self.resourceExtension) else {
                let where_ = "bundle \(bundle.bundleIdentifier ?? bundle.bundlePath)"
                throw ChallengeResourceError.missingResource(name: fileName, searchedIn: where_)
            }
            url = found

        case let .directory(directory):
            let candidate = directory.appendingPathComponent(fileName, isDirectory: false)
            guard FileManager.default.fileExists(atPath: candidate.path) else {
                throw ChallengeResourceError.missingResource(name: fileName, searchedIn: directory.path)
            }
            url = candidate
        }

        do {
            return try Data(contentsOf: url)
        } catch {
            throw ChallengeResourceError.unreadableResource(
                name: fileName,
                reason: String(describing: error)
            )
        }
    }
}

// MARK: - 条件文本占位符

/// 挑战条件文本中的水果占位符。
///
/// 只提供纯函数：本卡不决定 UI 如何渲染，也不决定占位符在哪种语言下替换成什么。
public enum ChallengeConditionText {

    /// [待验证] 真实字形需由 `Content/ch1challengeList.json` / `challengeConditions.json`
    /// 全文检索确认（测试 `testChallengeFruitPlaceholderAppearsInRealDocumentsAndIsReplaced` 会强制校验）。
    public static let challengeFruitPlaceholder = "{challengeFruit}"

    public static func containsFruitPlaceholder(_ text: String) -> Bool {
        text.contains(challengeFruitPlaceholder)
    }

    public static func replacingFruitPlaceholder(in text: String, with fruitName: String) -> String {
        text.replacingOccurrences(of: challengeFruitPlaceholder, with: fruitName)
    }
}

// MARK: - 装配结果

/// 挑战资源装配值：只承载「已解码的目录 + 已解码的条件」，不含任何选择/解锁决策。
public struct OriginalContentChallengeResources {
    public let catalog: ChallengeCatalog
    public let conditions: ChallengeConditions

    public init(catalog: ChallengeCatalog, conditions: ChallengeConditions) {
        self.catalog = catalog
        self.conditions = conditions
    }
}

// MARK: - 装载接线

public enum OriginalContentChallengeLoader {

    /// 生产路径：使用既有 Loader 的默认入口。
    public static func load(
        locator: ChallengeResourceLocator
    ) throws -> OriginalContentChallengeResources {
        try load(
            locator: locator,
            decodeCatalog: { try ChallengeCatalogLoader.decode($0) },        // [需核对] 既有入口
            decodeConditions: { try ChallengeConditionLoader.decode($0) }    // [需核对] 既有入口
        )
    }

    /// 可注入版本：仅供测试或将来替换解码实现，不改变语义。
    static func load(
        locator: ChallengeResourceLocator,
        decodeCatalog: (Data) throws -> ChallengeCatalog,
        decodeConditions: (Data) throws -> ChallengeConditions
    ) throws -> OriginalContentChallengeResources {
        // 先取齐两份 Data：缺失资源时优先报「缺文件」，而不是报解码错误。
        let catalogData = try locator.catalogData()
        let conditionsData = try locator.conditionsData()

        return OriginalContentChallengeResources(
            catalog: try decodeCatalog(catalogData),
            conditions: try decodeConditions(conditionsData)
        )
    }
}

//
//  ChallengeFruitWeights.swift
//  FNPBCore (T-041 草案)
//
//  从真实 FruitInfo 的**有效运行时字段**构造挑战水果权重表。
//  纯函数：无副作用、不读存档、不读随机源、不做单位换算。
//
//  [需核对] 假设 FruitInfo.id: Int32、FruitInfo.chance 与 ChallengeFruitWeightEntry.chance 同型、
//           ChallengeFruitWeightEntry(chance:noCritical:coinsMax:) 在模块内可见。
//

import Foundation

public enum ChallengeFruitWeightMap {

    /// 挑战权重条目的 coinsMax 取「已核实的 FruitInfo 有效默认值」0。
    ///
    /// 证据（见 build/staging/T-041/README.md §3.1）：
    /// - FruitInfo 由 fruitlist.xml 解码，原始 XML 无 coin 相关字段 → 有效运行时值即默认 0。
    /// - 本草案不发明其它值（不按 chance 推导、不默认 1、不读存档）。
    /// [待验证] T-014 落盘的 FruitInfo 默认值表原文需 Kimi 复核。
    public static let coinsMaxDefault: Int32 = 0

    /// 构造权重表。
    ///
    /// 重复 id 的处理：**先到先得**（保留首次出现），不静默覆盖；
    /// 需要诊断时调用 `duplicateIdentifiers(in:)`。
    public static func make(from fruits: [FruitInfo]) -> [Int32: ChallengeFruitWeightEntry] {
        var map: [Int32: ChallengeFruitWeightEntry] = [:]
        map.reserveCapacity(fruits.count)

        for fruit in fruits {
            let id = fruit.id
            if map[id] != nil { continue }
            map[id] = ChallengeFruitWeightEntry(
                chance: fruit.chance,
                noCritical: fruit.noCritical,
                coinsMax: coinsMaxDefault
            )
        }

        return map
    }

    /// 返回重复出现的 id（按首次重复出现顺序），用于资源体检与测试断言。
    public static func duplicateIdentifiers(in fruits: [FruitInfo]) -> [Int32] {
        var seen = Set<Int32>()
        var reported = Set<Int32>()
        var duplicates: [Int32] = []

        for fruit in fruits {
            if !seen.insert(fruit.id).inserted, reported.insert(fruit.id).inserted {
                duplicates.append(fruit.id)
            }
        }

        return duplicates
    }
}

//

//  FNPBCoreTests (T-041 草案)
//
//  原则：期望值一律从真实 Content/ 文档现场派生，不把猜测写成黄金值。

import XCTest
import Foundation
@testable import FNPBCore

final class ChallengeResourceTests: XCTestCase {

    // MARK: - 既有类型访问适配（Kimi 核对后只需改这一个块）

    private enum ExistingAPI {
        static func catalogTierNames(_ catalog: ChallengeCatalog) -> [String] {
            catalog.tiers.map(\.name)
        }

        static func catalogChallengeNames(_ catalog: ChallengeCatalog) -> [String] {
            catalog.tiers.flatMap(\.groups).flatMap(\.challenges).map(\.name)
        }

        static func conditionCount(_ conditions: ChallengeConditions) -> Int {
            conditions.conditions.count
        }

        static func conditionNames(_ conditions: ChallengeConditions) -> [String] {
            conditions.conditions.map(\.name)
        }
    }

    // MARK: - 真实文档键名（[需核对]，不符会被 XCTFail 点名）

    private enum DocumentKeys {
        static let tiers = "tiers"
        static let groups = "groups"
        static let challenges = "challenges"
        static let conditions = "conditions"
        static let name = "name"
    }

    // MARK: - 1. 真实资源加载

    func testLoadsRealChallengeResources() throws {
        let resources = try loadRealResources()

        XCTAssertFalse(
            ExistingAPI.catalogTierNames(resources.catalog).isEmpty,
            "ch1challengeList.json 解析后 tiers 为空——请核对 ChallengeCatalogLoader 或文档内容"
        )
        XCTAssertGreaterThan(
            ExistingAPI.conditionCount(resources.conditions), 0,
            "challengeConditions.json 解析后条件为空——请核对 ChallengeConditionLoader 或文档内容"
        )
    }

    // MARK: - 2. 条件数量与名称（期望值从文档派生）

    func testConditionCountAndNamesMatchDocument() throws {
        let resources = try loadRealResources()
        let expected = try expectedConditionNames()
        let actual = ExistingAPI.conditionNames(resources.conditions)

        XCTAssertEqual(expected.count, actual.count, "条件数量与真实文档不一致")
        XCTAssertEqual(expected, actual, "条件名称顺序与真实文档不一致")
        XCTAssertTrue(actual.allSatisfy { !$0.isEmpty }, "存在空条件名")
        XCTAssertEqual(Set(actual).count, actual.count, "条件名应唯一")
    }

    // MARK: - 3. challengeFruit 占位符

    func testChallengeFruitPlaceholderAppearsInRealDocumentsAndIsReplaced() throws {
        let placeholder = ChallengeConditionText.challengeFruitPlaceholder
        let content = try contentDirectory()

        let catalogText = try String(contentsOf: content.appendingPathComponent("ch1challengeList.json"), encoding: .utf8)
        let conditionsText = try String(contentsOf: content.appendingPathComponent("challengeConditions.json"), encoding: .utf8)

        XCTAssertTrue(
            catalogText.contains(placeholder) || conditionsText.contains(placeholder),
            "真实文档中未出现占位符 \(placeholder)——占位符字形需订正 ChallengeConditionText.challengeFruitPlaceholder"
        )

        let source = "切掉 \(placeholder) x3"
        XCTAssertTrue(ChallengeConditionText.containsFruitPlaceholder(source))

        let formatted = ChallengeConditionText.replacingFruitPlaceholder(in: source, with: "Banana")
        XCTAssertEqual(formatted, "切掉 Banana x3")
        XCTAssertFalse(ChallengeConditionText.containsFruitPlaceholder(formatted))
    }

        let resources = try loadRealResources()

        let expectedTiers = try expectedTierNames()
        XCTAssertEqual(expectedTiers, ExistingAPI.catalogTierNames(resources.catalog), "tier 顺序与文档不一致")

        let expectedChallenges = try expectedChallengeNames()
        XCTAssertFalse(expectedChallenges.isEmpty, "ch1challengeList.json 中未解析出任何 challenge")

            "group/challenge 顺序与文档不一致（若 loader 内部使用字典，会在此暴露）"

    }

    // MARK: - 5. 资源缺失错误

    func testMissingCatalogResourceThrowsExplicitError() throws {

## Kimi 审查风险点

1. **最大风险：既有 API 假设。** `ChallengeCatalogLoader` / `ChallengeConditionLoader` / `ChallengeCatalog` / `ChallengeConditions` / `ChallengeFruitWeightEntry` 的签名与类型名都不在共享上下文里（DECISIONS 只到 T-020）。草案把这 8 处假设集中标注，**签名不符时编译会直接失败，不会静默出错**，但需要你逐条核对后再放行。
2. **占位符字形未验证。** `{challengeFruit}` 是我按任务卡措辞推断的；测试会在真实文档中强制检索，字形不符会红并给出订正提示。若真实占位符在 `ch1challengeList.json` 而非条件文件里，测试已同时检查两份文件。
3. **JSON 键名未验证**（`tiers`/`groups`/`challenges`/`conditions`/`name`）。测试用 `XCTFail` 明确点名，不会静默通过；派生的期望值方式避免了把猜测写成黄金值，但也意味着**测试本身就是一份结构断言**——首次跑红属预期。
4. **`coinsMax` 证据链需你复核到落盘代码**：任务卡说「已核实为 0」，但 findings 里没有 T-014 的 `FruitInfo` 默认值表原文。若 `fruitlist.xml` 实际存在 coin 字段，此常量与 `testWeightMapUsesRuntimeFruitFields` 都要改。
5. **`FruitInfo` 缺省 chance = 0 的证据**同样只有任务卡口径；`testWeightMapKeepsMissingChanceAtZero` 会把这个假设变成显式断言，若 `FruitInfo` 有必填字段导致 `makeFruit` 解码失败，需要换成真实 fruitlist 条目。
6. **保序断言可能揭示 loader 缺陷**：若现有 loader 内部用字典聚合 group/challenge，`testCatalogPreservesTierGroupChallengeOrder` 会失败——这属于 loader 实现问题，超出本卡范围，需要你决定是「加卡修 loader」还是「本卡扩展」。
7. **构建配置缺口**：`project.yml` 是否已把 `Content/ch1challengeList.json` 与 `challengeConditions.json` 打进 App Bundle resources，我无法确认。若未打进去，App 侧会立刻抛 `missingResource`（这正是验收要求的显式失败），但功能上仍缺一条最小 diff。
8. **测试依赖仓库布局**：`#filePath` 向上查找要求 CI checkout 完整仓库；若 pipeline 只跑构建产物或测试在沙箱中执行，路径会失效，需要改为测试资源打入。
9. **`OriginalContent` 的落点未定**：我给了「加两个字段」的 diff 提案但没有直接改生产结构，实际形态（struct 字段 vs App 层注入）需要你拍板，否则 `OriginalContentBundle+Challenges.swift` 的返回值可能无人消费。
10. **错误文案未本地化**：`errorDescription` 为中文，原版文案是英文；若后续要 UI 展示，需要与本地化任务卡对齐（本卡不处理）。
